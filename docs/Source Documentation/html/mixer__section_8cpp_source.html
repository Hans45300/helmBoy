88 int title_width = getTitleWidth();

91 int text_y = getHeight() - text_section_width;
92 int slider_width = size_ratio_ * SLIDER_WIDTH;

104 Justification::centred, 1, 0.65f);
105
106 const String channel_labels[] = {
107 TRANS("osc1"), TRANS("osc2"), TRANS("sub"),
108 TRANS("Noise"), TRANS("RM1"), TRANS("RM2")};
109 for (int channel = 0; channel < MIXER_CHANNEL_COUNT; ++channel) {
110 g.drawFittedText(channel_labels[channel], channel * column_spacing, text_y,
111 slider_width, text_section_width,
112 Justification::centred, 1, 0.55f);

116void MixerSection::resized() {
117 int slider_width = size_ratio_ * SLIDER_WIDTH;
118 int buffer = (getWidth() - MIXER_CHANNEL_COUNT * slider_width)
119 / (MIXER_CHANNEL_COUNT - 1);
120 int column_spacing = slider_width + buffer;
121 int title_width = getTitleWidth();
122 int group_label_height = size_ratio_ * GROUP_LABEL_HEIGHT;
123 int text_section_width = size_ratio_ * TEXT_SECTION_WIDTH;
124 int slider_y = title_width + group_label_height;
125 int slider_height = getHeight() - text_section_width - slider_y;
126
127 osc_1_->setBounds(0, slider_y, slider_width, slider_height);
128 osc_2_->setBounds(column_spacing, slider_y, slider_width, slider_height);
129 sub_->setBounds(2 * column_spacing, slider_y, slider_width, slider_height);
130 noise_->setBounds(3 * column_spacing, slider_y, slider_width, slider_height);
131 ring_mod_osc1_->setBounds(4 * column_spacing, slider_y, slider_width, slider_height);
132 ring_mod_osc2_->setBounds(5 * column_spacing, slider_y, slider_width, slider_height);
133 SynthSection::resized();
134}
