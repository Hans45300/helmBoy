89void StepSequencerSection::paintBackground(Graphics& g) {
90 static const DropShadow component_shadow(Colour(Colors::Color_88000000), 2, Point<int>(0, 1));
91 int text_height = size_ratio_ * TEXT_HEIGHT;
92 float font_size = size_ratio_ * 10.0f;
93 int text_buffer = 6;
94
95 SynthSection::paintBackground(g);
96 g.setColour(Colors::control_label_text);
97 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(font_size));
98 drawTextForComponent(g, TRANS("STEPS"), num_steps_.get(), text_buffer);
99 g.drawText(TRANS("FREQUENCY"),
100 retrigger_->getBounds().getX(), frequency_->getBounds().getBottom() + text_buffer,
101 frequency_->getBounds().getWidth() + 2 * text_height, font_size,
102 Justification::centred, false);
103
104 drawTextForComponent(g, TRANS("SLIDE"), smoothing_.get(), text_buffer);
105
106 component_shadow.drawForRectangle(g, step_sequencer_->getBounds());
107}

109void StepSequencerSection::resized() {
110 int mod_button_width = getModButtonWidth();
111 int title_width = getTitleWidth();
112 int text_width = size_ratio_ * TEXT_WIDTH;
113 int text_height = size_ratio_ * TEXT_HEIGHT;
114 int control_section_width = size_ratio_ * CONTROL_SECTION_WIDTH;
115 int slider_width = size_ratio_* SLIDER_WIDTH;
116
117 step_sequencer_->setBounds(0, title_width, getWidth(),
118 getHeight() - title_width - control_section_width);
119 int y = getHeight() - control_section_width + size_ratio_ * 6.0f;
120
121 modulation_button_->setBounds(size_ratio_ * 10.0f, y, mod_button_width, mod_button_width);
122 num_steps_->setBounds(proportionOfWidth(0.22f), y, text_width, text_height);
123 retrigger_->setBounds(proportionOfWidth(0.51f) - text_height, y, text_height, text_height);
124 frequency_->setBounds(proportionOfWidth(0.51f), y, text_width, text_height);
125 sync_->setBounds(frequency_->getBounds().getX() + text_width, frequency_->getBounds().getY(),
126 text_height, text_height);
127
128 smoothing_->setBounds(proportionOfWidth(0.81f), y, slider_width, text_height);
129 tempo_->setBounds(frequency_->getBounds());
130
131 SynthSection::resized();
132
133 frequency_->setPopupDisplayEnabled(false, false, nullptr);
134 tempo_->setPopupDisplayEnabled(false, false, nullptr);
135}

137void StepSequencerSection::createStepSequencerSliders() {
138 for (int i = 0; i < num_steps_->getMaximum(); ++i) {
139 String num(i);
140 if (num.length() == 1)
141 num = String("0") + num;
142
143 SynthSlider* step = new SynthSlider(String("step_seq_") + num);
144 step->snapToValue(true, 0.0);
145 addSlider(step);
146 sequencer_sliders_.push_back(step);
147 }
148 step_sequencer_->setStepSliders(sequencer_sliders_);
149 step_sequencer_->setNumStepsSlider(num_steps_.get());
150 step_sequencer_->setName("step_sequencer_step");
151}
