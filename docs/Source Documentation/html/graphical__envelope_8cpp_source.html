67void GraphicalEnvelope::paint(Graphics& g) {
68 const float ratio = getHeight() / 100.0f;
69 const float line_width = 1.0f * getHeight() / 150.0f;
70 const PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
71 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
72
73 g.fillAll(Colour(Colors::Color_ff424242));
74
75 g.setColour(Colour(Colors::Color_ff4a4a4a));
76 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
77 g.drawLine(x, 0, x, getHeight());
78 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
79 g.drawLine(0, y, getWidth(), y);
80
81 shadow.drawForPath(g, envelope_line_);
82 g.setColour(Colors::graph_fill);
83 g.fillPath(envelope_line_);
84
85 if (effective_shape_differs_) {
86 g.setColour(Colors::modulation.withAlpha(0.5f));
87 g.fillPath(effective_envelope_line_);
88 }
89
90 g.setColour(Colour(Colors::Color_ff505050));
91 if (delay_slider_ != nullptr)
92 g.drawLine(getDelayX(), 0.0f, getDelayX(), getHeight());
93 g.drawLine(getAttackX(), 0.0f, getAttackX(), getHeight());
94 if (hold_slider_ != nullptr)
95 g.drawLine(getHoldX(), 0.0f, getHoldX(), getHeight());
96 g.drawLine(getDecayX(), getSustainY(), getDecayX(), getHeight());
97
98 g.setColour(Colors::modulation);
99 g.strokePath(envelope_line_, stroke);
100 if (effective_shape_differs_)
101 g.strokePath(effective_envelope_line_, stroke);
102
103 const float phase = telemetry_values_[0];
104 const float amplitude = telemetry_values_[1];
105 const auto geometry = geometry_snapshot_.load(std::memory_order_acquire);
106 if (geometry != nullptr && synth_ != nullptr && telemetry_indices_[0] >= 0 && telemetry_indices_[1] >= 0 &&
107 (amplitude > 0.0f || static_cast<int>(phase) ==
108 static_cast<int>(mopo::Envelope::State::Delaying))) {
109 const Point<float> marker = valuesToPosition(
110 *geometry, phase, amplitude, telemetry_values_[2]);
111 if (marker.x >= -1.0f && marker.x <= 1.0f) {
112 const float x = (marker.x + 1.0f) * getWidth() * 0.5f;
113 g.setColour(Colour(Colors::Color_77ffffff));
114 g.fillRect(x - MARKER_WIDTH * ratio * 0.5f, 0.0f,
115 MARKER_WIDTH * ratio, static_cast<float>(getHeight()));
116 }
117 }
118
119 float hover_line_x = -20;
120 if (delay_hover_)
121 hover_line_x = getDelayX();
122 else if (attack_hover_)
123 hover_line_x = getAttackX();
124 else if (hold_hover_)
125 hover_line_x = getHoldX();
126 else if (decay_hover_)
127 hover_line_x = getDecayX();
128 else if (release_hover_)
129 hover_line_x = getReleaseX();
130
131 g.setColour(Colour(Colors::Color_bbffffff));
132 g.fillRect(hover_line_x - 0.5f, 0.0f, 1.0f, 1.0f * getHeight());
133
134 if (sustain_hover_) {
135 if (mouse_down_) {
136 g.setColour(Colour(Colors::Color_11ffffff));
137 const float grab_radius = 20.0f * ratio;
138 g.fillEllipse(getDecayX() - grab_radius, getSustainY() - grab_radius,
139 grab_radius * 2.0f, grab_radius * 2.0f);
140 }
141
142 g.setColour(Colour(Colors::Color_bbffffff));
143 const float hover_radius = 7.0f * ratio;
144 g.drawEllipse(getDecayX() - hover_radius, getSustainY() - hover_radius,
145 hover_radius * 2.0f, hover_radius * 2.0f, 1.0f);
146 }
147 else if (mouse_down_) {
148 g.setColour(Colour(Colors::Color_11ffffff));
149 g.fillRect(hover_line_x - 10.0f, 0.0f, 20.0f, 1.0f * getHeight());
150 }
151

158void GraphicalEnvelope::mouseMove(const MouseEvent& e) {
159 float x = e.getPosition().x;
160 const float y = e.getPosition().y;
161 const float delay_delta = std::abs(x - getDelayX());
162 const float attack_delta = std::abs(x - getAttackX());
163 const float hold_delta = std::abs(x - getHoldX());
164 const float decay_delta = std::abs(x - getDecayX());
165 const float release_delta = std::abs(x - getReleaseX());
166 const float sustain_delta = std::abs(y - getSustainY());
167
168 const bool delay_is_hover = delay_slider_ != nullptr && delay_delta < HOVER_DISTANCE &&
169 y > getHeight() - HOVER_DISTANCE;
170 const bool hold_is_hover = hold_slider_ != nullptr && hold_delta < HOVER_DISTANCE &&
171 y < HOVER_DISTANCE;
172 const bool a_hover = !delay_is_hover && !hold_is_hover &&
173 attack_delta < decay_delta && attack_delta < HOVER_DISTANCE;
174 const bool d_hover = !delay_is_hover && !hold_is_hover && !a_hover &&
175 decay_delta < release_delta && decay_delta < HOVER_DISTANCE;
176 const bool r_hover = !delay_is_hover && !hold_is_hover && !a_hover && !d_hover &&
177 release_delta < HOVER_DISTANCE;
178 const bool s_hover = !delay_is_hover && !hold_is_hover && !a_hover && !r_hover &&
179 x > getDecayX() - HOVER_DISTANCE &&
180 x < getDecayX() + HOVER_DISTANCE && sustain_delta < HOVER_DISTANCE;
181
182 if (delay_is_hover != delay_hover_ || a_hover != attack_hover_ ||
183 hold_is_hover != hold_hover_ || d_hover != decay_hover_ ||
184 s_hover != sustain_hover_ || r_hover != release_hover_) {
185 delay_hover_ = delay_is_hover;
186 attack_hover_ = a_hover;
187 hold_hover_ = hold_is_hover;
188 decay_hover_ = d_hover;
189 sustain_hover_ = s_hover;
190 release_hover_ = r_hover;
191 repaint();
192 }
193}

244void GraphicalEnvelope::timerCallback() {
245 if (!isShowing())
246 return;
247
248 if (synth_ == nullptr)
249 resolveRealtimeTelemetry();
250 FullInterface* interface = findParentComponentOfClass<FullInterface>();
251 if (interface == nullptr)
252 return;
253
254 const std::array<SynthSlider*, 6> sliders {
255 delay_slider_, attack_slider_, hold_slider_, decay_slider_,
256 sustain_slider_, release_slider_};
257 const std::array<const char*, 6> stage_names {
258 "delay", "attack", "hold", "decay", "sustain", "release"};
259 const std::array<float, 6> base_values = getSliderValues();
260 std::array<float, 6> next_values = base_values;
261 bool has_modulation = false;
262
263 if (interface->isAnimating()) {
264 std::string prefix = getName().toStdString();
265 constexpr char suffix[] = "_envelope";
266 if (prefix.size() > sizeof(suffix) - 1 &&
267 prefix.compare(prefix.size() - (sizeof(suffix) - 1),
268 sizeof(suffix) - 1, suffix) == 0)
269 prefix.resize(prefix.size() - (sizeof(suffix) - 1));
270
271 for (size_t index = 0; index < sliders.size(); ++index) {
272 SynthSlider* slider = sliders[index];
273 if (slider == nullptr)
274 continue;
275 const std::string destination = prefix + "_" + stage_names[index];
276 if (!interface->isDestinationModulated(destination))
277 continue;
278 has_modulation = true;
279 float value = effective_controls_valid_ ? effective_controls_[index] : base_values[index];
280 if (interface->copyModulatedValue(destination, value) && std::isfinite(value))
281 next_values[index] = jlimit(static_cast<float>(slider->getMinimum()),
282 static_cast<float>(slider->getMaximum()), value);
283 else
284 next_values[index] = value;
285 }
286 }
287
288 bool values_changed = !effective_controls_valid_;
289 bool shape_differs = false;
290 for (size_t index = 0; index < sliders.size(); ++index) {
291 SynthSlider* slider = sliders[index];
292 if (slider == nullptr)
293 continue;
294 const float base = static_cast<float>(
295 slider->valueToProportionOfLength(base_values[index]));
296 const float effective = static_cast<float>(
297 slider->valueToProportionOfLength(next_values[index]));
298 const float previous = static_cast<float>(slider->valueToProportionOfLength(
299 effective_controls_valid_ ? effective_controls_[index] : base_values[index]));
300 values_changed = values_changed || std::abs(effective - previous) >= 0.0005f;
301 shape_differs = shape_differs || std::abs(effective - base) >= 0.001f;
302 }
303
304 shape_differs = shape_differs && has_modulation && interface->isAnimating();
305 if (values_changed || shape_differs != effective_shape_differs_) {
306 effective_controls_ = next_values;
307 effective_controls_valid_ = true;
308 effective_shape_differs_ = shape_differs;
309 resetEnvelopeLine();
310 repaint();
311 }
312
313 if (!interface->isAnimating() || synth_ == nullptr ||
314 telemetry_indices_[0] < 0 || telemetry_indices_[1] < 0 ||
315 telemetry_indices_[2] < 0)
316 return;
317
318 std::array<float, 3> next_telemetry {};
319 if (!synth_->copyVisualTelemetry(telemetry_indices_, next_telemetry) ||
320 !std::all_of(next_telemetry.begin(), next_telemetry.end(),
321 [](float value) { return std::isfinite(value); }))
322 return;
323
324 bool telemetry_changed = false;
325 for (size_t index = 0; index < next_telemetry.size(); ++index)
326 telemetry_changed = telemetry_changed ||
327 std::abs(next_telemetry[index] - telemetry_values_[index]) >= 0.0001f;
328 if (telemetry_changed) {
329 telemetry_values_ = next_telemetry;
330 repaint();
331 }
332}

347void GraphicalEnvelope::resolveRealtimeTelemetry() {
348 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
349 if (parent == nullptr)
350 return;
351 synth_ = parent->getSynth();
352 if (synth_ == nullptr)
353 return;
354
355 const std::string source_name = getName().toStdString();
356 telemetry_indices_[0] = synth_->getVisualTelemetryIndex(
357 synth_->getModSource(source_name + "_phase"));
358 telemetry_indices_[1] = synth_->getVisualTelemetryIndex(
359 synth_->getModSource(source_name + "_amp"));
360 telemetry_indices_[2] = synth_->getVisualTelemetryIndex(
361 synth_->getModSource(source_name + "_progress"));
362}

410float GraphicalEnvelope::getReleaseRange() {
411 const double delay_percent = delay_slider_
412 ? delay_slider_->valueToProportionOfLength(delay_slider_->getValue()) : 0.0;
413 const double hold_percent = hold_slider_
414 ? hold_slider_->valueToProportionOfLength(hold_slider_->getValue()) : 0.0;
415 return 1.0f - ATTACK_RANGE_PERCENT - DECAY_RANGE_PERCENT -
416 delay_percent * DELAY_RANGE_PERCENT - hold_percent * HOLD_RANGE_PERCENT;
417}

518void GraphicalEnvelope::resetEnvelopeLine() {
519 const std::array<float, 6> base_controls = getSliderValues();
520 if (!effective_controls_valid_) {
521 effective_controls_ = base_controls;
522 effective_controls_valid_ = true;
523 }
524
525 envelope_line_ = createEnvelopeLine(base_controls);
526 auto geometry = std::make_shared<GeometrySnapshot>();
527 effective_envelope_line_ = createEnvelopeLine(effective_controls_, geometry.get());
528 geometry->width = static_cast<float>(getWidth());
529 geometry->height = static_cast<float>(getHeight());
530 geometry->line = effective_shape_differs_ ? effective_envelope_line_ : envelope_line_;
531 geometry_snapshot_.store(std::move(geometry), std::memory_order_release);
532}

544Path GraphicalEnvelope::createEnvelopeLine(const std::array<float, 6>& controls,
545 GeometrySnapshot* geometry) {
546 const std::array<SynthSlider*, 6> sliders {
547 delay_slider_, attack_slider_, hold_slider_, decay_slider_,
548 sustain_slider_, release_slider_};
549 std::array<float, 6> proportions {};
550 for (size_t index = 0; index < sliders.size(); ++index) {
551 if (sliders[index] != nullptr)
552 proportions[index] = static_cast<float>(
553 sliders[index]->valueToProportionOfLength(controls[index]));
554 }
555
556 const float delay_x = delay_slider_
557 ? 1.0f + (getWidth() - 1.0f) * proportions[0] * DELAY_RANGE_PERCENT : 1.0f;
558 const float attack_x = delay_x + (getWidth() - 1.0f) *
559 proportions[1] * ATTACK_RANGE_PERCENT;
560 const float hold_x = attack_x + (getWidth() - 1.0f) *
561 proportions[2] * HOLD_RANGE_PERCENT;
562 const float decay_x = hold_x + getWidth() * proportions[3] * DECAY_RANGE_PERCENT;
563 const float sustain_y = getHeight() * (1.0f - proportions[4]);
564 const float release_range = std::max(0.0f, 1.0f - ATTACK_RANGE_PERCENT -
565 DECAY_RANGE_PERCENT - proportions[0] * DELAY_RANGE_PERCENT -
566 proportions[2] * HOLD_RANGE_PERCENT);
567 const float release_x = decay_x + getWidth() * proportions[5] * release_range;
568
569 if (geometry != nullptr) {
570 geometry->delay_x = delay_x;
571 geometry->attack_x = attack_x;
572 geometry->hold_x = hold_x;
573 geometry->decay_x = decay_x;
574 geometry->release_x = release_x;
575 geometry->sustain = proportions[4];
576 }
577
578 Path line;
579 line.startNewSubPath(1.0f, static_cast<float>(getHeight()));
580 line.lineTo(delay_x, static_cast<float>(getHeight()));
581 line.lineTo(attack_x, 0.0f);
582 line.lineTo(hold_x, 0.0f);
583 line.quadraticTo(0.5f * (hold_x + decay_x), sustain_y, decay_x, sustain_y);
584 line.quadraticTo(0.5f * (release_x + decay_x), static_cast<float>(getHeight()),
585 release_x, static_cast<float>(getHeight()));
586 return line;
587}

589Point<float> GraphicalEnvelope::valuesToPosition(const GeometrySnapshot& geometry,
590 float phase, float amp,
591 float progress) const {
592 if (phase < 0.0f || phase > static_cast<float>(mopo::Envelope::State::Holding) ||
593 geometry.width <= 0.0f || geometry.height <= 0.0f)
594 return Point<float>(-2.0f, -2.0f);
595
596 const int stage = static_cast<int>(phase);
597 float x = 0.0f;
598 const float delay_x = geometry.delay_x;
599 const float attack_x = geometry.attack_x;
600 const float hold_x = geometry.hold_x;
601 const float decay_x = geometry.decay_x;
602 const float release_x = geometry.release_x;
603
604 if (stage == static_cast<int>(mopo::Envelope::State::Delaying))
605 x = 1.0f + progress * (delay_x - 1.0f);
606 else if (stage == static_cast<int>(mopo::Envelope::State::Attacking))
607 x = delay_x + amp * (attack_x - delay_x);
608 else if (stage == static_cast<int>(mopo::Envelope::State::Holding))
609 x = attack_x + progress * (hold_x - attack_x);
610 else if (stage == static_cast<int>(mopo::Envelope::State::Decaying)) {
611 const float difference = 1.0f - geometry.sustain;
