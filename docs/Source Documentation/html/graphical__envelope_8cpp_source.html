49void GraphicalEnvelope::paint(Graphics& g) {
50 static const PathStrokeType stroke(1.5f, PathStrokeType::beveled, PathStrokeType::rounded);
51 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
52
53 g.fillAll(Colour(Colors::Color_ff424242));
54
55 g.setColour(Colour(Colors::Color_ff4a4a4a));
56 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
57 g.drawLine(x, 0, x, getHeight());
58 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
59 g.drawLine(0, y, getWidth(), y);
60
61 shadow.drawForPath(g, envelope_line_);
62 g.setColour(Colors::graph_fill);
63 g.fillPath(envelope_line_);
64
65 g.setColour(Colour(Colors::Color_ff505050));
66 g.drawLine(getAttackX(), 0.0f, getAttackX(), getHeight());
67 g.drawLine(getDecayX(), getSustainY(), getDecayX(), getHeight());
68
69 g.setColour(Colors::modulation);
70 g.strokePath(envelope_line_, stroke);
71
72 float hover_line_x = -20;
73 if (attack_hover_)
74 hover_line_x = getAttackX();
75 else if (decay_hover_)
76 hover_line_x = getDecayX();
77 else if (release_hover_)
78 hover_line_x = getReleaseX();
79
80 g.setColour(Colour(Colors::Color_bbffffff));
81 g.fillRect(hover_line_x - 0.5f, 0.0f, 1.0f, 1.0f * getHeight());
82
83 if (sustain_hover_) {
84 if (mouse_down_) {
85 g.setColour(Colour(Colors::Color_11ffffff));
86 g.fillEllipse(getDecayX() - 20.0, getSustainY() - 20.0, 40.0, 40.0);
87 }
88
89 g.setColour(Colour(Colors::Color_bbffffff));
90 g.drawEllipse(getDecayX() - 7.0, getSustainY() - 7.0, 14.0, 14.0, 1.0);
91 }
92 else if (mouse_down_) {
93 g.setColour(Colour(Colors::Color_11ffffff));
94 g.fillRect(hover_line_x - 10.0f, 0.0f, 20.0f, 1.0f * getHeight());
95 }
96
97 g.setColour(Colors::modulation);
98 g.fillEllipse(getDecayX() - MARKER_WIDTH / 2.0f, getSustainY() - MARKER_WIDTH / 2.0f,
99 MARKER_WIDTH, MARKER_WIDTH);
100 g.setColour(Colour(Colors::Color_ff000000));
101 g.fillEllipse(getDecayX() - MARKER_WIDTH / 4.0f, getSustainY() - MARKER_WIDTH / 4.0f,
102 MARKER_WIDTH / 2.0f, MARKER_WIDTH / 2.0f);
103}

109void GraphicalEnvelope::mouseMove(const MouseEvent& e) {
110 float x = e.getPosition().x;
111 float attack_delta = fabs(x - getAttackX());
112 float decay_delta = fabs(x - getDecayX());
113 float release_delta = fabs(x - getReleaseX());
114 float sustain_delta = fabs(e.getPosition().y - getSustainY());
115
116 bool a_hover = attack_delta < decay_delta && attack_delta < HOVER_DISTANCE;
117 bool d_hover = !attack_hover_ && decay_delta < release_delta && decay_delta < HOVER_DISTANCE;
118 bool r_hover = !decay_hover_ && release_delta < HOVER_DISTANCE;
119 bool s_hover = !a_hover && !r_hover && x > getDecayX() - HOVER_DISTANCE &&
120 x < getDecayX() + HOVER_DISTANCE && sustain_delta < HOVER_DISTANCE;
121
122 if (a_hover != attack_hover_ || d_hover != decay_hover_ ||
123 s_hover != sustain_hover_ || r_hover != release_hover_) {
124 attack_hover_ = a_hover;
125 decay_hover_ = d_hover;
126 sustain_hover_ = s_hover;
127 release_hover_ = r_hover;
128 repaint();
129 }
130}

172float GraphicalEnvelope::getAttackX() {
173 if (!attack_slider_)
174 return 0.0;
175
176 double percent = attack_slider_->valueToProportionOfLength(attack_slider_->getValue());
177 return 1 + (getWidth() - 1) * percent * ATTACK_RANGE_PERCENT;
178}
179
180float GraphicalEnvelope::getDecayX() {
181 if (!decay_slider_)
182 return 0.0;
183
184 double percent = decay_slider_->valueToProportionOfLength(decay_slider_->getValue());
185 return getAttackX() + getWidth() * percent * DECAY_RANGE_PERCENT;
186}
187
188float GraphicalEnvelope::getSustainY() {
189 if (!sustain_slider_)
190 return 0.0;
191
192 double percent = sustain_slider_->valueToProportionOfLength(sustain_slider_->getValue());
193 return getHeight() * (1.0 - percent);
194}
195
196float GraphicalEnvelope::getReleaseX() {
197 if (!release_slider_)
198 return 0.0;
199
200 double percent = release_slider_->valueToProportionOfLength(release_slider_->getValue());
201 return getDecayX() + getWidth() * percent * (1.0 - DECAY_RANGE_PERCENT - ATTACK_RANGE_PERCENT);
202}

227void GraphicalEnvelope::setReleaseX(double x) {
228 if (!release_slider_)

264void GraphicalEnvelope::resetEnvelopeLine() {
265 envelope_line_.clear();
266 envelope_line_.startNewSubPath(1, getHeight());
267 envelope_line_.lineTo(getAttackX(), 0.0f);
268 envelope_line_.quadraticTo(0.5f * (getAttackX() + getDecayX()), getSustainY(),
269 getDecayX(), getSustainY());
270
271 envelope_line_.quadraticTo(0.5f * (getReleaseX() + getDecayX()), getHeight(),
272 getReleaseX(), getHeight());
273}
