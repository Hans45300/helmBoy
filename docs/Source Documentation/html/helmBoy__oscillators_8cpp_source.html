90 void HelmBoyOscillators::loadBasePhaseInc() {
91 int samples = buffer_size_;
92
93 int* dest1 = oscillator1_phase_diffs_;
94 int* dest2 = oscillator2_phase_diffs_;
95
96 const mopo_float* src1 = input(kOscillator1PhaseInc)->source->buffer;
97 const mopo_float* src2 = input(kOscillator2PhaseInc)->source->buffer;
98
99 VECTORIZE_LOOP
100 for (auto i = 0; i < samples; ++i) {
101 dest1[i] = UINT_MAX * src1[i];
102 dest2[i] = UINT_MAX * src2[i];
103 }
104
105 for (auto i = 1; i < samples; ++i) {
106 dest1[i] += dest1[i - 1];
107 dest2[i] += dest2[i - 1];
108 }

111 void HelmBoyOscillators::computeDetuneRatios(std::span<int> detune_diffs,
112 int oscillator_diff,
113 bool harmonize, mopo_float detune,
114 int voices) {
115 int harmonize_mult = harmonize ? 1 : 0;
116 for (auto v = 0; v < MAX_UNISON; ++v) {
117 mopo_float amount = (detune * ((v + 1) / 2)) / ((voices + 1) / 2);
118
119 if (v % 2)
120 amount = -amount;
121
122 mopo_float harmonic = harmonize_mult * v;
123
124 mopo_float detune_ratio = harmonic + DetuneLookup::detuneLookup(amount);
125 detune_diffs[v] = detune_ratio * oscillator_diff - oscillator_diff;
126 }
127 }

139 void HelmBoyOscillators::processInitial() {
140 loadBasePhaseInc();
141
142 int voices1 = utils::iclamp(input(kUnisonVoices1)->source->buffer[0], 1, MAX_UNISON);
143 int voices2 = utils::iclamp(input(kUnisonVoices2)->source->buffer[0], 1, MAX_UNISON);
144 mopo_float detune1 = input(kUnisonDetune1)->source->buffer[0];
145 mopo_float detune2 = input(kUnisonDetune2)->source->buffer[0];
146 mopo_float harmonize1 = input(kHarmonize1)->source->buffer[0];
147 mopo_float harmonize2 = input(kHarmonize2)->source->buffer[0];
148
149 computeDetuneRatios(std::span<int>(detune_diffs1_, MAX_UNISON), oscillator1_phase_diffs_[0],
150 harmonize1, detune1, voices1);
151 computeDetuneRatios(std::span<int>(detune_diffs2_, MAX_UNISON), oscillator2_phase_diffs_[0],
152 harmonize2, detune2, voices2);
153
154 int wave1 = static_cast<int>(input(kOscillator1Waveform)->source->buffer[0] + 0.5);
155 int wave2 = static_cast<int>(input(kOscillator2Waveform)->source->buffer[0] + 0.5);
156 wave1 = utils::iclamp(wave1, 0, FixedPointWaveLookup::kWhiteNoise - 1);
157 wave2 = utils::iclamp(wave2, 0, FixedPointWaveLookup::kWhiteNoise - 1);
158
159 prepareBuffers(wave_buffers1_, std::span<const int>(detune_diffs1_, MAX_UNISON), std::span<const int>(oscillator1_phase_diffs_, MAX_BUFFER_SIZE), wave1);
160 prepareBuffers(wave_buffers2_, std::span<const int>(detune_diffs2_, MAX_UNISON), std::span<const int>(oscillator2_phase_diffs_, MAX_BUFFER_SIZE), wave2);
161 }

163 void HelmBoyOscillators::processCrossMod() {
164 mopo_float cross_mod = input(kCrossMod)->at(0);
165 mopo_float fm_amount = input(kFM_amount)->at(0);
166 const int* phase_diffs1 = oscillator1_phase_diffs_;
167 const int* phase_diffs2 = oscillator2_phase_diffs_;
168 int* dest_cross_mod2 = oscillator2_cross_mods_;
169 int* dest_cross_mod1 = oscillator1_cross_mods_;
170 int* dest_fm = fm_mods_;
171
172 if (cross_mod == 0.0 && fm_amount == 0.0) {
173 utils::zeroBuffer(std::span<int>(dest_cross_mod1, buffer_size_));
174 utils::zeroBuffer(std::span<int>(dest_cross_mod2, buffer_size_));
175 utils::zeroBuffer(std::span<int>(dest_fm, buffer_size_));
176 return;
177 }
178
179 int i = 0;
180 if (input(kReset)->source->triggered) {
181 int trigger_offset = input(kReset)->source->trigger_offset;
182 for (; i < trigger_offset; ++i) {
183 tickCrossMod(i, cross_mod, fm_amount, dest_cross_mod1, dest_cross_mod2, dest_fm,
184 oscillator1_phase_base_ + phase_diffs1[i],
185 oscillator2_phase_base_ + phase_diffs2[i]);
186 }
187
188 oscillator1_cross_mods_[i] = 0;
189 oscillator2_cross_mods_[i] = 0;
190 fm_mods_[i] = 0;
191 oscillator1_cross_mods_[i + 1] = 0;
192 oscillator2_cross_mods_[i + 1] = 0;
193 fm_mods_[i + 1] = 0;
194
195 oscillator1_phase_base_ = 0.0;
196 oscillator2_phase_base_ = 0.0;
197 }
198 for (; i < buffer_size_; ++i) {
199 tickCrossMod(i, cross_mod, fm_amount, dest_cross_mod1, dest_cross_mod2, dest_fm,
200 oscillator1_phase_base_ + phase_diffs1[i],
201 oscillator2_phase_base_ + phase_diffs2[i]);
202 }
203 }

205 void HelmBoyOscillators::processVoices() {
206 int voices1 = utils::iclamp(input(kUnisonVoices1)->source->buffer[0], 1, MAX_UNISON);
207 int voices2 = utils::iclamp(input(kUnisonVoices2)->source->buffer[0], 1, MAX_UNISON);
208 mopo_float stretch1 = input(kPhaseStretch1)->source->buffer[0];
209 mopo_float stretch2 = input(kPhaseStretch2)->source->buffer[0];
210
211 utils::zeroBuffer(std::span<mopo_float>(oscillator1_totals_, buffer_size_));
212 utils::zeroBuffer(std::span<mopo_float>(oscillator2_totals_, buffer_size_));
213
214 int j = 0;
215 if (input(kReset)->source->triggered) {
216 int trigger_offset = input(kReset)->source->trigger_offset;
217 for (; j < trigger_offset; ++j)
218 tickInitialVoices(j, stretch1, stretch2);
219
220 oscillator1_phases_[0] = 0;
221 oscillator2_phases_[0] = 0;
222 }
223
224 for (; j < buffer_size_; ++j)
225 tickInitialVoices(j, stretch1, stretch2);
226
227 for (int v = 1; v < voices1; ++v) {
228 const mopo_float* wave_buffer = wave_buffers1_[v];
229 unsigned int start_phase = oscillator1_phases_[v];
230 int detune = detune_diffs1_[v];
231
232 int i = 0;
233 if (input(kReset)->source->triggered) {
234 int trigger_offset = input(kReset)->source->trigger_offset;
235 for (; i < trigger_offset; ++i)
236 tickVoice1(i, v, wave_buffer, start_phase, detune, stretch1);
237
238 oscillator1_phases_[v] = (UINT_MAX / RAND_MAX) * rand();
239 }
240
241 for (; i < buffer_size_; ++i)
242 tickVoice1(i, v, wave_buffer, start_phase, detune, stretch1);
243 }
244
245 for (int v = 1; v < voices2; ++v) {
246 const mopo_float* wave_buffer = wave_buffers2_[v];
247 unsigned int start_phase = oscillator2_phases_[v];
248 int detune = detune_diffs2_[v];
249
250 int i = 0;
251 if (input(kReset)->source->triggered) {
252 int trigger_offset = input(kReset)->source->trigger_offset;
253 for (; i < trigger_offset; ++i)
254 tickVoice2(i, v, wave_buffer, start_phase, detune, stretch2);
255
256 oscillator2_phases_[v] = (UINT_MAX / RAND_MAX) * rand();
257 }
258 for (; i < buffer_size_; ++i)
259 tickVoice2(i, v, wave_buffer, start_phase, detune, stretch2);
260 }
261
262 finishVoices(voices1, voices2);
263 }

265 void HelmBoyOscillators::finishVoices(int voices1, int voices2) {
266 mopo_float scale1 = scales[voices1];
267 mopo_float scale2 = scales[voices2];
268
269 mopo_float* dest = output()->buffer;
270 const mopo_float* amp1 = input(kOscillator1Amplitude)->source->buffer;
271 const mopo_float* amp2 = input(kOscillator2Amplitude)->source->buffer;
272 const mopo_float* oscillator1_totals = oscillator1_totals_;
273 const mopo_float* oscillator2_totals = oscillator2_totals_;
274 mopo_float ring_mod_amount = input(kRingMod)->at(0);
275
276 VECTORIZE_LOOP
277 for (auto j = 0; j < buffer_size_; ++j)
278 tickOut(j, dest, amp1, amp2, oscillator1_totals, oscillator2_totals, scale1, scale2, ring_mod_amount);
279
280 oscillator1_cross_mods_[0] = oscillator1_cross_mods_[buffer_size_];
281 oscillator2_cross_mods_[0] = oscillator2_cross_mods_[buffer_size_];
282 fm_mods_[0] = fm_mods_[buffer_size_];
283
284 oscillator1_phase_base_ += oscillator1_phase_diffs_[buffer_size_ - 1];
285 oscillator2_phase_base_ += oscillator2_phase_diffs_[buffer_size_ - 1];
286
287 for (auto v = 0; v < MAX_UNISON; ++v) {
288 oscillator1_phases_[v] += oscillator1_phase_diffs_[buffer_size_ - 1] +
289 buffer_size_ * detune_diffs1_[v];
290 oscillator2_phases_[v] += oscillator2_phase_diffs_[buffer_size_ - 1] +
291 buffer_size_ * detune_diffs2_[v];
292 }
293 }
