91 void HelmBoyOscillators::loadBasePhaseInc() {
92 int samples = buffer_size_;
93
94 int* dest1 = oscillator1_phase_diffs_;
95 int* dest2 = oscillator2_phase_diffs_;
96
97 const mopo_float* src1 = input(kOscillator1PhaseInc)->source->buffer;
98 const mopo_float* src2 = input(kOscillator2PhaseInc)->source->buffer;
99
100 VECTORIZE_LOOP
101 for (auto i = 0; i < samples; ++i) {
102 dest1[i] = UINT_MAX * src1[i];
103 dest2[i] = UINT_MAX * src2[i];
104 }
105
106 for (auto i = 1; i < samples; ++i) {
107 dest1[i] += dest1[i - 1];
108 dest2[i] += dest2[i - 1];
109 }

112 void HelmBoyOscillators::computeDetuneRatios(std::span<int> detune_diffs,
113 int oscillator_diff,
114 bool harmonize, mopo_float detune,
115 int voices) {
116 int harmonize_mult = harmonize ? 1 : 0;
117 for (auto v = 0; v < MAX_UNISON; ++v) {
118 mopo_float amount = (detune * ((v + 1) / 2)) / ((voices + 1) / 2);
119
120 if (v % 2)
121 amount = -amount;
122
123 mopo_float harmonic = harmonize_mult * v;
124
125 mopo_float detune_ratio = harmonic + DetuneLookup::detuneLookup(amount);
126 detune_diffs[v] = detune_ratio * oscillator_diff - oscillator_diff;
127 }
128 }

140 void HelmBoyOscillators::processInitial() {
141 loadBasePhaseInc();
142
143 const unsigned int oscillator1_phase_offset =
144 utils::phaseToFixedPoint(input(kOscillator1PhaseOffset)->at(0));
145 const unsigned int oscillator2_phase_offset =
146 utils::phaseToFixedPoint(input(kOscillator2PhaseOffset)->at(0));
147 for (int i = 0; i < buffer_size_; ++i) {
148 oscillator1_phase_offsets_[i] = oscillator1_phase_offset;
149 oscillator2_phase_offsets_[i] = oscillator2_phase_offset;
150 }
151
152 int voices1 = utils::iclamp(input(kUnisonVoices1)->source->buffer[0], 1, MAX_UNISON);
153 int voices2 = utils::iclamp(input(kUnisonVoices2)->source->buffer[0], 1, MAX_UNISON);
154 mopo_float detune1 = input(kUnisonDetune1)->source->buffer[0];
155 mopo_float detune2 = input(kUnisonDetune2)->source->buffer[0];
156 mopo_float harmonize1 = input(kHarmonize1)->source->buffer[0];
157 mopo_float harmonize2 = input(kHarmonize2)->source->buffer[0];
158
159 computeDetuneRatios(std::span<int>(detune_diffs1_, MAX_UNISON), oscillator1_phase_diffs_[0],
160 harmonize1, detune1, voices1);
161 computeDetuneRatios(std::span<int>(detune_diffs2_, MAX_UNISON), oscillator2_phase_diffs_[0],
162 harmonize2, detune2, voices2);
163
164 int wave1 = static_cast<int>(input(kOscillator1Waveform)->source->buffer[0] + 0.5);
165 int wave2 = static_cast<int>(input(kOscillator2Waveform)->source->buffer[0] + 0.5);
166 wave1 = utils::iclamp(wave1, 0, FixedPointWaveLookup::kWhiteNoise - 1);
167 wave2 = utils::iclamp(wave2, 0, FixedPointWaveLookup::kWhiteNoise - 1);
168
169 prepareBuffers(wave_buffers1_, std::span<const int>(detune_diffs1_, MAX_UNISON), std::span<const int>(oscillator1_phase_diffs_, MAX_BUFFER_SIZE), wave1);
170 prepareBuffers(wave_buffers2_, std::span<const int>(detune_diffs2_, MAX_UNISON), std::span<const int>(oscillator2_phase_diffs_, MAX_BUFFER_SIZE), wave2);
171 }

173 void HelmBoyOscillators::processCrossMod() {
174 mopo_float cross_mod = input(kCrossMod)->at(0);
175 mopo_float fm_amount = input(kFM_amount)->at(0);
176 const int* phase_diffs1 = oscillator1_phase_diffs_;
177 const int* phase_diffs2 = oscillator2_phase_diffs_;
178 int* dest_cross_mod2 = oscillator2_cross_mods_;
179 int* dest_cross_mod1 = oscillator1_cross_mods_;
180 int* dest_fm = fm_mods_;
181
182 if (cross_mod == 0.0 && fm_amount == 0.0) {
183 utils::zeroBuffer(std::span<int>(dest_cross_mod1, buffer_size_));
184 utils::zeroBuffer(std::span<int>(dest_cross_mod2, buffer_size_));
185 utils::zeroBuffer(std::span<int>(dest_fm, buffer_size_));
186 return;
187 }
188
189 int i = 0;
190 if (input(kReset)->source->triggered) {
191 int trigger_offset = input(kReset)->source->trigger_offset;
192 for (; i < trigger_offset; ++i) {
193 tickCrossMod(i, cross_mod, fm_amount, dest_cross_mod1, dest_cross_mod2, dest_fm,
194 oscillator1_phase_base_ + phase_diffs1[i],
195 oscillator2_phase_base_ + phase_diffs2[i]);
196 }
197
198 oscillator1_cross_mods_[i] = 0;
199 oscillator2_cross_mods_[i] = 0;
200 fm_mods_[i] = 0;
201 oscillator1_cross_mods_[i + 1] = 0;
202 oscillator2_cross_mods_[i + 1] = 0;
203 fm_mods_[i + 1] = 0;
204
205 oscillator1_phase_base_ = 0.0;
206 oscillator2_phase_base_ = 0.0;
207 }
208 for (; i < buffer_size_; ++i) {
209 tickCrossMod(i, cross_mod, fm_amount, dest_cross_mod1, dest_cross_mod2, dest_fm,
210 oscillator1_phase_base_ + phase_diffs1[i],
211 oscillator2_phase_base_ + phase_diffs2[i]);
212 }
213 }

215 void HelmBoyOscillators::processVoices() {
216 int voices1 = utils::iclamp(input(kUnisonVoices1)->source->buffer[0], 1, MAX_UNISON);
217 int voices2 = utils::iclamp(input(kUnisonVoices2)->source->buffer[0], 1, MAX_UNISON);
218 mopo_float stretch1 = input(kPhaseStretch1)->source->buffer[0];
219 mopo_float stretch2 = input(kPhaseStretch2)->source->buffer[0];
220
221 utils::zeroBuffer(std::span<mopo_float>(oscillator1_totals_, buffer_size_));
222 utils::zeroBuffer(std::span<mopo_float>(oscillator2_totals_, buffer_size_));
223
224 if (input(kHardSync)->at(0) >= 0.5) {
225 processHardSyncVoices(voices1, voices2, stretch1, stretch2);
226 return;
227 }
228
229 int j = 0;
230 if (input(kReset)->source->triggered) {
231 int trigger_offset = input(kReset)->source->trigger_offset;
232 for (; j < trigger_offset; ++j)
233 tickInitialVoices(j, stretch1, stretch2);
234
235 oscillator1_phases_[0] = 0;
236 oscillator2_phases_[0] = 0;
237 }
238
239 for (; j < buffer_size_; ++j)
240 tickInitialVoices(j, stretch1, stretch2);
241
242 for (int v = 1; v < voices1; ++v) {
243 const mopo_float* wave_buffer = wave_buffers1_[v];
244 unsigned int start_phase = oscillator1_phases_[v];
245 int detune = detune_diffs1_[v];
246
247 int i = 0;
248 if (input(kReset)->source->triggered) {
249 int trigger_offset = input(kReset)->source->trigger_offset;
250 for (; i < trigger_offset; ++i)
251 tickVoice1(i, v, wave_buffer, start_phase, detune, stretch1);
252
253 oscillator1_phases_[v] = (UINT_MAX / RAND_MAX) * rand();
254 }
255
256 for (; i < buffer_size_; ++i)
257 tickVoice1(i, v, wave_buffer, start_phase, detune, stretch1);
258 }
259
260 for (int v = 1; v < voices2; ++v) {
261 const mopo_float* wave_buffer = wave_buffers2_[v];
262 unsigned int start_phase = oscillator2_phases_[v];
263 int detune = detune_diffs2_[v];
264
265 int i = 0;
266 if (input(kReset)->source->triggered) {
267 int trigger_offset = input(kReset)->source->trigger_offset;
268 for (; i < trigger_offset; ++i)
269 tickVoice2(i, v, wave_buffer, start_phase, detune, stretch2);
270
271 oscillator2_phases_[v] = (UINT_MAX / RAND_MAX) * rand();
272 }
273 for (; i < buffer_size_; ++i)
274 tickVoice2(i, v, wave_buffer, start_phase, detune, stretch2);
275 }
276
277 finishVoices(voices1, voices2);
278 }

280 void HelmBoyOscillators::processHardSyncVoices(int voices1, int voices2,
281 mopo_float stretch1,
282 mopo_float stretch2) {
283 int trigger_offset = -1;
284 if (input(kReset)->source->triggered)
285 trigger_offset = input(kReset)->source->trigger_offset;
286
287 unsigned int master_phase_base = oscillator1_phases_[0];
288 unsigned int previous_master_phase = master_phase_base;
289 for (int sample_index = 0; sample_index < buffer_size_; ++sample_index) {
290 if (sample_index == trigger_offset) {
291 master_phase_base = 0;
292 previous_master_phase = 0;
293 }
294
295 unsigned int current_master_phase = master_phase_base +
296 static_cast<unsigned int>(oscillator1_phase_diffs_[sample_index]);
297 oscillator1_cycle_wraps_[sample_index] = sample_index != trigger_offset &&
298 current_master_phase < previous_master_phase;
299 previous_master_phase = current_master_phase;
300 }
301
302 unsigned int oscillator1_phase_base = oscillator1_phases_[0];
303 unsigned int oscillator2_phase_base = oscillator2_phases_[0];
304 unsigned int oscillator2_sync_offset = 0;
305 unsigned int oscillator2_phase_diff_origin = 0;
306 for (int sample_index = 0; sample_index < buffer_size_; ++sample_index) {
307 if (sample_index == trigger_offset) {
308 oscillator1_phase_base = 0;
309 oscillator1_phases_[0] = 0;
310 oscillator2_phase_base = 0;
311 oscillator2_phases_[0] = 0;
312 oscillator2_phase_diff_origin =
313 static_cast<unsigned int>(oscillator2_phase_diffs_[sample_index]);
314 oscillator2_sync_offset = 0;
315 }
316
317 unsigned int phase1 = static_cast<unsigned int>(oscillator2_cross_mods_[sample_index]) +
318 oscillator1_phase_base +
319 static_cast<unsigned int>(oscillator1_phase_diffs_[sample_index]) +
320 oscillator1_phase_offsets_[sample_index];
321 unsigned int oscillator2_phase = oscillator2_phase_base +
322 static_cast<unsigned int>(oscillator2_phase_diffs_[sample_index]) -
323 oscillator2_phase_diff_origin;
324 if (oscillator1_cycle_wraps_[sample_index])
325 oscillator2_sync_offset = 0u - oscillator2_phase;
326
327 unsigned int phase2 = static_cast<unsigned int>(oscillator1_cross_mods_[sample_index]) +
328 static_cast<unsigned int>(fm_mods_[sample_index]) +
329 oscillator2_phase + oscillator2_sync_offset +
330 oscillator2_phase_offsets_[sample_index];
331 unsigned int warped_phase1 = remapPhase(phase1, stretch1);
332 unsigned int warped_phase2 = remapPhase(phase2, stretch2);
333 oscillator1_totals_[sample_index] +=
334 FixedPointWave::interpretWave(wave_buffers1_[0], warped_phase1);
335 oscillator2_totals_[sample_index] +=
336 FixedPointWave::interpretWave(wave_buffers2_[0], warped_phase2);
337 }
338
339 oscillator2_phases_[0] = oscillator2_phase_base +
340 static_cast<unsigned int>(oscillator2_phase_diffs_[buffer_size_ - 1]) -
341 oscillator2_phase_diff_origin + oscillator2_sync_offset;
342
343 for (int voice_index = 1; voice_index < voices1; ++voice_index) {
344 const mopo_float* wave_buffer = wave_buffers1_[voice_index];
345 unsigned int start_phase = oscillator1_phases_[voice_index];
346 int detune = detune_diffs1_[voice_index];
347
348 int sample_index = 0;
349 if (input(kReset)->source->triggered) {
350 for (; sample_index < trigger_offset; ++sample_index)
351 tickVoice1(sample_index, voice_index, wave_buffer, start_phase, detune, stretch1);
352
353 oscillator1_phases_[voice_index] = (UINT_MAX / RAND_MAX) * rand();
354 }
355
356 for (; sample_index < buffer_size_; ++sample_index)
357 tickVoice1(sample_index, voice_index, wave_buffer, start_phase, detune, stretch1);
358 }
359
360 for (int voice_index = 1; voice_index < HelmBoyOscillators::MAX_UNISON; ++voice_index) {
361 const mopo_float* wave_buffer = wave_buffers2_[voice_index];
362 unsigned int phase_base = oscillator2_phases_[voice_index];
363 unsigned int phase_diff_origin = 0;
364 unsigned int sync_offset = 0;
365 int phase_origin_index = 0;
366 int detune = detune_diffs2_[voice_index];
367 bool render_voice = voice_index < voices2;
368
369 for (int sample_index = 0; sample_index < buffer_size_; ++sample_index) {
370 if (sample_index == trigger_offset && render_voice) {
371 phase_base = (UINT_MAX / RAND_MAX) * rand();
372 oscillator2_phases_[voice_index] = phase_base;
373 phase_diff_origin = static_cast<unsigned int>(
374 oscillator2_phase_diffs_[sample_index]);
375 phase_origin_index = sample_index;
376 sync_offset = 0;
377 }
378
379 unsigned int phase = phase_base +
380 static_cast<unsigned int>(oscillator2_phase_diffs_[sample_index]) -
381 phase_diff_origin +
382 static_cast<unsigned int>(sample_index - phase_origin_index) *
383 static_cast<unsigned int>(detune);
384 if (oscillator1_cycle_wraps_[sample_index])
385 sync_offset = 0u - phase;
386
387 if (render_voice) {
388 phase += static_cast<unsigned int>(oscillator1_cross_mods_[sample_index]);
389 phase += static_cast<unsigned int>(fm_mods_[sample_index]);
390 phase += sync_offset;
391 unsigned int warped_phase = remapPhase(phase, stretch2);
392 oscillator2_totals_[sample_index] +=
393 FixedPointWave::interpretWave(wave_buffer, warped_phase);
394 }
395 }
396
397 oscillator2_phases_[voice_index] = phase_base +
398 static_cast<unsigned int>(oscillator2_phase_diffs_[buffer_size_ - 1]) -
399 phase_diff_origin +
400 static_cast<unsigned int>(buffer_size_ - phase_origin_index) *
401 static_cast<unsigned int>(detune) + sync_offset;
402 }
403
404 finishVoices(voices1, voices2, true);
405 }

407 void HelmBoyOscillators::finishVoices(int voices1, int voices2,
408 bool oscillator2_phases_updated) {
409 mopo_float scale1 = scales[voices1];
410 mopo_float scale2 = scales[voices2];
411
412 mopo_float* dest = output(0)->buffer;
413 mopo_float* oscillator1_output = output(1)->buffer;
414 mopo_float* oscillator2_output = output(2)->buffer;
415 const mopo_float* amp1 = input(kOscillator1Amplitude)->source->buffer;
416 const mopo_float* amp2 = input(kOscillator2Amplitude)->source->buffer;
417 const mopo_float* ring_mod_amp1 = input(kRingModOscillator1Amplitude)->source->buffer;
418 const mopo_float* ring_mod_amp2 = input(kRingModOscillator2Amplitude)->source->buffer;
419 const mopo_float* oscillator1_totals = oscillator1_totals_;
420 const mopo_float* oscillator2_totals = oscillator2_totals_;
421 mopo_float ring_mod_amount = input(kRingMod)->at(0);
422
423 VECTORIZE_LOOP
424 for (auto j = 0; j < buffer_size_; ++j)
425 tickOut(j, dest, oscillator1_output, oscillator2_output,
426 amp1, amp2, ring_mod_amp1, ring_mod_amp2,
427 oscillator1_totals, oscillator2_totals, scale1, scale2, ring_mod_amount);
428
429 oscillator1_cross_mods_[0] = oscillator1_cross_mods_[buffer_size_];
430 oscillator2_cross_mods_[0] = oscillator2_cross_mods_[buffer_size_];
431 fm_mods_[0] = fm_mods_[buffer_size_];
432
433 oscillator1_phase_base_ += oscillator1_phase_diffs_[buffer_size_ - 1];
434 oscillator2_phase_base_ += oscillator2_phase_diffs_[buffer_size_ - 1];
435
436 for (auto v = 0; v < MAX_UNISON; ++v) {
437 oscillator1_phases_[v] += oscillator1_phase_diffs_[buffer_size_ - 1] +
438 buffer_size_ * detune_diffs1_[v];
439 if (!oscillator2_phases_updated)
440 oscillator2_phases_[v] += oscillator2_phase_diffs_[buffer_size_ - 1] +
441 buffer_size_ * detune_diffs2_[v];
442 }
443 }
