39#define FLOATS_PER_METER 24
40#define INDICES_PER_METER 6
41#define POINTS_PER_METER 4

43OpenGLModulationManager::OpenGLModulationManager(
44 mopo::output_map modulation_sources,
45 std::map<std::string, ModulationButton*> modulation_buttons,
46 std::map<std::string, SynthSlider*> sliders,
47 mopo::output_map mono_modulations,
48 mopo::output_map poly_modulations) {
49 static const int quad_triangles[6] {
50 0, 1, 2,
51 2, 3, 0
52 };
53
54 modulation_buttons_ = modulation_buttons;
55 modulation_sources_ = modulation_sources;
56 setInterceptsMouseClicks(false, true);
57
58 current_modulator_ = "";
59
60 polyphonic_destinations_ = std::make_unique<Component>();
61 polyphonic_destinations_->setInterceptsMouseClicks(false, true);
62
63 monophonic_destinations_ = std::make_unique<Component>();
64 monophonic_destinations_->setInterceptsMouseClicks(false, true);
65
66 for (auto& mod_button : modulation_buttons_) {
67 mod_button.second->addListener(this);
68 mod_button.second->addDisconnectListener(this);
69
70 // Create modulation highlight overlays.
71 std::string name = mod_button.second->getName().toStdString();
72 ModulationHighlight* overlay = new ModulationHighlight();
73 addChildComponent(overlay);
74 overlay_lookup_[name] = overlay;
75 overlay->setName(name);
76 Rectangle<int> local_bounds = mod_button.second->getBoundsInParent();
77 overlay->setBounds(mod_button.second->getParentComponent()->localAreaToGlobal(local_bounds));
78 }
79
80 slider_model_lookup_ = sliders;
81 vertices_ = new float[FLOATS_PER_METER * slider_model_lookup_.size()];
82 triangles_ = new int[INDICES_PER_METER * slider_model_lookup_.size()];
83
84 int i = 0;
85 for (auto& slider : slider_model_lookup_) {
86 std::string name = slider.first;
87 const mopo::Output* mono_total = mono_modulations[name];
88 const mopo::Output* poly_total = poly_modulations[name];
89
90 float* meter_vertices = vertices_ + (i * FLOATS_PER_METER);
91 memset(meter_vertices, 0, FLOATS_PER_METER * sizeof(float));
92
93 int* meter_triangles = triangles_ + (i * INDICES_PER_METER);
94 for (int t = 0; t < INDICES_PER_METER; ++t)
95 meter_triangles[t] = i * POINTS_PER_METER + quad_triangles[t];
96
97 slider.second->addSliderListener(this);
98
99 // Create modulation meter.
100 if (mono_total) {
101 std::string name = slider.second->getName().toStdString();
102 OpenGLModulationMeter* meter = new OpenGLModulationMeter(mono_total, poly_total,
103 slider.second, std::span<float>(meter_vertices, FLOATS_PER_METER));
104 addChildComponent(meter);
105 meter_lookup_[name] = meter;
106 meter->setName(name);
107 Rectangle<int> local_bounds = slider.second->getBoundsInParent();
108 meter->setBounds(slider.second->getParentComponent()->localAreaToGlobal(local_bounds));
109 }
110
111 // Create modulation slider.
112 ModulationSlider* mod_slider = new ModulationSlider(slider.second);
113 mod_slider->setLookAndFeel(ModulationLookAndFeel::instance());
114 mod_slider->addListener(this);
115 if (poly_total)
116 polyphonic_destinations_->addAndMakeVisible(mod_slider);
117 else
118 monophonic_destinations_->addAndMakeVisible(mod_slider);
119
120 slider_lookup_[name] = mod_slider;
121 owned_sliders_.push_back(mod_slider);
122
123 ++i;
124 }
125
126 addAndMakeVisible(polyphonic_destinations_.get());
127 addAndMakeVisible(monophonic_destinations_.get());
128
129 forgetModulator();
130}

144void OpenGLModulationManager::resized() {
145 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
146 polyphonic_destinations_->setBounds(getBounds());
147 monophonic_destinations_->setBounds(getBounds());
148
149 // Update modulation slider locations.
150 for (auto& slider : slider_lookup_) {
151 SynthSlider* model = slider_model_lookup_[slider.first];
152 Point<float> local_top_left = getLocalPoint(model, Point<float>(0.0f, 0.0f));
153 slider.second->setVisible(model->isVisible());
154 slider.second->setBounds(local_top_left.x, local_top_left.y,
155 model->getWidth(), model->getHeight());
156 }
157
158 // Update modulation meter locations.
159 for (auto& meter : meter_lookup_) {
160 Slider* model = slider_model_lookup_[meter.first];
161 Point<float> local_top_left = getLocalPoint(model, Point<float>(0.0f, 0.0f));
162 meter.second->setBounds(local_top_left.x, local_top_left.y,
163 model->getWidth(), model->getHeight());
164 if (parent) {
165 int num_modulations = parent->getSynth()->getNumModulations(meter.first);
166 meter.second->setModulated(num_modulations);
167 meter.second->setVisible(num_modulations);
168 }
169 }
170
171 // Update modulation highlight overlay locations.
172 for (auto& overlay : overlay_lookup_) {
173 ModulationButton* model = modulation_buttons_[overlay.first];
174 Point<float> local_top_left = getLocalPoint(model, Point<float>(0.0f, 0.0f));
175 overlay.second->setBounds(local_top_left.x, local_top_left.y,
176 model->getWidth(), model->getHeight());
177 }
178
179 OpenGLComponent::resized();
180}

213void OpenGLModulationManager::init(OpenGLContext& open_gl_context) {
214 open_gl_context.extensions.glGenBuffers(1, &vertex_buffer_);
215 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
216
217 int num_meters = slider_model_lookup_.size();
218 GLsizeiptr vert_size = static_cast<GLsizeiptr>(num_meters * FLOATS_PER_METER * sizeof(float));
219 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
220 vertices_, GL_STATIC_DRAW);
221
222 open_gl_context.extensions.glGenBuffers(1, &triangle_buffer_);
223 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
224
225 GLsizeiptr tri_size = static_cast<GLsizeiptr>(num_meters * INDICES_PER_METER * sizeof(int));
226 open_gl_context.extensions.glBufferData(GL_ELEMENT_ARRAY_BUFFER, tri_size,
227 triangles_, GL_STATIC_DRAW);
228
229 const char* vertex_shader = Shaders::getShader(Shaders::kModulationVertex);
230 const char* fragment_shader = Shaders::getShader(Shaders::kModulationFragment);
231
232 shader_ = std::make_unique<OpenGLShaderProgram>(open_gl_context);
233
234 if (shader_->addVertexShader(OpenGLHelpers::translateVertexShaderToV3(vertex_shader)) &&
235 shader_->addFragmentShader(OpenGLHelpers::translateFragmentShaderToV3(fragment_shader)) &&
236 shader_->link()) {
237 shader_->use();
238 position_ = std::make_unique<OpenGLShaderProgram::Attribute>(*shader_, "position");
239 coordinates_ = std::make_unique<OpenGLShaderProgram::Attribute>(*shader_, "coordinates");
240 range_ = std::make_unique<OpenGLShaderProgram::Attribute>(*shader_, "range");
241 radius_uniform_ = std::make_unique<OpenGLShaderProgram::Uniform>(*shader_, "radius");
242 }
243}

245void OpenGLModulationManager::render(OpenGLContext& open_gl_context, bool animate) {
246 if (!animate)
247 return;
248
249 for (auto& meter : meter_lookup_) {
250 bool show = meter.second->isModulated() && slider_model_lookup_[meter.first]->isVisible();
251 if (show)
252 meter.second->updateDrawing();
253 }
254
255 glEnable(GL_BLEND);
256 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
257 setViewPort(open_gl_context);
258
259 shader_->use();
260 radius_uniform_->set(0.9f);
261
262 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
263
264 int num_meters = slider_model_lookup_.size();
265 GLsizeiptr vert_size = static_cast<GLsizeiptr>(num_meters * FLOATS_PER_METER * sizeof(float));
266 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
267 vertices_, GL_STATIC_DRAW);
268 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
269
270 if (position_ != nullptr) {
271 open_gl_context.extensions.glVertexAttribPointer(position_->attributeID, 2, GL_FLOAT,
272 GL_FALSE, 6 * sizeof(float), 0);
273 open_gl_context.extensions.glEnableVertexAttribArray(position_->attributeID);
274 }
275
276 if (coordinates_ != nullptr) {
277 open_gl_context.extensions.glVertexAttribPointer(coordinates_->attributeID, 2, GL_FLOAT,
278 GL_FALSE, 6 * sizeof(float),
279 (GLvoid*)(2 * sizeof(float)));
280 open_gl_context.extensions.glEnableVertexAttribArray(coordinates_->attributeID);
281 }
282
283 if (range_ != nullptr) {
284 open_gl_context.extensions.glVertexAttribPointer(range_->attributeID, 2, GL_FLOAT,
285 GL_FALSE, 6 * sizeof(float),
286 (GLvoid*)(4 * sizeof(float)));
287 open_gl_context.extensions.glEnableVertexAttribArray(range_->attributeID);
288 }
289
290 glDrawElements(GL_TRIANGLES, num_meters * INDICES_PER_METER, GL_UNSIGNED_INT, 0);
291
292 if (position_ != nullptr)
293 open_gl_context.extensions.glDisableVertexAttribArray(position_->attributeID);
294
295 if (coordinates_ != nullptr)
296 open_gl_context.extensions.glDisableVertexAttribArray(coordinates_->attributeID);
297
298 if (range_ != nullptr)
299 open_gl_context.extensions.glDisableVertexAttribArray(range_->attributeID);
300
301 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
302 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
303}

376void OpenGLModulationManager::setSliderValues() {
377 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
378 if (parent == nullptr)
379 return;
380
381 std::vector<mopo::ModulationConnection*> connections =
382 parent->getSynth()->getSourceConnections(current_modulator_);
383 for (auto& slider : slider_lookup_) {
384 std::string destination_name = slider.second->getName().toStdString();
385 float value = 0.0f;
386
387 for (mopo::ModulationConnection* connection : connections) {
388 if (connection->destination == destination_name) {
389 value = connection->amount.value();
390 break;
391 }
392 }
393 slider.second->setValue(value, NotificationType::dontSendNotification);
394 slider.second->repaint();
395 }
