81void ExtraModSection::resized() {
82 int button_width = getModButtonWidth();
83 int title_width = getTitleWidth();
84 int x = size_ratio_ * 30;
85 int x2 = getWidth() / 2 + size_ratio_ * 15.0f;
86 float space = (getHeight() - title_width - (3.0f * button_width)) / 4.0f;
87
88 aftertouch_mod_->setBounds(x, title_width + space, button_width, button_width);
89 note_mod_->setBounds(x, aftertouch_mod_->getBottom() + space, button_width, button_width);
90 velocity_mod_->setBounds(x, note_mod_->getBottom() + space, button_width, button_width);
91 mod_wheel_mod_->setBounds(x2, title_width + space, button_width, button_width);
92 pitch_wheel_mod_->setBounds(x2, mod_wheel_mod_->getBottom() + space, button_width, button_width);
93 random_mod_->setBounds(x2, pitch_wheel_mod_->getBottom() + space, button_width, button_width);
94
95 SynthSection::resized();
96}
