67void ExtraModSection::drawTextToRightOfComponent(Graphics& g, Component* component, String text) {
68 float space = size_ratio_ * 6.0f;
69 int column_right = component->getX() < getWidth() / 2
70 ? getWidth() / 2 - size_ratio_ * 4 : getWidth() - size_ratio_ * 4;
71 int text_x = component->getRight() + space;
72 int text_width = jmax(1, column_right - text_x);
73 g.drawFittedText(text, text_x, component->getY(), text_width,
74 component->getHeight(), Justification::centredLeft, 1, 0.65f);
75}

92void ExtraModSection::resized() {
93 int title_width = getTitleWidth();
94 int button_width = jmax(1, static_cast<int>(jmin(getModButtonWidth(), size_ratio_ * 28.0f,
95 (getHeight() - title_width) / 4.0f)));
96 int left_x = size_ratio_ * 10;
97 int right_x = getWidth() / 2 + size_ratio_ * 8;
98 float row_space = (getHeight() - title_width - (4.0f * button_width)) / 5.0f;
99 int row_y[4];
100 for (int row = 0; row < 4; ++row)
101 row_y[row] = title_width + row_space + row * (button_width + row_space);
102
103 channel_aftertouch_mod_->setBounds(left_x, row_y[0], button_width, button_width);
104 poly_aftertouch_mod_->setBounds(right_x, row_y[0], button_width, button_width);
105 note_mod_->setBounds(left_x, row_y[1], button_width, button_width);
106 mod_wheel_mod_->setBounds(right_x, row_y[1], button_width, button_width);
107 velocity_mod_->setBounds(left_x, row_y[2], button_width, button_width);
108 pitch_wheel_mod_->setBounds(right_x, row_y[2], button_width, button_width);
109 random_mod_->setBounds(left_x, row_y[3], button_width, button_width);
110
111 SynthSection::resized();
112}
