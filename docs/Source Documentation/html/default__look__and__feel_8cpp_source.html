32 // Gradient colors: 0% saphir, 25% émeraude, 50% jaune citron, 75% orange, 100% rubis
33 Colour getGradientColour(float position) {
34 // Clamp position entre 0 et 1
35 position = jlimit(0.0f, 1.0f, position);
36
37
38 if (position <= 0.20f) {
39 // Interpolation entre améthyste et saphir
40 float t = position / 0.20f;
41 return Colors::amethyst.interpolatedWith(Colors::sapphire, t);
42 }
43 else if (position <= 0.40f) {
44 // Interpolation entre saphir et émeraude
45 float t = (position - 0.20f) / 0.20f;
46 return Colors::sapphire.interpolatedWith(Colors::emerald, t);
47 }
48 else if (position <= 0.60f) {
49 // Interpolation entre émeraude et jaune citron
50 float t = (position - 0.40f) / 0.20f;
51 return Colors::emerald.interpolatedWith(Colors::topaze, t);
52 }
53 else if (position <= 0.80f) {
54 // Interpolation entre jaune citron et orange
55 float t = (position - 0.60f) / 0.20f;
56 return Colors::topaze.interpolatedWith(Colors::orange, t);
57 }
58 else {
59 // Interpolation entre orange et rubis
60 float t = (position - 0.80f) / 0.20f;
61 return Colors::orange.interpolatedWith(Colors::ruby, t);
62 }
63 }

76void DefaultLookAndFeel::drawLinearSlider(Graphics& g, int x, int y, int width, int height,
77 float slider_pos, float min, float max,
78 const Slider::SliderStyle style, Slider& slider) {
79 static const DropShadow thumb_shadow(Colors::shadow_thumb, 3, Point<int>(-1, 0));
80
81 bool bipolar = false;
82 bool flip_coloring = false;
83 bool active = true;
84 SynthSlider* s_slider = dynamic_cast<SynthSlider*>(&slider);
85 if (s_slider) {
86 bipolar = s_slider->isBipolar();
87 flip_coloring = s_slider->isFlippedColor();
88 active = s_slider->isActive();
89 }
90
91 Colour slider_color = Colors::slider_active;
92 Colour lighten_color = Colors::slider_lighten_active;
93 Colour thumb_color = Colors::slider_thumb_active;
94
95 if (!active) {
96 slider_color = Colors::slider_inactive;
97 thumb_color = Colors::slider_thumb_inactive;
98 lighten_color = Colors::slider_lighten_inactive;
99 }
100
101 float pos = slider_pos - 1.0f;
102 if (style == Slider::SliderStyle::LinearBar) {
103 g.setColour(Colors::shadow_light);
104 float w = slider.getWidth();
105 float h = slider.getHeight();
106 g.fillRect(0.0f, 0.0f, w, h);
107
108 g.setColour(Colors::slider_track_background);
109 fillSplitHorizontalRect(g, 0.0f, w, h, Colours::transparentBlack);
110
111 // Dessiner le gradient par segments
112 if (active) {
113 const int num_segments = 100;
114 float start_x, end_x;
115
116 if (bipolar) {
117 start_x = w / 2.0f;
118 end_x = pos;
119 }
120 else if (flip_coloring) {
121 start_x = pos;
122 end_x = 0.0f;
123 }
124 else {
125 start_x = 0.0f;
126 end_x = pos;
127 }
128
129 float segment_width = (end_x - start_x) / num_segments;
130
131 for (int i = 0; i < num_segments; ++i) {
132 float x1 = start_x + i * segment_width;
133 float x2 = start_x + (i + 1) * segment_width;
134
135 float gradient_pos;
136 if (bipolar) {
137 // Pour bipolaire, mapper x1 dans [start_x, end_x] vers la plage [0, 1]
138 // où la plage totale va du centre (w/2) à l'extrême appropriée
139 float center = w / 2.0f;
140 float max_extent = (pos > center) ? w : 0.0f;
141 gradient_pos = std::abs(x1 - center) / std::abs(max_extent - center);
142 }
143 else if (flip_coloring) {
144 // Pour flip, inverser le gradient : pos?w devrait aller de 100% à 0%
145 // x1 va de pos vers w, mapper sur [1, 0]
146 gradient_pos = 1.0f - ((x1 - pos) / (w - pos));
147 }
148 else {
149 // Pour normal, mapper depuis le début
150 gradient_pos = x1 / w;
151 }
152
153 g.setColour(getGradientColour(gradient_pos));
154 g.fillRect(x1, 0.0f, x2 - x1, h);
155 }
156 }
157 else {
158 g.setColour(slider_color);
159 if (bipolar)
160 fillSplitHorizontalRect(g, w / 2.0f, pos, h, lighten_color);
161 else if (flip_coloring)
162 fillSplitHorizontalRect(g, pos, w - pos, h, lighten_color);
163 else
164 fillSplitHorizontalRect(g, 0.0f, pos, h, lighten_color);
165 }
166
167 thumb_shadow.drawForRectangle(g, Rectangle<int>(pos + 0.5f, 0, 2, h));
168 g.setColour(thumb_color);
169 g.fillRect(pos, 0.0f, 2.0f, h);
170 }
171 else if (style == Slider::SliderStyle::LinearBarVertical) {
172 g.setColour(Colors::shadow_light);
173 float w = slider.getWidth();
174 float h = slider.getHeight();
175 g.fillRect(0.0f, 0.0f, w, h);
176
177 g.setColour(Colors::slider_track_background);
178 fillSplitVerticalRect(g, 0.0f, h, w, Colours::transparentBlack);
179
180 // Dessiner le gradient par segments
181 if (active) {
182 const int num_segments = 100;
183 float start_y, end_y;
184
185 if (bipolar) {
186 start_y = h / 2.0f;
187 end_y = pos;
188 }
189 else {
190 // Pour tous les sliders non-bipolaires, remplir du bas (h) vers la position (pos)
191 start_y = h;
192 end_y = pos;
193 }
194
195 float segment_height = (end_y - start_y) / num_segments;
196
197 for (int i = 0; i < num_segments; ++i) {
198 float y1 = start_y + i * segment_height;
199 float y2 = start_y + (i + 1) * segment_height;
200
201 // Gérer le cas où on dessine "en arrière" (segment_height négatif)
202 float rect_y = std::min(y1, y2);
203 float rect_height = std::abs(y2 - y1);
204
205 float gradient_pos;
206 if (bipolar) {
207 // Pour bipolaire vertical, mapper depuis le centre vers l'extrême approprié
208 // On veut saphir au centre et rubis à l'extrême (haut ou bas)
209 float center = h / 2.0f;
210 float max_extent = (pos > center) ? h : 0.0f;
211 // Utiliser rect_y (position réelle du segment) au lieu de y1
212 float segment_center = rect_y + rect_height / 2.0f;
213 gradient_pos = std::abs(segment_center - center) / std::abs(max_extent - center);
214 }
215 else {
216 // Pour tous les sliders non-bipolaires verticaux
217 // On veut saphir en bas (y=h) et rubis en haut (y=0)
218 // Utiliser le centre du segment pour le calcul
219 float segment_center = rect_y + rect_height / 2.0f;
220 gradient_pos = 1.0f - (segment_center / h);
221 }
222
223 g.setColour(getGradientColour(gradient_pos));
224 g.fillRect(0.0f, rect_y, w, rect_height);
225 }
226 }
227 else {
228 g.setColour(slider_color);
229 if (bipolar)
230 fillSplitVerticalRect(g, h / 2.0f, pos, w, lighten_color);
231 else if (flip_coloring)
232 fillSplitVerticalRect(g, h + 1, pos, w, lighten_color);
233 else
234 fillSplitVerticalRect(g, 0, pos, w, lighten_color);
235 }
236
237 thumb_shadow.drawForRectangle(g, Rectangle<int>(0, pos + 0.5f, w, 2));
238 g.setColour(thumb_color);
239 g.fillRect(0.0f, pos, w, 2.0f);
240 }

250void DefaultLookAndFeel::drawRotarySlider(Graphics& g, int x, int y, int width, int height,
251 float slider_t, float start_angle, float end_angle,
252 Slider& slider) {
253 static const float stroke_percent = 0.1f;
254
255 float full_radius = std::min(width / 2.0f, height / 2.0f);
256 float stroke_width = 2.0f * full_radius * stroke_percent;
257 float knob_radius = 0.63f * full_radius;
258 float small_outer_radius = knob_radius + stroke_width / 6.0f;
259 PathStrokeType outer_stroke =
260 PathStrokeType(stroke_width, PathStrokeType::beveled, PathStrokeType::butt);
261
262 float current_angle = start_angle + slider_t * (end_angle - start_angle);
263 float end_x = full_radius + 0.8f * knob_radius * sin(current_angle);
264 float end_y = full_radius - 0.8f * knob_radius * cos(current_angle);
265
266 if (slider.getInterval() == 1) {
267 static const float TEXT_W_PERCENT = 0.35f;
268 Rectangle<float> text_bounds(1.0f + width * (1.0f - TEXT_W_PERCENT) / 2.0f,
269 0.5f * height, width * TEXT_W_PERCENT, 0.5f * height);
270
271 g.setColour(Colour(0xff464646));
272 g.fillRoundedRectangle(text_bounds, 2.0f);
273
274 g.setColour(Colour(0xff999999));
275 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(0.2f * height));
276 g.drawFittedText(String(slider.getValue()), text_bounds.getSmallestIntegerContainer(),
277 Justification::horizontallyCentred | Justification::bottom, 1);
278 }
279
280 Path active_section;
281 bool bipolar = false;
282 bool active = true;
283 SynthSlider* s_slider = dynamic_cast<SynthSlider*>(&slider);
284 if (s_slider) {
285 bipolar = s_slider->isBipolar();
286 active = s_slider->isActive();
287 }
288
289 Path rail;
290 rail.addCentredArc(full_radius, full_radius, small_outer_radius, small_outer_radius,
291 0.0f, start_angle, end_angle, true);
292
293 if (active)
294 g.setColour(Colour(0xff4a4a4a));
295 else
296 g.setColour(Colour(0xff333333));
297
298 g.strokePath(rail, outer_stroke);
299
300 // Dessiner l'arc actif avec gradient par segments
301 if (active) {
302 const int num_segments = 100; // Nombre de segments pour un gradient lisse
303
304 if (bipolar) {
305 // Pour bipolaire, dessiner depuis le centre (0.0) vers current_angle
306 float center_angle = 0.0f;
307 float arc_span = current_angle - 2.0f * mopo::PI - center_angle;
308 float half_total_span = (end_angle - start_angle) / 2.0f;
309 // Déterminer l'extrême approprié selon la direction
310 float max_angle_extent = (current_angle > center_angle) ? end_angle : start_angle;
311
312 for (int i = 0; i < num_segments; ++i) {
313 float t1 = static_cast<float>(i) / num_segments;
314 float t2 = static_cast<float>(i + 1) / num_segments;
315
316 float angle1 = center_angle + t1 * arc_span;
317 float angle2 = center_angle + t2 * arc_span;
318
319 // Pour bipolaire : gradient symétrique
320 // Rubis à -max ? améthyste au centre ? rubis à +max
321 // Utiliser la distance depuis le centre normalisée sur la demi-plage totale
322 float distance_from_center = std::abs(angle1 - center_angle);
323 // Mapper sur [0, 1] où 0=centre (améthyste) et 1=extrêmes (rubis)
324 float gradient_pos = distance_from_center / half_total_span;
325
326 Path segment;
327 segment.addCentredArc(full_radius, full_radius, small_outer_radius, small_outer_radius,
328 0.0f, angle1, angle2, true);
329
330 g.setColour(getGradientColour(gradient_pos));
331 g.strokePath(segment, outer_stroke);
332 }
333 }
334 else {
335 // Pour unipolaire, dessiner depuis start_angle vers current_angle
336 float arc_span = current_angle - start_angle;
337 float total_arc_span = end_angle - start_angle;
338
339 for (int i = 0; i < num_segments; ++i) {
340 float t1 = static_cast<float>(i) / num_segments;
341 float t2 = static_cast<float>(i + 1) / num_segments;
342
343 float angle1 = start_angle + t1 * arc_span;
344 float angle2 = start_angle + t2 * arc_span;
345
346 // Mapper angle1 dans [start_angle, current_angle] vers [0, 1] où 1 correspond à end_angle
347 float gradient_pos = (angle1 - start_angle) / total_arc_span;
348
349 Path segment;
350 segment.addCentredArc(full_radius, full_radius, small_outer_radius, small_outer_radius,
351 0.0f, angle1, angle2, true);
352
353 g.setColour(getGradientColour(gradient_pos));
354 g.strokePath(segment, outer_stroke);
355 }
356 }
357 }
358 else {
359 // Slider inactif - couleur unie
360 if (bipolar) {
361 active_section.addCentredArc(full_radius, full_radius, small_outer_radius, small_outer_radius,
362 0.0f, 0.0f, current_angle - 2.0f * mopo::PI, true);
363 }
364 else {
365 active_section.addCentredArc(full_radius, full_radius, small_outer_radius, small_outer_radius,
366 0.0f, start_angle, current_angle, true);
367 }
368
369 g.setColour(Colour(0xff555555));
370 g.strokePath(active_section, outer_stroke);
371 }
372
373 if (active)
374 g.setColour(Colour(0xff000000));
375 else
376 g.setColour(Colour(0xff444444));
377
378 g.fillEllipse(full_radius - knob_radius,
379 full_radius - knob_radius,
380 2.0f * knob_radius,
381 2.0f * knob_radius);
382
383 if (active)
384 g.setColour(Colour(0xff666666));
385 else
386 g.setColour(Colour(0xff555555));
387
388 g.drawEllipse(full_radius - knob_radius + stroke_width / 4.0f + 0.5f,
389 full_radius - knob_radius + stroke_width / 4.0f + 0.5f,
390 2.0f * knob_radius - stroke_width / 2.0f - 1.0f,
391 2.0f * knob_radius - stroke_width / 2.0f - 1.0f, 1.5f);
392
393 g.setColour(Colour(0xff999999));
394 g.drawLine(full_radius, full_radius, end_x, end_y, 1.0f);
395}

397void DefaultLookAndFeel::drawToggleButton(Graphics& g, ToggleButton& button,
398 bool hover, bool is_down) {
399 static const DropShadow shadow(Colour(0x88000000), 1.0f, Point<int>(0, 0));
400 static float stroke_percent = 0.1f;
401 float ratio = button.getWidth() / 20.0f;
402 float padding = ratio * 3.0f;
403 float hover_padding = ratio;
404
405 float full_radius = std::min(button.getWidth(), button.getHeight()) / 2.0;
406 float stroke_width = 2.0f * full_radius * stroke_percent;
407 PathStrokeType stroke_type(stroke_width, PathStrokeType::beveled, PathStrokeType::rounded);
408 float outer_radius = full_radius - stroke_width - padding;
409 Path outer;
410 outer.addCentredArc(full_radius, full_radius, outer_radius, outer_radius,
411 mopo::PI, -POWER_ARC_ANGLE, POWER_ARC_ANGLE, true);
412
413 Path shadow_path;
414 stroke_type.createStrokedPath(shadow_path, outer);
415 shadow.drawForPath(g, shadow_path);
416 Rectangle<int> bar_shadow_rect(full_radius - 1.0f, padding, 2.0f, full_radius - padding);
417 shadow.drawForRectangle(g, bar_shadow_rect);
418
419 SynthButton* synth_button = dynamic_cast<SynthButton*>(&button);
420 if (button.getToggleState()) {
421 if (synth_button && synth_button->isModulated())
422 g.setColour(Colors::modulation.withAlpha(0.85f));
423 else
424 g.setColour(Colours::white);
425 }
426 else {
427 if (synth_button && synth_button->isModulated())
428 g.setColour(Colors::modulation.withAlpha(0.425f));
429 else
430 g.setColour(Colours::grey);
431 }
432
433 g.strokePath(outer, stroke_type);
434 g.fillRoundedRectangle(full_radius - 1.0f, padding, 2.0f, full_radius - padding, 1.0f);
435
436 if (is_down) {
437 g.setColour(Colour(0x11000000));
438 g.fillEllipse(hover_padding, hover_padding,
439 button.getWidth() - 2 * hover_padding, button.getHeight() - 2 * hover_padding);
440 }
441 else if (hover) {
442 g.setColour(Colour(0x11ffffff));
443 g.fillEllipse(hover_padding, hover_padding,
444 button.getWidth() - 2 * hover_padding, button.getHeight() - 2 * hover_padding); }
445
446 if (synth_button && synth_button->isModulationEditActive()) {
447 Rectangle<float> bounds = button.getLocalBounds().toFloat().reduced(1.5f);
448 g.setColour(Colors::modulation.withAlpha(0.85f));
449 g.drawRoundedRectangle(bounds, 4.0f, 2.0f);
450
451 Rectangle<float> accent(bounds.getRight() - 3.0f, bounds.getY() + 2.0f,
452 2.0f, bounds.getHeight() - 4.0f);
453 g.fillRect(accent);
454 }
455}

498void DefaultLookAndFeel::fillSplitHorizontalRect(Graphics& g, float x1, float x2, float height,
499 Colour fill_color) {
500 float h = (height - SynthSlider::linear_rail_width) / 2.0f;
501 float x = std::min(x1, x2);
502 float width = fabsf(x1 - x2);
503
504 g.saveState();
505 g.setColour(fill_color);
506 g.fillRect(x, 0.0f, width, height);
507 g.restoreState();
508
509 g.fillRect(x, 0.0f, width, h);
510 g.fillRect(x, h + SynthSlider::linear_rail_width, width, h);
511}
