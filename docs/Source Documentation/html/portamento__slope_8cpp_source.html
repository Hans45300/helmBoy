66 void PortamentoSlope::process() {
67 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Target)));
68
69 processTriggers();
70 auto state = static_cast<State>(static_cast<int>(input(static_cast<int>(Inputs::PortamentoType))->at(0)));
71 mopo_float run_seconds = input(static_cast<int>(Inputs::RunSeconds))->at(0);
72 if (state == State::Off || utils::closeToZero(run_seconds)) {
73 processBypass(0);
74 return;
75 }
76
77 mopo_float increment = 0.4f / (sample_rate_ * input(static_cast<int>(Inputs::RunSeconds))->at(0));
78 mopo_float decay = 0.07f / (sample_rate_ * input(static_cast<int>(Inputs::RunSeconds))->at(0));
79 const mopo_float* targets = input(static_cast<int>(Inputs::Target))->source->buffer;
80
81 int i = 0;
82 int note_number = static_cast<int>(input(static_cast<int>(Inputs::NoteNumber))->source->trigger_value);
83
84 if (state == State::Auto && note_number <= 1 && input(static_cast<int>(Inputs::TriggerJump))->source->triggered) {
85 int trigger_offset = input(static_cast<int>(Inputs::TriggerJump))->source->trigger_offset;
86 for (; i < trigger_offset; ++i)
87 tick(i, targets[i], increment, decay);
88
89 last_value_ = input(static_cast<int>(Inputs::Target))->at(trigger_offset);
90 }
91 else if (input(static_cast<int>(Inputs::TriggerStart))->source->triggered) {
92 int trigger_offset = input(static_cast<int>(Inputs::TriggerStart))->source->trigger_offset;
93 for (; i < trigger_offset; ++i)
94 tick(i, targets[i], increment, decay);
95
96 last_value_ = input(static_cast<int>(Inputs::TriggerStart))->source->trigger_value;
97 }
98
99 if (last_value_ == input(static_cast<int>(Inputs::Target))->at(0) &&
100 last_value_ == input(static_cast<int>(Inputs::Target))->at(buffer_size_ - 1)) {
101 processBypass(i);
102 }
103 else {
104 for (; i < buffer_size_; ++i)
105 tick(i, targets[i], increment, decay);
106 }
107 }

112 inline void PortamentoSlope::tick(int i, mopo_float target,
113 mopo_float increment, mopo_float decay) {
114 if (target <= last_value_)
115 last_value_ = utils::clamp(last_value_ - increment, target, last_value_);
116 else
117 last_value_ = utils::clamp(last_value_ + increment, last_value_, target);
118
119 mopo_float movement = target - last_value_;
120 movement *= std::fabs(movement);
121 last_value_ += movement * decay;
122 output()->buffer[i] = last_value_;
123 }
