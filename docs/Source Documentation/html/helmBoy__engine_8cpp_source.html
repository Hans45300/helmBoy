364 void HelmBoyEngine::connectModulation(ModulationConnection* connection) {
365 Output* source = getModulationSource(connection->source);
366 bool source_poly = source->owner->isPolyphonic();
367 MOPO_ASSERT(source != nullptr);
368
369 Processor* destination = getModulationDestination(connection->destination, source_poly);
370 MOPO_ASSERT(destination != nullptr);
371
372 ValueSwitch* mono_mod_switch = getMonoModulationSwitch(connection->destination);
373 MOPO_ASSERT(mono_mod_switch != nullptr);
374
375 connection->modulation_scale.plug(source, 0);
376 connection->modulation_scale.plug(&connection->amount, 1);
377 source->owner->router()->addProcessor(&connection->modulation_scale);
378 destination->plugNext(&connection->modulation_scale);
379
380 mono_mod_switch->set(1);
381 ValueSwitch* poly_mod_switch = getPolyModulationSwitch(connection->destination);
382 if (poly_mod_switch)
383 poly_mod_switch->set(1);
384
385 mod_connections_.insert(connection);
386 }

398 void HelmBoyEngine::disconnectModulation(ModulationConnection* connection) {
399 Output* source = getModulationSource(connection->source);
400 bool source_poly = source->owner->isPolyphonic();
401
402 Processor* destination = getModulationDestination(connection->destination, source_poly);
403 Processor* mono_destination = getMonoModulationDestination(connection->destination);
404 Processor* poly_destination = getPolyModulationDestination(connection->destination);
405 MOPO_ASSERT(destination != nullptr);
406
407 destination->unplug(&connection->modulation_scale);
408
409 if (mono_destination->connectedInputs() == 1 &&
410 (poly_destination == nullptr || poly_destination->connectedInputs() == 0)) {
411 ValueSwitch* mono_mod_switch = getMonoModulationSwitch(connection->destination);
412 mono_mod_switch->set(0);
413
414 ValueSwitch* poly_mod_switch = getPolyModulationSwitch(connection->destination);
415 if (poly_mod_switch)
416 poly_mod_switch->set(0);
417 }
418
419 source->owner->router()->removeProcessor(&connection->modulation_scale);
420 mod_connections_.erase(connection);
421 }

431 void HelmBoyEngine::process() {
432 bool playing_arp = arp_on_->value();
433 if (was_playing_arp_ != playing_arp)
434 arpeggiator_->allNotesOff();
435
436 was_playing_arp_ = playing_arp;
437 arpeggiator_->process();
438 ProcessorRouter::process();
439
440 if (getNumActiveVoices() == 0) {
441 for (auto& modulation : mod_connections_)
442 modulation->modulation_scale.process();
443 }
444 }
