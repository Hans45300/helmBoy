43void RetriggerSelector::mouseDown(const MouseEvent& e) {
44 if (e.mods.isPopupMenu()) {
45 SynthSlider::mouseDown(e);
46 return;
47 }
48 PopupMenu m;
49 m.setLookAndFeel(DefaultLookAndFeel::instance());

51 m.addItem(kFree, TRANS("Free"));
52 m.addItem(kRetrigger, TRANS("Retrigger"));
53 m.addItem(kSyncToPlayhead, TRANS("Sync to Playhead"));
54
55 m.showMenuAsync(PopupMenu::Options().withTargetComponent(this),
56 ModalCallbackFunction::forComponent(retriggerTypeSelectedCallback, this));

59void RetriggerSelector::paint(Graphics& g) {
60 g.setColour(Colour(Colors::Color_ffbbbbbb));
61 g.fillRect(0, 0, getWidth(), getHeight());
62
63 g.setColour(Colour(Colors::Color_ff222222));
64 g.fillPath(arrow_);
65
66 g.setFont(Fonts::instance()->monospace());
67 int value = getValue() + 1;
68 if (value == kFree)
69 g.drawText("F", getLocalBounds(), Justification::centred);
70 else if (value == kRetrigger)
71 g.drawText("R", getLocalBounds(), Justification::centred);
72 else if (value == kSyncToPlayhead)
73 g.drawText("S", getLocalBounds(), Justification::centred);
74}
