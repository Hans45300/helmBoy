64void OpenGLWaveViewer::paintBackground() {
65 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
66
67 if (getWidth() <= 0 || getHeight() <= 0)
68 return;
69
70 auto *display = Desktop::getInstance().getDisplays().getPrimaryDisplay();
71 jassert(display != nullptr);
72
73 float scale = display->scale;
74 background_image_ = Image(Image::ARGB, scale * getWidth(), scale * getHeight(), true);
75 Graphics g(background_image_);
76 g.addTransform(AffineTransform::scale(scale, scale));
77 // Fond de la grille d'affichage de la courbe d'onde
78 g.fillAll(Colour(Colors::Color_ff424242));
79 // Quadrillage de la grille d'affichage de la courbe d'onde
80 g.setColour(Colour(Colors::Color_ff4a4a4a));
81 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
82 g.drawLine(x, 0, x, getHeight());
83 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
84 g.drawLine(0, y, getWidth(), y);
85
86 shadow.drawForPath(g, wave_path_);
87
88 g.setColour(Colors::graph_fill);
89 g.fillPath(wave_path_);
90
91 g.setColour(Colors::modulation);
92 float line_width = 1.5f * getHeight() / 75.0f;
93 PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
94 g.strokePath(wave_path_, stroke);
95
96 background_.updateBackgroundImage(background_image_);
97}

99void OpenGLWaveViewer::paintPositionImage() {
100 int min_image_width = roundToInt(2 * MARKER_WIDTH);
101 int image_width = mopo::utils::nextPowerOfTwo(min_image_width);
102 int marker_width = MARKER_WIDTH;
103 int image_height = roundToInt(2 * IMAGE_HEIGHT);
104 position_image_ = Image(Image::ARGB, image_width, image_height, true);
105 Graphics g(position_image_);
106
107 g.setColour(Colour(Colors::Color_77ffffff));
108 g.fillRect(image_width / 2.0f - 0.5f, 0.0f, 1.0f, 1.0f * image_height);
109
110 g.setColour(Colors::modulation);
111 g.fillEllipse(image_width / 2 - marker_width / 2, image_height / 2 - marker_width / 2,
112 marker_width, marker_width);
113 g.setColour(Colour(Colors::Color_ff000000));
114 g.fillEllipse(image_width / 2 - marker_width / 4, image_height / 2 - marker_width / 4,
115 marker_width / 2, marker_width / 2);
116}

158void OpenGLWaveViewer::drawRandom() {
159 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
160 float draw_width = getWidth();
161 float padding = getRatio() * PADDING;
162 float draw_height = getHeight() - 2.0f * padding;
163
164 wave_path_.startNewSubPath(0, getHeight() / 2.0f);
165 for (int i = 0; i < NOISE_RESOLUTION; ++i) {
166 float t1 = (1.0f * i) / NOISE_RESOLUTION;
167 float t2 = (1.0f + i) / NOISE_RESOLUTION;
168 float val = amplitude * random_values[i];
169 wave_path_.lineTo(t1 * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
170 wave_path_.lineTo(t2 * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
171 }
172
173 wave_path_.lineTo(getWidth(), getHeight() / 2.0f);
174}

176void OpenGLWaveViewer::drawSmoothRandom() {
177 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
178 float draw_width = getWidth();
179 float padding = getRatio() * PADDING;
180 float draw_height = getHeight() - 2.0f * padding;
181
182 float start_val = amplitude * random_values[0];
183 wave_path_.startNewSubPath(-50, getHeight() / 2.0f);
184 wave_path_.lineTo(0, PADDING + draw_height * ((1.0f - start_val) / 2.0f));
185 for (int i = 1; i < resolution_ - 1; ++i) {
186 float t = (1.0f * i) / resolution_;
187 float phase = t * (NOISE_RESOLUTION - 1);
188 int index = (int)phase;
189 phase = mopo::PI * (phase - index);
190 float val = amplitude * mopo::utils::interpolate(random_values[index],
191 random_values[index + 1],
192 0.5f - cosf(phase) / 2.0f);
193 wave_path_.lineTo(t * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
194 }
195
196 float end_val = amplitude * random_values[NOISE_RESOLUTION - 1];
197 wave_path_.lineTo(getWidth(), padding + draw_height * ((1.0f - end_val) / 2.0f));
198 wave_path_.lineTo(getWidth() + 50, getHeight() / 2.0f);
199
200}

202void OpenGLWaveViewer::resetWavePath() {
203 wave_path_.clear();
204
205 if (wave_slider_ == nullptr)
206 return;
207
208 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
209 float draw_width = getWidth();
210 float padding = getRatio() * PADDING;
211 float draw_height = getHeight() - 2.0f * padding;
212
213 mopo::Wave::Type type = static_cast<mopo::Wave::Type>(static_cast<int>(wave_slider_->getValue()));
214
215 if (type < mopo::Wave::Type::WhiteNoise) {
216 wave_path_.startNewSubPath(0, getHeight() / 2.0f);
217 for (int i = 1; i < resolution_ - 1; ++i) {
218 float t = (1.0f * i) / resolution_;
219 float val = amplitude * mopo::Wave::wave(type, t);
220 wave_path_.lineTo(t * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
221 }
222
223 wave_path_.lineTo(getWidth(), getHeight() / 2.0f);
224 }
225 else if (type == mopo::Wave::Type::WhiteNoise)
226 drawRandom();
227 else
228 drawSmoothRandom();
229
230 paintBackground();
231}

242 paintPositionImage();

261void OpenGLWaveViewer::drawPosition(OpenGLContext& open_gl_context) {
262 if (position_texture_.getWidth() != position_image_.getWidth())
263 position_texture_.loadImage(position_image_);
264
265 if (wave_phase_ == nullptr || wave_amp_ == nullptr || wave_phase_->buffer[0] <= 0.0)
266 return;
267
268 float x = 2.0f * wave_phase_->buffer[0] - 1.0f;
269 float padding = getRatio() * PADDING;
270 float y = (getHeight() - 2 * padding) * wave_amp_->buffer[0] / getHeight();
271
272 glEnable(GL_BLEND);
273 glBlendFunc(GL_ONE, GL_ONE_MINUS_SRC_ALPHA);
274
275 int draw_width = getWidth();
276 int draw_height = getHeight();
277
278 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
279 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
280
281 float ratio = getHeight() / 75.0f;
282
283 float position_height = ratio * (0.5f * position_texture_.getHeight()) / draw_height;
284 float position_width = ratio * (0.5f * position_texture_.getWidth()) / draw_width;
285 position_vertices_[0] = x - position_width;
286 position_vertices_[1] = y + position_height;
287 position_vertices_[4] = x - position_width;
288 position_vertices_[5] = y - position_height;
289 position_vertices_[8] = x + position_width;
290 position_vertices_[9] = y - position_height;
291 position_vertices_[12] = x + position_width;
292 position_vertices_[13] = y + position_height;
293
294 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
295 GLsizeiptr vert_size = static_cast<GLsizeiptr>(static_cast<size_t>(16 * sizeof(float)));
296 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
297 position_vertices_, GL_STATIC_DRAW);
298
299 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
300 position_texture_.bind();
301
302 open_gl_context.extensions.glActiveTexture(GL_TEXTURE0);
303
304 if (background_.texture_uniform() != nullptr)
305 background_.texture_uniform()->set(0);
306
307 background_.shader()->use();
308
309 background_.enableAttributes(open_gl_context);
310 glDrawElements(GL_TRIANGLES, 6, GL_UNSIGNED_INT, 0);
311 background_.disableAttributes(open_gl_context);
312
313 position_texture_.unbind();
314
315 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
316 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
317}
