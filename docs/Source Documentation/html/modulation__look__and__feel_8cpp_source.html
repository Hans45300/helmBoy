37void ModulationLookAndFeel::drawLinearSlider(Graphics& g, int x, int y, int width, int height,
38 float slider_pos, float min, float max,
39 const Slider::SliderStyle style, Slider& slider) {
40 ButtonModulationSlider* button_slider = dynamic_cast<ButtonModulationSlider*>(&slider);
41 if (button_slider != nullptr) {
42 drawTextModulation(g, slider, slider.getValue());
43 return;
44 }
45
46 ModulationSlider* mod_slider = dynamic_cast<ModulationSlider*>(&slider);
47 if (!mod_slider)
48 return;
49 Slider* destination = mod_slider->getDestinationSlider();
50 float destination_percentage = destination->valueToProportionOfLength(destination->getValue());
51 float destination_range = destination->getMaximum() - destination->getMinimum();
52 float mod_percentage = slider.getValue() / destination_range;
53
54 if (mod_percentage == 0.0) {
55 g.setColour(Colour(0x11b9f6ca));
56 g.fillRect(0, 0, slider.getWidth(), slider.getHeight());
57 g.setColour(Colour(0xffb9f6ca));
58 g.drawRect(0, 0, slider.getWidth(), slider.getHeight(), 1);
59 }
60 else {
61 g.setColour(Colour(0x1100e676));
62 g.fillRect(0, 0, slider.getWidth(), slider.getHeight());
63 g.setColour(Colors::modulation);
64 g.drawRect(0.0f, 0.0f, float(slider.getWidth()), float(slider.getHeight()), 2.5f);
65 }
66
67 g.setColour(Colour(0x5500e676));
68
69 if (style == Slider::SliderStyle::LinearBar) {
70 float destination_position = width * destination_percentage;
71 float mod_diff = width * mod_percentage;
72 float from = std::min<float>(destination_position + mod_diff, destination_position);
73 float to = std::max<float>(destination_position + mod_diff, destination_position);
74
75 g.fillRect(x + from, float(y), to - from, float(height));
76 g.setColour(Colour(0xffffffff));
77 g.fillRect(x + destination_position + mod_diff, 1.0f * y, 2.0f, 1.0f * height);
78 }
79 else if (style == Slider::SliderStyle::LinearBarVertical) {
80 float destination_position = height * (1.0f - destination_percentage);
81 float mod_diff = height * mod_percentage;
82 float from = std::min<float>(destination_position - mod_diff, destination_position);
83 float to = std::max<float>(destination_position - mod_diff, destination_position);
84
85 g.fillRect(float(x), y + from, float(width), to - from);
86 g.setColour(Colour(0xffffffff));
87 g.fillRect(float(x), y + destination_position - mod_diff, 1.0f * width, 2.0f);
88 }
89}

121void ModulationLookAndFeel::drawRotarySlider(Graphics& g, int x, int y, int width, int height,
122 float slider_t, float start_angle, float end_angle,
123 Slider& slider) {
124 ModulationSlider* mod_slider = dynamic_cast<ModulationSlider*>(&slider);
125 if (!mod_slider)
126 return;
127
128 SynthSlider* destination = mod_slider->getDestinationSlider();
129 if (&destination->getLookAndFeel() == TextLookAndFeel::instance()) {
130 float destination_range = destination->getMaximum() - destination->getMinimum();
131 float mod_percentage = slider.getValue() / destination_range;
132 drawTextModulation(g, slider, mod_percentage);
133 return;
134 }
135
136 float destination_percentage = destination->valueToProportionOfLength(destination->getValue());
137 float destination_range = destination->getMaximum() - destination->getMinimum();
138 float destination_angle = start_angle + destination_percentage * (end_angle - start_angle);
139 float mod_diff = slider.getValue() * (end_angle - start_angle) / destination_range;
140
141 float draw_radius = std::min(width / 2.0f, height / 2.0f);
142 float knob_radius = 0.65f * draw_radius;
143 PathStrokeType stroke_type =
144 PathStrokeType(knob_radius, PathStrokeType::beveled, PathStrokeType::butt);
145
146 if (mod_diff == 0.0) {
147 g.setColour(Colour(0x33b9f6ca));
148 g.fillEllipse(width / 2.0f - knob_radius, height / 2.0f - knob_radius,
149 2.0 * knob_radius, 2.0 * knob_radius);
150 g.setColour(Colors::modulation);
151 g.drawEllipse(width / 2.0f - knob_radius + 0.5f, height / 2.0f - knob_radius + 0.5f,
152 2.0f * knob_radius - 1.0f, 2.0f * knob_radius - 1.0f, 1.0f);
153 }
154 else {
155 g.setColour(Colour(0xaa00e676));
156 g.fillEllipse(width / 2.0f - knob_radius, height / 2.0f - knob_radius,
157 2.0 * knob_radius, 2.0 * knob_radius);
158 g.setColour(Colors::modulation);
159 g.drawEllipse(width / 2.0f - knob_radius + 1.5f, height / 2.0f - knob_radius + 1.5f,
160 2.0f * knob_radius - 3.0f, 2.0f * knob_radius - 3.0f, 3.0f);
161 }
162
163
164 Path active_section;
165 float center_x = x + draw_radius;
166 float center_y = y + draw_radius;
167 if (destination_angle > mopo::PI)
168 destination_angle -= 2.0f * static_cast<float>(mopo::PI);
169 active_section.addCentredArc(center_x, center_y, knob_radius / 2.0, knob_radius / 2.0,
170 destination_angle, mod_diff, 0, true);
171
172 g.setColour(Colour(0xff69f0ae));
173 g.strokePath(active_section, stroke_type);
174
175 float end_x = draw_radius + 0.9f * knob_radius * sin(destination_angle + mod_diff);
176 float end_y = draw_radius - 0.9f * knob_radius * cos(destination_angle + mod_diff);
177 g.setColour(Colour(0xffffffff));
178 g.drawLine(draw_radius, draw_radius, end_x, end_y, 1.0f);
179}

181void ModulationLookAndFeel::drawToggleButton(Graphics& g, ToggleButton& button,
182 bool hover, bool is_down) {
183 static const DropShadow shadow(Colour(0x88000000), 2, Point<int>(0, 0));
184
185 int size = button.getWidth();
186
187 // Déterminer les états actif et sélectionné
188 bool selected = button.getToggleState();
189 bool active = false;
190
191 SynthGuiInterface* parent = button.findParentComponentOfClass<SynthGuiInterface>();
192 if (parent && parent->getSynth()->getSourceConnections(button.getName().toStdString()).size()) {
193 active = true;
194 }
195
196 // Créer l'image de modulation avec HelmBoyGraphics
197 Image button_image = HelmBoyGraphics::createModulationImage(size, selected, active);
198 int icon_width = button.getWidth() * 2 / 3;
199 int icon_height = button.getHeight() * 2 / 3;
200
201 g.setColour(Colours::white);
202 g.drawImage(button_image,
203 (button.getWidth() - icon_width) / 2,
204 (button.getHeight() - icon_height) / 2,
205 icon_width, icon_height,
206 0, 0, button_image.getWidth(), button_image.getHeight());
207
208 if (is_down) {
209 g.setColour(Colour(0x11000000));
210 g.fillEllipse(1, 2, button.getWidth() - 2, button.getHeight() - 2);
211 }
212 else if (hover) {
213 g.setColour(Colour(0x11ffffff));
214 g.fillEllipse(1, 2, button.getWidth() - 2, button.getHeight() - 2);
215 }
