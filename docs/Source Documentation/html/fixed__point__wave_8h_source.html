98 class FixedPointWave {
99 public:
100 static inline mopo_float harmonicWave(int waveform, unsigned int t, int harmonic) {
101 return lookup_.waves_[waveform][harmonic][getIndex(t)];
102 }
103
104 static inline mopo_float wave(int waveform, unsigned int t, int phase_inc) {
105 return lookup_.waves_[waveform][getHarmonicIndex(phase_inc)][getIndex(t)];
106 }
107
108 static inline mopo_float wave(int waveform, unsigned int t) {
109 unsigned int index = getIndex(t);
110 return lookup_.waves_[waveform][0][index];
111 }
112
113 static inline mopo_float* getBuffer(int waveform, int phase_inc) {
114 int clamped_inc = mopo::utils::iclamp(phase_inc, 1, INT_MAX);
115 return lookup_.waves_[waveform][getHarmonicIndex(clamped_inc)];
116 }
117
118 static inline int getHarmonicIndex(int phase_inc) {
119 int ratio = INT_MAX / phase_inc;
120 return mopo::utils::iclamp(FixedPointWaveLookup::HARMONICS + 1 - ratio,
121 0, FixedPointWaveLookup::HARMONICS - 1);
122 }
123
124 static inline mopo_float interpretWave(const mopo_float* buffer, unsigned int t) {
125 int index = getIndex(t);
126 mopo_float mult = getFractional(t);
127 mopo_float inc = mult * buffer[index + FixedPointWaveLookup::FIXED_LOOKUP_SIZE];
128 return buffer[index] + inc;
129 }
130
131 static inline unsigned int getIndex(unsigned int t) {
132 return t >> FixedPointWaveLookup::FRACTIONAL_BITS;
133 }
134
135 static inline unsigned int getFractional(unsigned int t) {
136 return t & FixedPointWaveLookup::FRACTIONAL_MASK;
137 }
138
139 protected:
140 static const FixedPointWaveLookup lookup_;
141 };
