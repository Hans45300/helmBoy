137 inline unsigned int remapPhase(unsigned int phase, mopo_float stretch) {
138 constexpr mopo_float mult = 1.0 / UINT_MAX;
139 mopo_float norm_phase = phase * mult; // 0.0 to 1.0
140
141 mopo_float remapped;
142 if (norm_phase < stretch) {
143 remapped = (norm_phase / stretch) * 0.5;
144 } else {
145 remapped = 0.5 + ((norm_phase - stretch) / (1.0 - stretch)) * 0.5;
146 }
147
148 return static_cast<unsigned int>(remapped * UINT_MAX);
149 }

151 inline void tickCrossMod(int i, const mopo_float cross_mod, const mopo_float fm_amount,
152 int* dest_cross_mod1, int* dest_cross_mod2, int* dest_fm,
153 unsigned int phase1, unsigned int phase2) {
154 constexpr mopo_float mult = (1.0 / UINT_MAX);
155
156 int master_phase1 = dest_cross_mod2[i] + phase1;
157 int master_phase2 = dest_cross_mod1[i] + phase2;
158 mopo_float sin1 = utils::quickerSin(mult * master_phase1);
159 mopo_float sin2 = utils::quickerSin(mult * master_phase2);
160 // Cross-modulation bidirectionnelle (inchangée)
161 dest_cross_mod1[i + 1] = sin1 * cross_mod * INT_MAX;
162 dest_cross_mod2[i + 1] = sin2 * cross_mod * INT_MAX;
163 // FM unidirectionnelle (OSC1 ? OSC2)
164 dest_fm[i + 1] = sin1 * fm_amount * INT_MAX;
165 }

167 inline void tickInitialVoices(int i, mopo_float stretch1, mopo_float stretch2) {
168 int phase1 = oscillator2_cross_mods_[i] + oscillator1_phases_[0] + oscillator1_phase_diffs_[i];
169 int phase2 = oscillator1_cross_mods_[i] + fm_mods_[i] + oscillator2_phases_[0] + oscillator2_phase_diffs_[i];
170
171 unsigned int warped_phase1 = remapPhase(phase1, stretch1);
172 unsigned int warped_phase2 = remapPhase(phase2, stretch2);
173
174 oscillator1_totals_[i] += FixedPointWave::interpretWave(wave_buffers1_[0], warped_phase1);
175 oscillator2_totals_[i] += FixedPointWave::interpretWave(wave_buffers2_[0], warped_phase2);
176 }

178 inline void tickVoice1(int i, int voice, const mopo_float* wave_buffer,
179 unsigned int start_phase, int detune, mopo_float stretch) {
180 int phase = oscillator1_cross_mods_[i] + start_phase +
181 i * detune + oscillator1_phase_diffs_[i];
182 unsigned int warped_phase = remapPhase(phase, stretch);
183 oscillator1_totals_[i] += FixedPointWave::interpretWave(wave_buffer, warped_phase);
184 }

194 inline void tickOut(int i, mopo_float* dest,
195 const mopo_float* amp1, const mopo_float* amp2,
196 const mopo_float* oscillator1_totals,
197 const mopo_float* oscillator2_totals,
198 mopo_float scale1, mopo_float scale2,
199 mopo_float ring_mod_amount) {
200 mopo_float osc1_out = amp1[i] * scale1 * oscillator1_totals[i];
201 mopo_float osc2_out = amp2[i] * scale2 * oscillator2_totals[i];
202 mopo_float ring_mod = scale1 * oscillator1_totals[i] * scale2 * oscillator2_totals[i] * ring_mod_amount * 2.0;
203 dest[i] = osc1_out + osc2_out + ring_mod;
204 MOPO_ASSERT(std::isfinite(dest[i]));
205 }
