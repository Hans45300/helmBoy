144 inline unsigned int remapPhase(unsigned int phase, mopo_float stretch) {
145 constexpr mopo_float mult = 1.0 / UINT_MAX;
146 mopo_float norm_phase = phase * mult; // 0.0 to 1.0
147 return static_cast<unsigned int>(utils::remapPhase(norm_phase, stretch) * UINT_MAX);
148 }

150 inline void tickCrossMod(int i, const mopo_float cross_mod, const mopo_float fm_amount,
151 int* dest_cross_mod1, int* dest_cross_mod2, int* dest_fm,
152 unsigned int phase1, unsigned int phase2) {
153 constexpr mopo_float mult = (1.0 / UINT_MAX);
154
155 unsigned int master_phase1 = static_cast<unsigned int>(dest_cross_mod2[i]) + phase1 +
156 oscillator1_phase_offsets_[i];
157 unsigned int master_phase2 = static_cast<unsigned int>(dest_cross_mod1[i]) + phase2 +
158 oscillator2_phase_offsets_[i];
159 mopo_float sin1 = utils::quickerSin(mult * master_phase1);
160 mopo_float sin2 = utils::quickerSin(mult * master_phase2);
161 // Cross-modulation bidirectionnelle (inchangée)
162 dest_cross_mod1[i + 1] = sin1 * cross_mod * INT_MAX;
163 dest_cross_mod2[i + 1] = sin2 * cross_mod * INT_MAX;
164 // FM unidirectionnelle (OSC1 ? OSC2)
165 dest_fm[i + 1] = sin1 * fm_amount * INT_MAX;
166 }

168 inline void tickInitialVoices(int i, mopo_float stretch1, mopo_float stretch2) {
169 unsigned int phase1 = static_cast<unsigned int>(oscillator2_cross_mods_[i]) +
170 oscillator1_phases_[0] + static_cast<unsigned int>(oscillator1_phase_diffs_[i]) +
171 oscillator1_phase_offsets_[i];
172 unsigned int phase2 = static_cast<unsigned int>(oscillator1_cross_mods_[i]) +
173 static_cast<unsigned int>(fm_mods_[i]) + oscillator2_phases_[0] +
174 static_cast<unsigned int>(oscillator2_phase_diffs_[i]) +
175 oscillator2_phase_offsets_[i];
176
177 unsigned int warped_phase1 = remapPhase(phase1, stretch1);
178 unsigned int warped_phase2 = remapPhase(phase2, stretch2);
179
180 oscillator1_totals_[i] += FixedPointWave::interpretWave(wave_buffers1_[0], warped_phase1);
181 oscillator2_totals_[i] += FixedPointWave::interpretWave(wave_buffers2_[0], warped_phase2);
182 }

184 inline void tickVoice1(int i, int voice, const mopo_float* wave_buffer,
185 unsigned int start_phase, int detune, mopo_float stretch) {
186 unsigned int phase = static_cast<unsigned int>(oscillator1_cross_mods_[i]) + start_phase +
187 static_cast<unsigned int>(i * detune) +
188 static_cast<unsigned int>(oscillator1_phase_diffs_[i]) +
189 oscillator1_phase_offsets_[i];
190 unsigned int warped_phase = remapPhase(phase, stretch);
191 oscillator1_totals_[i] += FixedPointWave::interpretWave(wave_buffer, warped_phase);
192 }
193
194 inline void tickVoice2(int i, int voice, const mopo_float* wave_buffer,
195 unsigned int start_phase, int detune, mopo_float stretch) {
196 unsigned int phase = static_cast<unsigned int>(oscillator1_cross_mods_[i]) +
197 static_cast<unsigned int>(fm_mods_[i]) + start_phase +
198 static_cast<unsigned int>(i * detune) +
199 static_cast<unsigned int>(oscillator2_phase_diffs_[i]) +
200 oscillator2_phase_offsets_[i];
201 unsigned int warped_phase = remapPhase(phase, stretch);
202 oscillator2_totals_[i] += FixedPointWave::interpretWave(wave_buffer, warped_phase);
203 }

205 inline void tickOut(int i, mopo_float* dest,
206 mopo_float* oscillator1_output,
207 mopo_float* oscillator2_output,
208 const mopo_float* amp1, const mopo_float* amp2,
209 const mopo_float* ring_mod_amp1,
210 const mopo_float* ring_mod_amp2,
211 const mopo_float* oscillator1_totals,
212 const mopo_float* oscillator2_totals,
213 mopo_float scale1, mopo_float scale2,
214 mopo_float ring_mod_amount) {
215 mopo_float osc1_out = amp1[i] * scale1 * oscillator1_totals[i];
216 mopo_float osc2_out = amp2[i] * scale2 * oscillator2_totals[i];
217 mopo_float ring_mod = scale1 * oscillator1_totals[i] * ring_mod_amp1[i]
218 * scale2 * oscillator2_totals[i] * ring_mod_amp2[i]
219 * ring_mod_amount * 2.0;
220 mopo_float oscillator1_ring_mod = ring_mod * 0.5;
221 mopo_float oscillator2_ring_mod = ring_mod - oscillator1_ring_mod;
222 oscillator1_output[i] = osc1_out + oscillator1_ring_mod;
223 oscillator2_output[i] = osc2_out + oscillator2_ring_mod;
224 dest[i] = oscillator1_output[i] + oscillator2_output[i];
225 MOPO_ASSERT(std::isfinite(dest[i]));
226 }
