41LfoSection::LfoSection(String name, std::string value_prepend, bool retrigger, bool can_animate) :
42 SynthSection(name), can_animate_(can_animate) {
43 static const int TEMPO_DRAG_SENSITIVITY = 150;
44
45 retrigger_ = std::make_unique<RetriggerSelector>(value_prepend + "_retrigger");
46 retrigger_->setSliderStyle(Slider::LinearBar);
47 retrigger_->setStringLookup(mopo::strings::freq_retrigger_styles);
48 if (retrigger)
49 addSlider(retrigger_.get());
50
51 addSlider((amplitude_ = std::make_unique<SynthSlider>(value_prepend + "_amplitude")).get());
52 amplitude_->setSliderStyle(Slider::LinearBarVertical);
53 amplitude_->setBipolar();
54 amplitude_->snapToValue(true, 0.0);
55
56 addSlider((frequency_ = std::make_unique<SynthSlider>(value_prepend + "_frequency")).get());
57 frequency_->setSliderStyle(Slider::RotaryHorizontalVerticalDrag);
58 frequency_->setLookAndFeel(TextLookAndFeel::instance());
59
60 addSlider((tempo_ = std::make_unique<SynthSlider>(value_prepend + "_tempo")).get());
61 tempo_->setSliderStyle(Slider::RotaryHorizontalVerticalDrag);
62 tempo_->setStringLookup(mopo::strings::synced_frequencies);
63 tempo_->setLookAndFeel(TextLookAndFeel::instance());
64 tempo_->setMouseDragSensitivity(TEMPO_DRAG_SENSITIVITY);
65
66 addSlider((sync_ = std::make_unique<TempoSelector>(value_prepend + "_sync")).get());
67 sync_->setSliderStyle(Slider::LinearBar);
68 sync_->setTempoSlider(tempo_.get());
69 sync_->setFreeSlider(frequency_.get());
70 sync_->setStringLookup(mopo::strings::freq_sync_styles);
71
72 addSlider((phase_offset_ = std::make_unique<SynthSlider>(value_prepend + "_phase_stretch")).get());
73 phase_offset_->setSliderStyle(Slider::RotaryHorizontalVerticalDrag);
74
75 addSlider((wave_selector_ = std::make_unique<WaveSelector>(value_prepend + "_waveform")).get());
76 wave_selector_->setSliderStyle(Slider::LinearBar);
77 wave_selector_->setStringLookup(mopo::strings::waveforms);
78
79 addOpenGLComponent((wave_viewer_ = std::make_unique<OpenGLWaveViewer>(WAVE_VIEWER_RESOLUTION)).get());
80 wave_viewer_->setAmplitudeSlider(amplitude_.get());
81 wave_viewer_->setWaveSlider(wave_selector_.get());
82 wave_viewer_->setName(value_prepend);
83
84 addModulationButton((modulation_button_ = std::make_unique<ModulationButton>(value_prepend)).get());
85 modulation_button_->setLookAndFeel(ModulationLookAndFeel::instance());
86}

104void LfoSection::paintBackground(Graphics& g) {
105 static const DropShadow component_shadow(Colour(Colors::Color_88000000), 2, Point<int>(0, 1));
106
107 SynthSection::paintBackground(g);
108 g.setColour(Colors::control_label_text);
109 float text_height = size_ratio_ * 10.0f;
110 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(text_height));
111 float text_slider_height = size_ratio_ * TEXT_HEIGHT;
112 float text_buffer = size_ratio_ * 6.0f;
113
114 if (retrigger_->isVisible()) {
115 g.drawText(TRANS("FREQUENCY"),
116 retrigger_->getBounds().getX(), frequency_->getBounds().getBottom() + text_buffer,
117 frequency_->getBounds().getWidth() + 2 * text_slider_height, text_height + 1,
118 Justification::centred, false);
119 }
120 else {
121 float extra_bump = size_ratio_ * 5.0f;
122 g.drawText(TRANS("FREQUENCY"),
123 frequency_->getBounds().getX() - extra_bump,
124 frequency_->getBounds().getBottom() + text_buffer,
125 frequency_->getBounds().getWidth() + text_slider_height + 2 * extra_bump,
126 text_height + 1,
127 Justification::centred, false);
128 }
129 drawTextForComponent(g, TRANS("PS"), phase_offset_.get());
130 component_shadow.drawForRectangle(g, wave_viewer_->getBounds());
131}

133void LfoSection::resized() {
134 int control_section_width = size_ratio_ * CONTROL_SECTION_WIDTH;
135 int y = getHeight() - control_section_width + size_ratio_ * 6.0f;
136 int knob_width = getStandardKnobSize();
137 int mod_button_width = getModButtonWidth();
138 int text_width = size_ratio_ * TEXT_WIDTH;
139 int text_height = size_ratio_ * TEXT_HEIGHT;
140 int slider_width = size_ratio_ * SLIDER_WIDTH;
141 int title_width = getTitleWidth();
142 int mod_button_x = size_ratio_ * 10.0f;
143
144 int wave_height = getHeight() - title_width - control_section_width - slider_width * 2;
145 wave_selector_->setBounds(slider_width, title_width, getWidth() - slider_width, slider_width * 2);
146 wave_viewer_->setBounds(slider_width, title_width + slider_width * 2,
147 getWidth() - slider_width, wave_height);
148
149 amplitude_->setBounds(0, title_width + slider_width * 2, slider_width, wave_height);
150
151 modulation_button_->setBounds(mod_button_x, y, mod_button_width, mod_button_width);
152 retrigger_->setBounds(proportionOfWidth(0.5f) - text_height, y, text_height, text_height);
153
154 frequency_->setBounds(proportionOfWidth(0.5f), y, text_width, text_height);
155 sync_->setBounds(frequency_->getBounds().getX() + text_width, frequency_->getBounds().getY(),
156 text_height, text_height);
157
158 tempo_->setBounds(frequency_->getBounds());
159
160 SynthSection::resized();
161
162 frequency_->setPopupDisplayEnabled(false, false, nullptr);
163 tempo_->setPopupDisplayEnabled(false, false, nullptr);
164
165 phase_offset_->setBounds((getWidth() - (knob_width / 2.0f)),
166 getHeight() - knob_width * .85f,
167 knob_width / 2.0f, knob_width / 2.0f);
168}
