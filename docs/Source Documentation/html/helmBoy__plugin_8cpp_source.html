31HelmPlugin::HelmPlugin() :
32 AudioProcessor(BusesProperties().withOutput("Output", juce::AudioChannelSet::stereo(), true)) {
33 set_state_time_ = 0;
34
35 current_program_ = 0;
36
37 loadPatches();
38
39 for (const auto& [name, control] : controls_) {
40 ValueBridge* bridge = new ValueBridge(name, control);
41 bridge->setListener(this);
42 bridge_lookup_[name] = bridge;
43 addParameter(bridge);
44 }
45
46 // Initialize MIDI 2.0 manager with high-resolution callbacks
47 midi2_manager_ = std::make_unique<helmboy::Midi2Manager>();
48
49 // Charger la configuration MIDI 2.0 sauvegardée
50 bool enable_midi2 = LoadSave::shouldEnableMidi2();
51 midi2_manager_->setMidiVersion(
52 enable_midi2 ? helmboy::Midi2Manager::MidiVersion::MIDI_2_0
53 : helmboy::Midi2Manager::MidiVersion::MIDI_1_0
54 );
55
56 // Connect MIDI 2.0 callbacks to synthesis engine
57 midi2_manager_->onNoteOn = [this](int channel, int note, float velocity) {
58 engine_.noteOn(note, velocity, 0, channel);
59 };
60
61 midi2_manager_->onNoteOff = [this](int channel, int note, float velocity) {
62 engine_.noteOff(note, 0);
63 };
64
65 midi2_manager_->onControlChange = [this](int channel, int cc, float value) {
66 // Route CC to MIDI manager (converts back to 7-bit for existing system)
67 juce::MidiMessage msg = juce::MidiMessage::controllerEvent(channel + 1, cc, static_cast<int>(value * 127.0f));
68 midi_manager_->processMidiMessage(msg, 0);
69 };
70
71 midi2_manager_->onPitchBend = [this](int channel, float value) {
72 // Convert -1.0 to +1.0 range to 0-16383 MIDI pitch bend range
73 int midi_value = static_cast<int>((value + 1.0f) * 8192.0f);
74 juce::MidiMessage msg = juce::MidiMessage::pitchWheel(channel + 1, midi_value);
75 midi_manager_->processMidiMessage(msg, 0);
76 };
77}

166void HelmPlugin::setCurrentProgram(int index) {
167 // Hack for some DAWs that set program on load for VSTs.
168 if (Time::getMillisecondCounter() - set_state_time_ < SET_PROGRAM_WAIT_MILLISECONDS) [[unlikely]]
169 return;
170
171 if (all_patches_.size() > index) [[likely]] {
172 current_program_ = index;
173 LoadSave::loadPatchFile(all_patches_[current_program_], this, save_info_);
174 auto editor = getGuiInterface();
175 if (editor.has_value())
176 editor.value()->updateFullGui();
177 }
178}

207void HelmPlugin::processBlock(AudioSampleBuffer& buffer, MidiBuffer& midi_messages) {
208 int total_samples = buffer.getNumSamples();
209 int num_channels = getTotalNumOutputChannels();
210 getPlayHead()->getCurrentPosition(position_info_);
211 if (position_info_.bpm)
212 engine_.setBpm(position_info_.bpm);
213
214 if (position_info_.isPlaying || position_info_.isLooping || position_info_.isRecording)
215 engine_.correctToTime(position_info_.timeInSamples);
216
217 processControlChanges();
218 processModulationChanges();
219
220 MidiBuffer keyboard_messages = midi_messages;
221 processKeyboardEvents(keyboard_messages, total_samples);
222
223 for (auto sample_offset = 0; sample_offset < total_samples;) {
224 auto num_samples = std::min<int>(total_samples - sample_offset, MAX_BUFFER_PROCESS);
225
226 // Route MIDI through MIDI 2.0 manager if enabled
227 if (midi2_manager_ && midi2_manager_->isMidi2Enabled()) {
228 for (auto it = midi_messages.cbegin(); it != midi_messages.cend(); ++it) {
229 int midi_sample = (*it).samplePosition;
230 if (midi_sample >= sample_offset && midi_sample < sample_offset + num_samples) {
231 midi2_manager_->processMidiMessage((*it).getMessage());
232 }
233 }
234 } else {
235 // Standard MIDI 1.0 processing via base class
236 processMidi(midi_messages, sample_offset, sample_offset + num_samples);
237 }
238
239 processAudioSafe(&buffer, num_channels, num_samples, sample_offset);
240
241 sample_offset += num_samples;
242 }
243}
