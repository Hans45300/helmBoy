28HelmEditor::HelmEditor(HelmPlugin& helm) : AudioProcessorEditor(&helm), SynthGuiInterface(&helm),
29 helm_(helm), was_animating_(true) {
30 setLookAndFeel(DefaultLookAndFeel::instance());
31
32 addAndMakeVisible(gui_.get());
33
34 gui_->setOutputMemory(helm.getOutputMemory());
35 gui_->animate(LoadSave::shouldAnimateWidgets());
36
37 constrainer_.setMinimumSize(2 * mopo::DEFAULT_WINDOW_WIDTH / 3,
38 2 * mopo::DEFAULT_WINDOW_HEIGHT / 3);
39 double ratio = (1.0 * mopo::DEFAULT_WINDOW_WIDTH) / mopo::DEFAULT_WINDOW_HEIGHT;
40 constrainer_.setFixedAspectRatio(ratio);
41 setConstrainer(&constrainer_);
42
43 float window_size = LoadSave::loadWindowSize();
44 setResizable(true, true);
45 setSize(window_size * mopo::DEFAULT_WINDOW_WIDTH, window_size * mopo::DEFAULT_WINDOW_HEIGHT);
46
47 repaint();
48}

89void HelmEditor::checkAnimate() {
90 Component* top_level = getTopLevelComponent();
91 bool should_animate = top_level->hasKeyboardFocus(true) && top_level->isShowing();
92 if (was_animating_ != should_animate) {
93 gui_->animate(should_animate && LoadSave::shouldAnimateWidgets());
94 was_animating_ = should_animate;
95 }
