32HelmEditor::HelmEditor(HelmPlugin& helm) : AudioProcessorEditor(&helm), SynthGuiInterface(&helm, false),
33 helm_(helm), was_animating_(true) {
34 appendPluginHostTraceEditor("HelmEditor::HelmEditor begin");
35 if (!initializeGui())
36 throw std::runtime_error("helmBoy editor GUI initialization failed");
37
38 setLookAndFeel(DefaultLookAndFeel::instance());
39
40 appendPluginHostTraceEditor("HelmEditor::HelmEditor addAndMakeVisible(gui_) start");
41 addAndMakeVisible(gui_.get());
42 appendPluginHostTraceEditor("HelmEditor::HelmEditor addAndMakeVisible(gui_) done");
43
44 appendPluginHostTraceEditor("HelmEditor::HelmEditor setOutputMemory start");
45 gui_->setOutputMemorySource(&helm);
46 appendPluginHostTraceEditor("HelmEditor::HelmEditor setOutputMemory done");
47
48 appendPluginHostTraceEditor("HelmEditor::HelmEditor animate start");
49 gui_->animate(LoadSave::shouldAnimateWidgets());
50 appendPluginHostTraceEditor("HelmEditor::HelmEditor animate done");
51
52 constrainer_.setMinimumSize(2 * mopo::DEFAULT_WINDOW_WIDTH / 3,
53 2 * mopo::DEFAULT_WINDOW_HEIGHT / 3);
54 double ratio = (1.0 * mopo::DEFAULT_WINDOW_WIDTH) / mopo::DEFAULT_WINDOW_HEIGHT;
55 constrainer_.setFixedAspectRatio(ratio);
56 setConstrainer(&constrainer_);
57
58 float window_size = LoadSave::loadWindowSize();
59 setResizable(true, true);
60 appendPluginHostTraceEditor("HelmEditor::HelmEditor setSize start");
61 setSize(window_size * mopo::DEFAULT_WINDOW_WIDTH, window_size * mopo::DEFAULT_WINDOW_HEIGHT);
62 appendPluginHostTraceEditor("HelmEditor::HelmEditor setSize done");
63
64 repaint();
65 appendPluginHostTraceEditor("HelmEditor::HelmEditor end");
66}

112void HelmEditor::checkAnimate() {
113 if (gui_ == nullptr)
114 return;
115
116 Component* top_level = getTopLevelComponent();
117 bool should_animate = top_level->isShowing();
118 if (was_animating_ != should_animate) {
119 gui_->animate(should_animate && LoadSave::shouldAnimateWidgets());
120 was_animating_ = should_animate;
121 }
