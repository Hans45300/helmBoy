78 void pushBlock(const mopo_float* samples, int num) {
79 std::atomic_ref<unsigned int> atomic_offset(offset_);
80 unsigned int current_offset = atomic_offset.load(std::memory_order_acquire);
81 unsigned int next_offset = (current_offset + num) & bitmask_;
82
83 if (next_offset < current_offset) [[unlikely]] {
84 int block1 = num - next_offset - 1;
85 memcpy(memory_ + current_offset + 1, samples, sizeof(mopo_float) * block1);
86 memcpy(memory_, samples + block1, sizeof(mopo_float) * next_offset);
87 } else {
88 memcpy(memory_ + current_offset + 1, samples, sizeof(mopo_float) * num);
89 }
90
91 atomic_offset.store(next_offset, std::memory_order_release);
92 }

121 [[nodiscard]] inline mopo_float getIndex(int index) const {
122 std::atomic_ref<const unsigned int> atomic_offset(offset_);
123 unsigned int current_offset = atomic_offset.load(std::memory_order_acquire);
124 int spot = (current_offset - index) & bitmask_;
125 MOPO_ASSERT(spot < size_);
126 return memory_[spot];

135 [[nodiscard]] inline mopo_float get(mopo_float past) const {
136 MOPO_ASSERT(past >= 0.0);
137 int index = utils::imax(past, 1);
138 mopo_float sample_fraction = past - index;
139
140 mopo_float from = getIndex(index - 1);
141 mopo_float to = getIndex(index);
142 return utils::interpolate(from, to, sample_fraction);
143 }
