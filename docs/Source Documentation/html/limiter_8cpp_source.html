48 void Limiter::process() {
49 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::AudioLeft)));
50 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::AudioRight)));
51
52 // Get input buffers
53 auto audio_left = std::span<const mopo_float>(
54 input(static_cast<int>(Inputs::AudioLeft))->source->buffer, buffer_size_);
55 auto audio_right = std::span<const mopo_float>(
56 input(static_cast<int>(Inputs::AudioRight))->source->buffer, buffer_size_);
57 const mopo_float on = input(static_cast<int>(Inputs::On))->at(0);
58 mopo_float ceiling_db = input(static_cast<int>(Inputs::Ceiling))->at(0);
59 mopo_float release_ms = input(static_cast<int>(Inputs::Release))->at(0);
60 if (!std::isfinite(ceiling_db))
61 ceiling_db = 0.0;
62 if (!std::isfinite(release_ms))
63 release_ms = 100.0;
64 ceiling_db = std::clamp(ceiling_db, -12.0, 0.0);
65 release_ms = std::clamp(release_ms, 5.0, 500.0);
66
67 // Get output buffers
68 auto dest_left = std::span<mopo_float>(
69 output(static_cast<int>(Outputs::OutputLeft))->buffer, buffer_size_);
70 auto dest_right = std::span<mopo_float>(
71 output(static_cast<int>(Outputs::OutputRight))->buffer, buffer_size_);
72
73 // Track coupled peak (max of both channels)
74 for (int i = 0; i < buffer_size_; ++i) {
75 peak_ = std::max({peak_, std::abs(audio_left[i]), std::abs(audio_right[i])});
76 }
77
78 const mopo_float ceiling = static_cast<mopo_float>(
79 std::pow(10.0, static_cast<double>(ceiling_db) / 20.0));
80 const mopo_float target_gain = peak_ > ceiling ? ceiling / peak_ : 1.0;
81
82 if (on > 0.5) {
83 if (target_gain < current_gain_)
84 current_gain_ = target_gain;
85
86 const mopo_float release_samples = std::max(
87 1.0, static_cast<mopo_float>(sample_rate_) * release_ms * 0.001);
88 const mopo_float release_coefficient = static_cast<mopo_float>(
89 std::exp(-1.0 / release_samples));
90
91 for (int i = 0; i < buffer_size_; ++i) {
92 if (current_gain_ < target_gain)
93 current_gain_ = target_gain + (current_gain_ - target_gain) * release_coefficient;
94 dest_left[i] = utils::quickTanh(audio_left[i] * current_gain_);
95 dest_right[i] = utils::quickTanh(audio_right[i] * current_gain_);
96 }
97 } else {
98 for (int i = 0; i < buffer_size_; ++i) {
99 dest_left[i] = audio_left[i];
100 dest_right[i] = audio_right[i];
101 }
102 current_gain_ = 1.0;
103 }
104
105 // Reset peak for next block
106 peak_ = 0.0;
107 }
