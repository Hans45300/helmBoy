44 class SampleDecayLookupSingleton {
45 public:
46 SampleDecayLookupSingleton() {
47 for (int i = 0; i < SAMPLE_DECAY_LOOKUP_RESOLUTION + 3; ++i) {
48 mopo_float percent = (1.0 * i) / SAMPLE_DECAY_LOOKUP_RESOLUTION;
49 sample_decay_lookup_[i] = std::pow(CLOSE_ENOUGH, percent);
50 }
51 }

58 [[nodiscard]] mopo_float sampleDecayLookup(mopo_float sample_length) const {
59 if (sample_length <= 1.0)
60 return 0.0;
61
62 mopo_float percent = 1.0 / sample_length;
63 mopo_float index = SAMPLE_DECAY_LOOKUP_RESOLUTION * percent;
64 int int_index = static_cast<int>(index);
65 mopo_float fraction = index - int_index;
66
67 return utils::interpolate(sample_decay_lookup_[int_index],
68 sample_decay_lookup_[int_index + 1], fraction);
69 }
70
71 private:
72 mopo_float sample_decay_lookup_[SAMPLE_DECAY_LOOKUP_RESOLUTION + 3] = {};
73 };
