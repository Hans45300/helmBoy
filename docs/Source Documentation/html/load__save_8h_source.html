51 static int compareElements(File a, File b) {
52 if (a.getFileName() == "Factory Presets")
53 return -1;
54 else if (b.getFileName() == "Factory Presets")
55 return 1;
56
57 if (a.getFileName() == "Old Factory Presets")
58 return 1;
59 else if (b.getFileName() == "Old Factory Presets")
60 return -1;
61
62 return a.getFullPathName().toLowerCase().compare(b.getFullPathName().toLowerCase());
63 }
64};
