47 static int compareElements(File a, File b) {
48 if (a.getFileName() == "Factory Presets")
49 return -1;
50 else if (b.getFileName() == "Factory Presets")
51 return 1;
52
53 if (a.getFileName() == "Old Factory Presets")
54 return 1;
55 else if (b.getFileName() == "Old Factory Presets")
56 return -1;
57
58 return a.getFullPathName().toLowerCase().compare(b.getFullPathName().toLowerCase());
59 }
60};
