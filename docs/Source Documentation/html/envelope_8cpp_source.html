66 void Envelope::trigger(mopo_float event) {
67 if (static_cast<int>(event) == static_cast<int>(kVoiceOn) || static_cast<int>(event) == static_cast<int>(kVoiceReset)) [[likely]] {
68 current_value_ = 0.0;
69 const Input* delay_input = input(static_cast<int>(Inputs::Delay));
70 mopo_float delay = delay_input->source == nullptr ? 0.0 :
71 utils::max(delay_input->at(0), 0.0);
72 stage_duration_samples_ = static_cast<int>(delay * sample_rate_ + 0.5);
73 stage_samples_remaining_ = stage_duration_samples_;
74 stage_progress_ = 0.0;
75 state_ = stage_samples_remaining_ > 0 ? State::Delaying : State::Attacking;
76
77 output(static_cast<int>(Outputs::Finished))->trigger(kVoiceReset);
78 }
79 else if (static_cast<int>(event) == static_cast<int>(kVoiceOff)) [[likely]] {
80 state_ = State::Releasing;
81 stage_samples_remaining_ = 0;
82 stage_duration_samples_ = 0;
83 stage_progress_ = 0.0;
84 }
85 else if (static_cast<int>(event) == static_cast<int>(kVoiceKill)) [[unlikely]] {
86 state_ = State::Killing;
87 stage_samples_remaining_ = 0;
88 stage_duration_samples_ = 0;
89 stage_progress_ = 0.0;
90 }
91 }

102 void Envelope::process() {
103 // Clear the finished flag from the previous block.
104 output(static_cast<int>(Outputs::Finished))->clearTrigger();
105
106 // If a trigger arrived on the input, handle it first.
107 if (input(static_cast<int>(Inputs::Trigger))->source->triggered)
108 trigger(input(static_cast<int>(Inputs::Trigger))->source->trigger_value);
109
110 // Keep the historical phase output stable for modulation routes.
111 output(static_cast<int>(Outputs::Phase))->buffer[0] = static_cast<mopo_float>(state_);
112 stage_progress_ = 0.0;
113
114 int samples_remaining = samples_to_process_;
115
116 if (state_ == State::Delaying) {
117 if (stage_samples_remaining_ > samples_remaining) {
118 stage_samples_remaining_ -= samples_remaining;
119 stage_progress_ = 1.0 - static_cast<mopo_float>(stage_samples_remaining_) /
120 stage_duration_samples_;
121 output(static_cast<int>(Outputs::Value))->buffer[0] = current_value_;
122 output(static_cast<int>(Outputs::Progress))->buffer[0] = stage_progress_;
123 return;
124 }
125
126 samples_remaining -= stage_samples_remaining_;
127 stage_samples_remaining_ = 0;
128 stage_progress_ = 1.0;
129 state_ = State::Attacking;
130 }
131
132 if (state_ == State::Attacking) {
133 // Attack is handled by linear incrementation until an ATTACK_DONE threshold
134 // is reached (ATTACK_DONE slightly less than 1.0 to avoid float edge cases).
135 mopo_float attack = utils::max(input(static_cast<int>(Inputs::Attack))->at(0), 0.000000001);
136 mopo_float attack_increment = 1.0 / (sample_rate_ * attack);
137 int samples = (ATTACK_DONE - current_value_) / attack_increment;
138
139 if (samples < samples_remaining) {
140 samples_remaining -= samples;
141 current_value_ = 1.0;
142 const Input* hold_input = input(static_cast<int>(Inputs::Hold));
143 mopo_float hold = hold_input->source == nullptr ? 0.0 :
144 utils::max(hold_input->at(0), 0.0);
145 stage_duration_samples_ = static_cast<int>(hold * sample_rate_ + 0.5);
146 stage_samples_remaining_ = stage_duration_samples_;
147 stage_progress_ = 0.0;
148 state_ = stage_samples_remaining_ > 0 ? State::Holding : State::Decaying;
149 }
150 else {
151 // Still attacking after this block: output current value and advance
152 // the envelope by samples_to_process_ samples worth of attack.
153 output(static_cast<int>(Outputs::Value))->buffer[0] = current_value_;
154 current_value_ += samples_remaining * attack_increment;
155 output(static_cast<int>(Outputs::Progress))->buffer[0] = stage_progress_;
156 return;
157 }
158 }
159
160 if (state_ == State::Holding) {
161 if (stage_samples_remaining_ > samples_remaining) {
162 stage_samples_remaining_ -= samples_remaining;
163 stage_progress_ = 1.0 - static_cast<mopo_float>(stage_samples_remaining_) /
164 stage_duration_samples_;
165 output(static_cast<int>(Outputs::Value))->buffer[0] = current_value_;
166 output(static_cast<int>(Outputs::Progress))->buffer[0] = stage_progress_;
167 return;
168 }
169
170 samples_remaining -= stage_samples_remaining_;
171 stage_samples_remaining_ = 0;
172 stage_duration_samples_ = 0;
173 stage_progress_ = 1.0;
174 state_ = State::Decaying;
175 }
176
177 if (state_ == State::Decaying) {
178 // Decay is exponential towards the sustain level. The SampleDecayLookup
179 // provides an efficient per-sample multiplier for a given decay time.
180 mopo_float decay_samples = sample_rate_ * input(static_cast<int>(Inputs::Decay))->at(0);
181 mopo_float sustain = input(static_cast<int>(Inputs::Sustain))->at(0);
182
183 mopo_float decay_decay_ = SampleDecayLookup::sampleDecayLookup(decay_samples);
184 mopo_float delta = current_value_ - sustain;
185 mopo_float end_delta = delta * pow(decay_decay_, samples_remaining);
186
187 current_value_ = sustain + end_delta;
188 output(static_cast<int>(Outputs::Value))->buffer[0] = current_value_;
189 }
190 else if (state_ == State::Releasing) {
191 // Release stage behaves similarly to decay, exponentially approaching
192 // zero with a time constant specified by the Release input.
193 mopo_float release_samples = sample_rate_ * input(static_cast<int>(Inputs::Release))->at(0);
194
195 mopo_float release_decay = SampleDecayLookup::sampleDecayLookup(release_samples);
196 current_value_ = current_value_ * pow(release_decay, samples_to_process_);
197 output(static_cast<int>(Outputs::Value))->buffer[0] = current_value_;
198 }
199 else if (state_ == State::Killing) {
200 // Kill mode is a hard ramp-down over VOICE_KILL_TIME seconds to ensure
201 // the voice is silenced quickly without producing hard clicks.
202 mopo_float decrement = samples_to_process_ / (VOICE_KILL_TIME * sample_rate_);
203 current_value_ = utils::max(0.0, current_value_ - decrement);
204 output(static_cast<int>(Outputs::Value))->buffer[0] = current_value_;
205 }
206
207 output(static_cast<int>(Outputs::Progress))->buffer[0] = stage_progress_;
208 }
