97void EnvelopeSection::paintBackground(Graphics& g) {
98 static const DropShadow component_shadow(Colour(Colors::Color_88000000), 2, Point<int>(0, 1));
99
100 int title_width = getTitleWidth();
101 int text_width = size_ratio_ * TEXT_WIDTH;
102 int slider_section_width = size_ratio_ * SLIDER_SECTION_WIDTH;
103
104 SynthSection::paintBackground(g);
105 component_shadow.drawForRectangle(g, envelope_->getBounds());
106
107 g.setColour(Colors::background);
108 g.fillRect(getWidth() - text_width - slider_section_width, title_width,
109 text_width, getHeight() - title_width);
110
111 g.setColour(Colors::control_label_text);
112 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(size_ratio_ * 10.0f));
113
114 const String labels[] = {TRANS("D"), TRANS("A"), TRANS("H"),
115 TRANS("D"), TRANS("S"), TRANS("R")};
116 int label_offset = has_delay_ ? 0 : 1;
117 for (int control = 0; control < control_count_; ++control) {
118 SynthSlider* slider = ordered_controls_[control];
119 g.drawText(labels[control + label_offset], slider->getX() - text_width,
120 slider->getY(), text_width, slider->getHeight(),
121 Justification::centred, true);
122 }
123}

125void EnvelopeSection::resized() {
126 int title_width = getTitleWidth();
127 int text_width = size_ratio_ * TEXT_WIDTH;
128 int slider_section_width = size_ratio_ * SLIDER_SECTION_WIDTH;
129 int slider_width = size_ratio_ * SLIDER_WIDTH;
130 int mod_section_width = size_ratio_ * MOD_SECTION_WIDTH;
131 int mod_button_width = getModButtonWidth();
132
133 int envelope_width = getWidth() - slider_section_width - text_width - mod_section_width;
134 envelope_->setBounds(mod_section_width, title_width, envelope_width, getHeight() - title_width);
135
136 int x = getWidth() - slider_section_width;
137 float space = (getHeight() - (control_count_ * slider_width) - title_width) /
138 static_cast<float>(control_count_ - 1);
139
140 float mod_button_x = (mod_section_width - mod_button_width) / 2.0f;
141 float mod_button_y = (getHeight() - mod_button_width + title_width) / 2.0f;
142 modulation_button_->setBounds(mod_button_x, mod_button_y, mod_button_width, mod_button_width);
143 for (int control = 0; control < control_count_; ++control)
144 ordered_controls_[control]->setBounds(x, title_width + control * (space + slider_width),
145 slider_section_width, slider_width);
146
147 SynthSection::resized();
148}
