75void EnvelopeSection::paintBackground(Graphics& g) {

78 int title_width = getTitleWidth();

106void EnvelopeSection::resized() {
107 int title_width = getTitleWidth();
108 int text_width = size_ratio_ * TEXT_WIDTH;
109 int slider_section_width = size_ratio_ * SLIDER_SECTION_WIDTH;
110 int slider_width = size_ratio_ * SLIDER_WIDTH;
111 int mod_section_width = size_ratio_ * MOD_SECTION_WIDTH;
112 int mod_button_width = getModButtonWidth();
113
114 int envelope_width = getWidth() - slider_section_width - text_width - mod_section_width;
115 envelope_->setBounds(mod_section_width, title_width, envelope_width, getHeight() - title_width);
116
117 float space = (getHeight() - (4.0f * slider_width) - title_width) / 3.0f;
118 int x = getWidth() - slider_section_width;
119
120 float mod_button_x = (mod_section_width - mod_button_width) / 2.0f;
121 float mod_button_y = (getHeight() - mod_button_width + title_width) / 2.0f;
122 modulation_button_->setBounds(mod_button_x, mod_button_y, mod_button_width, mod_button_width);
123 attack_->setBounds(x, title_width, slider_section_width, slider_width);
124 decay_->setBounds(x, title_width + space + slider_width, slider_section_width, slider_width);
125 sustain_->setBounds(x, title_width + 2 * (space + slider_width),
126 slider_section_width, slider_width);
127 release_->setBounds(x, title_width + 3 * (space + slider_width),
128 slider_section_width, slider_width);
129
130 SynthSection::resized();
131}
