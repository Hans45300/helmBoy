93 void Processor::plugNext(const Output* source) {
94 for (size_t i = 0; i < inputs_->size(); ++i) {
95 Input* input = inputs_->at(i);
96 if (input && input->source == &Processor::null_source_) {
97 plug(source, i);
98 return;
99 }
100 }
101
102 // If there are no empty inputs, create another.
103 Input* input = new Input();
104 owned_inputs_.push_back(input);
105 input->source = source;
106 registerInput(input);
107 }

189 Output* Processor::addOutput() {
190 Output* output = 0;
191 if (control_rate_)
192 output = new cr::Output();
193 else
194 output = new Output();
195
196 owned_outputs_.push_back(output);
197
198 // All outputs are owned by this Processor.
199 output->owner = this;
200 registerOutput(output);
201 return output;
202 }
