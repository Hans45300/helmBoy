39void XYPad::paintBackground(Graphics& g) {

49void XYPad::paint(Graphics& g) {
50 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
51
52 g.drawImage(background_,
53 0, 0, getWidth(), getHeight(),
54 0, 0, background_.getWidth(), background_.getHeight());
55
56 float x = x_slider_->getValue() * getWidth();
57 float y = (1.0f - y_slider_->getValue()) * getHeight();
58
59 Path target;
60 float handle_radius = 0.05f * getWidth();
61 target.addEllipse(x - handle_radius, y - handle_radius,
62 2.0f * handle_radius, 2.0f * handle_radius);
63
64 shadow.drawForPath(g, target);
65
66 g.setColour(Colors::graph_fill);
67 g.fillPath(target);
68
69 if (active_)
70 g.setColour(Colors::audio);
71 else
72 g.setColour(Colors::graph_disable);
73
74 PathStrokeType stroke(0.01f * getWidth(), PathStrokeType::beveled, PathStrokeType::rounded);
75 g.strokePath(target, stroke);
76 float dot_radius = 0.01f * getWidth();
77
78 g.fillEllipse(x - dot_radius, y - dot_radius, 2.0f * dot_radius, 2.0f * dot_radius);
79
80 if (mouse_down_) {
81 g.setColour(Colour(Colors::Color_11ffffff));
82 float hover_radius = 0.2 * getWidth();
83 g.fillEllipse(x - hover_radius, y - hover_radius, 2.0f * hover_radius, 2.0f * hover_radius);
84 }
85}

87void XYPad::resized() {
88 auto *display = Desktop::getInstance().getDisplays().getPrimaryDisplay();
89 jassert(display != nullptr);
90
91 float scale = display->scale;
92 background_ = Image(Image::RGB, scale * getWidth(), scale * getHeight(), true);
93 Graphics g(background_);
94 g.addTransform(AffineTransform::scale(scale, scale));
95 paintBackground(g);
96}

113void XYPad::setSlidersFromPosition(Point<int> position) {
114 if (x_slider_) {
115 double percent = mopo::utils::clamp((1.0 * position.x) / getWidth(), 0.0, 1.0);
116 x_slider_->setValue(percent);
117 }
118 if (y_slider_) {
119 double percent = mopo::utils::clamp(1.0 - (1.0 * position.y) / getHeight(), 0.0, 1.0);
120 y_slider_->setValue(percent);
121 }
122}
