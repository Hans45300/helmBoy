44 void Oscillator::process() {
45 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Frequency)));
46 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Phase)));
47
48 preprocess();
49
50 int i = 0;
51 // Gestion du reset : reste en scalaire pour la justesse
52 if (input(static_cast<int>(Inputs::Reset))->source->triggered &&
53 input(static_cast<int>(Inputs::Reset))->source->trigger_value == static_cast<mopo_float>(kVoiceReset)) {
54 int trigger_offset = input(static_cast<int>(Inputs::Reset))->source->trigger_offset;
55 for (; i < trigger_offset; ++i)
56 tick(i);
57 offset_ = 0.0;
58 }
59
60 // SIMD : traitement par paquets
61 using batch = xsimd::batch<mopo_float>;
62 constexpr std::size_t simd_size = batch::size;
63 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
64
65 auto* freq_buf = input(static_cast<int>(Inputs::Frequency))->source->buffer;
66 auto* phase_buf = input(static_cast<int>(Inputs::Phase))->source->buffer;
67 auto* out_audio = output(static_cast<int>(Outputs::Audio))->buffer;
68 auto* out_phase = output(static_cast<int>(Outputs::OscPhase))->buffer;
69
70 for (; i < simd_end; i += simd_size) {
71 batch freq = batch::load_unaligned(&freq_buf[i]);
72 batch phase = batch::load_unaligned(&phase_buf[i]);
73
74 // Calcul vectoris� de l'offset et du signal
75 batch offset = batch(offset_);
76 batch sample_rate = batch(sample_rate_);
77 offset += freq / sample_rate;
78 // Pour la justesse, on ne vectorise pas mod() ni blwave (trop de d�pendances lookup)
79 // On stocke les offsets interm�diaires, puis on boucle scalairement pour la g�n�ration
80 alignas(alignof(batch)) mopo_float offset_arr[simd_size];
81 offset.store_unaligned(offset_arr);
82 for (std::size_t j = 0; j < simd_size; ++j) {
83 offset_ = offset_arr[j];
84 mopo_float integral;
85 offset_ = utils::mod(offset_, &integral);
86 out_phase[i + j] = offset_;
87 out_audio[i + j] = Wave::blwave(waveform_, offset_ + phase.get(j), freq.get(j));
88 }
89 }
90 // Reste en scalaire
91 for (; i < buffer_size_; ++i)
92 tick(i);
93 }
