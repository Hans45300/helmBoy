110void DelaySection::resized() {
111 int title_width = getTitleWidth();
112 on_->setBounds(size_ratio_ * 2.0f, 0, title_width, title_width);
113 int knob_width = getSmallKnobSize();
114 int text_width = size_ratio_ * TEXT_WIDTH;
115 int text_height = size_ratio_ * TEXT_HEIGHT;
116
117 int horizontal_margin = size_ratio_ * 2;
118 int selector_width = text_width + text_height;
119 int selector_x = std::max(horizontal_margin,
120 getWidth() - horizontal_margin - selector_width);
121 int label_height = size_ratio_ * 10.0f;
122 int knob_y = getHeight() - knob_width - label_height - size_ratio_ * 3.0f;
123 int selector_y = title_width + (title_width - text_height) / 2;
124 float knob_space = (getWidth() - (3.0f * knob_width)) / 4.0f;
125
126 ping_pong_->setBounds(horizontal_margin, title_width, title_width, title_width);
127 frequency_->setBounds(selector_x, selector_y, text_width, text_height);
128 sync_->setBounds(selector_x + text_width, selector_y, text_height, text_height);
129 tempo_->setBounds(frequency_->getBounds());
130 feedback_->setBounds(knob_space, knob_y, knob_width, knob_width);
131 dry_wet_->setBounds(2 * knob_space + knob_width, knob_y, knob_width, knob_width);
132 lp_cutoff_->setBounds(3 * knob_space + 2 * knob_width, knob_y, knob_width, knob_width);
133
134 SynthSection::resized();
135
136 frequency_->setPopupDisplayEnabled(false, false, nullptr);
137 tempo_->setPopupDisplayEnabled(false, false, nullptr);
138}
