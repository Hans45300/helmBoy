59OpenGLOscilloscope::OpenGLOscilloscope() : output_memory_(nullptr) {
60 line_data_ = new float[2 * RESOLUTION];
61 line_indices_ = new int[2 * RESOLUTION];
62
63 for (int i = 0; i < RESOLUTION; ++i) {
64 float t = i / (RESOLUTION - 1.0f);
65 line_data_[2 * i] = 2.0f * t - 1.0f;
66 line_data_[2 * i + 1] = 0.0f;
67
68 line_indices_[2 * i] = i;
69 line_indices_[2 * i + 1] = i + 1;
70 }
71
72 line_indices_[2 * RESOLUTION - 1] = RESOLUTION - 1;
73}

75OpenGLOscilloscope::~OpenGLOscilloscope() {
76 delete[] line_data_;
77 delete[] line_indices_;
78}

93void OpenGLOscilloscope::init(OpenGLContext& open_gl_context) {
94 open_gl_context.extensions.glGenBuffers(1, &line_buffer_);
95 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, line_buffer_);
96
97 GLsizeiptr vert_size = static_cast<GLsizeiptr>(2 * RESOLUTION * sizeof(float));
98 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
99 line_data_, GL_STATIC_DRAW);
100
101 open_gl_context.extensions.glGenBuffers(1, &line_indices_buffer_);
102 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, line_indices_buffer_);
103
104 GLsizeiptr line_size = static_cast<GLsizeiptr>(2 * RESOLUTION * sizeof(int));
105 open_gl_context.extensions.glBufferData(GL_ELEMENT_ARRAY_BUFFER, line_size,
106 line_indices_, GL_STATIC_DRAW);
107
108 const char* vertex_shader = Shaders::getShader(Shaders::kOscilloscopeVertex);
109 const char* fragment_shader = Shaders::getShader(Shaders::kOscilloscopeFragment);
110
111 shader_ = std::make_unique<OpenGLShaderProgram>(open_gl_context);
112
113 if (shader_->addVertexShader(OpenGLHelpers::translateVertexShaderToV3(vertex_shader)) &&
114 shader_->addFragmentShader(OpenGLHelpers::translateFragmentShaderToV3(fragment_shader)) &&
115 shader_->link()) {
116 shader_->use();
117 position_ = std::make_unique<OpenGLShaderProgram::Attribute>(*shader_, "position");
118 }
119}

121void OpenGLOscilloscope::drawLines(OpenGLContext& open_gl_context) {
122 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
123 glEnable(GL_LINE_SMOOTH);
124 glHint(GL_LINE_SMOOTH_HINT, GL_NICEST);
125 glLineWidth(1.0f);
126
127 setViewPort(open_gl_context);
128
129 if (output_memory_) {
130 for (int i = 0; i < RESOLUTION; ++i) {
131 float memory_spot = (1.0f * i * mopo::MEMORY_RESOLUTION) / RESOLUTION;
132 int memory_index = memory_spot;
133 float remainder = memory_spot - memory_index;
134 line_data_[2 * i + 1] = mopo::utils::interpolate(output_memory_[memory_index],
135 output_memory_[memory_index + 1],
136 remainder);
137 }
138
139 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, line_buffer_);
140
141 GLsizeiptr vert_size = static_cast<GLsizeiptr>(2 * RESOLUTION * sizeof(float));
142 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
143 line_data_, GL_STATIC_DRAW);
144
145 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
146 }
147
148 shader_->use();
149 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, line_buffer_);
150 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, line_indices_buffer_);
151
152 open_gl_context.extensions.glVertexAttribPointer(position_->attributeID, 2, GL_FLOAT,
153 GL_FALSE, 2 * sizeof(float), 0);
154 open_gl_context.extensions.glEnableVertexAttribArray(position_->attributeID);
155 glDrawElements(GL_LINES, 2 * RESOLUTION, GL_UNSIGNED_INT, 0);
156
157 open_gl_context.extensions.glDisableVertexAttribArray(position_->attributeID);
158
159 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
160 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
161 glDisable(GL_LINE_SMOOTH);
162 MOPO_ASSERT(glGetError() == GL_NO_ERROR);
163}
