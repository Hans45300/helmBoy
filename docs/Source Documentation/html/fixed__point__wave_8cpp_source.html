51 void FixedPointWaveLookup::preprocessSin() {
52 for (auto h = 0; h < HARMONICS + 1; ++h) {
53 for (auto i = 0; i < FIXED_LOOKUP_SIZE; ++i)
54 sin_[h][i] = sin((2 * PI * i) / FIXED_LOOKUP_SIZE);

60 void FixedPointWaveLookup::preprocessTriangle() {
61 for (auto i = 0; i < FIXED_LOOKUP_SIZE; ++i) {
62 triangle_[0][i] = Wave::triangle((1.0 * i) / FIXED_LOOKUP_SIZE);
63
64 int p = i;
65 mopo_float scale = 8.0 / (PI * PI);
66 triangle_[HARMONICS][i] = scale * sin_[0][p];
67
68 for (auto h = 1; h < HARMONICS; ++h) {
69 p = (p + i) % FIXED_LOOKUP_SIZE;
70 triangle_[HARMONICS - h][i] = triangle_[HARMONICS - h + 1][i];
71 mopo_float harmonic = scale * sin_[0][p] / ((h + 1) * (h + 1));
72
73 if (h % 4 == 0)
74 triangle_[HARMONICS - h][i] += harmonic;
75 else if (h % 2 == 0)
76 triangle_[HARMONICS - h][i] -= harmonic;
77 }
78 }
79
80 preprocessDiffs(triangle_);
81 }

85 square_[0][i] = Wave::square((1.0 * i) / FIXED_LOOKUP_SIZE);

88 mopo_float scale = 4.0 / PI;
89 square_[HARMONICS][i] = scale * sin_[0][p];
90

92 p = (p + i) % FIXED_LOOKUP_SIZE;

112 void FixedPointWaveLookup::preprocessUpSaw() {
113 for (auto i = 0; i < FIXED_LOOKUP_SIZE; ++i) {
114 up_saw_[0][i] = Wave::upsaw((1.0 * i) / FIXED_LOOKUP_SIZE);
115
116 int index = (i + (FIXED_LOOKUP_SIZE / 2)) % FIXED_LOOKUP_SIZE;
117 int p = i;
118 mopo_float scale = 2.0 / PI;
119 up_saw_[HARMONICS][index] = scale * sin_[0][p];
120
121 for (auto h = 1; h < HARMONICS; ++h) {
122 p = (p + i) % FIXED_LOOKUP_SIZE;
123 mopo_float harmonic = scale * sin_[0][p] / (h + 1);
124
125 if (h % 2 == 0)
126 up_saw_[HARMONICS - h][index] = up_saw_[HARMONICS - h + 1][index] + harmonic;
127 else
128 up_saw_[HARMONICS - h][index] = up_saw_[HARMONICS - h + 1][index] - harmonic;
129 }
130 }
131
132 preprocessDiffs(up_saw_);
133 }

135 template<size_t steps> requires (steps > 1)
136 void FixedPointWaveLookup::preprocessStep(wave_type buffer) {
137 static int num_steps = steps;
138 static const mopo_float step_size = num_steps / (num_steps - 1.0);
139
140 for (auto h = 0; h < HARMONICS + 1; ++h) {
141 int base_num_harmonics = HARMONICS + 1 - h;
142 int harmony_num_harmonics = base_num_harmonics / num_steps;
143 int harmony_h = HARMONICS + 1 - harmony_num_harmonics;
144
145 for (auto i = 0; i < FIXED_LOOKUP_SIZE; ++i) {
146 buffer[h][i] = step_size * up_saw_[h][i];
147
148 if (harmony_num_harmonics) {
149 int harm_index = (num_steps * i) % FIXED_LOOKUP_SIZE;
150 buffer[h][i] += step_size * down_saw_[harmony_h][harm_index] / num_steps;
151 }
152 }
153 }
154
155 preprocessDiffs(buffer);
156 }

158 template<size_t steps> requires (steps > 1)
159 void FixedPointWaveLookup::preprocessPyramid(wave_type buffer) {
160 static const int squares = steps - 1;
161 static const int offset = 3 * FIXED_LOOKUP_SIZE / 4;
162
163 for (int h = 0; h < HARMONICS + 1; ++h) {
164 for (int i = 0; i < FIXED_LOOKUP_SIZE; ++i) {
165 buffer[h][i] = 0;
166
167 for (auto s = 0u; s < squares; ++s) {
168 int square_offset = (s * FIXED_LOOKUP_SIZE) / (2 * squares);
169 int phase = (i + offset + square_offset) % FIXED_LOOKUP_SIZE;
170 buffer[h][i] += square_[h][phase] / squares;
171 }
172 }
173 }
174
175 preprocessDiffs(buffer);
176 }

178 void FixedPointWaveLookup::preprocessDiffs(wave_type wave) {
179 for (auto h = 0; h < HARMONICS + 1; ++h) {
180 for (auto i = 0; i < FIXED_LOOKUP_SIZE - 1; ++i)
181 wave[h][i + FIXED_LOOKUP_SIZE] = FRACTIONAL_MULT * (wave[h][i + 1] - wave[h][i]);
182
183 mopo_float last_delta = wave[h][0] - wave[h][FIXED_LOOKUP_SIZE - 1];
184 wave[h][2 * FIXED_LOOKUP_SIZE - 1] = FRACTIONAL_MULT * last_delta;
185 }
186 }
