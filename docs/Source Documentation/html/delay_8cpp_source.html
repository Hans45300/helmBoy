110 void Delay::updateLowPassTarget(mopo_float cutoff) {
111 const mopo_float safe_sample_rate = std::max(1, sample_rate_);
112 const mopo_float safe_maximum = std::max(kMinimumDelayLowPassCutoff,
113 safe_sample_rate * 0.45);
114 mopo_float safe_cutoff = utils::clamp(cutoff, kMinimumDelayLowPassCutoff,
115 kMaximumDelayLowPassCutoff);
116 const bool fully_open = safe_cutoff >= kMaximumDelayLowPassCutoff - 0.5;
117 if (fully_open)
118 safe_cutoff = kMaximumDelayLowPassCutoff;
119 else
120 safe_cutoff = std::min(safe_cutoff, safe_maximum);
121 if (safe_cutoff == low_pass_target_cutoff_)
122 return;
123
124 if (fully_open) {
125 low_pass_target_coefficients_ = {};
126 }
127 else {
128 low_pass_target_coefficients_[0] =
129 BiquadFilter::lowPassCoefficients(safe_cutoff, kFirstButterworthQ, safe_sample_rate);
130 low_pass_target_coefficients_[1] =
131 BiquadFilter::lowPassCoefficients(safe_cutoff, kSecondButterworthQ, safe_sample_rate);
132 }
133 low_pass_target_cutoff_ = safe_cutoff;
134 low_pass_smoothing_remaining_ = low_pass_smoothing_samples_;
135 }

137 void Delay::smoothLowPassCoefficients() {
138 if (low_pass_smoothing_remaining_ <= 0)
139 return;
140
141 const mopo_float remaining = low_pass_smoothing_remaining_;
142 for (size_t stage = 0; stage < low_pass_coefficients_.size(); ++stage) {
143 auto& current = low_pass_coefficients_[stage];
144 const auto& target = low_pass_target_coefficients_[stage];
145 current.in_0 += (target.in_0 - current.in_0) / remaining;
146 current.in_1 += (target.in_1 - current.in_1) / remaining;
147 current.in_2 += (target.in_2 - current.in_2) / remaining;
148 current.out_1 += (target.out_1 - current.out_1) / remaining;
149 current.out_2 += (target.out_2 - current.out_2) / remaining;
150 }
151
152 if (--low_pass_smoothing_remaining_ == 0)
153 low_pass_coefficients_ = low_pass_target_coefficients_;
154 }

156 mopo_float Delay::processLowPassSample(int channel, mopo_float sample) {
157 if (low_pass_target_cutoff_ >= kMaximumDelayLowPassCutoff &&
158 low_pass_smoothing_remaining_ == 0) {
159 for (auto& state : low_pass_state_[channel]) {
160 state.past_in_2 = state.past_in_1;
161 state.past_in_1 = sample;
162 state.past_out_2 = state.past_out_1;
163 state.past_out_1 = sample;
164 }
165 return sample;
166 }
167
168 for (size_t stage = 0; stage < low_pass_coefficients_.size(); ++stage) {
169 auto& state = low_pass_state_[channel][stage];
170 const auto& coefficients = low_pass_coefficients_[stage];
171 const mopo_float filtered = sample * coefficients.in_0 +
172 state.past_in_1 * coefficients.in_1 +
173 state.past_in_2 * coefficients.in_2 -
174 state.past_out_1 * coefficients.out_1 -
175 state.past_out_2 * coefficients.out_2;
176 state.past_in_2 = state.past_in_1;
177 state.past_in_1 = sample;
178 state.past_out_2 = state.past_out_1;
179 state.past_out_1 = filtered;
180 sample = filtered;
181 }
182 return sample;
183 }

193 void Delay::process() {
194
195 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
196
197 auto audio = std::span<const mopo_float>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
198 const Output* right_source = input(static_cast<int>(Inputs::AudioRight))->source;
199 if (right_source == &null_source_)
200 right_source = input(static_cast<int>(Inputs::Audio))->source;
201 MOPO_ASSERT(right_source->buffer_size >= buffer_size_);
202 const mopo_float* right_audio = right_source->buffer;
203 mopo_float* right_dest = output(1)->buffer;
204 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
205
206 const Output* cutoff_source = input(static_cast<int>(Inputs::DelayLpCutoff))->source;
207 const mopo_float requested_cutoff =
208 cutoff_source == &null_source_ ? kMaximumDelayLowPassCutoff
209 : input(static_cast<int>(Inputs::DelayLpCutoff))->at(0);
210 const mopo_float cutoff = std::isfinite(requested_cutoff)
211 ? utils::clamp(requested_cutoff, kMinimumDelayLowPassCutoff,
212 kMaximumDelayLowPassCutoff)
213 : kMaximumDelayLowPassCutoff;
214 updateLowPassTarget(cutoff);
215
216 // Read target wet value and compute a square-root curve for perceptual
217 // loudness compensation. We also compute the corresponding dry value.
218 mopo_float wet = utils::clamp(input(static_cast<int>(Inputs::Wet))->at(0), 0.0, 1.0);
219 mopo_float new_wet = sqrt(wet);
220 mopo_float new_dry = sqrt(1.0 - wet);
221 // Compute per-sample increments to smoothly interpolate parameter values
222 // across the current buffer to avoid clicks.
223 mopo_float wet_inc = (new_wet - current_wet_) / buffer_size_;
224 mopo_float dry_inc = (new_dry - current_dry_) / buffer_size_;
225
226 // Feedback parameter interpolation
227 mopo_float new_feedback = input(static_cast<int>(Inputs::Feedback))->at(0);
228 mopo_float feedback_inc = (new_feedback - current_feedback_) / buffer_size_;
229
230 // Delay period (in samples) clamped to [2, memory_size - 1]
231 mopo_float new_period = utils::clamp(input(static_cast<int>(Inputs::SampleDelay))->at(0), 2.0, memory_->getSize() - 1.0);
232 mopo_float period_inc = (new_period - current_period_) / buffer_size_;
233 const bool ping_pong = input(static_cast<int>(Inputs::PingPong))->at(0) > 0.5;
234
235 // Process each sample in the block, updating smoothed parameters and
236 // calling tick() which performs the per-sample memory read/write.
237 std::ranges::for_each(std::views::iota(0, buffer_size_), [&](int i) {
238 current_feedback_ += feedback_inc;
239 current_wet_ += wet_inc;
240 current_dry_ += dry_inc;
241 current_period_ += period_inc;
242 const mopo_float left_read = memory_->get(current_period_);
243 const mopo_float right_read = right_memory_->get(current_period_);
244 smoothLowPassCoefficients();
245 const mopo_float left_tap = processLowPassSample(0, left_read);
246 const mopo_float right_tap = processLowPassSample(1, right_read);
247 const mopo_float left_feedback = ping_pong ? right_tap : left_tap;
248 const mopo_float right_feedback = ping_pong ? left_tap : right_tap;
249 memory_->push(audio[i] + left_feedback * current_feedback_);
250 right_memory_->push(right_audio[i] + right_feedback * current_feedback_);
251 dest[i] = current_dry_ * audio[i] + current_wet_ * (ping_pong ? right_tap : left_tap);
252 right_dest[i] = current_dry_ * right_audio[i] +
253 current_wet_ * (ping_pong ? left_tap : right_tap);
254 MOPO_ASSERT(std::isfinite(dest[i]));
255 MOPO_ASSERT(std::isfinite(right_dest[i]));
256 });
257 }

268 inline void Delay::tick(int i, const mopo_float* audio, mopo_float* dest) {
269 const mopo_float read = memory_->get(current_period_);
270 smoothLowPassCoefficients();
271 const mopo_float filtered = processLowPassSample(0, read);
272 memory_->push(audio[i] + filtered * current_feedback_);
273 dest[i] = current_dry_ * audio[i] + current_wet_ * filtered;
274 MOPO_ASSERT(std::isfinite(dest[i]));
275 }
