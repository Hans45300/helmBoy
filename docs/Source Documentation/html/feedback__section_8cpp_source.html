31FeedbackSection::FeedbackSection(String name) : SynthSection(name) {
32 static const int TRANSPOSE_MOUSE_SENSITIVITY = 800;
33 static const char* control_names[3][3] = {
34 {"osc_feedback_transpose", "osc_feedback_tune", "osc_feedback_amount"},
35 {"osc_2_feedback_transpose", "osc_2_feedback_tune", "osc_2_feedback_amount"},
36 {"sub_noise_feedback_transpose", "sub_noise_feedback_tune", "sub_noise_feedback_amount"}
37 };
38 static const char* group_names[] = {"Osc1", "Osc2", "Sub + Noise"};
39 static const char* control_names_for_display[] = {"Transpose", "Tune", "Amount"};
40
41 for (int group = 0; group < 3; ++group) {
42 for (int control = 0; control < 3; ++control) {
43 auto slider = std::make_unique<SynthSlider>(control_names[group][control]);
44 slider->setSliderStyle(Slider::RotaryHorizontalVerticalDrag);
45 slider->setBipolar();
46 slider->setPopupPlacement(BubbleComponent::above, 0);
47 slider->setTooltip(String(group_names[group]) + " feedback " +
48 control_names_for_display[control]);
49 if (control == 0)
50 slider->setMouseDragSensitivity(TRANSPOSE_MOUSE_SENSITIVITY);
51 addSlider(slider.get());
52 feedback_controls_[group][control] = std::move(slider);
53 }
54 }
55}

57FeedbackSection::~FeedbackSection() {
58 for (auto& group : feedback_controls_) {
59 for (auto& slider : group)
60 slider.reset();
61 }

64void FeedbackSection::paintBackground(Graphics& g) {
65 SynthSection::paintBackground(g);
66
67 g.setColour(Colors::control_label_text);
68 int title_height = getTitleWidth();
69 int group_header_height = size_ratio_ * 12.0f;
70 int label_height = size_ratio_ * 9.0f;
71 int group_width = getWidth() / 3;
72 int header_y = title_height;
73 int controls_y = header_y + group_header_height + size_ratio_ * 2.0f;
74 int second_row_label_height = size_ratio_ * 8.0f;
75 const String group_names[] = {TRANS("Osc1"), TRANS("Osc2"), TRANS("Sub + Noise")};
76 const String tune_label = TRANS("TUNE");
77 const String transpose_label = TRANS("TRANS");
78 const String amount_label = TRANS("AMT");
79
80 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(size_ratio_ * 9.0f));
81 for (int group = 0; group < 3; ++group) {
82 int group_x = group * group_width;
83 int current_group_width = group == 2 ? getWidth() - group_x : group_width;
84 int control_width = current_group_width / 3;
85 int second_cell_width = current_group_width / 2;
86 int preferred_knob_size = jmax(1, static_cast<int>(size_ratio_ * 22.0f));
87 int horizontal_padding = static_cast<int>(size_ratio_ * 2.0f);
88 int tune_cell_width = jmax(1, control_width - horizontal_padding);
89 int side_cell_width = jmax(1, second_cell_width - horizontal_padding);
90 int available_height = jmax(1, getHeight() - controls_y - label_height);
91 int knob_size = jmax(1, jmin(preferred_knob_size,
92 jmin(available_height, jmin(tune_cell_width, side_cell_width))));
93 int second_row_y = controls_y + knob_size;
94
95 g.drawFittedText(group_names[group], group_x, header_y, current_group_width,
96 group_header_height, Justification::centred, 1, 0.65f);
97
98 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(size_ratio_ * 8.0f));
99 g.drawFittedText(tune_label, group_x + control_width,
100 controls_y + knob_size, control_width, label_height,
101 Justification::centred, 1, 0.6f);
102 int second_row_label_y = second_row_y + knob_size;
103 g.drawFittedText(transpose_label, group_x, second_row_label_y,
104 second_cell_width, second_row_label_height,
105 Justification::centred, 1, 0.6f);
106 g.drawFittedText(amount_label, group_x + second_cell_width,
107 second_row_label_y,
108 current_group_width - second_cell_width,
109 second_row_label_height, Justification::centred, 1, 0.6f);
110 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(size_ratio_ * 9.0f));
111 }
112}

125 int control_width = current_group_width / 3;
126 int second_cell_width = current_group_width / 2;
127 int preferred_knob_size = jmax(1, static_cast<int>(size_ratio_ * 22.0f));
128 int horizontal_padding = static_cast<int>(size_ratio_ * 2.0f);
129 int tune_cell_width = jmax(1, control_width - horizontal_padding);
130 int side_cell_width = jmax(1, second_cell_width - horizontal_padding);
131 int available_height = jmax(1, getHeight() - controls_y - label_height);
132 int knob_size = jmax(1, jmin(preferred_knob_size,
133 jmin(available_height, jmin(tune_cell_width, side_cell_width))));
134 feedback_controls_[group][1]->setBounds(
135 group_x + control_width + (control_width - knob_size) / 2,
136 controls_y, knob_size, knob_size);
137
138 int second_row_y = controls_y + knob_size;
139 feedback_controls_[group][0]->setBounds(
140 group_x + (second_cell_width - knob_size) / 2,
141 second_row_y, knob_size, knob_size);
142 feedback_controls_[group][2]->setBounds(
143 group_x + second_cell_width +
144 (current_group_width - second_cell_width - knob_size) / 2,
145 second_row_y, knob_size, knob_size);
146 }
