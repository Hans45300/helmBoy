65void FeedbackSection::resized() {
66 int knob_width = getStandardKnobSize();
67 float space = (getWidth() - (3.0f * knob_width)) / 4.0f;
68 int y = size_ratio_ * 26;
69
70 transpose_->setBounds(space, y, knob_width, knob_width);
71 tune_->setBounds((knob_width + space) + space, y, knob_width, knob_width);
72 amount_->setBounds(2 * (knob_width + space) + space, y, knob_width, knob_width);
73
74 SynthSection::resized();
75}
