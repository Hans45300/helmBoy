60void DistortionSection::paintBackground(Graphics& g) {
61 SynthSection::paintBackground(g);
62 int knob_width = getStandardKnobSize();
63
64 g.setColour(Colors::control_label_text);
65 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(size_ratio_ * 8.0f));
66
67 g.drawText(TRANS("TYPE"),
68 type_->getBounds().getX() - size_ratio_ * 5.0f,
69 drive_->getBounds().getY() + knob_width + size_ratio_ * 4,
70 type_->getBounds().getWidth() + size_ratio_ * 10.0f,
71 size_ratio_ * 10.0f, Justification::centred, false);
72 drawTextForComponent(g, TRANS("DRIVE"), drive_.get());
73 drawTextForComponent(g, TRANS("MIX"), mix_.get());
74}

76void DistortionSection::resized() {
77 int title_width = getTitleWidth();
78 int knob_width = getStandardKnobSize();
79 int text_width = size_ratio_ * TEXT_WIDTH;
80 int text_y = size_ratio_ * 44;
81
82 on_->setBounds(size_ratio_ * 2.0f, 0, title_width, title_width);
83
84 float space = (getWidth() - (2.0f * knob_width + text_width)) / 4.0f;
85 int y = size_ratio_ * 30;
86
87 type_->setBounds(space, text_y, text_width, size_ratio_ * TEXT_HEIGHT);
88 drive_->setBounds(text_width + 2 * space, y, knob_width, knob_width);
89 mix_->setBounds(text_width + knob_width + 3 * space, y, knob_width, knob_width);
90
91 SynthSection::resized();
92}
