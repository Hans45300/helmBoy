119void UpdateCheckSection::checkUpdate() {
120 static const int TIMEOUT = 200;
121 URL version_url("http://tytel.org/static/dist/helm_version.txt");
122 auto in = version_url.createInputStream(false, nullptr, nullptr, "", TIMEOUT);
123
124 if (in == nullptr)
125 return;
126
127 version_ = in->readEntireStreamAsString().upToFirstOccurrenceOf("\n", false, false);
128
129 if (!version_.isEmpty() &&
130 LoadSave::compareVersionStrings(ProjectInfo::versionString, version_) < 0) {
131 repaint();
132 setVisible(true);
133 }
134}
