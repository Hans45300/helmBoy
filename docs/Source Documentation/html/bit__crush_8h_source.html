59 void tick(int i) {
60 mopo_float audio = input(static_cast<int>(Inputs::Audio))->at(i);
61 mopo_float wet = input(static_cast<int>(Inputs::Wet))->at(i);
62
63 mopo_float out = std::floor(magnification_ * (1.0 + audio) + 0.5) / magnification_ - 1.0;
64
65 output(0)->buffer[i] = utils::interpolate(audio, out, wet);
66 }

68 protected:
