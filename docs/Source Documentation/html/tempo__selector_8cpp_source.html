44void TempoSelector::mouseDown(const MouseEvent& e) {
45 if (e.mods.isPopupMenu()) {
46 SynthSlider::mouseDown(e);
47 return;
48 }
49 PopupMenu m;
50 m.setLookAndFeel(DefaultLookAndFeel::instance());
51
52 m.addItem(kSeconds, "Seconds");
53 m.addItem(kTempo, "Tempo");
54 m.addItem(kTempoDotted, "Tempo Dotted");
55 m.addItem(kTempoTriplet, "Tempo Triplets");
56
57 m.showMenuAsync(PopupMenu::Options().withTargetComponent(this),
58 ModalCallbackFunction::forComponent(tempoTypeSelectedCallback, this));
59}

75void TempoSelector::paint(Graphics& g) {
76 if (isActive())
77 g.setColour(Colour(Colors::Color_ffbbbbbb));
78 else
79 g.setColour(Colour(Colors::Color_ff555555));
80 g.fillRect(0, 0, getWidth(), getHeight());
81
82 g.setColour(Colour(Colors::Color_ff222222));
83 g.fillPath(arrow_);
84
85 int value = getValue() + 1;
86 if (value == kSeconds)
87 g.fillPath(clock_);
88 else if (value == kTempo || value == kTempoDotted) {
89 g.fillEllipse(getWidth() / 3.0f, getHeight() / 2.0f,
90 getWidth() / 3.0f, getHeight() / 4.0f);
91 g.fillRect(2.0f * getWidth() / 3.0f - 1.0f, getHeight() / 6.0f,
92 1.0f, 11.0f * getHeight() / 24.0f);
93 }
94 else if (value == kTempoTriplet) {
95 float width = getWidth() / 4.0f;
96 float height = getHeight() / 6.0f;
97 float x = getWidth() / 4.0f - getWidth() / 16.0f;
98 float y = getHeight() / 2.0f + getHeight() / 16.0f;
99
100 g.fillRect(x + width - 1.0f, y + height / 2.0f - getHeight() / 3.0f,
101 getWidth() / 2.0f, 1.0f);
102
103 g.fillEllipse(x, y, width, height);
104 g.fillRect(x + width - 1.0f, y + height / 2.0f - getHeight() / 3.0f,
105 1.0f, getHeight() / 3.0f);
106 g.fillEllipse(x + getWidth() / 4.0f, y, width, height);
107 g.fillRect(x + width - 1.0f + getWidth() / 4.0f, y + height / 2.0f - getHeight() / 3.0f,
108 1.0f, getHeight() / 3.0f);
109 g.fillEllipse(x + getWidth() / 2.0f, y, width, height);
110 g.fillRect(x + width - 1.0f + getWidth() / 2.0f, y + height / 2.0f - getHeight() / 3.0f,
111 1.0f, getHeight() / 3.0f);
112 }
113 if (value == kTempoDotted) {
114 g.fillEllipse(3.0f * getWidth() / 4.0f, getHeight() / 2.0f,
115 getWidth() / 6.0f, getHeight() / 6.0f);
116 }
117}

119void TempoSelector::resized() {
120 static const float clock_angle = 1.0f;
121
122 arrow_.clear();
123 arrow_.startNewSubPath(getWidth() / 8.0f, getHeight() / 8.0f);
124 arrow_.lineTo(3.0f * getWidth() / 8.0f, getHeight() / 8.0f);
125 arrow_.lineTo(2.0f * getWidth() / 8.0f, 2.0f * getHeight() / 8.0f);
126
127 clock_.clear();
128 clock_.addPieSegment(7.0f * getWidth() / 24.0f, getHeight() / 4.0f,
129 getWidth() / 2.0f, getHeight() / 2.0f,
130 0.0f, clock_angle - 2.0f * static_cast<float>(mopo::PI), 0.0f);
131}
