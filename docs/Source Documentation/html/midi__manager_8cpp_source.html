62void MidiManager::midiInput(int midi_id, mopo::mopo_float value) {
63 if (armed_value_) {
64 midi_learn_map_[midi_id][armed_value_->name] = armed_value_;
65 armed_value_ = nullptr;
66
67 // TODO: Probably shouldn't write this config on the audio thread.
68 LoadSave::saveMidiMapConfig(this);
69 }
70
71 if (midi_learn_map_.count(midi_id)) {
72 for (auto& control : midi_learn_map_[midi_id]) {
73 const mopo::ValueDetails* details = control.second;
74 mopo::mopo_float percent = value / (mopo::MIDI_SIZE - 1);
75 if (details->steps) {
76 mopo::mopo_float max_step = details->steps - 1;
77 percent = floor(percent * max_step + 0.5) / max_step;
78 }
79
80 mopo::mopo_float translated = percent * (details->max - details->min) + details->min;
81 listener_->valueChangedThroughMidi(control.first, translated);
82 }
83 }
84}

102void MidiManager::processMidiMessage(const MidiMessage& midi_message, int sample_position) {
103 if (midi_message.isProgramChange()) {
104 current_patch_ = midi_message.getProgramChangeNumber();
105 File patch = LoadSave::loadPatch(current_bank_, current_folder_, current_patch_,
106 synth_, *gui_state_);
107 PatchLoadedCallback* callback = new PatchLoadedCallback(listener_, patch);
108 callback->post();
109 return;
110 }
111
112 if (midi_message.isNoteOn()) {
113 engine_->noteOn(midi_message.getNoteNumber(),
114 midi_message.getVelocity() / (mopo::MIDI_SIZE - 1.0),
115 0, midi_message.getChannel() - 1);
116 }
117 else if (midi_message.isNoteOff())
118 engine_->noteOff(midi_message.getNoteNumber());
119 else if (midi_message.isAllNotesOff())
120 engine_->allNotesOff();
121 else if (midi_message.isSustainPedalOn())
122 engine_->sustainOn();
123 else if (midi_message.isSustainPedalOff())
124 engine_->sustainOff();
125 else if (midi_message.isAftertouch()) {
126 mopo::mopo_float note = midi_message.getNoteNumber();
127 mopo::mopo_float value = (1.0 * midi_message.getAfterTouchValue()) / mopo::MIDI_SIZE;
128 engine_->setAftertouch(note, value);
129 }
130 else if (midi_message.isChannelPressure()) {
131 int channel = midi_message.getChannel();
132 mopo::mopo_float value = midi_message.getChannelPressureValue() / (mopo::MIDI_SIZE - 1.0f);
133 engine_->setChannelAftertouch(channel, value);
134 }
135 else if (midi_message.isPitchWheel()) {
136 double percent = (1.0 * midi_message.getPitchWheelValue()) / PITCH_WHEEL_RESOLUTION;
137 double value = 2 * percent - 1.0;
138 engine_->setPitchWheel(value, midi_message.getChannel());
139 }
140 else if (midi_message.isController()) {
141 int controller_number = midi_message.getControllerNumber();
142 if (controller_number == MOD_WHEEL_CONTROL_NUMBER) {
143 double percent = (1.0 * midi_message.getControllerValue()) / MOD_WHEEL_RESOLUTION;
144 engine_->setModWheel(percent, midi_message.getChannel());
145 }
146 else if (controller_number == BANK_SELECT_NUMBER)
147 current_bank_ = midi_message.getControllerValue();
148 else if (controller_number == FOLDER_SELECT_NUMBER)
149 current_folder_ = midi_message.getControllerValue();
150 else {
151 // Create midi_id that includes channel: (channel << 8) | controller_number
152 int midi_id = (midi_message.getChannel() << 8) | midi_message.getControllerNumber();
153 midiInput(midi_id, midi_message.getControllerValue());
154 }
155 }
156}
