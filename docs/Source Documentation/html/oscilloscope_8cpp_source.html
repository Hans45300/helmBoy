72void Oscilloscope::resetWavePath() {
73 if (output_memory_ == nullptr)
74 return;
75
76 wave_path_.clear();
77
78 float draw_width = getWidth() - 2.0f * PADDING_X;
79 float draw_height = getHeight() - 2.0f * PADDING_Y;
80
81 wave_path_.startNewSubPath(PADDING_X, getHeight() / 2.0f);
82 for (int i = 0; i < mopo::MEMORY_RESOLUTION; ++i) {
83 float t = (1.0f * i) / mopo::MEMORY_RESOLUTION;
84 float val = output_memory_[i];
85 if (val != val)
86 val = 0.0f;
87 wave_path_.lineTo(PADDING_X + t * draw_width, PADDING_Y + draw_height * ((1.0f - val) / 2.0f));
88 }
89
90 wave_path_.lineTo(getWidth() - PADDING_X, getHeight() / 2.0f);
91}
