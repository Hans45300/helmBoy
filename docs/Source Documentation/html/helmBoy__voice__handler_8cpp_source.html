114 void HelmBoyVoiceHandler::init() {
115 // Create modulation and pitch wheels per channel.
116 choose_pitch_wheel_ = new Gate();
117 // Correction�: Gate::Inputs::Choice n'existe pas, remplacer par l'entr�e appropri�e ou commenter si non utilis�
118 // TODO: Remplacer par l'entr�e correcte si besoin
119
120 Gate* choose_mod_wheel = new Gate();
121 // Correction�: Gate::Inputs::Choice n'existe pas, remplacer par l'entr�e appropri�e ou commenter si non utilis�
122 // TODO: Remplacer par l'entr�e correcte si besoin
123
124 for (auto i = 0; i < mopo::NUM_MIDI_CHANNELS; ++i) {
125 pitch_wheel_amounts_[i] = new cr::Value(0);
126 choose_pitch_wheel_->plugNext(pitch_wheel_amounts_[i]);
127
128 mod_wheel_amounts_[i] = new cr::Value(0);
129 choose_mod_wheel->plugNext(mod_wheel_amounts_[i]);
130
131 addGlobalProcessor(pitch_wheel_amounts_[i]);
132 addGlobalProcessor(mod_wheel_amounts_[i]);
133 }
134
135 getMonoRouter()->addProcessor(choose_pitch_wheel_);
136 getMonoRouter()->addProcessor(choose_mod_wheel);
137
138 mod_sources_["pitch_wheel"] = choose_pitch_wheel_->output();
139 mod_sources_["mod_wheel"] = choose_mod_wheel->output();
140
141 // Create all synthesizer voice components.
142 createArticulation(note(), last_note(), velocity(), voice_event());
143 createOscillators(current_frequency_->output(),
144 amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)));
145 createModulators(amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)));
146 createFilter(osc_feedback_->output(0), note_from_center_->output(),
147 amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)));
148
149 Value* aftertouch_value = new cr::Value();
150 aftertouch_value->plug(aftertouch());
151
152 addProcessor(aftertouch_value);
153 mod_sources_["aftertouch"] = aftertouch_value->output();
154
155 output_->plug(formant_container_, 0);
156 output_->plug(amplitude_, 1);
157
158 addProcessor(output_);
159
160 setVoiceKiller(amplitude_->output());
161
162 HelmBoyModule::init();
163 setupPolyModulationReadouts();
164 }

385 addProcessor(osc_feedback_amount_audio);

463 // Filter envelope.
464 Output* filter_attack = createPolyModControl("fil_attack", true);
465 Output* filter_decay = createPolyModControl("fil_decay", true);

469 filter_envelope_ = new Envelope();

471 filter_envelope_->plug(filter_decay, static_cast<int>(Envelope::Inputs::Decay));

686 void HelmBoyVoiceHandler::createArticulation(
687 Output* note, Output* last_note, Output* velocity, Output* trigger) {
688 // Legato.
689 legato_ = createBaseControl("legato");
690 LegatoFilter* legato_filter = new LegatoFilter();
691 legato_filter->plug(legato_, static_cast<int>(LegatoFilter::Inputs::Legato));
692 legato_filter->plug(trigger, static_cast<int>(LegatoFilter::Inputs::Trigger));
693
694 addProcessor(legato_filter);
695
696 // Amplitude envelope.
697 Output* amplitude_attack = createPolyModControl("amp_attack", true);
698 Output* amplitude_decay = createPolyModControl("amp_decay", true);
699 Output* amplitude_sustain = createPolyModControl("amp_sustain", true);
700 Output* amplitude_release = createPolyModControl("amp_release", true);
701
702 amplitude_envelope_ = new Envelope();
703 amplitude_envelope_->plug(legato_filter->output(static_cast<int>(LegatoFilter::Outputs::Retrigger)),
704 static_cast<int>(Envelope::Inputs::Trigger));
705 amplitude_envelope_->plug(amplitude_attack, static_cast<int>(Envelope::Inputs::Attack));
706 amplitude_envelope_->plug(amplitude_decay, static_cast<int>(Envelope::Inputs::Decay));
707 amplitude_envelope_->plug(amplitude_sustain, static_cast<int>(Envelope::Inputs::Sustain));
708 amplitude_envelope_->plug(amplitude_release, static_cast<int>(Envelope::Inputs::Release));
709 addProcessor(amplitude_envelope_);
710
711 // Voice and frequency resetting logic.
712 TriggerCombiner* note_change_trigger = new TriggerCombiner();
713 note_change_trigger->plug(legato_filter->output(static_cast<int>(LegatoFilter::Outputs::Remain)), 0);
714 note_change_trigger->plug(amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)), 1);
715
716 TriggerWait* note_wait = new TriggerWait();
717 Value* current_note = new Value();
718 note_wait->plug(note, static_cast<int>(TriggerWait::Inputs::Wait));
719 note_wait->plug(note_change_trigger, static_cast<int>(TriggerWait::Inputs::Trigger));
720 current_note->plug(note_wait);
721
722 static const cr::Value max_midi_invert(1.0 / (MIDI_SIZE - 1));
723 cr::Multiply* note_percentage = new cr::Multiply();
724 note_percentage->plug(&max_midi_invert, 0);
725 note_percentage->plug(current_note, 1);
726
727 addProcessor(note_change_trigger);
728 addProcessor(note_wait);
729 addProcessor(current_note);
730
731 // Key tracking.
732 static const Value center_adjust(-MIDI_SIZE / 2);
733 note_from_center_ = new cr::Add();
734 note_from_center_->plug(&center_adjust, 0);
735 note_from_center_->plug(current_note, 1);
736
737 addProcessor(note_from_center_);
738 addProcessor(note_percentage);
739
740 // Velocity tracking.
741 TriggerWait* velocity_wait = new TriggerWait();
742 cr::Value* current_velocity = new cr::Value();
743 velocity_wait->plug(velocity, static_cast<int>(TriggerWait::Inputs::Wait));
744 velocity_wait->plug(note_change_trigger, static_cast<int>(TriggerWait::Inputs::Trigger));
745 current_velocity->plug(velocity_wait);
746
747 addProcessor(velocity_wait);
748 addProcessor(current_velocity);
749
750 Output* velocity_track_amount = createPolyModControl("velocity_track", true);
751 cr::Interpolate* velocity_track_mult = new cr::Interpolate();
752 velocity_track_mult->plug(&utils::value_one, static_cast<int>(Interpolate::Inputs::From));
753 velocity_track_mult->plug(current_velocity, static_cast<int>(Interpolate::Inputs::To));
754 velocity_track_mult->plug(velocity_track_amount, static_cast<int>(Interpolate::Inputs::Fractional));
755 addProcessor(velocity_track_mult);
756
757 // Current amplitude using envelope and velocity.
758 Processor* control_amplitude = new cr::Multiply();
759 control_amplitude->plug(amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Value)), 0);
760 control_amplitude->plug(velocity_track_mult, 1);
761
762 amplitude_ = new LinearSmoothBuffer();
763 amplitude_->plug(control_amplitude, static_cast<int>(LinearSmoothBuffer::Inputs::Value));
764 amplitude_->plug(amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)),
765 static_cast<int>(LinearSmoothBuffer::Inputs::Trigger));
766
767 addProcessor(control_amplitude);
768 addProcessor(amplitude_);
769
770 // Portamento.
771 Output* portamento = createPolyModControl("portamento", true);
772 Value* portamento_type = createBaseControl("portamento_type");
773
774 current_frequency_ = new PortamentoSlope();
775 current_frequency_->plug(current_note, static_cast<int>(PortamentoSlope::Inputs::Target));
776 current_frequency_->plug(portamento_type, static_cast<int>(PortamentoSlope::Inputs::PortamentoType));
777 current_frequency_->plug(note_pressed(), static_cast<int>(PortamentoSlope::Inputs::NoteNumber));
778 current_frequency_->plug(portamento, static_cast<int>(PortamentoSlope::Inputs::RunSeconds));
779 current_frequency_->plug(note_pressed(), static_cast<int>(PortamentoSlope::Inputs::TriggerJump));
780 current_frequency_->plug(last_note, static_cast<int>(PortamentoSlope::Inputs::TriggerStart));
781
782 addProcessor(current_frequency_);
783
784 mod_sources_["amp_envelope"] = amplitude_envelope_->output();
785 mod_sources_["amp_envelope_amp"] =
786 registerOutput(amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Value)));
787 mod_sources_["amp_envelope_phase"] =
788 registerOutput(amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Phase)));
789 mod_sources_["note"] = note_percentage->output();
790 mod_sources_["velocity"] = current_velocity->output();
791
792 // Envelope Trigger.
793 TriggerFilter* note_off = new TriggerFilter(kVoiceOff);
794 note_off->plug(trigger);
795 env_trigger_ = new TriggerCombiner();
796 env_trigger_->plug(note_off, 0);
797 env_trigger_->plug(amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)), 1);
798
799 addProcessor(note_off);
800 addProcessor(env_trigger_);
801 }
