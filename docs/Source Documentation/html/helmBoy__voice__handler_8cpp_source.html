120 void HelmBoyVoiceHandler::init() {
121 const std::lock_guard<std::mutex> initLock(init_mutex_);
122 if (init_done_.load(std::memory_order_acquire)) {
123 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init already done");
124 return;
125 }
126
127 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init start");
128
129 try {
130
131 // Create modulation and pitch wheels per channel.
132 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels start");
133 choose_pitch_wheel_ = new Gate();
134 if (choose_pitch_wheel_ == nullptr)
135 throw std::runtime_error("HelmBoyVoiceHandler::init: choose_pitch_wheel allocation failed");
136 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels choose_pitch_wheel alloc done");
137 choose_pitch_wheel_->plug(channel(), Gate::kChoice);
138 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels choose_pitch_wheel choice plug done");
139
140 for (auto i = 0; i < mopo::NUM_MIDI_CHANNELS; ++i) {
141 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels loop i=" + juce::String(i) + " pitch alloc start");
142 pitch_wheel_amounts_[i] = new cr::Value(0);
143 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels loop i=" + juce::String(i) + " pitch alloc done");
144 choose_pitch_wheel_->plugNext(pitch_wheel_amounts_[i]);
145 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels loop i=" + juce::String(i) + " pitch plugNext done");
146
147 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels loop i=" + juce::String(i) + " mod alloc start");
148 mod_wheel_amounts_[i] = new cr::Value(0);
149 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels loop i=" + juce::String(i) + " mod alloc done");
150
151 addGlobalProcessor(pitch_wheel_amounts_[i]);
152 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels loop i=" + juce::String(i) + " addGlobalProcessor pitch done");
153 addGlobalProcessor(mod_wheel_amounts_[i]);
154 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels loop i=" + juce::String(i) + " addGlobalProcessor mod done");
155 }
156
157 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels addGlobalProcessor choose_pitch_wheel start");
158 addGlobalProcessor(choose_pitch_wheel_);
159 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels addGlobalProcessor choose_pitch_wheel done");
160
161 mod_sources_["pitch_wheel"] = choose_pitch_wheel_->output();
162 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels mod source pitch set");
163 mod_sources_["mod_wheel"] = mod_wheel_amounts_[0]->output();
164 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels mod source mod set (fallback ch0)");
165 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init wheels done");
166
167 // Create all synthesizer voice components.
168 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init createArticulation start");
169 createArticulation(note(), last_note(), velocity(), voice_event());
170 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init createArticulation done");
171
172 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init createOscillators start");
173 createOscillators(current_frequency_->output(),
174 amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)));
175 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init createOscillators done");
176
177 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init createModulators start");
178 createModulators(amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)));
179 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init createModulators done");
180
181 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init createFilter start");
182 createFilter(osc_feedback_sum_->output(0), note_from_center_->output(),
183 amplitude_envelope_->output(static_cast<int>(Envelope::Outputs::Finished)));
184 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init createFilter done");
185
186 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init aftertouch start");
187 Value* channel_aftertouch_value = new cr::Value();
188 channel_aftertouch_value->plug(channel_aftertouch());
189 Value* poly_aftertouch_value = new cr::Value();
190 poly_aftertouch_value->plug(aftertouch());
191
192 addProcessor(channel_aftertouch_value);
193 addProcessor(poly_aftertouch_value);
194 mod_sources_["channel_aftertouch"] = channel_aftertouch_value->output();
195 mod_sources_["poly_aftertouch"] = poly_aftertouch_value->output();
196
197 output_->plug(formant_container_, 0);
198 output_->plug(amplitude_, 1);
199
200 addProcessor(output_);
201
202 setVoiceKiller(amplitude_->output());
203 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init aftertouch sources/output done");
204
205 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init submodule init start");
206 HelmBoyModule::init();
207 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init submodule init done");
208
209 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init setupPolyModulationReadouts start");
210 setupPolyModulationReadouts();
211 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init setupPolyModulationReadouts done");
212 init_done_.store(true, std::memory_order_release);
213 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init done");
214 }
215 catch (const std::exception& ex) {
216 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init exception: " + juce::String(ex.what()));
217 throw;
218 }
219 catch (...) {
220 appendStartupTraceVoiceHandler("HelmBoyVoiceHandler::init unknown exception");
221 throw std::runtime_error("HelmBoyVoiceHandler::init unknown exception");
222 }
