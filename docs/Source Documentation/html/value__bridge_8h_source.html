29class ValueBridge : public AudioProcessorParameter {

31 class Listener {
32 public:
33 virtual ~Listener() { }

37 ValueBridge(std::string name, mopo::Value* value) :
38 AudioProcessorParameter(), name_(name), value_(value), listener_(nullptr),
39 source_changed_(false) {
40 details_ = mopo::Parameters::getDetails(name);
41 span_ = details_.max - details_.min;
42 }
43
44 float getValue() const override {
45 return convertToPluginValue(value_->value());
46 }
47
48 void setValue(float value) override {
49 if (listener_ && !source_changed_) {
50 source_changed_ = true;
51 mopo::mopo_float synth_value = convertToSynthValue(value);
52 listener_->parameterChanged(name_.toStdString(), synth_value);
53 source_changed_ = false;
54 }
55 }
56
57 void setListener(Listener* listener) {
58 listener_ = listener;
59 }
60
61 float getDefaultValue() const override {
62 return convertToPluginValue(details_.default_value);
63 }
64
65 String getName(int maximumStringLength) const override {
66 return name_.substring(0, maximumStringLength);
67 }
68
69 String getLabel() const override {
70 return "";
71 }
72
73 String getText(float value, int maximumStringLength) const override {
74 return String(getSkewedValue()).substring(0, maximumStringLength);
75 }
76
77 float getValueForText(const String &text) const override {
78 return text.getFloatValue();
79 }
80
81 bool isAutomatable() const override {
82 return true;
83 }
84
85 // Converts internal value to value from 0.0 to 1.0.
86 float convertToPluginValue(mopo::mopo_float synth_value) const {
87 return (synth_value - details_.min) / span_;
88 }
89
90 // Converts from value from 0.0 to 1.0 to internal synth value.
91 float convertToSynthValue(mopo::mopo_float plugin_value) const {
92 mopo::mopo_float percent = plugin_value;
93 if (details_.steps) {
94 mopo::mopo_float max_step = details_.steps - 1;
95 percent = floor(percent * max_step + 0.5) / max_step;
96 }
97
98 return percent * span_ + details_.min;
99 }
100
101 void setValueNotifyHost(float new_value) {
102 if (!source_changed_) {
103 source_changed_ = true;
104 setValueNotifyingHost(new_value);
105 source_changed_ = false;
106 }
107 }
108
109 private:
110 float getSkewedValue() const {
111 switch (details_.display_skew) {
112 case mopo::DisplaySkew::kQuadratic:
113 return powf(value_->value(), 2.0f);
114 case mopo::DisplaySkew::kExponential:
115 return powf(2.0f, value_->value());
116 default:
117 return value_->value();
118 }
119 }
