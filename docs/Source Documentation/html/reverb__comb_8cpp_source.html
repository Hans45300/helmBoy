48 void ReverbComb::process() {
49 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
50 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Feedback)));
51 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Damping)));
52
53 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
54 auto audio_buffer = std::span<const mopo_float>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
55 const int raw_period = static_cast<int>(input(static_cast<int>(Inputs::SampleDelay))->source->buffer[0]);
56 int period = std::clamp(raw_period, 1, memory_->getSize() - 1);
57#if defined(HELMBOY_DEBUG_REVERB)
58 if (raw_period != period)
59 std::fprintf(stderr, "[HelmBoy][ReverbComb] period clamped raw=%d → %d (max=%d)\n",
60 raw_period, period, memory_->getSize() - 1);
61#endif
62 auto feedback_buffer = std::span<const mopo_float>(input(static_cast<int>(Inputs::Feedback))->source->buffer, buffer_size_);
63 auto damping_buffer = std::span<const mopo_float>(input(static_cast<int>(Inputs::Damping))->source->buffer, buffer_size_);
64
65 for (int i = 0; i < buffer_size_; ++i)
66 tick(i, dest.data(), period, audio_buffer.data(), feedback_buffer.data(), damping_buffer.data());
67 }
