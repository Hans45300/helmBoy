27void TextSlider::paint(Graphics& g) {
28 static const PathStrokeType stroke(1.000f, PathStrokeType::curved, PathStrokeType::rounded);
29 static const float text_percentage = 0.7f;
30
31 int num_types = getMaximum() - getMinimum() + 1;
32 float cell_width = float(getWidth()) / num_types;
33 float height = getHeight();
34
35 int selected = getValue();
36 // Couleur du fond du slider
37 g.setColour(Colour(Colors::Color_ff323232));
38 g.fillRect(selected * cell_width, 0.0f, cell_width, height);
39
40 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(height * text_percentage));
41
42 const std::string* lookup = short_lookup_;
43 if (lookup == nullptr)
44 lookup = string_lookup_;
45
46 for (int i = 0; i < num_types; ++i) {
47 if (selected == i)
48 // Couleur du texte lorsque le slider est actif
49 g.setColour(Colour(Colors::Color_ffffffff));
50 else
51 // Couleur du texte lorsque le slider est inactif
52 g.setColour(Colour(Colors::Color_ffaaaaaa));
53
54 g.drawText(lookup[i], i * cell_width, 0.0f, cell_width, height, Justification::centred);
55 }
56}
