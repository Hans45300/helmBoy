52void Startup::doStartupChecks(MidiManager* midi_manager, mopo::StringLayout* layout) {
53 if (!LoadSave::isInstalled())
54 return;
55
56 fixPatchesFolder();
57
58 if (LoadSave::wasUpgraded()) {
59 storeOldFactoryPatches();
60 copyFactoryPatches();
61 LoadSave::saveVersionConfig();
62 LoadSave::saveLastAskedForMoney();
63 }
64 else

72void Startup::storeOldFactoryPatches() {
73 String old_version = LoadSave::loadVersion();
74
75 if (LoadSave::isInstalled() && LoadSave::compareVersionStrings(old_version, "0.9.0") < 0) {
76 File bank_directory = LoadSave::getBankDirectory();
77 File old_presets_dest = bank_directory.getChildFile("Old Factory Presets");
78
79 if (old_presets_dest.exists() || old_presets_dest.createDirectory().ok()) {
80 File factory_directory = bank_directory.getChildFile("Factory Presets");
81 File factory_dest = old_presets_dest.getChildFile("Factory Presets");
82 if (factory_dest.exists() || factory_dest.createDirectory().ok()) {
83 Array<File> factory_presets;
84 factory_directory.findChildFiles(factory_presets, File::findFiles, true,
85 String("*.") + mopo::PATCH_EXTENSION);
86
87 for (const auto& factory_preset : factory_presets) {
88 File new_dest = factory_dest.getChildFile(factory_preset.getFileName());
89 factory_preset.moveFileTo(new_dest);
90 }
91
92 factory_directory.deleteRecursively();
93 }
94
95 File cris_directory = bank_directory.getChildFile("Cris Owl Alvarez");
96 File cris_dest = old_presets_dest.getChildFile("Cris Owl Alvarez");
97 if (cris_dest.exists() || cris_dest.createDirectory().ok()) {
98 Array<File> cris_presets;
99 cris_directory.findChildFiles(cris_presets, File::findFiles, true,
100 String("*.") + mopo::PATCH_EXTENSION);
101
102 for (const auto& cris_preset : cris_presets) {
103 File new_dest = cris_dest.getChildFile(cris_preset.getFileName());
104 cris_preset.moveFileTo(new_dest);
105 }
106
107 cris_directory.deleteRecursively();
108 }
109 }
110 }

113void Startup::copyFactoryPatches() {
114 File factory_bank_dir = LoadSave::getFactoryBankDirectory();
115 File destination = LoadSave::getBankDirectory();
116 Array<File> factory_banks;
117 factory_bank_dir.findChildFiles(factory_banks, File::findDirectories, false);
118
119 for (const auto& factory_bank : factory_banks) {
120 if (isBankDirectory(factory_bank)) {
121 Array<File> patches;
122 factory_bank.findChildFiles(patches, File::findFiles, true,
123 String("*.") + mopo::PATCH_EXTENSION);
124
125 for (const auto& patch : patches) {
126 String relative_path = patch.getRelativePathFrom(factory_bank_dir);
127 File patch_location = destination.getChildFile(relative_path);
128
129 if (!patch_location.getParentDirectory().exists())
130 patch_location.getParentDirectory().createDirectory();
131 patch.copyFileTo(patch_location);
132 }
133 }
134 }

137void Startup::fixPatchesFolder() {
138 File bank_directory = LoadSave::getBankDirectory();
139 if (!bank_directory.exists())
140 bank_directory.createDirectory();
141
142 File user_bank = LoadSave::getUserBankDirectory();
143 if (!user_bank.exists())
144 user_bank.createDirectory();
145
146 // Moving old patches from v0.4.1 and earlier to new locations.
147 File old_user_bank = getOldUserPatchesDirectory();
148 File default_user_folder = user_bank.getChildFile("Default");
149 Array<File> misplaced_files;
150 old_user_bank.findChildFiles(misplaced_files, File::findFiles, false,
151 String("*.") + mopo::PATCH_EXTENSION);
152 if (misplaced_files.size() && !default_user_folder.exists())
153 default_user_folder.createDirectory();
154
155 for (const auto& misplaced_file : misplaced_files) {
156 File new_patch_location = default_user_folder.getChildFile(misplaced_file.getFileName());
157 misplaced_file.moveFileTo(new_patch_location);
158 }
159}

161void Startup::updateAllPatches(SynthBase* synth,
162 std::map<std::string, String>* gui_state,
163 const CriticalSection& critical_section) {
164 File user_bank = LoadSave::getBankDirectory();
165 Array<File> all_patches;
166 user_bank.findChildFiles(all_patches, File::findFiles, true,
167 String("*.") + mopo::PATCH_EXTENSION);
168
169 for (const auto& patch : all_patches) {
170 var parsed_json_state;
171 if (JSON::parse(patch.loadFileAsString(), parsed_json_state).wasOk()) {
172 LoadSave::varToState(synth, *gui_state, parsed_json_state);
173 String author = LoadSave::getAuthor(parsed_json_state);
174 if (author.isEmpty())
175 author = patch.getParentDirectory().getParentDirectory().getFileName();
176
177 std::map<std::string, String> gui_state;
178 gui_state["author"] = author;
179 gui_state["patch_name"] = patch.getFileNameWithoutExtension();
180 gui_state["folder_name"] = patch.getParentDirectory().getFileName();
181
182 var save_var = LoadSave::stateToVar(synth, gui_state, critical_section);
183 patch.replaceWithText(JSON::toString(save_var));
184 }
185 }
