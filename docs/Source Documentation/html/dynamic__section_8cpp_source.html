75void DynamicSection::resized() {
76 int knob_width = getStandardKnobSize();
77 int text_width = size_ratio_ * TEXT_WIDTH;
78 int text_height = size_ratio_ * TEXT_HEIGHT;
79 int selector_width = size_ratio_ * SELECTOR_WIDTH;
80
81 float space_x = (getWidth() - (knob_width + selector_width + text_width)) / 4.0f;
82 float space_y = (getHeight() - (knob_width + text_height)) / 2.0f;
83 float extra_text_space = 2 * (knob_width - text_height) / 3;
84
85 portamento_->setBounds(space_x, space_y, knob_width, knob_width);
86 portamento_type_->setBounds(knob_width + 2 * space_x, space_y + extra_text_space,
87 selector_width, text_height);
88 legato_->setBounds(knob_width + selector_width + 3 * space_x, space_y + extra_text_space,
89 text_width, text_height);
90
91 SynthSection::resized();
92}
