47 class MagnitudeLookupSingleton {
48 public:
49 MagnitudeLookupSingleton() {
50 for (int i = 0; i < MAGNITUDE_LOOKUP_RESOLUTION + 2; ++i) {
51 mopo_float t = (1.0 * i) / MAGNITUDE_LOOKUP_RESOLUTION;
52 mopo_float decibels = utils::interpolate(MIN_DB_LOOKUP, MAX_DB_LOOKUP, t);
53 magnitude_lookup_[i] = utils::dbToGain(decibels);
54 }
55 }

58 [[nodiscard]] mopo_float magnitudeLookup(mopo_float decibels) const {
59 mopo_float t = (decibels - MIN_DB_LOOKUP) / DB_RANGE;
60 mopo_float index = MAGNITUDE_LOOKUP_RESOLUTION * utils::clamp(t, 0.0, 1.0);
61 int int_index = static_cast<int>(index);
62 mopo_float fraction = index - int_index;
63
64 return utils::interpolate(magnitude_lookup_[int_index],
65 magnitude_lookup_[int_index + 1], fraction);
66 }
67
68 private:
69 mopo_float magnitude_lookup_[MAGNITUDE_LOOKUP_RESOLUTION + 2];
70 };
