42 void LinearSlope::process() {
43 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Target)));
44 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::RunSeconds)));
45
46 int i = 0;
47 // Cas courant : pas de trigger, runSeconds constant, rampe vectorisable
48 bool can_vectorize = !input(static_cast<int>(Inputs::TriggerJump))->source->triggered &&
49 utils::closeToZero(input(static_cast<int>(Inputs::RunSeconds))->at(0) - input(static_cast<int>(Inputs::RunSeconds))->at(buffer_size_ - 1));
50 if (can_vectorize) [[likely]] {
51 using batch = xsimd::batch<mopo_float>;
52 constexpr std::size_t simd_size = batch::size;
53 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
54 mopo_float increment = 1.0 / (sample_rate_ * input(static_cast<int>(Inputs::RunSeconds))->at(0));
55 mopo_float value = last_value_;
56 for (int idx = 0; idx < simd_end; idx += simd_size) {
57 alignas(alignof(batch)) mopo_float ramp[simd_size];
58 for (std::size_t j = 0; j < simd_size; ++j) {
59 ramp[j] = value + (idx + j) * increment;
60 }
61 batch b = batch::load_unaligned(ramp);
62 b.store_unaligned(&output(0)->buffer[idx]);
63 }
64 for (int idx = simd_end; idx < buffer_size_; ++idx) {
65 output(0)->buffer[idx] = value + idx * increment;
66 }
67 last_value_ = value + (buffer_size_ - 1) * increment;
68 } else {
69 if (input(static_cast<int>(Inputs::TriggerJump))->source->triggered) [[unlikely]] {
70 int trigger_offset = input(static_cast<int>(Inputs::TriggerJump))->source->trigger_offset;
71 for (; i < trigger_offset; ++i)
72 tick(i);
73 last_value_ = input(static_cast<int>(Inputs::Target))->at(i);
74 }
75 for (; i < buffer_size_; ++i)
76 tick(i);
77 }
78 }

80 inline void LinearSlope::tick(int i) {
81 mopo_float target = input(static_cast<int>(Inputs::Target))->at(i);
82 if (utils::closeToZero(input(static_cast<int>(Inputs::RunSeconds))->at(i))) [[unlikely]]
83 last_value_ = input(static_cast<int>(Inputs::Target))->at(i);
84
85 mopo_float increment = 1.0 / (sample_rate_ * input(static_cast<int>(Inputs::RunSeconds))->at(0));
86 if (target <= last_value_) [[likely]]
87 last_value_ = utils::clamp(last_value_ - increment, target, last_value_);
88 else
89 last_value_ = utils::clamp(last_value_ + increment, last_value_, target);
90 output(0)->buffer[i] = last_value_;
91 }
