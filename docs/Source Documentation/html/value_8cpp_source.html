41 void Value::process() {
42 using enum Inputs;
43 output()->clearTrigger();
44 if (output()->buffer[0] == value_ &&
45 output()->buffer[buffer_size_ - 1] == value_ &&
46 !input(static_cast<int>(Inputs::Set))->source->triggered) {
47 return;
48 }
49
50 if (input(static_cast<int>(Inputs::Set))->source->triggered) {
51 int i = 0;
52 int offset = std::min(buffer_size_, input(static_cast<int>(Inputs::Set))->source->trigger_offset);
53 for (; i < offset; ++i)
54 output()->buffer[i] = value_;
55
56 value_ = input(static_cast<int>(Inputs::Set))->source->trigger_value;
57
58 for (; i < buffer_size_; ++i)
59 output()->buffer[i] = value_;
60
61 output()->trigger(value_, input(static_cast<int>(Inputs::Set))->source->trigger_offset);
62 }
63 else {
64 for (int i = 0; i < buffer_size_; ++i)
65 output()->buffer[i] = value_;
66 }
67 }
