37 Value* HelmBoyModule::createBaseControl(std::string_view name, bool smooth_value) {
38 std::string name_str{name}; // Convert to string for map key
39 mopo_float default_value = Parameters::getDetails(name_str).default_value;
40 Value* val = 0;
41 if (smooth_value) {
42 val = new cr::SmoothValue(default_value);
43 getMonoRouter()->addProcessor(val);
44 }
45 else {
46 val = new cr::Value(default_value);
47 getMonoRouter()->addIdleProcessor(val);
48 }
49
50 controls_[name_str] = val;
51 return val;

64 Output* HelmBoyModule::createBaseModControl(std::string_view name, bool smooth_value) {
65 Processor* base_val = createBaseControl(name, smooth_value);
66
67 std::string name_str{name}; // Convert to string for map key
68
69 cr::VariableAdd* mono_total = new cr::VariableAdd();
70 mono_total->plugNext(base_val);
71 getMonoRouter()->addProcessor(mono_total);
72 mono_mod_destinations_[name_str] = mono_total;
73 mono_modulation_readout_[name_str] = mono_total->output();
74
75 ValueSwitch* control_switch = new ValueSwitch(0.0);
76 control_switch->plugNext(base_val);
77 control_switch->plugNext(mono_total);
78 control_switch->addProcessor(mono_total);
79 getMonoRouter()->addProcessor(control_switch);
80 control_switch->set(0);
81 mono_modulation_switches_[name_str] = control_switch;
82
83 return control_switch->output(ValueSwitch::kSwitch);
84 }

112 Output* HelmBoyModule::createMonoModControl(std::string_view name, bool control_rate,
113 bool smooth_value) {
114 ProcessorRouter* mono_owner = getMonoRouter();
115 std::string name_str{name}; // Convert to string for map key
116 ValueDetails details = Parameters::getDetails(name_str);
117 Output* control_rate_total = createBaseModControl(name, smooth_value);
118
119 if (details.display_skew == mopo::DisplaySkew::kQuadratic) {
120 Processor* scale = nullptr;
121 if (details.post_offset)
122 scale = new cr::Quadratic(details.post_offset);
123 else
124 scale = new cr::Square();
125
126 scale->plug(control_rate_total);
127 mono_owner->addProcessor(scale);
128 control_rate_total = scale->output();
129 }
130 else if (details.display_skew == mopo::DisplaySkew::kExponential) {
131 cr::ExponentialScale* exponential = new cr::ExponentialScale(2.0);
132 exponential->plug(control_rate_total);
133 mono_owner->addProcessor(exponential);
134 control_rate_total = exponential->output();
135 }
136 else if (details.display_skew == mopo::DisplaySkew::kSquareRoot) {
137 cr::Root* root = new cr::Root(details.post_offset);
138 root->plug(control_rate_total);
139 mono_owner->addProcessor(root);
140 control_rate_total = root->output();
141 }
142
143 if (control_rate)
144 return control_rate_total;
145
146 SampleAndHoldBuffer* audio_rate = new SampleAndHoldBuffer();
147 audio_rate->plug(control_rate_total);
148 mono_owner->addProcessor(audio_rate);
149
150 return audio_rate->output();
151 }

153 Output* HelmBoyModule::createPolyModControl(std::string_view name, bool control_rate,
154 bool smooth_value) {
155 std::string name_str{name}; // Convert to string for map key
156 ValueDetails details = Parameters::getDetails(name_str);
157 Output* base_control = createBaseModControl(name, smooth_value);
158 ProcessorRouter* poly_owner = getPolyRouter();
159
160 cr::VariableAdd* poly_total = new cr::VariableAdd();
161 poly_owner->addProcessor(poly_total);
162 poly_mod_destinations_[name_str] = poly_total;
163
164 cr::Add* modulation_total = new cr::Add();
165 modulation_total->plug(base_control, 0);
166 modulation_total->plug(poly_total, 1);
167 poly_owner->addProcessor(modulation_total);
168
169 poly_modulation_readout_[name_str] = poly_total->output();
170
171 ValueSwitch* control_switch = new ValueSwitch(0.0);
172 control_switch->plugNext(base_control);
173 control_switch->plugNext(modulation_total);
174 control_switch->addProcessor(poly_total);
175 control_switch->addProcessor(modulation_total);
176 control_switch->set(0);
177 poly_owner->addProcessor(control_switch);
178 poly_modulation_switches_[name_str] = control_switch;
179
180 Output* control_rate_total = control_switch->output(ValueSwitch::kSwitch);
181 if (details.display_skew == mopo::DisplaySkew::kQuadratic) {
182 Processor* scale = nullptr;
183 if (details.post_offset)
184 scale = new cr::Quadratic(details.post_offset);
185 else
186 scale = new cr::Square();
187
188 scale->plug(control_rate_total);
189 poly_owner->addProcessor(scale);
190 control_rate_total = scale->output();
191 }
192 else if (details.display_skew == mopo::DisplaySkew::kExponential) {
193 cr::ExponentialScale* exponential = new cr::ExponentialScale(2.0, details.post_offset);
194 exponential->plug(control_rate_total);
195 poly_owner->addProcessor(exponential);
196 control_rate_total = exponential->output();
197 }
198 else if (details.display_skew == mopo::DisplaySkew::kSquareRoot) {
199 cr::Root* root = new cr::Root(details.post_offset);
200 root->plug(control_rate_total);
201 poly_owner->addProcessor(root);
202 control_rate_total = root->output();
203 }
204
205 if (control_rate)
206 return control_rate_total;
207
208 SampleAndHoldBuffer* audio_rate = new SampleAndHoldBuffer();
209 audio_rate->plug(control_rate_total);
210 poly_owner->addProcessor(audio_rate);
211 return audio_rate->output();
212 }

214 Output* HelmBoyModule::createTempoSyncSwitch(std::string_view name, Processor* frequency,
215 Output* bps, bool poly, ValueSwitch* owner) {
216 static const Value dotted_ratio(2.0 / 3.0);
217 static const Value triplet_ratio(3.0 / 2.0);
218
219 std::string name_str{name}; // Convert to string for map key
220 ProcessorRouter* router = poly ? getPolyRouter() : getMonoRouter();
221 Output* tempo = nullptr;
222 if (poly)
223 tempo = createPolyModControl(name_str + "_tempo", frequency->isControlRate());
224 else
225 tempo = createMonoModControl(name_str + "_tempo", frequency->isControlRate());
226
227 Gate* choose_tempo = new Gate();
228 choose_tempo->plug(tempo, Gate::kChoice);
229
230 const bool is_lfo = name.starts_with("mono_lfo_") || name.starts_with("poly_lfo_");
231 const Value* tempo_ratios = is_lfo ? lfo_synced_freq_ratios : synced_freq_ratios;
232 const size_t num_tempo_ratios = is_lfo
233 ? sizeof(lfo_synced_freq_ratios) / sizeof(Value)
234 : sizeof(synced_freq_ratios) / sizeof(Value);
235 for (size_t i = 0; i < num_tempo_ratios; ++i)
236 choose_tempo->plugNext(&tempo_ratios[i]);
237
238 Gate* choose_modifier = new Gate();
239 Value* sync = new cr::Value(1);
240 router->addIdleProcessor(sync);
241 choose_modifier->plug(sync, Gate::kChoice);
242 choose_modifier->plugNext(&utils::value_one);
243 choose_modifier->plugNext(&utils::value_one);
244 choose_modifier->plugNext(&dotted_ratio);
245 choose_modifier->plugNext(&triplet_ratio);
246
247 Processor* modified_tempo = new cr::Multiply();
248 Processor* tempo_frequency = new cr::Multiply();
249
250 modified_tempo->plug(choose_tempo, 0);
251 modified_tempo->plug(choose_modifier, 1);
252
253 tempo_frequency->plug(modified_tempo, 0);
254 tempo_frequency->plug(bps, 1);
255
256 Gate* choose_frequency = new Gate();
257 choose_frequency->plug(sync, Gate::kChoice);
258 choose_frequency->plugNext(frequency->output());
259 choose_frequency->plugNext(tempo_frequency);
260 choose_frequency->plugNext(tempo_frequency);
261 choose_frequency->plugNext(tempo_frequency);
262
263 if (owner) {
264 owner->addProcessor(choose_tempo);
265 owner->addProcessor(choose_modifier);
266 owner->addProcessor(modified_tempo);
267 owner->addProcessor(tempo_frequency);
268 owner->addProcessor(choose_frequency);
269 owner->set(owner->value());
270 } else {
271 router->addProcessor(choose_tempo);
272 router->addProcessor(choose_modifier);
273 router->addProcessor(modified_tempo);
274 router->addProcessor(tempo_frequency);
275 router->addProcessor(choose_frequency);
276 }
277
278 controls_[name_str + "_sync"] = sync;
279 return choose_frequency->output();
280 }

421 void HelmBoyModule::updateAllModulationSwitches() {
422 for (auto& [destination, switch_control] : mono_modulation_switches_) {
423 bool enable = mono_mod_destinations_[destination]->connectedInputs() > 1;
424 if (poly_mod_destinations_.count(destination))
425 enable = enable || poly_mod_destinations_[destination]->connectedInputs() > 0;
426 switch_control->set(enable);
427 }
428
429 for (auto& [destination, switch_control] : poly_modulation_switches_)
430 switch_control->set(poly_mod_destinations_[destination]->connectedInputs() > 0);
431
432 for (HelmBoyModule* sub_module : sub_modules_)
433 sub_module->updateAllModulationSwitches();
434 }
