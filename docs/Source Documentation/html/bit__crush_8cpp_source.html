42 void BitCrush::process() {
43 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
44 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Wet)));
45
46 mopo_float bits = input(static_cast<int>(Inputs::Bits))->at(0);
47 magnification_ = std::pow(2.0, bits / 2.0);
48
49 using batch = xsimd::batch<mopo_float>;
50 constexpr std::size_t simd_size = batch::size;
51 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
52 auto audio = std::span<const mopo_float>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
53 auto wet = std::span<const mopo_float>(input(static_cast<int>(Inputs::Wet))->source->buffer, buffer_size_);
54 auto dest = std::span<mopo_float>(output(0)->buffer, buffer_size_);
55 batch magnif = batch(magnification_);
56 int i = 0;
57 for (; i < simd_end; i += simd_size) {
58 batch a = batch::load_unaligned(audio.data() + i);
59 batch w = batch::load_unaligned(wet.data() + i);
60 batch out = xsimd::floor(magnif * (batch(1.0) + a) + batch(0.5)) / magnif - batch(1.0);
61 batch result = a + w * (out - a);
62 result.store_unaligned(dest.data() + i);
63 }
64 std::ranges::for_each(std::views::iota(simd_end, buffer_size_), [this](int idx) { tick(idx); });
65 }
