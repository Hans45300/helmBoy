159void SynthesisInterface::resized() {
160 int column_1_x = padding_;
161 int column_2_x = column_1_x + padding_ + section_one_width_;
162 int column_3_x = column_2_x + padding_ + section_two_width_;
163 int section_three_left_width = (section_three_width_ - padding_) / 2;
164 int section_three_right_width = section_three_width_ - padding_ - section_three_left_width;
165 int column_4_x = column_3_x + padding_ + section_three_left_width;
166 int column_5_x = column_3_x + padding_ + section_three_width_;
167 int section_four_half_width = (section_four_width_ - padding_) / 2;
168
169 int sub_width = 0.53125f * section_one_width_;
170 int mixer_width = section_one_width_ - padding_ - sub_width;
171
172 int audio_height = size_ratio_ * 290.0f;
173 int oscillators_height = size_ratio_ * 180.0f;
174 int sub_mixer_height = audio_height - oscillators_height - padding_;
175 int envelopes_height = size_ratio_ * 105.0f;
176 int step_lfo_height = size_ratio_ * 148.0f;
177 int dynamics_height = size_ratio_ * 64.0f;
178 int keyboard_mod_height = dynamics_height + padding_;
179 int keyboard_padding = size_ratio_ * 5.0f;
180 int effect_height = (audio_height - 2 * padding_) / 3;
181 int last_effect_gap = audio_height - 3 * effect_height - padding_;
182 int feedback_height = effect_height;
183 int filter_height = audio_height - feedback_height - padding_;
184
185 oscillator_section_->setBounds(column_1_x, padding_, section_one_width_, oscillators_height);
186 sub_section_->setBounds(column_1_x, oscillator_section_->getBottom() + padding_,
187 sub_width, sub_mixer_height);
188 mixer_section_->setBounds(sub_section_->getRight() + padding_, sub_section_->getY(),
189 mixer_width, sub_mixer_height);
190 amplitude_envelope_section_->setBounds(column_1_x, sub_section_->getBottom() + padding_,
191 section_one_width_, envelopes_height);
192
193 feedback_section_->setBounds(column_2_x, padding_, section_two_width_, feedback_height);
194 filter_section_->setBounds(column_2_x, feedback_section_->getBottom() + padding_,
195 section_two_width_, filter_height);
196 filter_envelope_section_->setBounds(column_1_x, amplitude_envelope_section_->getBottom() + padding_,
197 section_one_width_, envelopes_height);
198
199 int lfo_width = 0.421875f * section_one_width_;
200 int step_sequencer_width = section_one_width_ + section_two_width_ + padding_ -
201 3 * (lfo_width + padding_);
202
203 int step_lfo_y = filter_envelope_section_->getBottom() + padding_;
204
205 // Calculer hauteur: 2/3 pour step_sequencer, 1/3 pour keyboard (rï¿½duit de moitiï¿½)
206 int available_height = getHeight() - step_lfo_y - 2 * padding_;
207 int sequencer_height = ((available_height - padding_) * 2 / 3);
208 int keyboard_height = ((available_height - padding_) / 3);
209
210 // Step sequencer ï¿½tendu sur 3 colonnes
211 step_sequencer_section_->setBounds(column_1_x, step_lfo_y,
212 column_3_x + section_three_width_ - column_1_x, sequencer_height);
213
214 distortion_section_->setBounds(column_4_x, padding_,
215 section_three_right_width, effect_height);
216 delay_section_->setBounds(column_4_x, distortion_section_->getBottom() + padding_,
217 section_three_right_width, effect_height);
218 reverb_section_->setBounds(column_4_x, delay_section_->getBottom() + last_effect_gap,
219 section_three_right_width, effect_height);
220
221 // LFOs dans la nouvelle colonne 5 (2 par ligne)
222 mono_lfo_1_section_->setBounds(column_5_x, padding_,
223 section_four_half_width, step_lfo_height);
224 mono_lfo_2_section_->setBounds(mono_lfo_1_section_->getRight() + padding_, padding_,
225 section_four_half_width, step_lfo_height);
226 // Ligne 2 de LFOs mono: mono_lfo_3 et mono_lfo_4
227 mono_lfo_3_section_->setBounds(column_5_x, mono_lfo_1_section_->getBottom() + padding_,
228 section_four_half_width, step_lfo_height);
229 mono_lfo_4_section_->setBounds(mono_lfo_3_section_->getRight() + padding_,
230 mono_lfo_1_section_->getBottom() + padding_,
231 section_four_half_width, step_lfo_height);
232 mono_lfo_5_section_->setBounds(column_5_x, mono_lfo_3_section_->getBottom() + padding_,
233 section_four_half_width, step_lfo_height);
234 mono_lfo_6_section_->setBounds(mono_lfo_5_section_->getRight() + padding_,
235 mono_lfo_5_section_->getY(),
236 section_four_half_width, step_lfo_height);
237
238 poly_lfo_1_section_->setBounds(column_5_x,
239 mono_lfo_5_section_->getBottom() + padding_,
240 section_four_half_width, step_lfo_height);
241
242 poly_lfo_2_section_->setBounds(poly_lfo_1_section_->getRight() + padding_,
243 poly_lfo_1_section_->getY(),
244 section_four_half_width, step_lfo_height);
245
246 stutter_section_->setBounds(column_3_x, distortion_section_->getY(),
247 distortion_section_->getWidth(), effect_height);
248 chorus_section_->setBounds(column_3_x, delay_section_->getY(),
249 distortion_section_->getWidth(), effect_height);
250 formant_section_->setBounds(column_3_x, reverb_section_->getY(),
251 distortion_section_->getWidth(), effect_height);
252
253 extra_envelope_section_->setBounds(column_2_x, amplitude_envelope_section_->getY(),
254 section_two_width_, envelopes_height);
255 mod_envelope_2_section_->setBounds(column_3_x, amplitude_envelope_section_->getY(),
256 section_three_width_, envelopes_height);
257 int second_mod_envelope_y = extra_envelope_section_->getBottom() + padding_;
258 mod_envelope_3_section_->setBounds(column_2_x, second_mod_envelope_y,
259 section_two_width_, envelopes_height);
260 mod_envelope_4_section_->setBounds(column_3_x, second_mod_envelope_y,
261 section_three_width_, envelopes_height);
262
263 // Leave one LFO-height gap before the poly LFO row, then compact the lower controls.
264 extra_mod_section_->setBounds(column_5_x, poly_lfo_1_section_->getBottom() + padding_,
265 section_four_width_, keyboard_mod_height);
266 int voice_dynamic_y = extra_mod_section_->getBottom() + padding_;
267 voice_section_->setBounds(column_5_x, voice_dynamic_y,
268 section_four_half_width, dynamics_height);
269 dynamic_section_->setBounds(voice_section_->getRight() + padding_, voice_dynamic_y,
270 section_four_half_width, dynamics_height);
271
272 // Keyboard ï¿½tendu sur les 3 colonnes d'origine, sous le step sequencer (1/3 de hauteur)
273 int keyboard_y = step_sequencer_section_->getBottom() + padding_;
274 keyboard_->setBounds(column_1_x, keyboard_y + keyboard_padding,
275 column_3_x + section_three_width_ - column_1_x - keyboard_padding,
276 keyboard_height - 2 * keyboard_padding);
277 keyboard_->setKeyWidth(size_ratio_ * 16.0f);
278
279 SynthSection::resized();
280}
