72void SubSection::resized() {
73 int knob_y = size_ratio_ * 24;
74 int knob_width = getStandardKnobSize();
75 int title_width = getTitleWidth();
76 int wave_section_width = size_ratio_ * WAVE_SECTION_WIDTH;
77 int wave_selector_width = size_ratio_ * WAVE_SELECTOR_WIDTH;
78 int space = (getWidth() - knob_width - wave_section_width) / 2;
79
80 wave_selector_->setBounds(0, title_width, wave_section_width, wave_selector_width * 2);
81 wave_viewer_->setBounds(0, wave_selector_->getBottom(),
82 wave_section_width, getHeight() - wave_selector_->getBottom());
83
84 shuffle_->setBounds(wave_section_width + space / 2.0f, knob_y,
85 knob_width, knob_width);
86
87 int button_y = size_ratio_ * 82;
88 sub_octave_->setBounds(wave_section_width + space / 2.0f, button_y, knob_width, size_ratio_ * 16);
89
90 phase_offset_->setBounds((getWidth() - (knob_width / 2.0f)),
91 getHeight() - knob_width * .85f,
92 knob_width / 2.0f, knob_width / 2.0f);
93
94 SynthSection::resized();
95}
