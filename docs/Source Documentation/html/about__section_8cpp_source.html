260void AboutSection::buttonClicked(Button* clicked_button) {
261 if (clicked_button == enable_midi_2_.get())
262 {
263 SynthGuiInterface* gui_interface = findParentComponentOfClass<SynthGuiInterface>();
264 if (gui_interface) {
265 bool enable = enable_midi_2_->getToggleState();
266
267 LoadSave::saveMidi2Config(enable);
268
269 if (auto* midi2 = gui_interface->getMidi2Manager()) {
270 midi2->setMidiVersion(
271 enable ? helmboy::Midi2Manager::MidiVersion::MIDI_2_0
272 : helmboy::Midi2Manager::MidiVersion::MIDI_1_0
273 );
274 DBG("MIDI 2.0 " << (enable ? "enabled" : "disabled"));
275 }
276 }
277 }
278 else if (clicked_button == animate_.get())
279 {
280 LoadSave::saveAnimateWidgets(animate_->getToggleState());
281
282 SynthSection* parent = findParentComponentOfClass<SynthSection>();
283 for (SynthSection* s = parent; s; s = parent->findParentComponentOfClass<SynthSection>())
284 parent = s;
285
286 parent->animate(animate_->getToggleState());
287 }
288 else if (clicked_button == size_button_small_.get())
289 setGuiSize(MULT_SMALL);
290 else if (clicked_button == size_button_normal_.get())
291 setGuiSize(1.0);
292 else if (clicked_button == size_button_large_.get())
293 setGuiSize(MULT_LARGE);
294 else if (clicked_button == size_button_extra_large_.get())
295 setGuiSize(MULT_EXTRA_LARGE);
296 else if (clicked_button == size_button_over_large_.get())
297 setGuiSize(MULT_OVER_LARGE);
298 else if (clicked_button == save_midi_profile_.get() || clicked_button == load_midi_profile_.get())
299 {
300 SynthGuiInterface* gui_interface = findParentComponentOfClass<SynthGuiInterface>();
301 SynthBase* synth = gui_interface ? gui_interface->getSynth() : nullptr;
302 MidiManager* midi_manager = synth ? synth->getMidiManager() : nullptr;
303 if (midi_manager == nullptr) {
304 AlertWindow::showMessageBoxAsync(AlertWindow::WarningIcon,
305 TRANS("MIDI profile"),
306 TRANS("Unable to access MIDI mappings."));
307 return;
308 }
309
310 if (clicked_button == save_midi_profile_.get()) {
311 AudioDeviceManager* device_manager = gui_interface ? gui_interface->getAudioDeviceManager() : nullptr;
312 auto chooser = std::make_shared<FileChooser>(TRANS("Save MIDI Profile"), File(), "*.hbMidiProfile");
313 auto chooser_flags = FileBrowserComponent::saveMode |
314 FileBrowserComponent::canSelectFiles |
315 FileBrowserComponent::warnAboutOverwriting;
316 chooser->launchAsync(chooser_flags, [chooser, midi_manager, device_manager](const FileChooser& file_chooser) {
317 File selected = file_chooser.getResult();
318 if (selected == File())
319 return;
320
321 File output_file = selected.withFileExtension(".hbMidiProfile");
322 bool success = LoadSave::saveMidiProfileToFile(output_file, midi_manager, device_manager);
323 AlertWindow::showMessageBoxAsync(success ? AlertWindow::InfoIcon : AlertWindow::WarningIcon,
324 TRANS("MIDI profile"),
325 success ? TRANS("MIDI profile saved.")
326 : TRANS("Failed to save MIDI profile."));
327 });
328 }
329 else {
330 AudioDeviceManager* device_manager = gui_interface ? gui_interface->getAudioDeviceManager() : nullptr;
331 auto chooser = std::make_shared<FileChooser>(TRANS("Load MIDI Profile"), File(), "*.hbMidiProfile");
332 auto chooser_flags = FileBrowserComponent::openMode | FileBrowserComponent::canSelectFiles;
333 chooser->launchAsync(chooser_flags, [gui_interface, chooser, midi_manager, device_manager](const FileChooser& file_chooser) {
334 File selected = file_chooser.getResult();
335 if (selected == File())
336 return;
337
338 bool success = LoadSave::loadMidiProfileFromFile(selected, midi_manager, device_manager);
339 if (success && gui_interface)
340 gui_interface->updateFullGui();
341
342 AlertWindow::showMessageBoxAsync(success ? AlertWindow::InfoIcon : AlertWindow::WarningIcon,
343 TRANS("MIDI profile"),
344 success ? TRANS("MIDI profile loaded.")
345 : TRANS("Failed to load MIDI profile."));
346 });
347 }
348 }
349}
