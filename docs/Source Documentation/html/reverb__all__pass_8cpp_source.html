43 void ReverbAllPass::process() {
44 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
45 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Feedback)));
46
47 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
48 auto audio_buffer = std::span<const mopo_float>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
49 auto feedback_buffer = std::span<const mopo_float>(input(static_cast<int>(Inputs::Feedback))->source->buffer, buffer_size_);
50 const int raw_period = static_cast<int>(input(static_cast<int>(Inputs::SampleDelay))->at(0));
51 int period = std::clamp(raw_period, 1, memory_->getSize() - 1);
52#if defined(HELMBOY_DEBUG_REVERB)
53 if (raw_period != period)
54 std::fprintf(stderr, "[HelmBoy][ReverbAllPass] period clamped raw=%d → %d (max=%d)\n",
55 raw_period, period, memory_->getSize() - 1);
56#endif
57
58 for (int i = 0; i < buffer_size_; ++i)
59 tick(i, dest.data(), period, audio_buffer.data(), feedback_buffer.data());
60 }
