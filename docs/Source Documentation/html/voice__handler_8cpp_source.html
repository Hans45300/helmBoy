155void mopo::VoiceHandler::prepareVoiceTriggers(mopo::Voice* voice) {
156 note_.clearTrigger();
157 last_note_.clearTrigger();
158 note_pressed_.clearTrigger();
159 channel_.clearTrigger();
160 velocity_.clearTrigger();
161 voice_event_.clearTrigger();
162 aftertouch_.clearTrigger();
163 channel_aftertouch_.clearTrigger();
164 channel_.buffer[0] = voice->state().channel;
165
166 if (voice->hasNewEvent()) {
167 voice_event_.trigger(voice->state().event, voice->event_sample());
168 if (voice->state().event == kVoiceOn) {
169 note_.trigger(voice->state().note, 0);
170 last_note_.trigger(voice->state().last_note, 0);
171 velocity_.trigger(voice->state().velocity, 0);
172 note_pressed_.trigger(voice->state().note_pressed, 0);
173 channel_.trigger(voice->state().channel, 0);
174 }
175 }
176
177 if (voice->hasNewAftertouch())
178 aftertouch_.trigger(voice->aftertouch(), voice->aftertouch_sample());
179 if (voice->hasNewChannelAftertouch())
180 channel_aftertouch_.trigger(voice->channelAftertouch(), voice->channelAftertouchSample());
181
182 voice->clearEvents();
183}

227void mopo::VoiceHandler::process() {
228 global_router_.process();
229
230 int num_voices = static_cast<int>(active_voices_.size());
231 if (num_voices == 0) {
232 if (last_num_voices_) {
233 clearNonaccumulatedOutputs();
234 clearAccumulatedOutputs();
235 }
236 last_num_voices_ = num_voices;
237 return;
238 }
239
240 int polyphony = static_cast<int>(input(static_cast<int>(Inputs::Polyphony))->at(0));
241 setPolyphony(utils::iclamp(polyphony, 1, polyphony));
242 clearAccumulatedOutputs();
243
244 auto iter = active_voices_.begin();
245 while (iter != active_voices_.end()) {
246 mopo::Voice* voice = *iter;
247 prepareVoiceTriggers(voice);
248 processVoice(voice);
249 accumulateOutputs();
250
251 // Remove voice if the right processor has a full silent buffer.
252 if (voice_killer_ && voice->state().event != mopo::kVoiceOn &&
253 utils::isSilent(std::span<const mopo_float>(voice_killer_->buffer, buffer_size_))) {
254 free_voices_.push_back(voice);
255 iter = active_voices_.erase(iter);
256 } else {
257 ++iter;
258 }
259 }
260
261 if (active_voices_.size())
262 writeNonaccumulatedOutputs();
263
264 last_num_voices_ = num_voices;
265}

313 mopo::Voice* mopo::VoiceHandler::grabVoice() {
314 mopo::Voice* voice = nullptr;
315 // First check free voices.
316 if (free_voices_.size() &&
317 (!legato_ || pressed_notes_.size() < polyphony_ || active_voices_.size() < polyphony_)) {
318 voice = free_voices_.front();
319 free_voices_.pop_front();
320 return voice;
321 }
322 // Next check released voices.
323 for (auto iter = active_voices_.begin(); iter != active_voices_.end(); ++iter) {
324 voice = *iter;
325 if (voice->key_state() == mopo::Voice::KeyState::Released) {
326 active_voices_.erase(iter);
327 return voice;
328 }
329 }
330 // Then check sustained voices.
331 for (auto iter = active_voices_.begin(); iter != active_voices_.end(); ++iter) {
332 voice = *iter;
333 if (voice->key_state() == mopo::Voice::KeyState::Sustained) {
334 active_voices_.erase(iter);
335 return voice;
336 }
337 }
338 // If all are active just grab the oldest voice.
339 MOPO_ASSERT(active_voices_.size());
340 voice = active_voices_.front();
341 active_voices_.pop_front();
342 return voice;
343}

345mopo::Voice* mopo::VoiceHandler::getVoiceToKill() {
346 int excess_voices = static_cast<int>(active_voices_.size()) - static_cast<int>(polyphony_);
347 mopo::Voice* oldest_released = nullptr;
348 mopo::Voice* oldest_sustained = nullptr;
349 mopo::Voice* oldest_held = nullptr;
350 for (auto iter = active_voices_.begin(); iter != active_voices_.end(); ++iter) {
351 mopo::Voice* voice = *iter;
352 if (voice->state().event == mopo::kVoiceKill)
353 excess_voices--;
354 else if (oldest_released == nullptr && voice->key_state() == mopo::Voice::KeyState::Released)
355 oldest_released = voice;
356 else if (oldest_sustained == nullptr && voice->key_state() == mopo::Voice::KeyState::Sustained)
357 oldest_sustained = voice;
358 else if (oldest_held == nullptr)
359 oldest_held = voice;
360 }
361 // Return null if we've killed enough voices.
362 if (excess_voices <= 0)
363 return nullptr;
364 // If there were any released notes kill the oldest.
365 if (oldest_released)
366 return oldest_released;
367 // Then if there were any sustained notes kill the oldest.
368 if (oldest_sustained)
369 return oldest_sustained;
370 // If all are active just grab the oldest held voice.
371 if (oldest_held)
372 return oldest_held;
373 return nullptr;
374}

376void mopo::VoiceHandler::noteOn(mopo::mopo_float note, mopo::mopo_float velocity, int sample, int channel) {
377 MOPO_ASSERT(sample >= 0 && sample < buffer_size_);
378 MOPO_ASSERT(channel >= 0 && channel < mopo::NUM_MIDI_CHANNELS);
379 mopo::Voice* voice = grabVoice();
380 pressed_notes_.remove(note);
381 pressed_notes_.push_front(note);
382 if (last_played_note_ < 0)
383 last_played_note_ = note;
384 voice->activate(note, velocity, last_played_note_, static_cast<int>(pressed_notes_.size()), sample, channel);
385 active_voices_.push_back(voice);
386 last_played_note_ = note;
387}

389mopo::VoiceEvent mopo::VoiceHandler::noteOff(mopo::mopo_float note, int sample) {
390 pressed_notes_.removeAll(note);
391 mopo::VoiceEvent voice_event = mopo::kVoiceOff;
392 for (mopo::Voice* voice : active_voices_) {
393 if (voice->state().note == note) {
394 if (sustain_)
395 voice->sustain();
396 else {
397 if (polyphony_ <= pressed_notes_.size() && voice->state().event != mopo::kVoiceKill) {
398 voice->kill();
399 mopo::Voice* new_voice = grabVoice();
400 active_voices_.push_back(new_voice);
401 mopo::mopo_float old_note = pressed_notes_.back();
402 pressed_notes_.pop_back();
403 pressed_notes_.push_front(old_note);
404 new_voice->activate(old_note, voice->state().velocity, last_played_note_,
405 static_cast<int>(pressed_notes_.size()) + 1, sample);
406 last_played_note_ = old_note;
407 voice_event = mopo::kVoiceReset;
408 } else {
409 voice->deactivate(sample);
410 }
411 }
412 }
413 }
414 return voice_event;
415}

433void mopo::VoiceHandler::setPolyphony(size_t polyphony) {
434 while (all_voices_.size() < polyphony) {
435 mopo::Voice* new_voice = createVoice();
436 all_voices_.push_back(new_voice);
437 active_voices_.push_back(new_voice);
438 }
439 int num_voices_to_kill = static_cast<int>(active_voices_.size()) - static_cast<int>(polyphony);
440 for (int i = 0; i < num_voices_to_kill; ++i) {
441 mopo::Voice* sacrifice = getVoiceToKill();
442 if (sacrifice)
443 sacrifice->kill();
444 }
445 polyphony_ = polyphony;
446}
