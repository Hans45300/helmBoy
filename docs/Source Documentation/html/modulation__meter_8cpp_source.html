34ModulationMeter::ModulationMeter(SynthBase* synth,
35 const mopo::Output* mono_total,
36 const mopo::Output* poly_total,
37 const SynthSlider* slider) :
38 synth_(synth), mono_total_(mono_total), poly_total_(poly_total),
39 destination_(slider), current_value_(0.0), knob_percent_(0.0), mod_percent_(0.0),
40 knob_stroke_(0.0f, PathStrokeType::beveled, PathStrokeType::butt),
41 full_radius_(0.0), outer_radius_(0.0) {
42 setInterceptsMouseClicks(false, false);
43 telemetry_indices_[0] = synth_->getVisualTelemetryIndex(mono_total_);
44 telemetry_indices_[1] = synth_->getVisualTelemetryIndex(poly_total_);
45 telemetry_count_ = telemetry_indices_[1] >= 0 ? 2 : telemetry_indices_[0] >= 0 ? 1 : 0;
46 updateValue();
47 updateDrawing();
48}

72void ModulationMeter::resized() {
73 static const float stroke_percent = 0.12f;
74
75 full_radius_ = std::min(getWidth() / 2.0f, getHeight() / 2.0f);
76 float stroke_width = 2.0f * full_radius_ * stroke_percent;
77 knob_stroke_ = PathStrokeType(stroke_width, PathStrokeType::beveled, PathStrokeType::butt);
78 outer_radius_ = full_radius_ - stroke_width;
79
80 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
81 if (parent) {
82 std::vector<mopo::ModulationConnection*> connections;
83 connections = parent->getSynth()->getSourceConnections(getName().toStdString());
84 setModulated(connections.size());
85 }
86}

88void ModulationMeter::updateValue() {
89 std::array<float, 2> values {};
90 if (synth_ == nullptr || telemetry_count_ == 0 ||
91 !synth_->copyVisualTelemetry(
92 std::span<const int>(telemetry_indices_.data(), telemetry_count_),
93 std::span<float>(values.data(), telemetry_count_)))
94 return;
95 current_value_ = values[0] + (telemetry_count_ > 1 ? values[1] : 0.0f);
96}

98bool ModulationMeter::copyModulatedValue(float& value) const {
99 std::array<float, 2> values {};
100 if (!modulated_ || synth_ == nullptr || telemetry_count_ == 0 ||
101 !synth_->copyVisualTelemetry(
102 std::span<const int>(telemetry_indices_.data(), telemetry_count_),
103 std::span<float>(values.data(), telemetry_count_)))
104 return false;
105 value = values[0] + (telemetry_count_ > 1 ? values[1] : 0.0f);
106 return std::isfinite(value);
107}

109void ModulationMeter::updateDrawing() {
110 double range = destination_->getMaximum() - destination_->getMinimum();
111 double value = (current_value_ - destination_->getMinimum()) / range;
112 double new_mod_percent = mopo::utils::clamp(value , 0.0, 1.0);
113 double new_knob_percent = (destination_->getValue() - destination_->getMinimum()) / range;
114
115 if (new_mod_percent != mod_percent_ || new_knob_percent != knob_percent_) {
116 mod_percent_ = new_mod_percent;
117 knob_percent_ = new_knob_percent;
118 repaint();
119 }
120}

122void ModulationMeter::drawTextSlider(Graphics& g) {
123 g.setColour(getMeterColour());
124
125 float diff_percent = mod_percent_ - knob_percent_;
126
127 if (diff_percent < 0.0f) {
128 float mod_position = -getHeight() * diff_percent;
129 fillVerticalRect(g, 0.0f, mod_position, getWidth() / 2.0f);
130 }
131 else if (diff_percent > 0.0f) {
132 float mod_position = getHeight() * (1.0f - diff_percent);
133 fillVerticalRect(g, mod_position, getHeight(), getWidth() / 2.0f);
134 }
135}

137void ModulationMeter::drawSlider(Graphics& g) {
138 g.setColour(getMeterColour());
139
140 if (destination_->getSliderStyle() == Slider::LinearBar) {
141 if (const auto* selector = dynamic_cast<const WaveSelector*>(destination_)) {
142 float effective_value = static_cast<float>(destination_->getValue());
143 if (modulated_ && !copyModulatedValue(effective_value))
144 return;
145
146 const int waveform_index = selector->getWaveformIndex(effective_value);
147 const auto selector_bounds = selector->getCellBounds(waveform_index);
148 const auto top_left = getLocalPoint(selector, selector_bounds.getTopLeft());
149 const auto bottom_right = getLocalPoint(selector, selector_bounds.getBottomRight());
150 const Rectangle<float> meter_bounds(
151 top_left.x, top_left.y,
152 bottom_right.x - top_left.x, bottom_right.y - top_left.y);
153 g.setColour(Colour(Colors::Color_aaffffff));
154 g.drawRect(meter_bounds, 1.0f);
155 return;
156 }
157
158 float knob_position = getWidth() * knob_percent_;
159 float mod_position = getWidth() * mod_percent_;
160
161 if (destination_->getInterval() == 1.0 && destination_->getMinimum() == 0.0) {
162 int index = mod_percent_ * destination_->getMaximum() + 0.5;
163 float width = getWidth() / (destination_->getMaximum() + 1.0);
164
165 g.setColour(Colour(Colors::Color_aaffffff));
166 g.drawRect(index * width, 0.0f, width, float(getHeight()), 1.0f);
167 }
168 else
169 fillHorizontalRect(g, knob_position, mod_position, getHeight() / 2.0f);
170 }
171 else {
172 float mod_position = getHeight() * (1.0f - mod_percent_);
173 float knob_position = getHeight() * (1.0f - knob_percent_);
174
175 fillVerticalRect(g, mod_position, knob_position, getWidth() / 2.0f);
176 }
177}

179void ModulationMeter::drawKnob(Graphics& g) {
180 float current_mod_angle = mopo::utils::interpolate(-ANGLE, ANGLE, mod_percent_);
181 float current_knob_angle = mopo::utils::interpolate(-ANGLE, ANGLE, knob_percent_);
182
183 if (current_mod_angle != current_knob_angle) {
184 Path mod_section;
185 mod_section.addCentredArc(full_radius_, full_radius_, outer_radius_, outer_radius_,
186 0.0f, current_mod_angle, current_knob_angle, true);
187 g.setColour(getMeterColour());
188 g.strokePath(mod_section, knob_stroke_);
189 }
190}
