58void ModulationMeter::resized() {
59 static const float stroke_percent = 0.12f;
60
61 full_radius_ = std::min(getWidth() / 2.0f, getHeight() / 2.0f);
62 float stroke_width = 2.0f * full_radius_ * stroke_percent;
63 knob_stroke_ = PathStrokeType(stroke_width, PathStrokeType::beveled, PathStrokeType::butt);
64 outer_radius_ = full_radius_ - stroke_width;
65
66 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
67 if (parent) {
68 std::vector<mopo::ModulationConnection*> connections;
69 connections = parent->getSynth()->getSourceConnections(getName().toStdString());
70 setModulated(connections.size());
71 }
72}

82void ModulationMeter::updateDrawing() {
83 double range = destination_->getMaximum() - destination_->getMinimum();
84 double value = (current_value_ - destination_->getMinimum()) / range;
85 double new_mod_percent = mopo::utils::clamp(value , 0.0, 1.0);
86 double new_knob_percent = (destination_->getValue() - destination_->getMinimum()) / range;
87
88 if (new_mod_percent != mod_percent_ || new_knob_percent != knob_percent_) {
89 mod_percent_ = new_mod_percent;
90 knob_percent_ = new_knob_percent;
91 repaint();
92 }
93}

95void ModulationMeter::drawTextSlider(Graphics& g) {
96 g.setColour(Colour(SLIDER_MOD_COLOR));
97
98 float diff_percent = mod_percent_ - knob_percent_;
99
100 if (diff_percent < 0.0f) {
101 float mod_position = -getHeight() * diff_percent;
102 fillVerticalRect(g, 0.0f, mod_position, getWidth() / 2.0f);
103 }
104 else if (diff_percent > 0.0f) {
105 float mod_position = getHeight() * (1.0f - diff_percent);
106 fillVerticalRect(g, mod_position, getHeight(), getWidth() / 2.0f);
107 }
108}

110void ModulationMeter::drawSlider(Graphics& g) {
111 g.setColour(Colour(SLIDER_MOD_COLOR));
112
113 if (destination_->getSliderStyle() == Slider::LinearBar) {
114 float knob_position = getWidth() * knob_percent_;
115 float mod_position = getWidth() * mod_percent_;
116
117 if (destination_->getInterval() == 1.0 && destination_->getMinimum() == 0.0) {
118 int index = mod_percent_ * destination_->getMaximum() + 0.5;
119 float width = getWidth() / (destination_->getMaximum() + 1.0);
120
121 g.setColour(Colour(Colors::Color_aaffffff));
122 g.drawRect(index * width, 0.0f, width, float(getHeight()), 1.0f);
123 }
124 else
125 fillHorizontalRect(g, knob_position, mod_position, getHeight() / 2.0f);
126 }
127 else {
128 float mod_position = getHeight() * (1.0f - mod_percent_);
129 float knob_position = getHeight() * (1.0f - knob_percent_);
130
131 fillVerticalRect(g, mod_position, knob_position, getWidth() / 2.0f);
132 }
133}

135void ModulationMeter::drawKnob(Graphics& g) {
136 float current_mod_angle = mopo::utils::interpolate(-ANGLE, ANGLE, mod_percent_);
137 float current_knob_angle = mopo::utils::interpolate(-ANGLE, ANGLE, knob_percent_);
138
139 if (current_mod_angle != current_knob_angle) {
140 Path mod_section;
141 mod_section.addCentredArc(full_radius_, full_radius_, outer_radius_, outer_radius_,
142 0.0f, current_mod_angle, current_knob_angle, true);
143 g.setColour(Colors::modulation);
144 g.strokePath(mod_section, knob_stroke_);
145 }
146}
