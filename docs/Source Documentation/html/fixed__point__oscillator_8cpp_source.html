35 void FixedPointOscillator::process() {
36 auto amplitude = std::span<const mopo_float>(input(kAmplitude)->source->buffer, buffer_size_);
37 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
38
39 int phase_inc = UINT_MAX * input(kPhaseInc)->at(0);
40 if (input(kLowOctave)->at(0))
41 phase_inc /= 2.0;
42
43 if (amplitude[0] == 0.0 && amplitude[amplitude.size() - 1] == 0.0) {
44 phase_ += phase_inc * buffer_size_;
45 utils::zeroBuffer(dest);
46 return;
47 }
48
49 mopo_float shuffle = utils::clamp(1.0 - input(kShuffle)->source->buffer[0], 0.0, 1.0);
50 mopo_float stretch = input(kPhaseStretch)->source->buffer[0];
51 const unsigned int phase_offset =
52 utils::phaseToFixedPoint(input(kPhaseOffset)->at(0));
53 unsigned int shuffle_index = INT_MAX * shuffle;
54
55 int waveform = utils::roundAndClampWaveformIndex(
56 input(kWaveform)->source->buffer[0], 0, FixedPointWaveLookup::kWhiteNoise - 1);
57 mopo_float* wave_buffer = FixedPointWave::getBuffer(waveform, 2.0 * phase_inc);
58
59 mopo_float first_adjust = bool(shuffle) * 2.0 / shuffle;
60 mopo_float second_adjust = 1.0 / (1.0 - 0.5 * shuffle);
61
62 if (input(kReset)->source->triggered)
63 phase_ = 0;
64
65 unsigned int i = 0;
66 unsigned int buffer_size = buffer_size_;
67 unsigned int current_phase = 0;
68 while (i < buffer_size) {
69 if (phase_ < shuffle_index) {
70 unsigned int max_samples = (shuffle_index - phase_) / phase_inc + 1;
71 unsigned int samples = std::min(buffer_size, i + max_samples);
72 std::ranges::for_each(std::views::iota(static_cast<unsigned int>(i), static_cast<unsigned int>(samples)), [&](unsigned int idx) {
73 phase_ += phase_inc;
74 current_phase = phase_ * first_adjust;
75 current_phase += phase_offset;
76 unsigned int warped_phase = remapPhase(current_phase, stretch);
77 mopo_float wave_read = FixedPointWave::interpretWave(wave_buffer, warped_phase);
78 dest[idx] = amplitude[idx] * wave_read;
79 });
80 i = samples;
81 }
82
83 unsigned int max_samples = -phase_ / phase_inc + 1;
84 unsigned int samples = std::min(buffer_size, i + max_samples);
85 std::ranges::for_each(std::views::iota(static_cast<unsigned int>(i), static_cast<unsigned int>(samples)), [&](unsigned int idx) {
86 phase_ += phase_inc;
87 current_phase = (phase_ - shuffle_index) * second_adjust;
88 current_phase += phase_offset;
89 unsigned int warped_phase = remapPhase(current_phase, stretch);
90 mopo_float wave_read = FixedPointWave::interpretWave(wave_buffer, warped_phase);
91 dest[idx] = amplitude[idx] * wave_read;
92 });
93 i = samples;
94 }
95 }
