67void FilterResponse::paint(Graphics& g) {
68 static const DropShadow shadow(Colour(Colors::shadow_dark), 5, Point<int>(0, 0));
69
70 g.drawImage(background_,
71 0, 0, getWidth(), getHeight(),
72 0, 0, background_.getWidth(), background_.getHeight());
73
74 shadow.drawForPath(g, filter_response_path_);
75
76 g.setColour(Colors::graph_fill);
77 g.fillPath(filter_response_path_);
78
79 if (active_)
80 g.setColour(Colors::audio);
81 else
82 g.setColour(Colors::graph_disable);
83
84 float line_width = 1.5f * getHeight() / 150.0f;
85 PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
86 g.strokePath(filter_response_path_, stroke);
87}

113float FilterResponse::getPercentForMidiNote(float midi_note) {
114 float frequency = mopo::utils::midiNoteToFrequency(midi_note);
115
116 float response = 0.0f;
117 if (style_ == mopo::StateVariableFilter::Styles::Shelf)
118 response = fabs(filter_shelf_.getAmplitudeResponse(frequency));
119 else {
120 float blend = filter_blend_slider_->getValue();
121 float low_pass_amount = mopo::utils::clamp(1.0 - blend, 0.0, 1.0);
122 float band_pass_amount = mopo::utils::clamp(1.0 - fabs(blend - 1.0), 0.0, 1.0);
123 float high_pass_amount = mopo::utils::clamp(blend - 1.0, 0.0, 1.0);
124
125 response = low_pass_amount * filter_low_.getAmplitudeResponse(frequency) +
126 band_pass_amount * filter_band_.getAmplitudeResponse(frequency) +
127 high_pass_amount * filter_high_.getAmplitudeResponse(frequency);
128
129 response = fabs(response);
130 if (style_ == mopo::StateVariableFilter::Styles::dB24)
131 response *= response;
132 }
133
134 float gain_db = mopo::utils::gainToDb(response);
135 return (gain_db - MIN_GAIN_DB) / (MAX_GAIN_DB - MIN_GAIN_DB);
136}

138void FilterResponse::resetResponsePath() {
139 static const int wrap_size = 10;
140
141 if (cutoff_slider_ == nullptr)
142 return;
143
144 filter_response_path_.clear();
145 filter_response_path_.startNewSubPath(-wrap_size, getHeight() + wrap_size);
146 float start_percent = getPercentForMidiNote(0.0);
147 float last_y = getHeight() * (1.0f - start_percent);
148 float last_slope = 0.0f;
149 float last_x = 0.0f;
150
151 filter_response_path_.lineTo(-wrap_size, last_y);
152
153 for (int i = 0; i < resolution_; ++i) {
154 float t = (1.0f * i) / (resolution_ - 1);
155 float midi_note = cutoff_slider_->proportionOfLengthToValue(t);
156 float percent = getPercentForMidiNote(midi_note);
157
158 float new_x = getWidth() * t;
159 float new_y = getHeight() * (1.0f - percent);
160 float new_slope = (new_y - last_y) / (new_x - last_x);
161 if (fabs(last_slope - new_slope) > DELTA_SLOPE_REDRAW_THRESHOLD ||
162 new_x - last_x > X_REDRAW_THRESHOLD) {
163 last_x = new_x;
164 last_y = new_y;
165 last_slope = new_slope;
166 filter_response_path_.lineTo(new_x, new_y);
167 }
168 }
169
170 float end_percent = getPercentForMidiNote(cutoff_slider_->getMaximum());
171
172 filter_response_path_.lineTo(getWidth() + wrap_size, getHeight() * (1.0f - end_percent));
173 filter_response_path_.lineTo(getWidth() + wrap_size, getHeight() + wrap_size);
174}

176void FilterResponse::computeFilterCoefficients() {
177 if (cutoff_slider_ == nullptr || resonance_slider_ == nullptr ||
178 filter_blend_slider_ == nullptr || filter_shelf_slider_ == nullptr) {
179 return;
180 }
181
182 mopo::StateVariableFilter::Shelves shelf = static_cast<mopo::StateVariableFilter::Shelves>(
183 static_cast<int>(filter_shelf_slider_->getValue()));
184 double frequency = mopo::utils::midiNoteToFrequency(cutoff_slider_->getValue());
185 double resonance = mopo::utils::magnitudeToQ(resonance_slider_->getValue());
186 double decibels = mopo::utils::interpolate(MIN_GAIN_DB, MAX_GAIN_DB,
187 resonance_slider_->getValue());
188 double gain = mopo::utils::dbToGain(decibels);
189
190 if (style_ == mopo::StateVariableFilter::Styles::dB24) {
191 resonance = sqrt(resonance);
192 gain = sqrt(gain);
193 }
194
195 if (style_ == mopo::StateVariableFilter::Styles::Shelf) {
196 mopo::BiquadFilter::Type type = mopo::BiquadFilter::Type::LowShelf;
197 if (shelf == mopo::StateVariableFilter::Shelves::BandShelf)
198 type = mopo::BiquadFilter::Type::BandShelf;
199 else if (shelf == mopo::StateVariableFilter::Shelves::HighShelf)
200 type = mopo::BiquadFilter::Type::HighShelf;
201 filter_shelf_.computeCoefficients(type, frequency, 1.0, gain);
202 }
203 else {
204 filter_low_.computeCoefficients(mopo::BiquadFilter::Type::LowPass, frequency, resonance, 1.0);
205 filter_band_.computeCoefficients(mopo::BiquadFilter::Type::BandPass, frequency, resonance, 1.0);
206 filter_high_.computeCoefficients(mopo::BiquadFilter::Type::HighPass, frequency, resonance, 1.0);
207 }
208 resetResponsePath();
209}

211void FilterResponse::setFilterSettingsFromPosition(Point<int> position) {
212 if (cutoff_slider_) {
213 double percent = mopo::utils::clamp((1.0 * position.x) / getWidth(), 0.0, 1.0);
214 double frequency = cutoff_slider_->proportionOfLengthToValue(percent);
215 cutoff_slider_->setValue(frequency);
216 }
217 if (resonance_slider_) {
218 double percent = mopo::utils::clamp(1.0 - (1.0 * position.y) / getHeight(), 0.0, 1.0);
219 resonance_slider_->setValue(resonance_slider_->proportionOfLengthToValue(percent));
220 }
221
222 computeFilterCoefficients();
223}
