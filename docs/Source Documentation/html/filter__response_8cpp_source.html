74void FilterResponse::paint(Graphics& g) {
75 static const DropShadow shadow(Colour(Colors::shadow_dark), 5, Point<int>(0, 0));
76
77 g.drawImage(background_,
78 0, 0, getWidth(), getHeight(),
79 0, 0, background_.getWidth(), background_.getHeight());
80
81 shadow.drawForPath(g, base_filter_response_path_);
82
83 g.setColour(Colors::graph_fill);
84 g.fillPath(base_filter_response_path_);
85
86 if (active_)
87 g.setColour(Colors::modulation);
88 else
89 g.setColour(Colors::graph_disable);
90
91 float line_width = 1.0f * getHeight() / 150.0f;
92 PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
93 g.strokePath(base_filter_response_path_, stroke);
94
95 if (effective_shape_differs_) {
96 g.setColour(Colors::modulation.withAlpha(0.5f));
97 g.fillPath(filter_response_path_);
98 g.setColour(Colors::modulation);
99 g.strokePath(filter_response_path_, stroke);
100 }

125float FilterResponse::getPercentForMidiNote(float midi_note) {
126 float frequency = mopo::utils::midiNoteToFrequency(midi_note);
127
128 float response = 0.0f;
129 if (style_ == mopo::StateVariableFilter::Styles::Shelf)
130 response = fabs(filter_shelf_.getAmplitudeResponse(frequency));
131 else if (style_ == mopo::StateVariableFilter::Styles::Notch)
132 response = fabs(filter_notch_.getAmplitudeResponse(frequency));
133 else if (style_ == mopo::StateVariableFilter::Styles::Comb) {
134 const float phase = 2.0f * static_cast<float>(mopo::PI) * frequency * comb_period_seconds_;
135 const float denominator = std::sqrt(1.0f + comb_feedback_ * comb_feedback_ -
136 2.0f * comb_feedback_ * std::cos(phase));
137 response = 1.0f / denominator;
138 }
139 else {
140 float blend = filter_blend_slider_->getValue();
141 float low_pass_amount = mopo::utils::clamp(1.0 - blend, 0.0, 1.0);
142 float band_pass_amount = mopo::utils::clamp(1.0 - fabs(blend - 1.0), 0.0, 1.0);
143 float high_pass_amount = mopo::utils::clamp(blend - 1.0, 0.0, 1.0);
144
145 response = low_pass_amount * filter_low_.getAmplitudeResponse(frequency) +
146 band_pass_amount * filter_band_.getAmplitudeResponse(frequency) +
147 high_pass_amount * filter_high_.getAmplitudeResponse(frequency);
148
149 response = fabs(response);
150 if (style_ == mopo::StateVariableFilter::Styles::dB24)
151 response *= response;
152 }
153
154 float gain_db = mopo::utils::gainToDb(response);
155 return (gain_db - MIN_GAIN_DB) / (MAX_GAIN_DB - MIN_GAIN_DB);
156}

158void FilterResponse::resetResponsePath() {
159 if (cutoff_slider_ == nullptr || resonance_slider_ == nullptr ||
160 filter_blend_slider_ == nullptr || filter_shelf_slider_ == nullptr)
161 return;
162
163 const double base_cutoff = cutoff_slider_->getValue();
164 const double base_resonance = resonance_slider_->getValue();
165 const double effective_cutoff = displayed_parameters_valid_
166 ? displayed_cutoff_ : base_cutoff;
167 const double effective_resonance = displayed_parameters_valid_
168 ? displayed_resonance_ : base_resonance;
169
170 setFilterCoefficients(effective_cutoff, effective_resonance);
171 filter_response_path_ = createResponsePath();
172
173 const double cutoff_change = std::abs(
174 cutoff_slider_->valueToProportionOfLength(effective_cutoff) -
175 cutoff_slider_->valueToProportionOfLength(base_cutoff));
176 const double resonance_change = std::abs(
177 resonance_slider_->valueToProportionOfLength(effective_resonance) -
178 resonance_slider_->valueToProportionOfLength(base_resonance));
179 effective_shape_differs_ = cutoff_change >= 0.0005 || resonance_change >= 0.0005;
180
181 setFilterCoefficients(base_cutoff, base_resonance);
182 base_filter_response_path_ = createResponsePath();
183 setFilterCoefficients(effective_cutoff, effective_resonance);
184}

186Path FilterResponse::createResponsePath() {
187 static const int wrap_size = 10;
188 Path response_path;
189 response_path.startNewSubPath(-wrap_size, getHeight() + wrap_size);
190 float start_percent = getPercentForMidiNote(0.0);
191 float last_y = getHeight() * (1.0f - start_percent);
192 float last_slope = 0.0f;
193 float last_x = 0.0f;
194
195 response_path.lineTo(-wrap_size, last_y);
196
197 for (int i = 0; i < resolution_; ++i) {
198 float t = (1.0f * i) / (resolution_ - 1);
199 float midi_note = cutoff_slider_->proportionOfLengthToValue(t);
200 float percent = getPercentForMidiNote(midi_note);
201
202 float new_x = getWidth() * t;
203 float new_y = getHeight() * (1.0f - percent);
204 float new_slope = (new_y - last_y) / (new_x - last_x);
205 if (fabs(last_slope - new_slope) > DELTA_SLOPE_REDRAW_THRESHOLD ||
206 new_x - last_x > X_REDRAW_THRESHOLD) {
207 last_x = new_x;
208 last_y = new_y;
209 last_slope = new_slope;
210 response_path.lineTo(new_x, new_y);
211 }
212 }
213
214 float end_percent = getPercentForMidiNote(cutoff_slider_->getMaximum());
215
216 response_path.lineTo(getWidth() + wrap_size, getHeight() * (1.0f - end_percent));
217 response_path.lineTo(getWidth() + wrap_size, getHeight() + wrap_size);
218 return response_path;
219}

242void FilterResponse::setFilterCoefficients(double cutoff, double resonance_value) {
243 if (cutoff_slider_ == nullptr || resonance_slider_ == nullptr ||
244 filter_blend_slider_ == nullptr || filter_shelf_slider_ == nullptr)
245 return;
246
247 mopo::StateVariableFilter::Shelves shelf = static_cast<mopo::StateVariableFilter::Shelves>(
248 static_cast<int>(filter_shelf_slider_->getValue()));
249 double frequency = mopo::utils::midiNoteToFrequency(cutoff);
250 double resonance = mopo::utils::magnitudeToQ(resonance_value);
251 double decibels = mopo::utils::interpolate(MIN_GAIN_DB, MAX_GAIN_DB,
252 resonance_value);
253 double gain = mopo::utils::dbToGain(decibels);
254
255 if (style_ == mopo::StateVariableFilter::Styles::dB24) {
256 resonance = sqrt(resonance);
257 gain = sqrt(gain);
258 }
259
260 const double cutoff_frequency = mopo::utils::midiNoteToFrequency(cutoff);
261 comb_period_seconds_ = static_cast<float>(mopo::utils::clamp(1.0 / cutoff_frequency,
262 0.00001, 0.1));
263 comb_feedback_ = static_cast<float>(mopo::utils::clamp(
264 0.95 * resonance_value, 0.0, 0.95));
265
266 if (style_ == mopo::StateVariableFilter::Styles::Shelf) {
267 mopo::BiquadFilter::Type type = mopo::BiquadFilter::Type::LowShelf;
268 if (shelf == mopo::StateVariableFilter::Shelves::BandShelf)
269 type = mopo::BiquadFilter::Type::BandShelf;
270 else if (shelf == mopo::StateVariableFilter::Shelves::HighShelf)
271 type = mopo::BiquadFilter::Type::HighShelf;
272 filter_shelf_.computeCoefficients(type, frequency, 1.0, gain);
273 }
274 else if (style_ == mopo::StateVariableFilter::Styles::Notch) {
275 filter_notch_.computeCoefficients(mopo::BiquadFilter::Type::Notch,
276 frequency, resonance, 1.0);
277 }
278 else if (style_ != mopo::StateVariableFilter::Styles::Comb) {
279 filter_low_.computeCoefficients(mopo::BiquadFilter::Type::LowPass, frequency, resonance, 1.0);
280 filter_band_.computeCoefficients(mopo::BiquadFilter::Type::BandPass, frequency, resonance, 1.0);
281 filter_high_.computeCoefficients(mopo::BiquadFilter::Type::HighPass, frequency, resonance, 1.0);
282 }
283}

285void FilterResponse::timerCallback() {
286 if (!isShowing() || cutoff_slider_ == nullptr || resonance_slider_ == nullptr)
287 return;
288
289 const double base_cutoff = cutoff_slider_->getValue();
290 const double base_resonance = resonance_slider_->getValue();
291 double effective_cutoff = base_cutoff;
292 double effective_resonance = base_resonance;
293
294 FullInterface* interface = findParentComponentOfClass<FullInterface>();
295 if (interface != nullptr && interface->isAnimating()) {
296 const auto readEffectiveValue = [interface](
297 const char* name, SynthSlider* slider, double previous) {
298 if (!interface->isDestinationModulated(name))
299 return static_cast<double>(slider->getValue());
300
301 float value = static_cast<float>(previous);
302 if (!interface->copyModulatedValue(name, value) || !std::isfinite(value))
303 return previous;
304 return jlimit(slider->getMinimum(), slider->getMaximum(),
305 static_cast<double>(value));
306 };
307
308 effective_cutoff = readEffectiveValue(
309 "cutoff", cutoff_slider_, displayed_parameters_valid_ ? displayed_cutoff_ : base_cutoff);
310 effective_resonance = readEffectiveValue(
311 "resonance", resonance_slider_,
312 displayed_parameters_valid_ ? displayed_resonance_ : base_resonance);
313 }
314
315 const double cutoff_change = std::abs(
316 cutoff_slider_->valueToProportionOfLength(effective_cutoff) -
317 cutoff_slider_->valueToProportionOfLength(
318 displayed_parameters_valid_ ? displayed_cutoff_ : base_cutoff));
319 const double resonance_change = std::abs(
320 resonance_slider_->valueToProportionOfLength(effective_resonance) -
321 resonance_slider_->valueToProportionOfLength(
322 displayed_parameters_valid_ ? displayed_resonance_ : base_resonance));
323 if (!displayed_parameters_valid_ || cutoff_change >= 0.0005 ||
324 resonance_change >= 0.0005) {
325 computeFilterCoefficients(effective_cutoff, effective_resonance);
326 repaint();
327 }
328}

330void FilterResponse::setFilterSettingsFromPosition(Point<int> position) {
