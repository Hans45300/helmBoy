37template<typename T>
38concept AudioBufferLike = requires(T a) {
39 typename T::SampleType;
40 { a.getArrayOfWritePointers() } -> std::same_as<typename T::SampleType* const*>;
41 { a.getNumChannels() } -> std::convertible_to<int>;
42 { a.getNumSamples() } -> std::convertible_to<int>;
43};

85 SafeAudioBuffer(SampleType* const* data, int num_channels, int num_samples)
86 : num_channels_(num_channels), num_samples_(num_samples) {
87
88 assert(data != nullptr);
89 assert(num_channels > 0);
90 assert(num_samples > 0);
91
92 for (int channel = 0; channel < num_channels; ++channel) {
93 assert(data[channel] != nullptr);
94 channels_.emplace_back(data[channel], num_samples);
95 }
96 }

140 void addFrom(const SafeAudioBuffer& other, int start_sample = 0, int num_samples_to_process = -1) {
141 if (num_samples_to_process == -1) {
142 num_samples_to_process = std::min(num_samples_ - start_sample,
143 other.num_samples_ - start_sample);
144 }
145
146 assert(start_sample >= 0);
147 assert(start_sample + num_samples_to_process <= num_samples_);
148 assert(start_sample + num_samples_to_process <= other.num_samples_);
149 assert(num_channels_ == other.num_channels_);
150
151 for (int channel = 0; channel < num_channels_; ++channel) {
152 auto dest = channels_[channel].subspan(start_sample, num_samples_to_process);
153 auto src = other.channels_[channel].subspan(start_sample, num_samples_to_process);
154
155 // Use SIMD-friendly transform with vectorization hints
156 if constexpr (std::is_same_v<SampleType, float>) {
157 // Optimized path for float (most common case)
158 #ifdef _MSC_VER
159 __pragma(loop(ivdep))
160 #endif
161 for (size_t i = 0; i < dest.size(); ++i) {
162 dest[i] += src[i];
163 }
164 } else {
165 // Generic fallback
166 std::ranges::transform(src, dest, dest.begin(), [](SampleType b, SampleType a) { return a + b; });
167 }
168 }
169 }

177 void copyFrom(const SafeAudioBuffer& other, int start_sample = 0, int num_samples_to_process = -1) {
178 if (num_samples_to_process == -1) {
179 num_samples_to_process = std::min(num_samples_ - start_sample,
180 other.num_samples_ - start_sample);
181 }
182
183 assert(start_sample >= 0);
184 assert(start_sample + num_samples_to_process <= num_samples_);
185 assert(start_sample + num_samples_to_process <= other.num_samples_);
186 assert(num_channels_ == other.num_channels_);
187
188 for (int channel = 0; channel < num_channels_; ++channel) {
189 auto dest = channels_[channel].subspan(start_sample, num_samples_to_process);
190 auto src = other.channels_[channel].subspan(start_sample, num_samples_to_process);
191
192 // Use optimized copy for contiguous memory
193 std::ranges::copy(src, dest.begin());
194 }

224 void applyGain(SampleType gain, int start_sample = 0, int num_samples_to_process = -1) {
225 if (num_samples_to_process == -1) {
226 num_samples_to_process = num_samples_ - start_sample;
227 }
228
229 assert(start_sample >= 0);
230 assert(start_sample + num_samples_to_process <= num_samples_);
231
232 for (int channel = 0; channel < num_channels_; ++channel) {
233 auto span = channels_[channel].subspan(start_sample, num_samples_to_process);
234
235 // Optimized gain application with vectorization hints
236 if constexpr (std::is_same_v<SampleType, float>) {
237 // Optimized path for float with compiler hints
238 #ifdef _MSC_VER
239 __pragma(loop(ivdep))
240 #endif
241 for (size_t i = 0; i < span.size(); ++i) {
242 span[i] *= gain;
243 }
244 } else {
245 // Generic path with parallel execution
246 std::ranges::transform(span, span.begin(), [gain](SampleType sample) { return sample * gain; });
247 }
248 }
249 }
