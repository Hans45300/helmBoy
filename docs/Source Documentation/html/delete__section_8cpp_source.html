49void DeleteSection::paint(Graphics& g) {
50 static const DropShadow shadow(Colour(Colors::Color_ff000000), 5, Point<int>(0, 0));
51
52 g.setColour(Colors::overlay_screen);
53 g.fillAll();
54
55 Rectangle<int> delete_rect = getDeleteRect();
56 shadow.drawForRectangle(g, delete_rect);
57 g.setColour(Colour(Colors::Color_ff303030));
58 g.fillRect(delete_rect);
59
60 g.saveState();
61 g.setOrigin(delete_rect.getX() + PADDING_X, delete_rect.getY() + PADDING_Y);
62
63 g.setFont(Fonts::instance()->proportional_light().withPointHeight(14.0f));
64 g.setColour(Colour(Colors::Color_ffaaaaaa));
65
66 String text;
67 if (file_.isDirectory())
68 text = TRANS("Are you sure you want to delte this folder?");
69 else
70 text = TRANS("Are you sure you want to delte this patch?");
71 g.drawText(text,
72 0, 0.0f, delete_rect.getWidth() - 2 * PADDING_X, 22.0f,
73 Justification::centred, false);
74
75 g.setFont(Fonts::instance()->monospace().withPointHeight(16.0f));
76 g.setColour(Colors::audio);
77 g.drawText(file_.getFileNameWithoutExtension(),
78 0, 20.0f, delete_rect.getWidth() - 2 * PADDING_X, 22.0f,
79 Justification::centred, false);
80
81 g.restoreState();
82}

84void DeleteSection::resized() {
85 Rectangle<int> delete_rect = getDeleteRect();
86
87 float button_width = (delete_rect.getWidth() - 3 * PADDING_X) / 2.0f;
88 delete_button_->setBounds(delete_rect.getX() + PADDING_X,
89 delete_rect.getBottom() - PADDING_Y - BUTTON_HEIGHT,
90 button_width, BUTTON_HEIGHT);
91 cancel_button_->setBounds(delete_rect.getX() + button_width + 2 * PADDING_X,
92 delete_rect.getBottom() - PADDING_Y - BUTTON_HEIGHT,
93 button_width, BUTTON_HEIGHT);
94}

101void DeleteSection::buttonClicked(Button* clicked_button) {
102 if (clicked_button == delete_button_.get()) {
103 file_.deleteRecursively();
104 setVisible(false);
105 for (int i = 0; i < listeners_.size(); ++i)
106 listeners_[i]->fileDeleted(file_);
107 }
108 else if (clicked_button == cancel_button_.get())
109 setVisible(false);
110}
