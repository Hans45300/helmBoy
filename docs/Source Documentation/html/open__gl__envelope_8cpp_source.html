72void OpenGLEnvelope::paintBackground() {
73 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
74
75 if (getWidth() <= 0 || getHeight() <= 0)
76 return;
77
78 float ratio = getHeight() / 100.0f;
79
80 auto *display = Desktop::getInstance().getDisplays().getPrimaryDisplay();
81 jassert(display != nullptr);
82
83 float scale = display->scale;
84 background_image_ = Image(Image::ARGB, scale * getWidth(), scale * getHeight(), true);
85 Graphics g(background_image_);
86 g.addTransform(AffineTransform::scale(scale, scale));
87
88 g.fillAll(Colour(Colors::Color_ff424242));
89
90 g.setColour(Colour(Colors::Color_ff4a4a4a));
91 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
92 g.drawLine(x, 0, x, getHeight());
93 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
94 g.drawLine(0, y, getWidth(), y);
95
96 shadow.drawForPath(g, envelope_line_);
97 g.setColour(Colors::graph_fill);
98 g.fillPath(envelope_line_);
99
100 g.setColour(Colour(Colors::Color_ff505050));
101 g.drawLine(getAttackX(), 0.0f, getAttackX(), getHeight());
102 g.drawLine(getDecayX(), getSustainY(), getDecayX(), getHeight());
103
104 g.setColour(Colors::modulation);
105 float line_width = 1.5f * ratio;
106 PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
107 g.strokePath(envelope_line_, stroke);
108
109 float hover_line_x = -20;
110 if (attack_hover_)
111 hover_line_x = getAttackX();
112 else if (decay_hover_)
113 hover_line_x = getDecayX();
114 else if (release_hover_)
115 hover_line_x = getReleaseX();
116
117 g.setColour(Colour(Colors::Color_bbffffff));
118 g.fillRect(hover_line_x - 0.5f, 0.0f, 1.0f, 1.0f * getHeight());
119
120 float grab_radius = 20.0f * ratio;
121 float hover_radius = 7.0f * ratio;
122 if (sustain_hover_) {
123 if (mouse_down_) {
124 g.setColour(Colour(Colors::Color_11ffffff));
125 g.fillEllipse(getDecayX() - grab_radius, getSustainY() - grab_radius,
126 2.0f * grab_radius, 2.0f * grab_radius);
127 }
128
129 g.setColour(Colour(Colors::Color_bbffffff));
130 g.drawEllipse(getDecayX() - hover_radius, getSustainY() - hover_radius,
131 2.0f * hover_radius, 2.0f * hover_radius, 1.0);
132 }
133 else if (mouse_down_) {
134 g.setColour(Colour(Colors::Color_11ffffff));
135 g.fillRect(hover_line_x - 10.0f, 0.0f, 20.0f, 1.0f * getHeight());
136 }
137
138 g.setColour(Colors::modulation);
139 float marker_radius = ratio * MARKER_WIDTH / 2.0f;
140 g.fillEllipse(getDecayX() - marker_radius, getSustainY() - marker_radius,
141 2.0f * marker_radius, 2.0f * marker_radius);
142 g.setColour(Colour(Colors::Color_ff000000));
143 g.fillEllipse(getDecayX() - marker_radius / 2.0f, getSustainY() - marker_radius / 2.0f,
144 marker_radius, marker_radius);
145
146 background_.updateBackgroundImage(background_image_);
147}

149void OpenGLEnvelope::paintPositionImage() {
150 int min_image_width = roundToInt(4 * MARKER_WIDTH);
151 int image_width = mopo::utils::nextPowerOfTwo(min_image_width);
152 int marker_width = 2 * MARKER_WIDTH;
153 int image_height = roundToInt(2 * IMAGE_HEIGHT);
154 position_image_ = Image(Image::ARGB, image_width, image_height, true);
155 Graphics g(position_image_);
156
157 g.setColour(Colour(Colors::Color_77ffffff));
158 g.fillRect(image_width / 2.0f - 0.5f, 0.0f, 1.0f, 1.0f * image_height);
159
160 g.setColour(Colors::modulation);
161 g.fillEllipse(image_width / 2 - marker_width / 2, image_height / 2 - marker_width / 2,
162 marker_width, marker_width);
163 g.setColour(Colour(Colors::Color_ff000000));
164 g.fillEllipse(image_width / 2 - marker_width / 4, image_height / 2 - marker_width / 4,
165 marker_width / 2, marker_width / 2);
166}

179void OpenGLEnvelope::mouseMove(const MouseEvent& e) {
180 float x = e.getPosition().x;
181 float attack_delta = fabs(x - getAttackX());
182 float decay_delta = fabs(x - getDecayX());
183 float release_delta = fabs(x - getReleaseX());
184 float sustain_delta = fabs(e.getPosition().y - getSustainY());
185
186 bool a_hover = attack_delta < decay_delta && attack_delta < HOVER_DISTANCE;
187 bool d_hover = !attack_hover_ && decay_delta < release_delta && decay_delta < HOVER_DISTANCE;
188 bool r_hover = !decay_hover_ && release_delta < HOVER_DISTANCE;
189 bool s_hover = !a_hover && !r_hover && x > getDecayX() - HOVER_DISTANCE &&
190 x < getDecayX() + HOVER_DISTANCE && sustain_delta < HOVER_DISTANCE;
191
192 if (a_hover != attack_hover_ || d_hover != decay_hover_ ||
193 s_hover != sustain_hover_ || r_hover != release_hover_) {
194 attack_hover_ = a_hover;
195 decay_hover_ = d_hover;
196 sustain_hover_ = s_hover;
197 release_hover_ = r_hover;
198 paintBackground();
199 }

242float OpenGLEnvelope::getAttackX() {
243 if (!attack_slider_)
244 return 0.0;
245
246 double percent = attack_slider_->valueToProportionOfLength(attack_slider_->getValue());
247 return 1 + (getWidth() - 1) * percent * ATTACK_RANGE_PERCENT;
248}
249
250float OpenGLEnvelope::getDecayX() {
251 if (!decay_slider_)
252 return 0.0;
253
254 double percent = decay_slider_->valueToProportionOfLength(decay_slider_->getValue());
255 return getAttackX() + getWidth() * percent * DECAY_RANGE_PERCENT;
256}
257
258float OpenGLEnvelope::getSustainY() {
259 if (!sustain_slider_)
260 return 0.0;
261
262 double percent = sustain_slider_->valueToProportionOfLength(sustain_slider_->getValue());
263 return getHeight() * (1.0 - percent);
264}
265
266float OpenGLEnvelope::getReleaseX() {
267 if (!release_slider_)
268 return 0.0;
269
270 double percent = release_slider_->valueToProportionOfLength(release_slider_->getValue());
271 return getDecayX() + getWidth() * percent * (1.0 - DECAY_RANGE_PERCENT - ATTACK_RANGE_PERCENT);
272}

330void OpenGLEnvelope::resetEnvelopeLine() {
331 envelope_line_.clear();
332 envelope_line_.startNewSubPath(1, getHeight());
333 envelope_line_.lineTo(getAttackX(), 0.0f);
334 envelope_line_.quadraticTo(0.5f * (getAttackX() + getDecayX()), getSustainY(),
335 getDecayX(), getSustainY());
336
337 envelope_line_.quadraticTo(0.5f * (getReleaseX() + getDecayX()), getHeight(),
338 getReleaseX(), getHeight());
339
340 paintBackground();
341}

343Point<float> OpenGLEnvelope::valuesToPosition(float phase, float amp) {
344 float y = (1.0 - amp) * getHeight();
345 float x = 0.0;
346
347 if (phase < 0.0 || phase > 2.5)
348 return Point<float>(-2.0, -2.0);
349
350 float attack_x = getAttackX();
351 float decay_x = getDecayX();
352 float release_x = getReleaseX();
353 float sustain = sustain_slider_->getValue();
354
355 if (phase <= 0.5f) {
356 x = amp * attack_x;
357 }
358 else if (phase <= 1.5f) {
359 float delta = decay_x - attack_x;
360 float amp_diff = 1.0f - sustain;
361 if (amp_diff == 0.0f)
362 x = decay_x;
363 else {
364 float percent = (1.0f - amp) / amp_diff;
365 x = attack_x + percent * delta;
366 }
367 }
368 else {
369 float delta = release_x - decay_x;
370 x = decay_x + delta - delta * (amp / sustain);
371 }
372
373 Point<float> closest;
374 envelope_line_.getNearestPoint(Point<float>(x, y), closest);
375 if (phase > 1.5f && phase < 2.5f && closest.x < decay_x) {
376 closest.x = decay_x;
377 closest.y = (1.0 - amp) * getHeight();
378 }
379 if (phase > 0.5f && phase < 1.5f && closest.x > decay_x) {
380 closest.x = decay_x;
381 closest.y = (1.0 - amp) * getHeight();
382 }
383 return Point<float>(2.0f * closest.x / getWidth() - 1.0f, 1.0f - 2.0f * closest.y / getHeight());
384}

386void OpenGLEnvelope::init(OpenGLContext& open_gl_context) {
387 paintPositionImage();
388
389 open_gl_context.extensions.glGenBuffers(1, &vertex_buffer_);
390 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
391
392 GLsizeiptr vert_size = static_cast<GLsizeiptr>(static_cast<size_t>(16 * sizeof(float)));
393 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
394 position_vertices_, GL_STATIC_DRAW);
395
396 open_gl_context.extensions.glGenBuffers(1, &triangle_buffer_);
397 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
398
399 GLsizeiptr tri_size = static_cast<GLsizeiptr>(static_cast<size_t>(6 * sizeof(float)));
400 open_gl_context.extensions.glBufferData(GL_ELEMENT_ARRAY_BUFFER, tri_size,
401 position_triangles_, GL_STATIC_DRAW);
402
403 background_.init(open_gl_context);
404}

406void OpenGLEnvelope::drawPosition(OpenGLContext& open_gl_context) {
407 if (position_texture_.getWidth() != position_image_.getWidth())
408 position_texture_.loadImage(position_image_);
409
410 if (envelope_phase_ == nullptr || envelope_amp_ == nullptr || envelope_amp_->buffer[0] <= 0.0)
411 return;
412
413 Point<float> point = valuesToPosition(envelope_phase_->buffer[0], envelope_amp_->buffer[0]);
414 float x = point.x;
415 float y = point.y;
416
417 glEnable(GL_BLEND);
418 glBlendFunc(GL_ONE, GL_ONE_MINUS_SRC_ALPHA);
419
420 int draw_width = getWidth();
421 int draw_height = getHeight();
422
423 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
424 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
425
426 float ratio = getHeight() / 100.0f;
427
428 float position_height = ratio * (0.5f * position_texture_.getHeight()) / draw_height;
429 float position_width = ratio * (0.5f * position_texture_.getWidth()) / draw_width;
430 position_vertices_[0] = x - position_width;
431 position_vertices_[1] = y + position_height;
432 position_vertices_[4] = x - position_width;
433 position_vertices_[5] = y - position_height;
434 position_vertices_[8] = x + position_width;
435 position_vertices_[9] = y - position_height;
436 position_vertices_[12] = x + position_width;
437 position_vertices_[13] = y + position_height;
438
439 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
440 GLsizeiptr vert_size = static_cast<GLsizeiptr>(static_cast<size_t>(16 * sizeof(float)));
441 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
442 position_vertices_, GL_STATIC_DRAW);
443
444 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
445 position_texture_.bind();
446
447 open_gl_context.extensions.glActiveTexture(GL_TEXTURE0);
448
449 if (background_.texture_uniform() != nullptr)
450 background_.texture_uniform()->set(0);
451
452 background_.shader()->use();
453
454 background_.enableAttributes(open_gl_context);
455 glDrawElements(GL_TRIANGLES, 6, GL_UNSIGNED_INT, 0);
456 background_.disableAttributes(open_gl_context);
457
458 position_texture_.unbind();
459
460 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
461 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
462}
