29void FilterSelector::paint(Graphics& g) {
30 static const PathStrokeType stroke(1.000f, PathStrokeType::curved, PathStrokeType::rounded);
31
32 int selected = getValue();
33 int num_types = getMaximum() - getMinimum() + 1;
34 float cell_width = float(getWidth()) / num_types;
35
36 g.setColour(Colour(Colors::background));
37 g.fillRect(getLocalBounds());
38
39 g.setColour(Colour(Colors::info_background));
40 g.fillRect(selected * cell_width, 0.0f, cell_width, float(getHeight()));
41
42 g.setColour(selected == 0 ? Colour(Colors::grid_full_white) : Colour(Colors::popup_background));
43 g.strokePath(low_shelf_, stroke);
44
45 g.setColour(selected == 1 ? Colour(Colors::grid_full_white) : Colour(Colors::popup_background));
46 g.strokePath(band_shelf_, stroke);
47
48 g.setColour(selected == 2 ? Colour(Colors::grid_full_white) : Colour(Colors::popup_background));
49 g.strokePath(high_shelf_, stroke);
50}

52void FilterSelector::resized() {
53 SynthSlider::resized();
54 int num_types = getMaximum() - getMinimum() + 1;
55 float cell_width = float(getWidth()) / num_types;
56 float padding_x = (cell_width - TYPE_WIDTH) / 2.0f;
57 float type_height = getHeight() - 2 * TYPE_PADDING_Y;
58
59 resizeLowShelf(0.0f * cell_width + padding_x, TYPE_PADDING_Y, TYPE_WIDTH, type_height);
60 resizeBandShelf(1.0f * cell_width + padding_x, TYPE_PADDING_Y, TYPE_WIDTH, type_height);
61 resizeHighShelf(2.0f * cell_width + padding_x, TYPE_PADDING_Y, TYPE_WIDTH, type_height);
62}
