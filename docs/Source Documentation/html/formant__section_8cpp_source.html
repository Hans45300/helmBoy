94void FormantSection::timerCallback() {
95 if (!isShowing())
96 return;
97
98 FullInterface* interface = findParentComponentOfClass<FullInterface>();
99 if (interface == nullptr || !interface->isAnimating()) {
100 effective_x_ = static_cast<float>(x_->getValue());
101 effective_y_ = static_cast<float>(y_->getValue());
102 effective_x_valid_ = true;
103 effective_y_valid_ = true;
104 xy_pad_->setEffectivePosition(Point<float>(effective_x_, effective_y_), false);
105 return;
106 }
107
108 const auto updatePosition = [interface](SynthSlider* slider,
109 const std::string& name,
110 float previous,
111 bool& position_valid) {
112 if (!interface->isDestinationModulated(name)) {
113 position_valid = true;
114 return static_cast<float>(slider->getValue());
115 }
116
117 float effective_value = previous;
118 if (interface->copyModulatedValue(name, effective_value) &&
119 std::isfinite(effective_value)) {
120 position_valid = true;
121 return jlimit(0.0f, 1.0f, effective_value);
122 }
123
124 if (!position_valid) {
125 position_valid = true;
126 return static_cast<float>(slider->getValue());
127 }
128 return previous;
129 };
130
131 effective_x_ = updatePosition(x_.get(), "formant_x", effective_x_,
132 effective_x_valid_);
133 effective_y_ = updatePosition(y_.get(), "formant_y", effective_y_,
134 effective_y_valid_);
135
136 const bool modulated =
137 interface->isDestinationModulated("formant_x") ||
138 interface->isDestinationModulated("formant_y");
139 xy_pad_->setEffectivePosition(Point<float>(effective_x_, effective_y_),
140 on_->getToggleState() && modulated);
141}
