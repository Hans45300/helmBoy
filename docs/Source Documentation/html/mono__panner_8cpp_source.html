50 void MonoPanner::process() {
51 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
52 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Pan)));
53
54 using batch = xsimd::batch<mopo_float>;
55 constexpr std::size_t simd_size = batch::size;
56 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
57 auto audio = std::span<const mopo_float>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
58 auto pan = std::span<const mopo_float>(input(static_cast<int>(Inputs::Pan))->source->buffer, buffer_size_);
59 auto out_left = std::span<mopo_float>(output(static_cast<int>(Outputs::Left))->buffer, buffer_size_);
60 auto out_right = std::span<mopo_float>(output(static_cast<int>(Outputs::Right))->buffer, buffer_size_);
61 int i = 0;
62 for (; i < simd_end; i += simd_size) {
63 batch a = batch::load_unaligned(&audio[i]);
64 batch p = batch::load_unaligned(&pan[i]);
65 alignas(alignof(batch)) mopo_float left_arr[simd_size];
66 alignas(alignof(batch)) mopo_float right_arr[simd_size];
67 for (std::size_t j = 0; j < simd_size; ++j) {
68 mopo_float integral;
69 left_arr[j] = Wave::fullsin(utils::mod(p.get(j) + LEFT_ROTATION, &integral));
70 right_arr[j] = Wave::fullsin(utils::mod(p.get(j) + RIGHT_ROTATION, &integral));
71 }
72 batch left_gain = batch::load_unaligned(left_arr);
73 batch right_gain = batch::load_unaligned(right_arr);
74 (a * left_gain).store_unaligned(&out_left[i]);
75 (a * right_gain).store_unaligned(&out_right[i]);
76 }
77 for (; i < buffer_size_; ++i) {
78 mopo_float integral;
79 mopo_float left_gain = Wave::fullsin(utils::mod(pan[i] + LEFT_ROTATION, &integral));
80 mopo_float right_gain = Wave::fullsin(utils::mod(pan[i] + RIGHT_ROTATION, &integral));
81 out_left[i] = audio[i] * left_gain;
82 out_right[i] = audio[i] * right_gain;
83 }
84 }
