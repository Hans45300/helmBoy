55 void tick(int i) {
56 mopo_float audio = input(static_cast<int>(Inputs::Audio))->at(i);
57 mopo_float wet = input(static_cast<int>(Inputs::Wet))->at(i);
58 mopo_float period = sample_rate_ / input(static_cast<int>(Inputs::Frequency))->at(i);
59
60 static_samples_ += 1.0;
61 if (static_samples_ >= period) {
62 static_samples_ -= period;
63 current_sample_ = audio;
64 }
65
66 output(0)->buffer[i] = utils::interpolate(audio, current_sample_, wet);
67 }

69 protected:
