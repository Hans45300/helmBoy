54void SynthSection::paintBackground(Graphics& g) {
55 static const DropShadow button_shadow(Colour(Colors::Color_ff000000), size_ratio_ * 3.0f, Point<int>(0, 0));
56
57 paintContainer(g);
58 // Draw shadow divider.
59 float shadow_top = size_ratio_ * (TITLE_WIDTH - SHADOW_WIDTH);
60 int title_width = getTitleWidth();
61 g.setGradientFill(ColourGradient(Colour(Colors::Color_22000000), 0.0f, shadow_top,
62 Colour(Colors::Color_66000000), 0.0f, title_width,
63 false));
64 g.fillRoundedRectangle(0, 0, getWidth(), title_width, 1.0f);
65
66 // Draw text title.
67 g.setColour(Colors::tab_heading_text);
68 g.setFont(Fonts::instance()->proportional_light().withPointHeight(size_ratio_ * 14.0f));
69 g.drawText(TRANS(getName()), 0, 0, getWidth(), title_width,
70 Justification::centred, true);
71
72 paintKnobShadows(g);
73 paintChildrenBackgrounds(g);
74}

213void SynthSection::setAllValues(mopo::control_map& controls) {
214 for (auto& slider : all_sliders_) {
215 if (controls.count(slider.first)) {
216 slider.second->setValue(controls[slider.first]->value(),
217 NotificationType::dontSendNotification);
218 slider.second->valueChanged();
219 }
220 }
221
222 for (auto& button : all_buttons_) {
223 if (controls.count(button.first)) {
224 bool toggle = controls[button.first]->value();
225 button.second->setToggleState(toggle, NotificationType::dontSendNotification);
226 }
227 }
228
229 repaint();
230}

232void SynthSection::setAllValues(
233 const std::vector<std::pair<std::string, mopo::mopo_float>>& values) {
234 auto findValue = [&values](const std::string& name) -> const mopo::mopo_float* {
235 auto value = std::lower_bound(values.begin(), values.end(), name,
236 [](const auto& entry, const std::string& key) { return entry.first < key; });
237 return value != values.end() && value->first == name ? &value->second : nullptr;
238 };
239
240 for (auto& [name, slider] : all_sliders_) {
241 if (const auto* value = findValue(name)) {
242 slider->setValue(*value, NotificationType::dontSendNotification);
243 slider->valueChanged();
244 }
245 }
246
247 for (auto& [name, button] : all_buttons_) {
248 if (const auto* value = findValue(name))
249 button->setToggleState(*value, NotificationType::dontSendNotification);
250 }
251
252 repaint();
253}
