34 inline mopo_float computeAmplitude(mopo_float offset, mopo_float period, mopo_float softness) {
35 mopo_float progress = offset / period;
36 mopo_float phase_setup = std::fabs(utils::interpolate(-softness, softness, progress));
37 mopo_float phase = utils::clamp(phase_setup - softness + PI, 0.0, PI);
38 return 0.5 * cos(phase) + 0.5;
39 }

72 void Stutter::process() {
73 using enum Inputs;
74 const Input* audio_input = input(static_cast<int>(Inputs::Audio));
75 if (audio_input == nullptr || audio_input->source == nullptr ||
76 audio_input->source->buffer == nullptr ||
77 audio_input->source->buffer_size < buffer_size_) {
78 utils::zeroBuffer(std::span<mopo_float>(output()->buffer, buffer_size_));
79 output()->clearTrigger();
80 return;
81 }
82
83 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
84
85 // A hack to save memory until stutter is used.
86 if (memory_ == nullptr)
87 memory_ = new Memory(size_);
88
89 mopo_float max_memory_write = memory_->getSize();
90 auto audio = std::span<const mopo_float>(audio_input->source->buffer, buffer_size_);
91 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
92
93 constexpr mopo_float kMinFrequency = 1.0e-4f;
94 mopo_float resample_frequency = input(static_cast<int>(Inputs::ResampleFrequency))->at(0);
95 mopo_float stutter_frequency = input(static_cast<int>(Inputs::StutterFrequency))->at(0);
96 if (!std::isfinite(resample_frequency) || resample_frequency < kMinFrequency)
97 resample_frequency = kMinFrequency;
98 if (!std::isfinite(stutter_frequency) || stutter_frequency < kMinFrequency)
99 stutter_frequency = kMinFrequency;
100
101 mopo_float sample_period = sample_rate_ / resample_frequency;
102 mopo_float end_stutter_period = sample_rate_ / stutter_frequency;
103 if (!std::isfinite(sample_period) || sample_period < 1.0f)
104 sample_period = 1.0f;
105 if (!std::isfinite(end_stutter_period) || end_stutter_period < 1.0f)
106 end_stutter_period = 1.0f;
107 end_stutter_period = utils::min(sample_period, end_stutter_period);
108 end_stutter_period = utils::min(max_memory_write, end_stutter_period);
109 mopo_float stutter_period = end_stutter_period;
110 if (last_stutter_period_)
111 stutter_period = last_stutter_period_;
112 mopo_float stutter_period_diff = (end_stutter_period - stutter_period) / buffer_size_;
113
114 mopo_float read_softness = utils::max(input(static_cast<int>(Inputs::WindowSoftness))->at(0), MIN_SOFTNESS);
115 mopo_float end_softness = PI * utils::max(1.0, 1.0 / read_softness);
116
117 int buffer_size = buffer_size_;
118 if (input(static_cast<int>(Inputs::Reset))->source->triggered) {
119 startResampling(sample_period);
120
121 stutter_period = end_stutter_period;
122 stutter_period_diff = 0.0;
123 }
124 else if (resample_countdown_ > sample_period)
125 resample_countdown_ = sample_period;
126
127 int i = 0;
128 while (i < buffer_size_) {
129 MOPO_ASSERT(stutter_period >= 0);
130 int max_samples = std::ceil(stutter_period - offset_);
131
132 if (!resampling_)
133 max_samples = std::ceil(std::min(stutter_period - offset_, resample_countdown_));
134
135 MOPO_ASSERT(max_samples >= 0);
136
137 int samples = std::min(buffer_size, i + max_samples);
138 int num_samples = samples - i;
139
140 MOPO_ASSERT(samples >= 0);
141 MOPO_ASSERT(num_samples >= 0);
142
143 if (num_samples <= 0) {
144 startResampling(sample_period);
145 stutter_period = end_stutter_period;
146 stutter_period_diff = 0.0f;
147 continue;
148 }
149
150 if (memory_offset_ < max_memory_write) {
151 int mem_samples = std::min<int>(max_memory_write - memory_offset_, num_samples);
152 MOPO_ASSERT(buffer_size_ - i >= mem_samples);
153 memory_->pushBlock(audio.data() + i, std::max<int>(0, mem_samples));
154 memory_offset_ += mem_samples;
155 }
156
157 stutter_period += num_samples * stutter_period_diff;
158 resample_countdown_ -= num_samples;
159 mopo_float next_offset = offset_ + num_samples;
160
161 mopo_float softness_offset = std::min(stutter_period - next_offset, next_offset);
162 softness_offset = std::min(softness_offset, resample_countdown_);
163
164 mopo_float end_amplitude = computeAmplitude(softness_offset, stutter_period, end_softness);
165 mopo_float amplitude = last_amplitude_;
166 mopo_float amplitude_diff = (end_amplitude - amplitude) / num_samples;
167
168 if (resampling_) {
169 for (; i < samples; ++i) {
170 amplitude += amplitude_diff;
171 dest[i] = amplitude * audio[i];
172 }
173 }
174 else {
175 mopo_float* playback_dest = dest.data() + i;
176 for (int s = 0; s < num_samples; ++s) {
177 amplitude += amplitude_diff;
178 playback_dest[s] = amplitude * memory_->get(memory_offset_ - offset_ - s);
179 }
180 }
181
182 i = samples;
183 offset_ = next_offset;
184 last_amplitude_ = end_amplitude;
185
186 if (offset_ >= stutter_period) {
187 resampling_ = false;
188 offset_ = 0.0f;
189 }
190
191 if (resample_countdown_ <= 0.0f)
192 startResampling(sample_period);
193
194 MOPO_ASSERT(stutter_period > 0);
195 MOPO_ASSERT(offset_ >= 0);
196 MOPO_ASSERT(samples >= 0 && samples <= buffer_size_);
197 }
198 last_stutter_period_ = end_stutter_period;
199 }
