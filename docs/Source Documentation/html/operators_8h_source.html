36#define PROCESS_TICK_FUNCTION \
37void process() override { \
38 for (int i = 0; i < buffer_size_; ++i) \
39 tick(i); \
40 processTriggers(); \
41}

45 // A base class for arithmetic operators.
52 class Operator : public Processor {
53 public:
54 Operator(int num_inputs, int num_outputs, bool control_rate = false) :
55 Processor(num_inputs, num_outputs, control_rate) { }
56
57 virtual void process() override;
58 virtual void tick(int i) = 0;
59 inline void processTriggers() {
60 output()->clearTrigger();
61 int num_inputs = numInputs();
62 for (int i = 0; i < num_inputs; ++i) {
63 if (input(i)->source->triggered) {
64 int offset = input(i)->source->trigger_offset;
65 tick(offset);
66 output()->trigger(output()->buffer[offset], offset);
67 }
68 }
69 }
70
71 private:
72 Operator() : Processor(0, 0) { }
73 };

392 // A processor that will interpolate two streams by an interpolation stream.
393 class BilinearInterpolate : public Operator {
394 public:
396 enum class Inputs {
397 TopLeft,
398 TopRight,
399 BottomLeft,
400 BottomRight,
401 XPosition,
402 YPosition,
403 NumInputs
404 };
405
406 BilinearInterpolate() : Operator(static_cast<int>(Inputs::NumInputs), 1) { }
407
408 virtual Processor* clone() const override {
409 return new BilinearInterpolate(*this);
410 }
411
412 void process() override;
413
414 inline void tick(int i) override {
415 mopo_float top = utils::interpolate(input(static_cast<int>(Inputs::TopLeft))->at(i),
416 input(static_cast<int>(Inputs::TopRight))->at(i),
417 input(static_cast<int>(Inputs::XPosition))->at(i));
418 mopo_float bottom = utils::interpolate(input(static_cast<int>(Inputs::BottomLeft))->at(i),
419 input(static_cast<int>(Inputs::BottomRight))->at(i),
420 input(static_cast<int>(Inputs::XPosition))->at(i));
421 output()->buffer[i] = utils::interpolate(top, bottom,
422 input(static_cast<int>(Inputs::YPosition))->at(i));
423 }
424 };
