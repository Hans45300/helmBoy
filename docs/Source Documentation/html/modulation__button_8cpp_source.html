41void ModulationButton::mouseDown(const MouseEvent& e) {
42 if (e.mods.isPopupMenu()) {
43 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
44 if (parent == nullptr)
45 return;
46
47 std::vector<mopo::ModulationConnection*> connections =
48 parent->getSynth()->getSourceConnections(getName().toStdString());
49
50 if (connections.size() == 0)
51 return;
52
53 PopupMenu m;
54 m.setLookAndFeel(DefaultLookAndFeel::instance());
55
56 String disconnect("Disconnect from ");
57 for (int i = 0; i < connections.size(); ++i)
58 m.addItem(kModulationList + i, disconnect + connections[i]->destination);
59
60 if (connections.size() > 1)
61 m.addItem(kDisconnect, "Disconnect all");
62
63 m.showMenuAsync(PopupMenu::Options(),
64 ModalCallbackFunction::forComponent(initPatchCallback, this));
65 }
66 else
67 ToggleButton::mouseDown(e);
68}

79void ModulationButton::disconnectIndex(int index) {
80 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
81 if (parent == nullptr)
82 return;
83
84 std::vector<mopo::ModulationConnection*> connections =
85 parent->getSynth()->getSourceConnections(getName().toStdString());
86
87 if (index == kDisconnect) {
88 for (auto* connection : connections)
89 disconnectModulation(connection);
90 repaint();
91 }
92 else if (index >= kModulationList) {
93 int connection_index = index - kModulationList;
94 disconnectModulation(connections[connection_index]);
95 repaint();
96 }
97}

99void ModulationButton::disconnectModulation(mopo::ModulationConnection* connection) {
100 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
101 parent->getSynth()->disconnectModulation(connection);
102 int modulations_left = parent->getSynth()->getNumModulations(connection->destination);
103
104 for (auto* listener : listeners_)
105 listener->modulationDisconnected(connection, modulations_left == 0);
106}
