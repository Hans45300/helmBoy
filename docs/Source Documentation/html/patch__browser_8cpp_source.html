49 Array<File> getSelectedFolders(ListBox* view, FileListBoxModel* model) {
50 SparseSet<int> selected_rows = view->getSelectedRows();
51
52 Array<File> selected_folders;
53 for (int i = 0; i < selected_rows.size(); ++i)
54 selected_folders.add(model->getFileAtRow(selected_rows[i]));
55
56 return selected_folders;
57 }
58
59 Array<File> getFoldersToScan(ListBox* view, FileListBoxModel* model) {
60 if (view->getSelectedRows().size())
61 return getSelectedFolders(view, model);
62
63 return model->getAllFiles();
64 }
65
66 void setSelectedRows(ListBox* view, FileListBoxModel* model, Array<File> selected) {
67 view->deselectAllRows();
68
69 for (int sel = 0, row = 0; sel < selected.size() && row < model->getNumRows();) {
70 String selected_path = selected[sel].getFullPathName();
71 int compare = model->getFileAtRow(row).getFullPathName().compare(selected_path);
72
73 if (compare < 0)
74 ++row;
75 else if (compare > 0)
76 ++sel;
77 else {
78 view->selectRow(row, true, false);
79 ++row;
80 ++sel;
81 }
82 }
83 }

380void PatchBrowser::buttonClicked(Button* clicked_button) {
381 if (clicked_button == save_as_button_.get() && save_section_)
382 save_section_->setVisible(true);
383 else if (clicked_button == delete_patch_button_.get() && delete_section_) {
384 File selected_patch = getSelectedPatch();
385 if (selected_patch.exists()) {
386 delete_section_->setFileToDelete(selected_patch);
387 delete_section_->setVisible(true);
388 }
389 }
390 else if (clicked_button == move_rename_patch_button_.get()) {
391 File selected_patch = getSelectedPatch();
392 if (!selected_patch.existsAsFile())
393 return;
394
395 String source_extension = selected_patch.getFileExtension();
396 String destination_extension = source_extension;
397 if (source_extension.equalsIgnoreCase(".helm"))
398 destination_extension = ".helmboy";
399
400 file_chooser_ = std::make_unique<FileChooser>(TRANS("Move / Rename Patch"),
401 selected_patch,
402 "*" + destination_extension);
403
404 file_chooser_->launchAsync(FileBrowserComponent::saveMode |
405 FileBrowserComponent::canSelectFiles |
406 FileBrowserComponent::warnAboutOverwriting,
407 [safe_browser = Component::SafePointer<PatchBrowser>(this), selected_patch, destination_extension](const FileChooser& fc) {
408 auto* browser = safe_browser.getComponent();
409 if (browser == nullptr)
410 return;
411 File target = fc.getResult();
412 if (target == File())
413 return;
414
415 if (target.getFileExtension().isEmpty())
416 target = target.withFileExtension(destination_extension);
417
418 if (target == selected_patch)
419 return;
420
421 if (!selected_patch.moveFileTo(target)) {
422 AlertWindow::showMessageBoxAsync(AlertWindow::WarningIcon,
423 TRANS("Move / Rename Failed"),
424 TRANS("Unable to move or rename the selected patch."));
425 return;
426 }
427
428 if (browser->external_patch_ == selected_patch)
429 browser->external_patch_ = target;
430
431 browser->banks_view_->deselectAllRows();
432 browser->folders_view_->deselectAllRows();
433 browser->patches_view_->deselectAllRows();
434 browser->scanAll();
435
436 int target_index = browser->patches_model_->getIndexOfFile(target);
437 if (target_index >= 0)
438 browser->patches_view_->selectRow(target_index);
439 });
440 }
441 else if (clicked_button == done_button_.get())
442 loadSelectedPatch();
443 else if (clicked_button == hide_button_.get())
444 {
445 setVisible(false);
446 if (selector_)
447 selector_->ensureBrowseButtonIsBrowse();
448 }
449 else if (clicked_button == import_bank_button_.get()) {
450 LoadSave::importBank([safe_browser = Component::SafePointer<PatchBrowser>(this)]() {
451 if (auto* browser = safe_browser.getComponent()) {
452 browser->scanAll();
453 browser->updateExportButtonState();
454 }
455 });
456 }
457 else if (clicked_button == export_bank_button_.get()) {
458 Array<File> selected_items = getSelectedFolders(folders_view_.get(), folders_model_.get());
459 selected_items.addArray(getSelectedFolders(patches_view_.get(), patches_model_.get()));
460
461 if (selected_items.isEmpty())
462 selected_items = getSelectedFolders(banks_view_.get(), banks_model_.get());
463 if (!selected_items.isEmpty())
464 LoadSave::exportBank(std::move(selected_items));
465 }
466 else if (clicked_button == preset_root_button_.get()) {
467 file_chooser_ = std::make_unique<FileChooser>(TRANS("Choose Preset Root Folder"),
468 LoadSave::getBankDirectory());
469 file_chooser_->launchAsync(FileBrowserComponent::openMode | FileBrowserComponent::canSelectDirectories,
470 [safe_browser = Component::SafePointer<PatchBrowser>(this)](const FileChooser& fc) {
471 auto* browser = safe_browser.getComponent();
472 if (browser == nullptr)
473 return;
474 File selected = fc.getResult();
475 if (selected == File())
476 return;
477
478 ScopedValueSetter<bool> lock_notifications(browser->suppress_model_notifications_, true);
479 LoadSave::saveCustomPatchRootDirectory(selected);
480 browser->search_box_->setText("", false);
481 browser->folders_model_->setShowRelativeDirectoryPaths(true, LoadSave::getBankDirectory());
482 browser->banks_view_->deselectAllRows();
483 browser->folders_view_->deselectAllRows();
484 browser->patches_view_->deselectAllRows();
485 browser->scanAll();
486 });
487 }
488}

539void PatchBrowser::jumpToPatch(int indices) {
540 ensureInitialScan();
541 static const FileSorterAscending file_sorter;
542
543 File parent = external_patch_.getParentDirectory();
544 if (parent.exists()) {
545 Array<File> patches;
546 LoadSave::findPatchFiles(parent, patches, false);
547 patches.sort(file_sorter);
548 if (patches.isEmpty())
549 return;
550 int index = patches.indexOf(external_patch_);
551 index = (index + indices + patches.size()) % patches.size();
552
553 File new_patch = patches[index];
554 loadFromFile(new_patch);
555 externalPatchLoaded(new_patch);
556 }
557 else {
558 SparseSet<int> selected_rows = patches_view_->getSelectedRows();
559 if (selected_rows.size()) {
560 int num_rows = patches_model_->getNumRows();
561 if (num_rows == 0)
562 return;
563 int row = (selected_rows[0] + indices + num_rows) % num_rows;
564 patches_view_->selectRow(row);
565 }
566 else
567 patches_view_->selectRow(0);
568 }
569}

585bool PatchBrowser::loadFromFile(File& patch) {
586 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
587 if (parent == nullptr)
588 return false;
589
590 SynthBase* synth = parent->getSynth();
591 if (synth->loadFromFile(patch)) {
592 setPatchInfo(patch);
593 synth->setPatchName(patch.getFileNameWithoutExtension());
594 synth->setFolderName(patch.getParentDirectory().getFileName());
595 synth->setAuthor(author_);
596 return true;
597 }
598 return false;
599}

626void PatchBrowser::scanBanks() {
627 Array<File> top_level;
