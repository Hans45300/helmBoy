49 Array<File> getSelectedFolders(ListBox* view, FileListBoxModel* model) {
50 SparseSet<int> selected_rows = view->getSelectedRows();
51
52 Array<File> selected_folders;
53 for (int i = 0; i < selected_rows.size(); ++i)
54 selected_folders.add(model->getFileAtRow(selected_rows[i]));
55
56 return selected_folders;
57 }
58
59 Array<File> getFoldersToScan(ListBox* view, FileListBoxModel* model) {
60 if (view->getSelectedRows().size())
61 return getSelectedFolders(view, model);
62
63 return model->getAllFiles();
64 }
65
66 void setSelectedRows(ListBox* view, FileListBoxModel* model, Array<File> selected) {
67 view->deselectAllRows();
68
69 for (int sel = 0, row = 0; sel < selected.size() && row < model->getNumRows();) {
70 String selected_path = selected[sel].getFullPathName();
71 int compare = model->getFileAtRow(row).getFullPathName().compare(selected_path);
72
73 if (compare < 0)
74 ++row;
75 else if (compare > 0)
76 ++sel;
77 else {
78 view->selectRow(row, true, false);
79 ++row;
80 ++sel;
81 }
82 }
83 }

241 float start_x = BROWSE_PADDING;
242 int button_padding = BROWSE_PADDING / 2;

244 int banks_width = getBanksWidth();
245 int folders_width = getFoldersWidth();
246 int patches_width = getPatchesWidth();
247 int patch_info_width = getPatchInfoWidth();
248 float height = size_ratio_ * BROWSING_HEIGHT - 2.0f * BROWSE_PADDING;
249 float height_with_buttons = height - BUTTON_HEIGHT - button_padding;
250 float search_box_height = 28.0f;

252 int import_export_width = (banks_width - button_padding) / 2.0f;
253 banks_view_->setBounds(start_x, BROWSE_PADDING, banks_width, height_with_buttons);
254 import_bank_button_->setBounds(start_x, banks_view_->getBottom() + button_padding,

259 folders_view_->setBounds(banks_view_->getRight() + BROWSE_PADDING,

263 search_box_->setBounds(patches_x, BROWSE_PADDING, patches_width, search_box_height);
264 patches_view_->setBounds(patches_x, search_box_height + BROWSE_PADDING,
265 patches_width, height_with_buttons - search_box_height);
266

268 save_as_button_->setBounds(patches_view_->getX(), patches_view_->getBottom() + button_padding,
269 button_width, BUTTON_HEIGHT);
270 delete_patch_button_->setBounds(patches_view_->getRight() - button_width,
271 patches_view_->getBottom() + button_padding,
272 button_width, BUTTON_HEIGHT);
273 delete_patch_button_->setEnabled(false);
274
275 float data_x = patches_view_->getRight() + BROWSE_PADDING;
276 float division = size_ratio_ * 90.0f;
277 float buffer = 20.0f;
278 cc_license_link_->setBounds(data_x + division + buffer, BROWSE_PADDING + 160.0f,

305void PatchBrowser::selectedFilesChanged(FileListBoxModel* model) {
306 if (model == banks_model_.get()) {
307 scanFolders();
308 export_bank_button_->setEnabled(banks_view_->getSelectedRows().size() == 1);
309 }
310 if (model == banks_model_.get() || model == folders_model_.get())
311 scanPatches();
312 else if (model == patches_model_.get()) {
313 SparseSet<int> selected_rows = patches_view_->getSelectedRows();
314 delete_patch_button_->setEnabled(selected_rows.size());
315
316 if (selected_rows.size()) {
317 external_patch_ = File();

323 }
324 else {
325 cc_license_link_->setVisible(false);
326 gpl_license_link_->setVisible(false);
327 }

354void PatchBrowser::buttonClicked(Button* clicked_button) {
355 if (clicked_button == save_as_button_.get() && save_section_)
356 save_section_->setVisible(true);
357 else if (clicked_button == delete_patch_button_.get() && delete_section_) {
358 File selected_patch = getSelectedPatch();
359 if (selected_patch.exists()) {
360 delete_section_->setFileToDelete(selected_patch);
361 delete_section_->setVisible(true);
362 }
363 }
364 else if (clicked_button == hide_button_.get() || clicked_button == done_button_.get())
365 {
366 setVisible(false);
367 // If a selector is linked, ensure its browse button returns to BROWSE when browser hides
368 if (selector_)
369 selector_->ensureBrowseButtonIsBrowse();
370 }
371 else if (clicked_button == import_bank_button_.get()) {
372 LoadSave::importBank();
373 scanAll();
374 }
375 else if (clicked_button == export_bank_button_.get()) {
376 Array<File> banks = getFoldersToScan(banks_view_.get(), banks_model_.get());
377 if (banks.size())
378 LoadSave::exportBank(banks[0].getFileName());
379 }

415void PatchBrowser::jumpToPatch(int indices) {
416 static const FileSorterAscending file_sorter;
417
418 File parent = external_patch_.getParentDirectory();
419 if (parent.exists()) {
420 Array<File> patches;
421 parent.findChildFiles(patches, File::findFiles, false, String("*.") + mopo::PATCH_EXTENSION);
422 patches.sort(file_sorter);
423 int index = patches.indexOf(external_patch_);
424 index = (index + indices + patches.size()) % patches.size();
425
426 File new_patch = patches[index];
427 loadFromFile(new_patch);
428 externalPatchLoaded(new_patch);
429 }
430 else {
431 SparseSet<int> selected_rows = patches_view_->getSelectedRows();
432 if (selected_rows.size()) {
433 int num_rows = patches_model_->getNumRows();
434 int row = (selected_rows[0] + indices + num_rows) % num_rows;
435 patches_view_->selectRow(row);
436 }
437 else
438 patches_view_->selectRow(0);
439 }
440}

456bool PatchBrowser::loadFromFile(File& patch) {
457 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
458 if (parent == nullptr)
459 return false;
460
461 SynthBase* synth = parent->getSynth();
462 if (synth->loadFromFile(patch)) {
463 setPatchInfo(patch);
464 synth->setPatchName(patch.getFileNameWithoutExtension());
465 synth->setFolderName(patch.getParentDirectory().getFileName());
466 synth->setAuthor(author_);
467 return true;
468 }
469 return false;
470}
