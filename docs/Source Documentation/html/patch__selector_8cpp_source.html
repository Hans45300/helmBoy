78void PatchSelector::paint(Graphics& g) {
79 SynthSection::paint(g);
80
81 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
82 patch_text_ = parent->getSynth()->getPatchName();
83 if (patch_text_ == "")
84 patch_text_ = TRANS("init");
85
86 if (modified_)
87 patch_text_ = "*" + patch_text_;
88 folder_text_ = parent->getSynth()->getFolderName();
89
90 int browse_height = proportionOfHeight(BROWSE_PERCENT);
91 Rectangle<int> top(proportionOfWidth(0.1f) + TEXT_PADDING, 0,
92 proportionOfWidth(0.8f) - TEXT_PADDING, browse_height);
93 Rectangle<int> bottom(proportionOfWidth(0.1f) + TEXT_PADDING, browse_height,
94 proportionOfWidth(0.8f) - TEXT_PADDING, browse_height);
95
96 g.setFont(Fonts::instance()->monospace().withPointHeight(size_ratio_ * 12.0f));
97 g.setColour(Colors::control_label_text);
98 g.drawFittedText(folder_text_, top, Justification::centredLeft, 1);
99 g.setColour(Colour(Colors::Color_ffffffff));
100 g.drawFittedText(patch_text_, bottom, Justification::centredLeft, 1);
101}

121void PatchSelector::resized() {
122 int full_browse_height = 2 * proportionOfHeight(BROWSE_PERCENT);
123 prev_patch_->setBounds(0, 0, proportionOfWidth(0.1f), full_browse_height);
124 next_patch_->setBounds(getWidth() - proportionOfWidth(0.1f), 0,
125 proportionOfWidth(0.1f), full_browse_height);
126
127 int button_width = (getWidth() - 2.0) / 3.0 - 1;
128 int button_height = getHeight() - full_browse_height;
129 int last_button_width = getWidth() - 2 * button_width - 2;
130 save_->setBounds(0, full_browse_height, button_width, button_height);
131 export_->setBounds(button_width + 1, full_browse_height, button_width, button_height);
132 browse_->setBounds(2 * button_width + 2, full_browse_height, last_button_width, button_height);
133 SynthSection::resized();
134}

149void PatchSelector::buttonClicked(Button* clicked_button) {
150 if (browser_ == nullptr)
151 return;
152
153 if (clicked_button == save_.get() && save_section_)
154 save_section_->setVisible(true);
155 else if (clicked_button == browse_.get()) {
156 browse_mode_ = !browse_mode_;
157 browser_->setVisible(browse_mode_);
158 if (browse_mode_)
159 browse_->setButtonText(TRANS("EDIT"));
160 else
161 browse_->setButtonText(TRANS("BROWSE"));
162 }
163 else if (clicked_button == export_.get()) {
164 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
165 if (parent == nullptr)
166 return;
167
168 SynthBase* synth = parent->getSynth();
169 // exportToFile() is async and will handle the file dialog
170 // The callback in exportToFile will update the UI when done
171 synth->exportToFile();
172 }
173 else if (clicked_button == prev_patch_.get())
174 browser_->loadPrevPatch();
175 else if (clicked_button == next_patch_.get())
176 browser_->loadNextPatch();
177}
