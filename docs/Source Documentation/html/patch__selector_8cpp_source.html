128void PatchSelector::paint(Graphics& g) {
129 SynthSection::paint(g);
130
131 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
132 patch_text_ = parent->getSynth()->getPatchName();
133 if (patch_text_ == "")
134 patch_text_ = TRANS("init");
135
136 if (modified_)
137 patch_text_ = "*" + patch_text_;
138 folder_text_ = parent->getSynth()->getFolderName();
139
140 int browse_height = proportionOfHeight(BROWSE_PERCENT);
141 Rectangle<int> top(proportionOfWidth(0.1f) + TEXT_PADDING, 0,
142 proportionOfWidth(0.8f) - TEXT_PADDING, browse_height);
143 Rectangle<int> bottom(proportionOfWidth(0.1f) + TEXT_PADDING, browse_height,
144 proportionOfWidth(0.8f) - TEXT_PADDING, browse_height);
145
146 g.setFont(Fonts::instance()->monospace().withPointHeight(size_ratio_ * 12.0f));
147 g.setColour(Colors::control_label_text);
148 g.drawFittedText(folder_text_, top, Justification::centredLeft, 1);
149 g.setColour(Colour(Colors::Color_ffffffff));
150 g.drawFittedText(patch_text_, bottom, Justification::centredLeft, 1);
151}

171void PatchSelector::resized() {
172 int full_browse_height = 2 * proportionOfHeight(BROWSE_PERCENT);
173 const int nav_button_height = full_browse_height / 2;
174 const int left_button_width = proportionOfWidth(0.1f);
175 const int right_button_x = getWidth() - left_button_width;
176 prev_patch_->setBounds(0, 0, left_button_width, nav_button_height);
177 init_->setBounds(0, nav_button_height, left_button_width,
178 full_browse_height - nav_button_height);
179 next_patch_->setBounds(right_button_x, 0, left_button_width, nav_button_height);
180 ab_comparison_->setBounds(right_button_x, nav_button_height, left_button_width,
181 full_browse_height - nav_button_height);
182
183 int button_width = (getWidth() - 2.0) / 3.0 - 1;
184 int button_height = getHeight() - full_browse_height;
185 int last_button_width = getWidth() - 2 * button_width - 2;
186 save_->setBounds(0, full_browse_height, button_width, button_height);
187 export_->setBounds(button_width + 1, full_browse_height, button_width, button_height);
188 browse_->setBounds(2 * button_width + 2, full_browse_height, last_button_width, button_height);
189 SynthSection::resized();
190}

205void PatchSelector::buttonClicked(Button* clicked_button) {
206 if (clicked_button == save_.get() && save_section_)
207 save_section_->setVisible(true);
208 else if (clicked_button == browse_.get()) {
209 if (browser_ == nullptr)
210 return;
211 browse_mode_ = !browse_mode_;
212 browser_->setVisible(browse_mode_);
213 if (browse_mode_)
214 browse_->setButtonText(TRANS("EDIT"));
215 else
216 browse_->setButtonText(TRANS("BROWSE"));
217 }
218 else if (clicked_button == export_.get()) {
219 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
220 if (parent == nullptr)
221 return;
222
223 SynthBase* synth = parent->getSynth();
224 // exportToFile() is async and will handle the file dialog
225 // The callback in exportToFile will update the UI when done
226 synth->exportToFile();
227 }
228 else if (clicked_button == prev_patch_.get() && browser_ != nullptr)
229 browser_->loadPrevPatch();
230 else if (clicked_button == next_patch_.get() && browser_ != nullptr)
231 browser_->loadNextPatch();
232 else if (clicked_button == init_.get())
233 initPatch();
234 else if (clicked_button == ab_comparison_.get())
235 switchABState();
236}

259void PatchSelector::initPatch() {
260 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
261 if (parent == nullptr)
262 return;
263
264 ensureABSnapshots();
265 SynthBase* synth = parent->getSynth();
266 const SynthGuiStateSnapshot init_snapshot = synth->captureInitStateSnapshot();
267 if (active_state_b_)
268 state_b_ = init_snapshot;
269 else
270 state_a_ = init_snapshot;
271
272 synth->applyParameterSnapshot(init_snapshot);
273 synth->setPatchName(TRANS("init"));
274 synth->setFolderName("");
275 if (browser_ != nullptr)
276 browser_->externalPatchLoaded(File());
277 parent->updateParameterSnapshot(init_snapshot);
278}

280void PatchSelector::ensureABSnapshots() {

294void PatchSelector::switchABState() {
295 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
296 if (parent == nullptr)
297 return;
298
299 ensureABSnapshots();
300 if (!ab_snapshots_initialized_)
301 return;
302
303 SynthBase* synth = parent->getSynth();
304 if (active_state_b_)
305 state_b_ = synth->captureGuiStateSnapshot();
306 else
307 state_a_ = synth->captureGuiStateSnapshot();
308
309 active_state_b_ = !active_state_b_;
310 const SynthGuiStateSnapshot& target = active_state_b_ ? state_b_ : state_a_;
311 synth->applyParameterSnapshot(target);
312 parent->updateParameterSnapshot(target);
313 updateABButtonState();
314}

316void PatchSelector::updateABButtonState() {

318 return;
319 ab_comparison_->setColour(TextButton::buttonColourId,
320 Colour(active_state_b_ ? Colors::helm_tertiary_accent : Colors::Color_ff464646));
321 ab_comparison_->setTooltip(active_state_b_
322 ? TRANS("State B active - click to switch to State A")
323 : TRANS("State A active - click to switch to State B"));
324 ab_comparison_->repaint();
325}
