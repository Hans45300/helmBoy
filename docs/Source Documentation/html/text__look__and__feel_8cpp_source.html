37void TextLookAndFeel::drawRotarySlider(Graphics& g, int x, int y, int width, int height,
38 float slider_t, float start_angle, float end_angle,
39 Slider& slider) {
40 float text_percentage = 0.7f;
41 static const int mod_buffer = 5;
42 Rectangle<int> text_bounds(x + mod_buffer, y, width - 2 * mod_buffer, height);
43
44 bool active = true;
45 SynthSlider* s_slider = dynamic_cast<SynthSlider*>(&slider);
46 if (s_slider) {
47 active = s_slider->isActive();
48 if (s_slider->getName() == "osc_1_unison_detune" ||
49 s_slider->getName() == "osc_2_unison_detune")
50 text_percentage = 0.5f;
51 }
52
53 g.setColour(Colors::text_background);
54 g.fillRect(x + mod_buffer, y, width - 2 * mod_buffer, height);
55
56 if (active)
57 g.setColour(Colors::text_background_highlighted);
58 else
59 g.setColour(Colors::text_background);
60 g.drawRect(slider.getLocalBounds());
61
62 if (active)
63 g.setColour(Colours::white);
64 else
65 g.setColour(Colors::text_button_normal);
66
67 g.setFont(Fonts::instance()->monospace().withPointHeight(height * text_percentage));
68 g.drawFittedText(slider.getTextFromValue(slider.getValue()), text_bounds,
69 Justification::centred, 1, 0.7f);
70}

72void TextLookAndFeel::drawToggleButton(Graphics& g, ToggleButton& button,
73 bool hover, bool is_down) {
74 static const float text_percentage = 0.7f;
75
76 if (button.getToggleState())
77 g.setColour(Colour(0xffffc400));
78 else
79 g.setColour(Colour(0xff313131));
80 g.fillRect(button.getLocalBounds());
81
82 g.setColour(Colours::white);
83 int height = button.getHeight();
84 g.setFont(Fonts::instance()->monospace().withPointHeight(height * text_percentage));
85 g.drawText(button.getButtonText(), 0, 0,
86 button.getWidth(), button.getHeight(), Justification::centred);
87
88 g.setColour(Colour(0xff565656));
89 g.drawRect(button.getLocalBounds());
90
91 if (is_down) {
92 g.setColour(Colour(0x11000000));
93 g.fillRect(button.getLocalBounds());
94 }
95 else if (hover) {
96 g.setColour(Colour(0x11ffffff));
97 g.fillRect(button.getLocalBounds());
98 }
99}
