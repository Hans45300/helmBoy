41 // An output port from the Processor.
42 struct Output {
43 Output(int size = MAX_BUFFER_SIZE) {
44 if (size <= 0)
45 size = 1;
46
47 owner = 0;
48 #if MOPO_SIMD_AVAILABLE && defined(_MSC_VER)
49 // AVX2 requires 32-byte alignment; new[] only guarantees 16-byte on MSVC x64.
50 // Misaligned vmovdqa causes a #GP exception (silent __fastfail) during init.
51 buffer = static_cast<mopo_float*>(_aligned_malloc(size * sizeof(mopo_float), 32));
52 #elif MOPO_SIMD_AVAILABLE
53 buffer = static_cast<mopo_float*>(std::aligned_alloc(32, size * sizeof(mopo_float)));
54 #else
55 buffer = new mopo_float[size];
56 #endif
57
58 if (buffer == nullptr)
59 throw std::bad_alloc();
60
61 buffer_size = size;
62 clearBuffer();
63 clearTrigger();
64 }
65
66 virtual ~Output() {
67 #if MOPO_SIMD_AVAILABLE && defined(_MSC_VER)
68 _aligned_free(buffer);
69 #elif MOPO_SIMD_AVAILABLE
70 std::free(buffer);
71 #else
72 delete[] buffer;
73 #endif
74 }

76 void trigger(mopo_float value, int offset = 0) {
77 triggered = true;
78 trigger_offset = offset;
79 trigger_value = value;
80 }
81
82 void clearTrigger() {
83 triggered = false;
84 trigger_offset = 0;
85 trigger_value = 0.0;
86 }
87
88 void clearBuffer() {
89 VECTORIZE_LOOP
90 for (int i = 0; i < buffer_size; ++i)
91 buffer[i] = 0.0;
92 }
93
94 mopo_float* buffer;
95 Processor* owner;
96
97 int buffer_size;
98 bool triggered;
99 int trigger_offset;
100 mopo_float trigger_value;
101 };
