38 const mopo_float MIN_LOOKUP_CENTS = -100.0;
39 const mopo_float MAX_LOOKUP_CENTS = 100.0;
40 const mopo_float CENTS_RANGE = MAX_LOOKUP_CENTS - MIN_LOOKUP_CENTS;
41 const int DETUNE_LOOKUP_RESOLUTION = 2046;
42 } // namespace
43
44 class DetuneLookupSingleton {
45 public:
46 DetuneLookupSingleton() {
47 for (auto i = 0; i < DETUNE_LOOKUP_RESOLUTION + 2; ++i) {
48 mopo_float t = (1.0 * i) / DETUNE_LOOKUP_RESOLUTION;
49 mopo_float cents = utils::interpolate(MIN_LOOKUP_CENTS, MAX_LOOKUP_CENTS, t);
50 detune_lookup_[i] = utils::centsToRatio(cents);
51 }
52 }
53
54 mopo_float detuneLookup(mopo_float cents) const {
55 mopo_float t = (cents - MIN_LOOKUP_CENTS) / CENTS_RANGE;
56 mopo_float index = DETUNE_LOOKUP_RESOLUTION * utils::clamp(t, 0.0, 1.0);
57 int int_index = index;
58 mopo_float fraction = index - int_index;
59
60 return utils::interpolate(detune_lookup_[int_index],
61 detune_lookup_[int_index + 1], fraction);
62 }
63
64 private:
65 mopo_float detune_lookup_[DETUNE_LOOKUP_RESOLUTION + 2];
66 };
