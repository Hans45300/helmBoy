38 void SwitchModulationProcessor::process() {
39 bool base_active = input(kBase)->source != &Processor::null_source_ && input(kBase)->at(0) != 0.0;
40 bool toggled = false;
41
42 int toggle_count = std::max(0, (numInputs() - kNumFixedInputs) / 2);
43 ensureToggleStorageSize(toggle_count);
44
45 int toggle_index = 0;
46 for (int input_index = kNumFixedInputs; input_index + 1 < numInputs(); input_index += 2, ++toggle_index) {
47 if (input(input_index)->source == &Processor::null_source_ ||
48 input(input_index + 1)->source == &Processor::null_source_) {
49 toggle_states_[toggle_index] = false;
50 hold_counters_[toggle_index] = 0;
51 continue;
52 }
53
54 bool toggle_active = updateToggleState(toggle_index,
55 input(input_index)->at(0),
56 input(input_index + 1)->at(0));
57 toggled ^= toggle_active;
58 }
59
60 output()->buffer[0] = (base_active != toggled) ? 1.0 : 0.0;
61 output()->clearTrigger();
62 }

64 bool SwitchModulationProcessor::updateToggleState(int toggle_index,
65 mopo_float source_value,
66 mopo_float signed_threshold) {
67 mopo_float threshold = utils::clamp(std::abs(signed_threshold), 0.0, 1.0);
68 mopo_float magnitude = std::abs(source_value);
69 mopo_float low = utils::clamp(threshold - kHysteresisWidth, 0.0, 1.0);
70 mopo_float high = utils::clamp(threshold + kHysteresisWidth, 0.0, 1.0);
71
72 bool current_state = toggle_states_[toggle_index] != 0;
73 bool desired_state = current_state;
74
75 if (signed_threshold < 0.0) {
76 // Negative threshold: active inside the center band.
77 if (current_state)
78 desired_state = magnitude < high;
79 else
80 desired_state = magnitude <= low;
81 }
82 else {
83 // Positive threshold: active outside the center band.
84 if (current_state)
85 desired_state = magnitude > low;
86 else
87 desired_state = magnitude >= high;
88 }
89
90 int& hold_counter = hold_counters_[toggle_index];
91 if (hold_counter > 0)
92 --hold_counter;
93
94 if (desired_state != current_state && hold_counter == 0) {
95 toggle_states_[toggle_index] = desired_state;
96 hold_counter = kMinHoldTicks;
97 }
98
99 return toggle_states_[toggle_index] != 0;
100 }

102 void SwitchModulationProcessor::ensureToggleStorageSize(int toggle_count) {
103 if (toggle_count <= static_cast<int>(toggle_states_.size()))
104 return;
105
106 toggle_states_.resize(toggle_count, false);
107 hold_counters_.resize(toggle_count, 0);
108 }
