60 void Clamp::process() {
61 MOPO_ASSERT(inputMatchesBufferSize());
62
63 mopo_float* dest = output()->buffer;
64 const mopo_float* source = input()->source->buffer;
65#ifdef USE_APPLE_ACCELERATE
66 vDSP_vclipD(source, 1, &min_, &max_, dest, 1, buffer_size_);
67#else
68 #include <xsimd/xsimd.hpp>
69 using batch = xsimd::batch<mopo_float>;
70 constexpr std::size_t simd_size = batch::size;
71 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
72 batch minv = batch(min_);
73 batch maxv = batch(max_);
74 int i = 0;
75 for (; i < simd_end; i += simd_size) {
76 batch s = batch::load_unaligned(&source[i]);
77 batch clipped = xsimd::min(xsimd::max(s, minv), maxv);
78 clipped.store_unaligned(&dest[i]);
79 }
80 for (; i < buffer_size_; ++i)
81 dest[i] = utils::clamp(source[i], min_, max_);
82#endif
83 processTriggers();
84 }

133 void Add::process() {
134 MOPO_ASSERT(inputMatchesBufferSize(0));
135 MOPO_ASSERT(inputMatchesBufferSize(1));
136
137 mopo_float* dest = output()->buffer;
138 const mopo_float* source_left = input(0)->source->buffer;
139 const mopo_float* source_right = input(1)->source->buffer;
140
141 #include <xsimd/xsimd.hpp>
142 using batch = xsimd::batch<mopo_float>;
143 constexpr std::size_t simd_size = batch::size;
144 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
145 int i = 0;
146 for (; i < simd_end; i += simd_size) {
147 batch l = batch::load_unaligned(&source_left[i]);
148 batch r = batch::load_unaligned(&source_right[i]);
149 (l + r).store_unaligned(&dest[i]);
150 }
151 for (; i < buffer_size_; ++i)
152 dest[i] = source_left[i] + source_right[i];
153
154 processTriggers();

196 void Interpolate::process() {
197 MOPO_ASSERT(inputMatchesBufferSize(0));
198 MOPO_ASSERT(inputMatchesBufferSize(1));
199 MOPO_ASSERT(inputMatchesBufferSize(2));
200
201 mopo_float* dest = output()->buffer;
202 const mopo_float* from = input(static_cast<int>(Interpolate::Inputs::From))->source->buffer;
203 const mopo_float* to = input(static_cast<int>(Interpolate::Inputs::To))->source->buffer;
204 const mopo_float* fractional = input(static_cast<int>(Interpolate::Inputs::Fractional))->source->buffer;
205
206 #include <xsimd/xsimd.hpp>
207 using batch = xsimd::batch<mopo_float>;
208 constexpr std::size_t simd_size = batch::size;
209 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
210 int i = 0;
211 for (; i < simd_end; i += simd_size) {
212 batch f = batch::load_unaligned(&from[i]);
213 batch t = batch::load_unaligned(&to[i]);
214 batch frac = batch::load_unaligned(&fractional[i]);
215 (f + frac * (t - f)).store_unaligned(&dest[i]);
216 }
217 for (; i < buffer_size_; ++i)
218 dest[i] = from[i] + fractional[i] * (to[i] - from[i]);
219
220 processTriggers();
221 }

229 void VariableAdd::process() {
230#if DEBUG
231 for (int i = 0; i < inputs_->size(); ++i)
232 MOPO_ASSERT(inputMatchesBufferSize(i));
233#endif
234
235 mopo_float* dest = output()->buffer;
236
237 if (isControlRate()) {
238 dest[0] = 0.0;
239
240 int num_inputs = inputs_->size();
241 for (int i = 0; i < num_inputs; ++i)
242 dest[0] += input(i)->at(0);
243 }
244 else {
245 utils::zeroBuffer(std::span<double>(dest, buffer_size_));
246
247 int num_inputs = inputs_->size();
248 for (int i = 0; i < num_inputs; ++i) {
249 if (input(i)->source != &Processor::null_source_) {
250#ifdef USE_APPLE_ACCELERATE
251 vDSP_vaddD(input(i)->source->buffer, 1,
252 output()->buffer, 1,
253 output()->buffer, 1, buffer_size_);
254#else
255 const mopo_float* source = input(i)->source->buffer;
256
257 VECTORIZE_LOOP
258 for (int s = 0; s < buffer_size_; ++s)
259 dest[s] += source[s];
260#endif
261 }
262 }
263 }
264 processTriggers();

319 void LinearSmoothBuffer::process() {
320 mopo_float new_value = input(static_cast<int>(LinearSmoothBuffer::Inputs::Value))->source->buffer[0];
321 mopo_float* dest = output()->buffer;
322
323 if (input(static_cast<int>(LinearSmoothBuffer::Inputs::Trigger))->source->triggered) {
324 int trigger_samples = input(static_cast<int>(LinearSmoothBuffer::Inputs::Trigger))->source->trigger_offset;
325 int i = 0;
326
327 mopo_float val = last_value_;
328 VECTORIZE_LOOP
329 for (; i < trigger_samples; ++i)
330 dest[i] = val;
331
332 val = new_value;
333
334 VECTORIZE_LOOP
335 for (; i < buffer_size_; ++i)
336 dest[i] = val;
337 }
338 else if (last_value_ == new_value &&
339 new_value == output()->buffer[0] &&
340 new_value == output()->buffer[buffer_size_ - 1] &&
341 (buffer_size_ <= 1 || new_value == output()->buffer[buffer_size_ - 2])) {
342 last_value_ = new_value;
343 return;
344 }
345 else {
346 mopo_float inc = (new_value - last_value_) / buffer_size_;
347 mopo_float val = last_value_ + inc;
348
349 VECTORIZE_LOOP
350 for (int i = 0; i < buffer_size_; ++i)
351 dest[i] = val + i * inc;
352 }
353
354 last_value_ = new_value;
355 processTriggers();
356 }
