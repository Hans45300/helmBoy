37 FormantManager::FormantManager(int num_formants) : ProcessorRouter(0, 0) {
38 Bypass* audio_input = new Bypass();
39 cr::Bypass* reset_input = new cr::Bypass();
40
41 registerInput(audio_input->input(), static_cast<int>(Inputs::Audio));
42 registerInput(reset_input->input(), static_cast<int>(Inputs::Reset));
43
44 addProcessor(audio_input);
45 addProcessor(reset_input);
46
47 VariableAdd* total = new VariableAdd(num_formants);
48 for (int i = 0; i < num_formants; ++i) {
49 BiquadFilter* formant = new BiquadFilter();
50 formant->plug(audio_input, static_cast<int>(mopo::BiquadFilter::Inputs::Audio));
51 formant->plug(reset_input, static_cast<int>(mopo::BiquadFilter::Inputs::Reset));
52 formants_.push_back(formant);
53
54 addProcessor(formant);
55 total->plugNext(formant);
56 }
57
58 addProcessor(total);
59 registerOutput(total->output());
60 }

62 std::complex<mopo_float> FormantManager::getResponse(mopo_float frequency) {
63 std::complex<mopo_float> total;
64 for (int i = 0; i < formants_.size(); ++i)
65 total += formants_[i]->getResponse(frequency);
66
67 return total;
