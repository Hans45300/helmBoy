50void WaveViewer::paint(juce::Graphics &g) {
51 g.drawImageWithin(background_,
52 0, 0, getWidth(), getHeight(), RectanglePlacement());
53
54 if (wave_phase_) {
55 if (phase_ >= 0.0 && phase_ < 1.0) {
56 float x = phaseToX(phase_);
57 g.setColour(Colour(Colors::Color_33ffffff));
58 g.fillRect(x - 0.5f, 0.0f, 1.0f, (float)getHeight());
59
60 float y = PADDING + (getHeight() - 2 * PADDING) * (1.0f - amp_) / 2.0f;
61
62 g.setColour(Colors::modulation);
63 g.fillEllipse(x - MARKER_WIDTH / 2.0f, y - MARKER_WIDTH / 2.0f,
64 MARKER_WIDTH, MARKER_WIDTH);
65 g.setColour(Colour(Colors::Color_ff000000));
66 g.fillEllipse(x - MARKER_WIDTH / 4.0f, y - MARKER_WIDTH / 4.0f,
67 MARKER_WIDTH / 2.0f, MARKER_WIDTH / 2.0f);
68 }
69 }
70}

72void WaveViewer::paintBackground(Graphics& g) {
73 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
74
75 g.fillAll(Colour(Colors::Color_ff424242));
76
77 g.setColour(Colour(Colors::Color_ff4a4a4a));
78 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
79 g.drawLine(x, 0, x, getHeight());
80 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
81 g.drawLine(0, y, getWidth(), y);
82
83 shadow.drawForPath(g, wave_path_);
84
85 g.setColour(Colors::graph_fill);
86 g.fillPath(wave_path_);
87
88 if (is_control_rate_)
89 g.setColour(Colors::modulation);
90 else
91 g.setColour(Colors::audio);
92
93 float line_width = 1.5f * getRatio();
94 PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
95 g.strokePath(wave_path_, stroke);
96}

120void WaveViewer::timerCallback() {
121 if (wave_phase_) {
122 float phase = wave_phase_->buffer[0];
123 amp_ = wave_amp_->buffer[0];
124 if (phase != phase_) {
125 float last_x = phaseToX(phase_);
126 float new_x = phaseToX(phase);
127 phase_ = phase;
128 repaint(last_x - MARKER_WIDTH / 2.0f - 1, 0.0, MARKER_WIDTH + 2, getHeight());
129 repaint(new_x - MARKER_WIDTH / 2.0f - 1, 0.0, MARKER_WIDTH + 2, getHeight());
130 }

150void WaveViewer::drawRandom() {
151 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
152 float draw_width = getWidth();
153 float padding = getRatio() * PADDING;
154 float draw_height = getHeight() - 2.0f * padding;
155
156 wave_path_.startNewSubPath(0, getHeight() / 2.0f);
157 for (int i = 0; i < NOISE_RESOLUTION; ++i) {
158 float t1 = (1.0f * i) / NOISE_RESOLUTION;
159 float t2 = (1.0f + i) / NOISE_RESOLUTION;
160 float val = amplitude * random_values[i];
161 wave_path_.lineTo(t1 * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
162 wave_path_.lineTo(t2 * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
163 }
164
165 wave_path_.lineTo(getWidth(), getHeight() / 2.0f);
166}

168void WaveViewer::drawSmoothRandom() {
169 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
170 float draw_width = getWidth();
171 float padding = getRatio() * PADDING;
172 float draw_height = getHeight() - 2.0f * padding;
173
174 float start_val = amplitude * random_values[0];
175 wave_path_.startNewSubPath(-50, getHeight() / 2.0f);
176 wave_path_.lineTo(0, padding + draw_height * ((1.0f - start_val) / 2.0f));
177 for (int i = 1; i < resolution_ - 1; ++i) {
178 float t = (1.0f * i) / resolution_;
179 float phase = t * (NOISE_RESOLUTION - 1);
180 int index = (int)phase;
181 phase = mopo::PI * (phase - index);
182 float val = amplitude * mopo::utils::interpolate(random_values[index],
183 random_values[index + 1],
184 0.5f - cosf(phase) / 2.0f);
185 wave_path_.lineTo(t * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
186 }
187
188 float end_val = amplitude * random_values[NOISE_RESOLUTION - 1];
189 wave_path_.lineTo(getWidth(), padding + draw_height * ((1.0f - end_val) / 2.0f));
190 wave_path_.lineTo(getWidth() + 50, getHeight() / 2.0f);
191
192}

194void WaveViewer::resetWavePath() {
195 if (!background_.isValid())
196 return;
197
198 wave_path_.clear();
199
200 if (wave_slider_ == nullptr)
201 return;
202
203 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
204 float draw_width = getWidth();
205 float padding = getRatio() * PADDING;
206 float draw_height = getHeight() - 2.0f * padding;
207
208 mopo::Wave::Type type = static_cast<mopo::Wave::Type>(static_cast<int>(wave_slider_->getValue()));
209
210 if (type < mopo::Wave::Type::WhiteNoise) {
211 wave_path_.startNewSubPath(0, getHeight() / 2.0f);
212 for (int i = 1; i < resolution_ - 1; ++i) {
213 float t = (1.0f * i) / resolution_;
214 float val = amplitude * mopo::Wave::wave(type, t);
215 wave_path_.lineTo(t * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
216 }
217
218 wave_path_.lineTo(getWidth(), getHeight() / 2.0f);
219 }
220 else if (type == mopo::Wave::Type::WhiteNoise)
221 drawRandom();
222 else
223 drawSmoothRandom();
224
225 auto *display = Desktop::getInstance().getDisplays().getPrimaryDisplay();
226 jassert(display != nullptr);
227
228 float scale = display->scale;
229 Graphics g(background_);
230 g.addTransform(AffineTransform::scale(scale, scale));
231 paintBackground(g);
232
233 repaint();
234}

241void WaveViewer::showRealtimeFeedback(bool show_feedback) {
242 if (show_feedback) {
243 if (wave_phase_ == nullptr) {
244 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
245 if (parent) {
246 wave_amp_ = parent->getSynth()->getModSource(getName().toStdString());
247 wave_phase_ = parent->getSynth()->getModSource(getName().toStdString() + "_phase");
248 startTimerHz(FRAMES_PER_SECOND);
249 }
250 }
251 }
252 else {
253 wave_phase_ = nullptr;
254 stopTimer();
