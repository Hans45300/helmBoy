65void FileListBoxModel::rescanFiles(const Array<File>& folders,
66 String search,
67 bool find_files) {
68 static const FileSorterAscending file_sorter;
69 files_.clear();
70
71 for (File folder : folders) {
72 if (folder.isDirectory()) {
73 Array<File> child_folders;
74 if (find_files)
75 folder.findChildFiles(child_folders, File::findFiles, false, search);
76 else
77 folder.findChildFiles(child_folders, File::findDirectories, false);
78
79 child_folders.sort(file_sorter);
80 files_.addArray(child_folders);
81 }
82 }
83}
