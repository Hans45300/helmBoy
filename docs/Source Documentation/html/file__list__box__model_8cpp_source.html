32void FileListBoxModel::paintListBoxItem(int row_number, Graphics& g,
33 int width, int height, bool selected) {
34 g.fillAll(Colour(Colors::background));
35 g.setColour(Colour(Colors::tab_body));
36 if (selected) {
37 g.fillAll(Colour(Colors::tab_body));
38 g.setColour(Colors::audio);
39 }
40
41 const File& row_file = files_[row_number];
42 String display_name = row_file.getFileNameWithoutExtension();
43 if (show_relative_directory_paths_ && row_file.isDirectory()) {
44 File root = relative_root_.isDirectory() ? relative_root_ : LoadSave::getBankDirectory();
45 display_name = row_file.getRelativePathFrom(root).replaceCharacter('\\', '/');
46 }
47
48 g.setFont(Fonts::instance()->monospace().withPointHeight(12.0f));
49 g.drawText(display_name,
50 5, 0, width, height,
51 Justification::centredLeft, true);
52
53 g.setColour(Colour(Colors::shadow_dark));
54 g.fillRect(0.0f, height - 1.0f, 1.0f * width, 1.0f);
55}

78void FileListBoxModel::rescanFiles(const Array<File>& folders,
79 String search,
80 bool find_files,
81 bool recursive) {
82 static const FileSorterAscending file_sorter;
83 files_.clear();
84 std::set<String> seen_paths;
85
86 for (File folder : folders) {
87 if (folder.isDirectory()) {
88 Array<File> child_folders;
89 if (find_files)
90 LoadSave::findPatchFiles(folder, child_folders, recursive, search);
91 else
92 folder.findChildFiles(child_folders, File::findDirectories, recursive);
93
94 child_folders.sort(file_sorter);
95 for (const auto& child : child_folders) {
96 String path = child.getFullPathName();
97 if (seen_paths.insert(path).second)
98 files_.add(child);
99 }
100 }
101 }
102}
