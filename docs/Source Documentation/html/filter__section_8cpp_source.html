48FilterSection::FilterSection(String name) : SynthSection(name) {
49 addSlider((filter_shelf_ = std::make_unique<FilterSelector>("filter_shelf")).get());
50 filter_shelf_->setSliderStyle(Slider::LinearBar);
51 filter_shelf_->setStringLookup(mopo::strings::filter_shelves);
52 filter_shelf_->setVisible(false);
53 filter_shelf_->setPopupPlacement(BubbleComponent::above);
54
55 addSlider((cutoff_ = std::make_unique<SynthSlider>("cutoff")).get());
56 cutoff_->setSliderStyle(Slider::LinearBar);
57 cutoff_->setPopupPlacement(BubbleComponent::below, 0);
58
59 addSlider((blend_ = std::make_unique<SynthSlider>("filter_blend")).get());
60 blend_->snapToValue(true, 1.0);
61 blend_->setBipolar(true);
62 blend_->setSliderStyle(Slider::LinearBar);
63 blend_->setPopupPlacement(BubbleComponent::above);
64
65 addSlider((resonance_ = std::make_unique<SynthSlider>("resonance")).get());
66 resonance_->setSliderStyle(Slider::LinearBarVertical);
67 resonance_->setPopupPlacement(BubbleComponent::right, 0);
68
69 addAndMakeVisible((filter_response_ = std::make_unique<FilterResponse>(300)).get());
70 filter_response_->setCutoffSlider(cutoff_.get());
71 filter_response_->setResonanceSlider(resonance_.get());
72 filter_response_->setFilterBlendSlider(blend_.get());
73 filter_response_->setFilterShelfSlider(filter_shelf_.get());
74
75 addSlider((drive_ = std::make_unique<SynthSlider>("filter_drive")).get());
76 drive_->setSliderStyle(Slider::RotaryHorizontalVerticalDrag);
77
78 addSlider((fil_env_depth_ = std::make_unique<SynthSlider>("fil_env_depth")).get());
79 fil_env_depth_->setSliderStyle(Slider::RotaryHorizontalVerticalDrag);
80 fil_env_depth_->setBipolar();
81
82 addSlider((keytrack_ = std::make_unique<SynthSlider>("keytrack")).get());
83 keytrack_->setSliderStyle(Slider::LinearBar);
84 keytrack_->snapToValue(true, 0.0);
85 keytrack_->setBipolar();
86
87 addSlider((filter_style_ = std::make_unique<TextSlider>("filter_style")).get());
88 filter_style_->setSliderStyle(Slider::LinearBar);
89 filter_style_->setStringLookup(mopo::strings::filter_style);
90 filter_style_->setShortStringLookup(mopo::strings::filter_style_short);
91 filter_style_->setPopupPlacement(BubbleComponent::above);
92
93 addButton((filter_on_ = std::make_unique<SynthButton>("filter_on")).get());
94 setActivator(filter_on_.get());

114void FilterSection::paintBackground(Graphics& g) {
115 static const PathStrokeType stroke(1.000f, PathStrokeType::curved, PathStrokeType::rounded);
116 static const DropShadow component_shadow(Colour(Colors::Color_bb000000), size_ratio_ * 4, Point<int>(0, 0));
117 SynthSection::paintBackground(g);
118
119 g.setColour(Colors::control_label_text);
120 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(size_ratio_ * 10.0f));
121
122 drawTextForComponent(g, TRANS("ENV DEPTH"), fil_env_depth_.get());
123 drawTextForComponent(g, TRANS("KEY TRACK"), keytrack_.get());
124 drawTextForComponent(g, TRANS("DRIVE"), drive_.get());
125
126 if (blend_->isVisible()) {
127 g.setColour(Colour(Colors::Color_ffaaaaaa));
128 g.strokePath(low_pass_, stroke);
129 g.strokePath(high_pass_, stroke);
130 }
131
132 component_shadow.drawForRectangle(g, filter_response_->getBounds());
133}

135void FilterSection::resized() {
136 int title_width = getTitleWidth();
137 int knob_width = getStandardKnobSize();
138 int knob_section_width = size_ratio_ * KNOB_SECTION_WIDTH;
139 int slider_width = size_ratio_ * SLIDER_WIDTH;
140 int style_slider_width = size_ratio_ * STYLE_SLIDER_WIDTH;
141 int style_label_width = size_ratio_ * STYLE_LABEL_WIDTH;
142 int style_slider_padding = size_ratio_ * STYLE_SLIDER_PADDING;
143 int filter_type_width = size_ratio_ * FILTER_TYPE_WIDTH;
144 int keytrack_top_padding = size_ratio_ * KEYTRACK_TOP_PADDING;
145 int keytrack_width = size_ratio_ * KEYTRACK_WIDTH;
146 int keytrack_height = size_ratio_ * KEYTRACK_HEIGHT;
147 filter_on_->setBounds(size_ratio_ * 2.0f, 0, title_width, title_width);
148
149 int response_width = getWidth() - knob_section_width - slider_width;
150 int response_height = getHeight() - slider_width - filter_type_width - title_width;
151 int knob_center_x = getWidth() - knob_section_width / 2;
152
153 int style_slider_remove = style_slider_width + style_slider_padding;
154 int blend_buffer = size_ratio_ * 2;
155 blend_->setBounds(style_label_width, title_width + blend_buffer,
156 getWidth() - knob_section_width - 2 * style_label_width - style_slider_remove,
157 filter_type_width - 2 * blend_buffer);
158 filter_shelf_->setBounds(0, title_width, getWidth() - knob_section_width - style_slider_remove,
159 filter_type_width);
160
161 cutoff_->setBounds(0, getHeight() - slider_width, response_width, slider_width);
162 resonance_->setBounds(response_width, title_width + filter_type_width, slider_width, response_height);
163 filter_response_->setBounds(0, title_width + filter_type_width, response_width, response_height);
164
165 int knob_vertical = knob_width + size_ratio_ * 14;
166 int keytrack_vertical = keytrack_top_padding + keytrack_height + size_ratio_ * 14;
167 int space = (getHeight() - title_width - 2 * knob_vertical - keytrack_vertical) / 4;
168
169 drive_->setBounds(knob_center_x - knob_width / 2, title_width + space,
170 knob_width, knob_width);
171 fil_env_depth_->setBounds(knob_center_x - knob_width / 2,
172 title_width + knob_vertical + 2 * space,
173 knob_width, knob_width);
174 keytrack_->setBounds(knob_center_x - keytrack_width / 2,
175 title_width + 2 * knob_vertical + 3 * space + keytrack_top_padding,
176 keytrack_width, keytrack_height);
177 filter_style_->setBounds(filter_shelf_->getRight() + style_slider_padding, title_width,
178 style_slider_width, filter_type_width);
179
180 int style_label_padding_x = size_ratio_ * STYLE_LABEL_PADDING_X;
181 int style_label_padding_y = size_ratio_ * STYLE_LABEL_PADDING_Y;
182 resizeLowPass(style_label_padding_x, title_width + style_label_padding_y,
183 style_label_width - 2 * style_label_padding_x,
184 filter_type_width - 2 * style_label_padding_y);
185 resizeHighPass(blend_->getRight() + style_label_padding_x, title_width + style_label_padding_y,
186 style_label_width - 2 * style_label_padding_x,
187 filter_type_width - 2 * style_label_padding_y);
188
189 SynthSection::resized();
190}

211void FilterSection::resetResponse() {
212 float value = filter_style_->getValue();
213 mopo::StateVariableFilter::Styles style =
214 static_cast<mopo::StateVariableFilter::Styles>(static_cast<int>(value));
215 filter_response_->setStyle(style);
216
217 bool pass_family = style == mopo::StateVariableFilter::Styles::dB12 ||
218 style == mopo::StateVariableFilter::Styles::dB24;
219 blend_->setVisible(pass_family);
220 filter_shelf_->setVisible(style == mopo::StateVariableFilter::Styles::Shelf);
221}

223void FilterSection::resizeLowPass(float x, float y, float width, float height) {
224 low_pass_.clear();
225 low_pass_.startNewSubPath(x, y + height / 2.0f);
226 low_pass_.lineTo(x + width / 2.0f, y + height / 2.0f);
227 low_pass_.lineTo(x + 4.0f * width / 6.0f, y + height / 4.0f);
228 low_pass_.lineTo(x + width, y + height);
229}

231void FilterSection::resizeHighPass(float x, float y, float width, float height) {
232 high_pass_.clear();
233 high_pass_.startNewSubPath(x + width, y + height / 2.0f);
234 high_pass_.lineTo(x + width / 2.0f, y + height / 2.0f);
235 high_pass_.lineTo(x + 2.0f * width / 6.0f, y + height / 4.0f);
236 high_pass_.lineTo(x, y + height);
237}
