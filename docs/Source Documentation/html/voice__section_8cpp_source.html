77void VoiceSection::resized() {
78 int knob_width = getStandardKnobSize();
79 int text_height = size_ratio_ * TEXT_HEIGHT;
80
81 float space_x = (getWidth() - (3.0f * knob_width)) / 4.0f;
82 float space_y = (getHeight() - (knob_width + text_height)) / 2.0f;
83
84 polyphony_->setBounds(space_x, space_y, knob_width, knob_width);
85 pitch_bend_->setBounds(knob_width + 2 * space_x, space_y, knob_width, knob_width);
86 velocity_track_->setBounds(2 * knob_width + 3 * space_x, space_y, knob_width, knob_width);
87
88 SynthSection::resized();
89}
