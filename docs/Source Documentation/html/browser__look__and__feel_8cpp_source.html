36void BrowserLookAndFeel::drawToggleButton(Graphics& g, ToggleButton& button,
37 bool hover, bool is_down) {
38 static const float text_percentage = 0.7f;
39 static const float max_text_height = 16.0f;
40
41 if (button.getToggleState()) [[unlikely]]
42 g.setColour(Colour(0xffffab00));
43 else [[likely]]
44 g.setColour(Colour(0xff313131));
45 g.fillRect(button.getLocalBounds());
46
47 if (button.getToggleState()) [[unlikely]]
48 g.setColour(Colours::white);
49 else [[likely]]
50 g.setColour(Colour(0xffbbbbbb));
51
52 int height = button.getHeight();
53 float text_height = std::min(max_text_height, height * text_percentage);
54 g.setFont(Fonts::instance()->monospace().withPointHeight(text_height));
55 g.drawText(button.getButtonText(), 0, 0,
56 button.getWidth(), button.getHeight(), Justification::centred);
57
58 g.setColour(Colour(0xff565656));
59 g.drawRect(button.getLocalBounds());
60
61 if (is_down) [[unlikely]] {
62 g.setColour(Colour(0x11000000));
63 g.fillRect(button.getLocalBounds());
64 }
65 else if (hover) [[likely]] {
66 g.setColour(Colour(0x11ffffff));
67 g.fillRect(button.getLocalBounds());
68 }
69}

71Font BrowserLookAndFeel::getTextButtonFont(TextButton& button, int button_height) {
72 static const float percent = 0.55f;
73 int min = std::min(button_height, button.getWidth());
74 return Fonts::instance()->monospace().withPointHeight(min * percent);
