50 void LadderFilter::process() {
51 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
52
53 mopo_float cutoff = utils::clamp(input(static_cast<int>(Inputs::Cutoff))->at(0), MIN_CUTTOFF, sample_rate_);
54
55 mopo_float g = g_;
56 computeCoefficients(cutoff);
57 mopo_float resonance = utils::clamp(resonance_multiple_ * input(static_cast<int>(Inputs::Resonance))->at(0) / 4.0,
58 MIN_RESONANCE, MAX_RESONANCE);
59 mopo_float drive = -input(static_cast<int>(Inputs::Drive))->at(0);
60 mopo_float delta_drive = (drive - current_drive_) / buffer_size_;
61
62 mopo_float delta_resonance = (resonance - current_resonance_) / buffer_size_;
63 mopo_float delta_g = (g_ - g) / buffer_size_;
64
65
66 auto audio_buffer = std::span<const mopo_float>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
67 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
68 double two_sr = sample_rate_ * 2.0;
69
70 // Pr�paration SIMD du buffer d'entr�e (audio * drive)
71 std::vector<mopo_float> driven_buffer(buffer_size_);
72 using batch = xsimd::batch<mopo_float>;
73 constexpr std::size_t simd_size = batch::size;
74 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
75 int i = 0;
76 for (; i < simd_end; i += simd_size) {
77 batch audio = batch::load_unaligned(&audio_buffer[i]);
78 batch drv = batch(current_drive_);
79 batch result = audio * drv;
80 result.store_unaligned(&driven_buffer[i]);
81 }
82 for (; i < buffer_size_; ++i) {
83 driven_buffer[i] = audio_buffer[i] * current_drive_;
84 }
85 if (input(static_cast<int>(Inputs::Reset))->source->triggered &&
86 input(static_cast<int>(Inputs::Reset))->source->trigger_value == static_cast<mopo_float>(kVoiceReset)) [[unlikely]] {
87
88 int trigger_offset = input(static_cast<int>(Inputs::Reset))->source->trigger_offset;
89 int j = 0;
90 for (; j < trigger_offset; ++j) {
91 g += delta_g;
92 current_resonance_ += delta_resonance;
93 current_drive_ += delta_drive;
94 tick(j, dest.data(), driven_buffer.data(), g, current_resonance_, two_sr);
95 tick(j, dest.data(), driven_buffer.data(), g, current_resonance_, two_sr);
96 }
97
98 reset();
99 current_resonance_ = resonance;
100 current_drive_ = drive;
101
102 for (; j < buffer_size_; ++j) {
103 tick(j, dest.data(), driven_buffer.data(), g_, resonance, two_sr);
104 tick(j, dest.data(), driven_buffer.data(), g_, resonance, two_sr);
105 }
106 }
107 else {
108 for (int j = 0; j < buffer_size_; ++j) {
109 g += delta_g;
110 current_resonance_ += delta_resonance;
111 current_drive_ += delta_drive;
112 tick(j, dest.data(), driven_buffer.data(), g, current_resonance_, two_sr);
113 tick(j, dest.data(), driven_buffer.data(), g, current_resonance_, two_sr);
114 }
115 }
116
117 current_resonance_ = resonance;
118 current_drive_ = drive;
119 }

121 inline void LadderFilter::tick(int i, mopo_float* dest, const mopo_float* audio_buffer,
122 mopo_float g, mopo_float resonance, mopo_float two_sr) {
123 mopo_float audio = audio_buffer[i] * current_drive_;
124
125 mopo_float new_tan = utils::quickTanh((audio + resonance * v_[3]) / TWO_THERMAL_VOLTAGE);
126 mopo_float delta_v0 = -g * (new_tan + tanh_v_[0]);
127 v_[0] += (delta_v0 + delta_v_[0]) / two_sr;
128 delta_v_[0] = delta_v0;
129 tanh_v_[0] = utils::quickTanh(v_[0] / TWO_THERMAL_VOLTAGE);
130
131 mopo_float delta_v1 = g * (tanh_v_[0] - tanh_v_[1]);
132 v_[1] += (delta_v1 + delta_v_[1]) / two_sr;
133 delta_v_[1] = delta_v1;
134 tanh_v_[1] = utils::quickTanh(v_[1] / TWO_THERMAL_VOLTAGE);
135
136 mopo_float delta_v2 = g * (tanh_v_[1] - tanh_v_[2]);
137 v_[2] += (delta_v2 + delta_v_[2]) / two_sr;
138 delta_v_[2] = delta_v2;
139 tanh_v_[2] = utils::quickTanh(v_[2] / TWO_THERMAL_VOLTAGE);
140
141 mopo_float delta_v3 = g * (tanh_v_[2] - tanh_v_[3]);
142 v_[3] += (delta_v3 + delta_v_[3]) / two_sr;
143 delta_v_[3] = delta_v3;
144 tanh_v_[3] = utils::quickTanh(v_[3] / TWO_THERMAL_VOLTAGE);
145
146 dest[i] = v_[3];
147 }

149 void LadderFilter::computeCoefficients(mopo_float cutoff) {
150 MOPO_ASSERT(cutoff > 0.0);
151
152 mopo_float delta_phase = (mopo::PI * cutoff * 0.5) / sample_rate_;
153 resonance_multiple_ = 1.0 / (-1.273 * delta_phase * delta_phase + 3.5 * delta_phase + 0.7);
154 g_ = mopo::PI * TWO_THERMAL_VOLTAGE * cutoff * (1.0 - delta_phase) / (1.0 + delta_phase);
155 }

161 }
