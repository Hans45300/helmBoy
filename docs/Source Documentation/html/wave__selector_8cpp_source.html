30 template<size_t steps> requires (steps > 1)
31 void resizeSteps(Path& path, float x, float y, float width, float height) {
32 path.clear();
33 path.startNewSubPath(x, y + height / 2.0f);
34
35 float inc_x = width / steps;
36 float inc_y = height / (steps - 1);
37 for (int i = 0; i < steps; ++i) {
38 path.lineTo(x + i * inc_x, y + height - i * inc_y);
39 path.lineTo(x + (i + 1) * inc_x, y + height - i * inc_y);
40 }
41
42 path.lineTo(x + width, y + height / 2.0f);
43 }

45 template<size_t steps> requires (steps > 1)
46 void resizePyramid(Path& path, float x, float y, float width, float height) {
47 static int parts = 2 * (steps - 1);
48
49 path.clear();
50 path.startNewSubPath(x, y + height / 2.0f);
51
52 float inc_x = width / parts;
53 float inc_y = height / (steps - 1);
54 float cur_x = x + inc_x;
55
56 for (int i = 0; i < (steps - 1) / 2; ++i) {
57 path.lineTo(cur_x, y + height / 2.0f - i * inc_y);
58 path.lineTo(cur_x, y + height / 2.0f - (i + 1) * inc_y);
59 cur_x += inc_x;
60 }
61
62 for (int i = 0; i < steps - 1; ++i) {
63 path.lineTo(cur_x, y + i * inc_y);
64 path.lineTo(cur_x, y + (i + 1) * inc_y);
65 cur_x += inc_x;
66 }
67
68 for (int i = 0; i < (steps - 1) / 2; ++i) {
69 path.lineTo(cur_x, y + height - i * inc_y);
70 path.lineTo(cur_x, y + height - (i + 1) * inc_y);
71 cur_x += inc_x;
72 }
73
74 path.lineTo(x + width, y + height / 2.0f);
75 }

140void WaveSelector::resized() {
141 SynthSlider::resized();
142 cell_width = float(getWidth()) / getNumberOfTypesPerLine(getMinimum(), getMaximum());
143 type_width = cell_width - 2 * TYPE_PADDING_X;
144 type_height = getHeight() / 2 - 2 * TYPE_PADDING_Y;
145 int actual_type_number = 0;
146 resizeSin(getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number), type_width, type_height);
147 actual_type_number++;
148 resizeTriangle(getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number), type_width, type_height);
149 actual_type_number++;
150 resizeSquare(getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number), type_width, type_height);
151 actual_type_number++;
152 resizeDownSaw(getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number), type_width, type_height);
153 actual_type_number++;
154 resizeUpSaw(getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number), type_width, type_height);
155 actual_type_number++;
156 resizeSteps<3>(three_step_, getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number),
157 type_width, type_height);
158 actual_type_number++;
159 resizeSteps<4>(four_step_, getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number),
160 type_width, type_height);
161 actual_type_number++;
162 resizeSteps<8>(eight_step_, getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number),
163 type_width, type_height);
164 actual_type_number++;
165 resizePyramid<3>(three_pyramid_, getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number),
166 type_width, type_height);
167 actual_type_number++;
168 resizePyramid<5>(five_pyramid_, getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number),
169 type_width, type_height);
170 actual_type_number++;
171 resizePyramid<9>(nine_pyramid_, getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number),
172 type_width, type_height);
173 if (getMaximum() >= oscillator_wf_types)
174 {
175 actual_type_number++;
176 resizeSampleAndHold(getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number), type_width, type_height);
177 actual_type_number++;
178 resizeNoise(getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X, getHeightFromTypeNumber(actual_type_number), type_width, type_height);
179 if (getMaximum() >= 13) {
180 actual_type_number++;
181 resizeSmoothRandom(getCellOffsetFromTypeNumber(actual_type_number) * cell_width + TYPE_PADDING_X,
182 getHeightFromTypeNumber(actual_type_number), type_width, type_height);
183 }
184 }

215void WaveSelector::mouseEvent(const juce::MouseEvent &e)
216{
217 //Choix de la waveform en fonction de la position du clic
218 float x = e.getPosition().getX();
219 float y = e.getPosition().getY();
220 int index;
221 index = static_cast<int>(getMinimum()) +
222 ((y > getHeight() / 2) ? getNumberOfTypesPerLine(getMinimum(), getMaximum()) : 0);
223 index += int(x / cell_width);
224 if(index > getMaximum()) { return;}
225 setValue(index);
226}

294void WaveSelector::resizeNoise(float x, float y, float width, float height) {
295 static const int noise_icon_resolution = 14;
296 srand(0);
297 noise_.clear();
298 noise_.startNewSubPath(x, y + height / 2.0f);
299 float inc_x = width / noise_icon_resolution;
