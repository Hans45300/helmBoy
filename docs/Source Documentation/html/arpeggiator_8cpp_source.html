35 Arpeggiator::Arpeggiator(NoteHandler* note_handler) :
36 Processor(static_cast<int>(Inputs::NumInputs), 1), note_handler_(note_handler),
37 sustain_(false), phase_(1.0), note_index_(-1),
38 current_octave_(0), octave_up_(true), last_played_note_(0) {

40 pressed_notes_.reserve(MIDI_SIZE);
41 sustained_notes_.reserve(MIDI_SIZE);

44 void Arpeggiator::process() {
45 if (input(static_cast<int>(Inputs::On))->at(0) == 0.0)
46 return;
47
48 mopo_float frequency = input(static_cast<int>(Inputs::Frequency))->at(0);
49 mopo_float min_gate = (MIN_VOICE_TIME + VOICE_KILL_TIME) * frequency;
50 mopo_float gate = utils::interpolate(min_gate, 1.0, input(static_cast<int>(Inputs::Gate))->at(0));
51
52 mopo_float delta_phase = frequency / sample_rate_;
53 mopo_float new_phase = phase_ + buffer_size_ * delta_phase;
54
55 // If we're past the gate phase and we're playing a note, turn it off.
56 if (new_phase >= gate && last_played_note_ >= 0) {
57 int offset = utils::iclamp((gate - phase_) / delta_phase, 0, buffer_size_ - 1);
58 note_handler_->noteOff(last_played_note_, offset);
59 last_played_note_ = -1;
60 }
61
62 // Check if it's time to play the next note.
63 if (getNumNotes() && new_phase >= 1.0) {
64 int offset = utils::iclamp((1 - phase_) / delta_phase, 0, buffer_size_ - 1);
65 std::pair<mopo_float, mopo_float> note = getNextNote();
66 note_handler_->noteOn(note.first, note.second, offset);
67 last_played_note_ = note.first;
68 phase_ = new_phase - 1.0;
69 }
70 else
71 phase_ = new_phase;
72 }

74 std::pair<mopo_float, mopo_float> Arpeggiator::getNextNote() {
75 int octaves = utils::imax(1, input(static_cast<int>(Inputs::Octaves))->at(0));
76 Pattern type = static_cast<Pattern>(static_cast<int>(input(static_cast<int>(Inputs::Pattern))->at(0)));
77 std::vector<mopo_float>* pattern = &as_played_;
78
79 note_index_++;
80
81 switch (type) {
82 case Pattern::Up:
83 pattern = &ascending_;
84 octave_up_ = true;
85 break;
86 case Pattern::Down:
87 pattern = &decending_;
88 octave_up_ = false;
89 break;
90 case Pattern::Random:
91 pattern = &ascending_;
92 note_index_ = rand() % ascending_.size();
93 current_octave_ = rand() % octaves;
94 break;
95 case Pattern::UpDown:
96 if (note_index_ >= static_cast<int>(ascending_.size()) - 1) {
97 if ((current_octave_ == 0 && !octave_up_) ||
98 (current_octave_ >= octaves - 1 && octave_up_)) {
99 note_index_ = 0;
100 octave_up_ = !octave_up_;
101 }
102 }
103 pattern = octave_up_ ? &ascending_ : &decending_;
104 break;
105 case Pattern::AsPlayed:
106 pattern = &as_played_;
107 break;
108 default:
109 break;
110 }
111
112 if (note_index_ >= pattern->size()) {
113 note_index_ = 0;
114 if (octave_up_)
115 current_octave_ = (current_octave_ + 1) % octaves;
116 else
117 current_octave_ = (current_octave_ + octaves - 1) % octaves;
118 }
119 mopo_float base_note = pattern->at(note_index_);
120 mopo_float note = base_note + mopo::NOTES_PER_OCTAVE * current_octave_;
121 mopo_float velocity = active_notes_[base_note];
122 return std::pair<mopo_float, mopo_float>(note, velocity);
123 }

129 void Arpeggiator::addNoteToPatterns(mopo_float note) {
130 as_played_.push_back(note);
131 ascending_.push_back(note);
132 std::ranges::sort(ascending_);
133 decending_.push_back(note);
134 std::ranges::sort(decending_, std::greater<>());

137 void Arpeggiator::removeNoteFromPatterns(mopo_float note) {

167 void Arpeggiator::noteOn(mopo_float note, mopo_float velocity, int sample, int channel) {
168 if (active_notes_.count(note))
169 return;
170 if (pressed_notes_.size() == 0) {
171 note_index_ = -1;
172 current_octave_ = 0;
173 phase_ = 1.0;
174 }
175 active_notes_[note] = velocity;
176 pressed_notes_.push_back(note);
177 addNoteToPatterns(note);
178 }
179
180 VoiceEvent Arpeggiator::noteOff(mopo_float note, int sample) {
181 if (pressed_notes_.count(note) == 0)
182 return kVoiceOff;
183
184 if (sustain_)
185 sustained_notes_.push_back(note);
186 else {
187 active_notes_.erase(note);
188 removeNoteFromPatterns(note);
189 }
190
191 pressed_notes_.removeAll(note);
192 return kVoiceOff;
193 }
