144void FullInterface::resized() {
145 int left = 0;
146 int width = getWidth();
147 int height = getHeight();
148 float ratio = 1.0f;
149 float width_ratio = getWidth() / (1.0f * mopo::DEFAULT_WINDOW_WIDTH);
150 float height_ratio = getHeight() / (1.0f * mopo::DEFAULT_WINDOW_HEIGHT);
151 if (width_ratio > height_ratio) {
152 ratio = height_ratio;
153 width = height_ratio * mopo::DEFAULT_WINDOW_WIDTH;
154 left = (getWidth() - width) / 2;
155 }
156 else {
157 ratio = width_ratio;
158 height = width_ratio * mopo::DEFAULT_WINDOW_HEIGHT;
159 }
160
161 setSizeRatio(ratio);
162 save_section_->setSizeRatio(ratio);
163 delete_section_->setSizeRatio(ratio);
164 patch_browser_->setSizeRatio(ratio);
165 about_section_->setSizeRatio(ratio);
166 int padding = 8 * ratio;
167 int top_height = TOP_HEIGHT * ratio;
168
169 int section_one_width = 320 * ratio;
170 int section_two_width = section_one_width;
171 int section_four_width = section_one_width;
172 int section_three_width = width - section_one_width - section_two_width - section_four_width - 5 * padding;
173
174 synthesis_interface_->setPadding(padding);
175 synthesis_interface_->setSectionOneWidth(section_one_width);
176 synthesis_interface_->setSectionTwoWidth(section_two_width);
177 synthesis_interface_->setSectionThreeWidth(section_three_width);
178 synthesis_interface_->setSectionFourWidth(section_four_width);
179
180 int logo_padding = 2 * ratio;
181 int logo_width = top_height + 2 * logo_padding;
182
183 int patch_selector_width = section_one_width - logo_width - padding;
184
185 logo_graphics_->setBounds(left + padding + logo_padding, padding, top_height, top_height);
186 patch_selector_->setBounds(logo_graphics_->getRight() + padding + logo_padding, padding,
187 patch_selector_width, top_height);
188 global_tool_tip_->setBounds(patch_selector_->getX() + 0.11 * patch_selector_->getWidth(),
189 patch_selector_->getY(),
190 0.78 * patch_selector_->getWidth(),
191 patch_selector_->getBrowseHeight());
192
193 int volume_width = (section_two_width - padding) / 2;
194 int bpm_width = 40 * ratio;
195 int oscilloscope_width = section_two_width + section_three_width - volume_width;
196
197 volume_section_->setBounds(patch_selector_->getRight() + padding, padding,
198 volume_width, top_height);
199
200 oscilloscope_->setBounds(volume_section_->getRight() + padding, padding,
201 oscilloscope_width, top_height);
202
203 // BPM aligné à gauche de la colonne 5
204 int column_5_x = patch_selector_->getX() + section_one_width + section_two_width + section_three_width + 4 * padding;
205 bpm_section_->setBounds(oscilloscope_->getRight() + padding, padding,
206 bpm_width, top_height);
207
208 // Arp juste à droite du BPM
209 int arp_width = section_four_width - bpm_width - padding;
210 arp_section_->setBounds(bpm_section_->getRight() + padding, padding,
211 arp_width, top_height);
212
213 synthesis_interface_->setBounds(left, top_height + padding,
214 width, height - top_height - padding);
215
216 about_section_->setBounds(getBounds());
217 update_check_section_->setBounds(getBounds());
218 save_section_->setBounds(getBounds());
219 delete_section_->setBounds(getBounds());
220
221 patch_browser_->setBounds(synthesis_interface_->getX() + padding, synthesis_interface_->getY(),
222 arp_section_->getRight() - synthesis_interface_->getX() - padding,
223 synthesis_interface_->getHeight() - padding);
224
225 SynthSection::resized();
226 modulation_manager_->setBounds(getBounds());
227
228 checkBackground();
229}

235void FullInterface::createModulationSliders(mopo::output_map modulation_sources,
236 mopo::output_map mono_modulations,
237 mopo::output_map poly_modulations) {
238 std::map<std::string, SynthSlider*> all_sliders = getAllSliders();
239 std::map<std::string, SynthSlider*> modulatable_sliders;
240
241 for (auto& destination : mono_modulations) {
242 if (all_sliders.count(destination.first))
243 modulatable_sliders[destination.first] = all_sliders[destination.first];
244 }
245
246 modulation_manager_ = std::make_unique<OpenGLModulationManager>(modulation_sources,
247 getAllModulationButtons(),
248 modulatable_sliders,
249 mono_modulations, poly_modulations);
250 modulation_manager_->setOpaque(false);
251 addOpenGLComponent(modulation_manager_.get());
252}

270void FullInterface::checkBackground() {
271 auto *display = Desktop::getInstance().getDisplays().getPrimaryDisplay();
272 jassert(display != nullptr);
273
274 float scale = display->scale;
275 int width = scale * getWidth();
276 int height = scale * getHeight();
277
278 if (background_image_.getWidth() != width || background_image_.getHeight() != height) {
279 background_image_ = Image(Image::ARGB, width, height, true);
280 Graphics g(background_image_);
281 g.addTransform(AffineTransform::scale(scale, scale));
282 paintBackground(g);
283 background_.updateBackgroundImage(background_image_);
284 }
285}
