75void Midi2Manager::processMidiMessage(const juce::MidiMessage& msg) {
76 try {
77 // MIDI 1.0 message processing (always supported)
78 if (msg.isController()) {
79 processMidi1ControlChange(msg);
80 }
81 else if (msg.isNoteOn()) {
82 processMidi1NoteOn(msg);
83 }
84 else if (msg.isNoteOff()) {
85 processMidi1NoteOff(msg);
86 }
87 else if (msg.isPitchWheel()) {
88 processMidi1PitchBend(msg);
89 }
90 else if (msg.isAftertouch() || msg.isChannelPressure()) {
91 processMidi1Aftertouch(msg);
92 }
93 else if (isMidi2Enabled() && msg.getRawDataSize() >= 4) {
94 // Check for MIDI 2.0 UMP format (detected via message type nibble)
95 const uint8_t* raw_data = msg.getRawData();
96
97 // UMP detection: Message type in upper 4 bits of first byte
98 // Types 0x2-0x5 indicate UMP packets (32/64/128-bit)
99 uint8_t message_type = (raw_data[0] >> 4) & 0x0F;
100
101 if (message_type >= 0x2 && message_type <= 0x5) {
102 // Reinterpret as 32-bit words for zero-copy UMP processing
103 const uint32_t* ump_data = reinterpret_cast<const uint32_t*>(raw_data);
104 int num_words = msg.getRawDataSize() / 4;
105 processUMP(ump_data, num_words);
106 }
107 }
108 }
109 catch (const std::exception& e) {
110 DBG("MIDI2: Error processing message: " << e.what());
111 }
112}

125void Midi2Manager::processUMP(const uint32_t* ump_data, int num_words) {
126 if (num_words < 1)
127 return;
128
129 try {
130 // Extract message type from first word (bits 28-31)
131 uint8_t message_type = (ump_data[0] >> 28) & 0x0F;
132
133 switch (message_type) {
134 case 0x2: // MIDI 1.0 Channel Voice (32-bit) - skip, handled by processMidiMessage
135 break;
136
137 case 0x4: // MIDI 2.0 Channel Voice (64-bit) - high-resolution
138 if (num_words >= 2) {
139 parseUMP_ChannelVoice64(ump_data);
140 }
141 break;
142
143 case 0x5: // Data Messages (128-bit) - Per-Note features
144 if (num_words >= 4) {
145 parseUMP_Data128(ump_data);
146 }
147 break;
148
149 default:
150 DBG("MIDI2: Unsupported UMP message type: " << (int)message_type);
151 break;
152 }
153 }
154 catch (const std::exception& e) {
155 DBG("MIDI2: Error parsing UMP: " << e.what());
156 }
157}

159void Midi2Manager::parseUMP_ChannelVoice64(const uint32_t* ump) {
160 // Word 1: [MT:4][Group:4][Status:4][Channel:4][Index:8][Data1:8]
161 // Word 2: [Data2:32]
162
163 uint8_t status = (ump[0] >> 20) & 0x0F;
164 uint8_t channel = (ump[0] >> 16) & 0x0F;
165 uint8_t index = (ump[0] >> 8) & 0xFF; // Note or CC number
166 uint8_t data1 = ump[0] & 0xFF;
167 uint32_t data2 = ump[1];
168
169 switch (status) {
170 case 0x8: // Note Off (64-bit)
171 processMidi2NoteOff(channel, index, data2 >> 16);
172 DBG("MIDI2: UMP Note Off ch=" << (int)channel << " note=" << (int)index
173 << " vel=" << (data2 >> 16) << " (64-bit native)");
174 break;
175
176 case 0x9: // Note On (64-bit)
177 if ((data2 >> 16) == 0) {
178 processMidi2NoteOff(channel, index, 0);
179 } else {
180 processMidi2NoteOn(channel, index, data2 >> 16);
181 }
182 DBG("MIDI2: UMP Note On ch=" << (int)channel << " note=" << (int)index
183 << " vel=" << (data2 >> 16) << " (64-bit native)");
184 break;
185
186 case 0xA: // Poly Pressure (64-bit)
187 processMidi2PerNotePressure(channel, index, data2);
188 DBG("MIDI2: UMP Poly Pressure ch=" << (int)channel << " note=" << (int)index
189 << " (32-bit native)");
190 break;
191
192 case 0xB: // Control Change (64-bit)
193 processMidi2ControlChange(channel, index, data2);
194 DBG("MIDI2: UMP CC ch=" << (int)channel << " cc=" << (int)index
195 << " (32-bit native)");
196 break;
197
198 case 0xD: // Channel Pressure (64-bit)
199 // Apply to all notes (use -1 as indicator)
200 if (onPerNotePressure) {
201 float pressure = convertMidi2ToFloat(data2);
202 onPerNotePressure(channel, -1, pressure);
203 }
204 break;
205
206 case 0xE: // Pitch Bend (64-bit)
207 processMidi2PitchBend(channel, data2);
208 DBG("MIDI2: UMP Pitch Bend ch=" << (int)channel << " (32-bit native)");
209 break;
210
211 case 0x0: // Per-Note Pitch Bend
212 processMidi2PerNotePitchBend(channel, index, data2);
213 DBG("MIDI2: UMP Per-Note Pitch Bend ch=" << (int)channel
214 << " note=" << (int)index << " (32-bit native)");
215 break;
216
217 default:
218 DBG("MIDI2: Unknown 64-bit status: " << (int)status);
219 break;
220 }
221}

223void Midi2Manager::parseUMP_Data128(const uint32_t* ump) {
224 // 128-bit messages for extended data
225 // Word 1: [MT:4][Group:4][Status:4][Channel:4][Index:8][Data1:8]
226 // Words 2-4: Extended data
227
228 uint8_t status = (ump[0] >> 20) & 0x0F;
229 uint8_t channel = (ump[0] >> 16) & 0x0F;
230 uint8_t index = (ump[0] >> 8) & 0xFF;
231
232 if (status == 0xF) {
233 // Per-Note Management or Per-Note Controller
234 uint8_t sub_status = (ump[1] >> 24) & 0xFF;
235
236 if (sub_status == 0x00) {
237 // Per-Note Management
238 uint8_t option_flags = (ump[1] >> 16) & 0xFF;
239 processMidi2PerNoteManagement(channel, index, option_flags, ump[2]);
240 }
241 else if (sub_status <= 0x7F) {
242 // Per-Note Assignable Controller
243 processMidi2PerNoteController(channel, index, sub_status, ump[2]);
244 }
245 }

250void Midi2Manager::processMidi1ControlChange(const juce::MidiMessage& msg) {
251 if (!onControlChange)
252 return;
253
254 int channel = msg.getChannel() - 1; // Convert to 0-based
255 int controller = msg.getControllerNumber();
256 int value_7bit = msg.getControllerValue();
257
258 // Convert to normalized float (upscaled in MIDI 2.0 mode)
259 float normalized_value = convertMidi1ToFloat(value_7bit);
260
261 onControlChange(channel, controller, normalized_value);
262
263 if (isMidi2Enabled()) {
264 DBG("MIDI2: CC ch=" << channel << " cc=" << controller
265 << " value=" << normalized_value << " (upscaled from 7-bit)");
266 }
267}

269void Midi2Manager::processMidi1NoteOn(const juce::MidiMessage& msg) {
270 if (!onNoteOn)
271 return;
272
273 int channel = msg.getChannel() - 1;
274 int note = msg.getNoteNumber();
275 int velocity_7bit = msg.getVelocity();
276
277 // Velocity 0 = note off in MIDI 1.0
278 if (velocity_7bit == 0) {
279 if (onNoteOff) {
280 onNoteOff(channel, note, 0.0f);
281 }
282 return;
283 }
284
285 float normalized_velocity = convertMidi1ToFloat(velocity_7bit);
286 onNoteOn(channel, note, normalized_velocity);
287
288 if (isMidi2Enabled()) {
289 DBG("MIDI2: Note On ch=" << channel << " note=" << note
290 << " vel=" << normalized_velocity << " (upscaled from 7-bit)");
291 }

322void Midi2Manager::processMidi1Aftertouch(const juce::MidiMessage& msg) {
323 if (!onPerNotePressure)
324 return;
325
326 int channel = msg.getChannel() - 1;
327
328 if (msg.isAftertouch()) {
329 // Per-note aftertouch (polyphonic)
330 int note = msg.getNoteNumber();
331 int pressure_7bit = msg.getAfterTouchValue();
332 float normalized_pressure = convertMidi1ToFloat(pressure_7bit);
333
334 onPerNotePressure(channel, note, normalized_pressure);
335 }
336 else if (msg.isChannelPressure()) {
337 // Channel aftertouch (monophonic) - apply to all notes
338 int pressure_7bit = msg.getChannelPressureValue();
339 float normalized_pressure = convertMidi1ToFloat(pressure_7bit);
340
341 // For channel pressure, use note -1 as a special indicator
342 onPerNotePressure(channel, -1, normalized_pressure);
343 }

422void Midi2Manager::processMidi2PerNoteManagement(int channel, int note, uint8_t option_flags, uint32_t data) {
423 if (!onPerNoteManagement)
424 return;
425
426 // Decode option flags for detune and tuning
427 // Bit 0: Detune present
428 // Bit 1: Tuning present
429
430 float detune = 0.0f;
431 float tuning = 0.0f;
432
433 if (option_flags & 0x01) {
434 // Detune: 7-bit signed value in cents (-64 to +63)
435 int8_t detune_raw = (data >> 25) & 0x7F;
436 if (detune_raw & 0x40) detune_raw |= 0x80; // Sign extend
437 detune = static_cast<float>(detune_raw);
438 }
439
440 if (option_flags & 0x02) {
441 // Tuning: 7-bit signed value in semitones (-64 to +63)
442 int8_t tuning_raw = (data >> 18) & 0x7F;
443 if (tuning_raw & 0x40) tuning_raw |= 0x80; // Sign extend
444 tuning = static_cast<float>(tuning_raw);
445 }
446
447 onPerNoteManagement(channel, note, detune, tuning);
448
449 DBG("MIDI2: Per-Note Management ch=" << channel << " note=" << note
450 << " detune=" << detune << " cents, tuning=" << tuning << " semitones");
451}
