59void Midi2Manager::processMidiMessage(const juce::MidiMessage& msg) {
60 try {
61 // MIDI 1.0 message processing (always supported)
62 if (msg.isController()) {
63 processMidi1ControlChange(msg);
64 }
65 else if (msg.isNoteOn()) {
66 processMidi1NoteOn(msg);
67 }
68 else if (msg.isNoteOff()) {
69 processMidi1NoteOff(msg);
70 }
71 else if (msg.isPitchWheel()) {
72 processMidi1PitchBend(msg);
73 }
74 else if (msg.isAftertouch() || msg.isChannelPressure()) {
75 processMidi1Aftertouch(msg);
76 }
77 }
78 catch (const std::exception& e) {
79 DBG("MIDI2: Error processing message: " << e.what());
80 }
81}

84uint32_t readUmpWord(const uint8_t* bytes) noexcept {
85 return (static_cast<uint32_t>(bytes[0]) << 24)
86 | (static_cast<uint32_t>(bytes[1]) << 16)
87 | (static_cast<uint32_t>(bytes[2]) << 8)
88 | static_cast<uint32_t>(bytes[3]);
89}
90}

92Midi2Manager::UMPDecodeResult Midi2Manager::decodeUMP(
93 std::span<const uint8_t> packet, int sample_offset, UMPEvent& event) noexcept {
94 if (sample_offset < 0)
95 return UMPDecodeResult::invalidArgument;
96 if (packet.size() != 8)
97 return UMPDecodeResult::invalidLength;
98 if ((packet[0] >> 4) != 0x4)
99 return UMPDecodeResult::unsupportedMessageType;
100
101 const uint8_t status = packet[1] >> 4;
102 const uint8_t channel = packet[1] & 0x0f;
103 const uint8_t index = packet[2];
104 const uint32_t data = readUmpWord(packet.data() + 4);
105 UMPEvent decoded;
106 decoded.group = packet[0] & 0x0f;
107 decoded.channel = channel;
108 decoded.index = index;
109 decoded.sampleOffset = sample_offset;
110
111 switch (status) {
112 case 0x8:
113 case 0x9:
114 if (index > 127)
115 return UMPDecodeResult::invalidData;
116 if (packet[3] != 0 || (data & 0xffff) != 0)
117 return UMPDecodeResult::unsupportedAttribute;
118 decoded.type = status == 0x8 || (data >> 16) == 0
119 ? UMPEvent::Type::noteOff : UMPEvent::Type::noteOn;
120 decoded.value = data >> 16;
121 break;
122 case 0xb:
123 if (index > 127 || packet[3] != 0)
124 return UMPDecodeResult::invalidData;
125 decoded.type = UMPEvent::Type::controlChange;
126 decoded.value = data;
127 break;
128 case 0xe:
129 if (packet[2] != 0 || packet[3] != 0)
130 return UMPDecodeResult::invalidData;
131 decoded.type = UMPEvent::Type::pitchBend;
132 decoded.index = 0;
133 decoded.value = data;
134 break;
135 default:
136 return UMPDecodeResult::unsupportedStatus;
137 }
138
139 event = decoded;
140 return UMPDecodeResult::decoded;
141}

143Midi2Manager::UMPDecodeResult Midi2Manager::processUMP(
144 std::span<const uint8_t> packet, int sample_offset) {
145 if (!isMidi2Enabled())
146 return UMPDecodeResult::disabled;
147
148 UMPEvent event;
149 const auto result = decodeUMP(packet, sample_offset, event);
150 if (result == UMPDecodeResult::decoded)
151 dispatchUMPEvent(event);
152 return result;
153}

155void Midi2Manager::processUMP(const uint32_t* ump_data, int num_words) {
156 if (ump_data == nullptr || num_words != 2)
157 return;
158
159 std::array<uint8_t, 8> bytes{};
160 for (int word = 0; word < num_words; ++word) {
161 const uint32_t value = ump_data[word];
162 for (int byte = 0; byte < 4; ++byte)
163 bytes[static_cast<size_t>(word * 4 + byte)] =
164 static_cast<uint8_t>(value >> (24 - byte * 8));
165 }
166 processUMP(bytes, 0);
167}

176void Midi2Manager::processMidi1ControlChange(const juce::MidiMessage& msg) {
177 if (!onControlChange)
178 return;
179
180 int channel = msg.getChannel() - 1; // Convert to 0-based
181 int controller = msg.getControllerNumber();
182 int value_7bit = msg.getControllerValue();
183
184 // Normalize the original 7-bit MIDI 1.0 value for callbacks.
185 float normalized_value = convertMidi1ToFloat(value_7bit);
186
187 onControlChange(channel, controller, normalized_value);
188
189 if (isMidi2Enabled()) {
190 DBG("MIDI2: MIDI 1.0 CC normalized from 7-bit value");
191 }
192}

194void Midi2Manager::processMidi1NoteOn(const juce::MidiMessage& msg) {
195 if (!onNoteOn)
196 return;
197
198 int channel = msg.getChannel() - 1;
199 int note = msg.getNoteNumber();
200 int velocity_7bit = msg.getVelocity();
201
202 // Velocity 0 = note off in MIDI 1.0
203 if (velocity_7bit == 0) {
204 if (onNoteOff) {
205 onNoteOff(channel, note, 0.0f);
206 }
207 return;
208 }
209
210 float normalized_velocity = convertMidi1ToFloat(velocity_7bit);
211 onNoteOn(channel, note, normalized_velocity);
212
213 if (isMidi2Enabled()) {
214 DBG("MIDI2: MIDI 1.0 Note On normalized from 7-bit velocity");
215 }
216}

245void Midi2Manager::processMidi1Aftertouch(const juce::MidiMessage& msg) {
246 if (!onPerNotePressure)
247 return;
248
249 int channel = msg.getChannel() - 1;
250
251 if (msg.isAftertouch()) {
252 // Per-note aftertouch (polyphonic)
253 int note = msg.getNoteNumber();
254 int pressure_7bit = msg.getAfterTouchValue();
255 float normalized_pressure = convertMidi1ToFloat(pressure_7bit);
256
257 onPerNotePressure(channel, note, normalized_pressure);
258 }
259 else if (msg.isChannelPressure()) {
260 // Channel aftertouch (monophonic) - apply to all notes
261 int pressure_7bit = msg.getChannelPressureValue();
262 float normalized_pressure = convertMidi1ToFloat(pressure_7bit);
263
264 // For channel pressure, use note -1 as a special indicator
265 onPerNotePressure(channel, -1, normalized_pressure);
266 }
267}
