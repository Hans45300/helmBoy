88 void StateVariableFilter::process() {
89 using enum Inputs;
90 using enum Styles;
91 using enum Shelves;
92 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Audio)));
93
94#if HELMBOY_DEBUG_DSP_PREP
95 static bool warned_invalid_spec = false;
96 if (!hasValidProcessingSpec() && !warned_invalid_spec) {
97 std::fprintf(stderr,
98 "[HELMBOY_DEBUG_DSP_PREP] StateVariableFilter invalid spec: sr=%f bs=%d\n",
99 static_cast<double>(sample_rate_), buffer_size_);
100 warned_invalid_spec = true;
101 }
102#endif
103
104 // Hard runtime guard: keep signal passing instead of processing with invalid spec.
105 if (!hasValidProcessingSpec()) {
106 auto invalid_src = input(static_cast<int>(Audio))->source;
107 if (invalid_src != nullptr) {
108 processAllPass(invalid_src->buffer, output()->buffer);
109 }
110 return;
111 }
112
113 auto audio_buffer = std::span<const mopo_float>(input(static_cast<int>(Audio))->source->buffer, buffer_size_);
114 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
115
116 if (input(static_cast<int>(On))->at(0) == 0.0) {
117 processAllPass(audio_buffer.data(), dest.data());
118 return;
119 }
120
121 const int style_raw = static_cast<int>(input(static_cast<int>(Style))->at(0));
122 const int style_index = std::clamp(style_raw, 0, static_cast<int>(Styles::NumStyles) - 1);
123 #if defined(HELMBOY_DEBUG_SVF_TRANSITIONS)
124 if (style_raw != style_index) {
125 std::fprintf(stderr,
126 "[HelmBoy][SVF] style clamp raw=%d clamped=%d\n",
127 style_raw,
128 style_index);
129 }
130 #endif
131 Styles style = static_cast<Styles>(style_index);
132 bool db24 = (style == Styles::dB24);
133
134 mopo_float cutoff_in = input(static_cast<int>(Cutoff))->at(0);
135 mopo_float resonance_in = input(static_cast<int>(Resonance))->at(0);
136 mopo_float drive_in = input(static_cast<int>(Drive))->at(0);
137
138 if (!isFiniteValue(cutoff_in))
139 cutoff_in = MIN_CUTTOFF;
140 if (!isFiniteValue(resonance_in))
141 resonance_in = MIN_RESONANCE;
142 if (!isFiniteValue(drive_in))
143 drive_in = 1.0;
144
145 mopo_float cutoff = utils::clamp(cutoff_in, MIN_CUTTOFF, sample_rate_);
146 mopo_float resonance = utils::clamp(resonance_in, MIN_RESONANCE, MAX_RESONANCE);
147 target_drive_ = drive_in;
148
149 if (style == Styles::Comb) {
150 if (style != last_style_) {
151 reset();
152 last_style_ = style;
153 }
154
155 const Output* comb_audio = input(static_cast<int>(CombAudio))->source;
156 if (comb_audio == nullptr) {
157 processAllPass(audio_buffer.data(), dest.data());
158 return;
159 }
160
161 if (input(static_cast<int>(On))->at(0) == 0.0) {
162 processAllPass(audio_buffer.data(), dest.data());
163 return;
164 }
165
166 mopo_float delta_drive = (target_drive_ - drive_) / buffer_size_;
167 for (int sample = 0; sample < buffer_size_; ++sample) {
168 drive_ += delta_drive;
169 dest[sample] = utils::quickTanh(drive_ * comb_audio->buffer[sample]);
170 }
171 return;
172 }
173
174 Shelves shelf_choice = Shelves::LowShelf;
175 const bool use_juce_path = shouldUseJucePath(style, shelf_choice);
176
177 if (style == Styles::Shelf) {
178 const int shelf_raw = static_cast<int>(input(static_cast<int>(ShelfChoice))->at(0));
179 const int shelf_index = std::clamp(shelf_raw, 0, static_cast<int>(Shelves::NumShelves) - 1);
180#if defined(HELMBOY_DEBUG_SVF_TRANSITIONS)
181 if (shelf_raw != shelf_index) {
182 std::fprintf(stderr,
183 "[HelmBoy][SVF] shelf clamp raw=%d clamped=%d\n",
184 shelf_raw,
185 shelf_index);
186 }
187#endif
188 shelf_choice = static_cast<Shelves>(shelf_index);
189
190 mopo_float gain_in = input(static_cast<int>(Gain))->at(0);
191 if (!isFiniteValue(gain_in))
192 gain_in = 1.0;
193
194 computeShelfCoefficients(shelf_choice, cutoff, gain_in);
195 }
196 else if (style == Styles::Notch) {
197 computeNotchCoefficients(cutoff, resonance);
198 }
199 else {
200 mopo_float blend = input(static_cast<int>(PassBlend))->at(0);
201 if (!isFiniteValue(blend))
202 blend = 1.0;
203 if (use_juce_path)
204 computePassCoefficientsJucePath(blend, cutoff, resonance, db24);
205 else
206 computePassCoefficients(blend, cutoff, resonance, db24);
207 }
208
209 if (style != last_style_) {
210#if defined(HELMBOY_DEBUG_SVF_TRANSITIONS)
211 std::fprintf(stderr,
212 "[HelmBoy][SVF] style transition old=%d new=%d (sr=%d bs=%d)\n",
213 static_cast<int>(last_style_),
214 static_cast<int>(style),
215 sample_rate_,
216 buffer_size_);
217#endif
218 reset();
219 last_style_ = style;
220 }
221
222#if defined(HELMBOY_DEBUG_SVF_TRANSITIONS)
223 static Styles last_route_style = Styles::NumStyles;
224 static bool last_route_juce = false;
225 if (style != last_route_style || use_juce_path != last_route_juce) {
226 std::fprintf(stderr,
227 "[HelmBoy][SVF] route style=%d mode=%s\n",
228 static_cast<int>(style),
229 use_juce_path ? "juce-path" : "legacy-path");
230 last_route_style = style;
231 last_route_juce = use_juce_path;
232 }
233#endif
234
235 if (use_juce_path)
236 processJucePath(audio_buffer.data(), dest.data(), style, shelf_choice);
237 else if (db24)
238 process24db(audio_buffer.data(), dest.data());
239 else
240 process12db(audio_buffer.data(), dest.data());
241 }

243 void StateVariableFilter::process12db(const mopo_float* audio_buffer, mopo_float* dest) {
244 using enum Inputs;
245 mopo_float delta_m0 = (target_m0_ - m0_) / buffer_size_;
246 mopo_float delta_m1 = (target_m1_ - m1_) / buffer_size_;
247 mopo_float delta_m2 = (target_m2_ - m2_) / buffer_size_;
248 mopo_float delta_drive = (target_drive_ - drive_) / buffer_size_;
249
250 if (input(static_cast<int>(Reset))->source->triggered &&
251 input(static_cast<int>(Reset))->source->trigger_value == static_cast<mopo_float>(kVoiceReset)) {
252 const int trigger_offset_raw = input(static_cast<int>(Reset))->source->trigger_offset;
253 int trigger_offset = std::clamp(trigger_offset_raw, 0, buffer_size_);
254 #if defined(HELMBOY_DEBUG_SVF_TRANSITIONS)
255 if (trigger_offset_raw != trigger_offset) {
256 std::fprintf(stderr,
257 "[HelmBoy][SVF] reset offset clamp raw=%d clamped=%d bs=%d\n",
258 trigger_offset_raw,
259 trigger_offset,
260 buffer_size_);
261 }
262 #endif
263 int i = 0;
264 for (; i < trigger_offset; ++i) {
265 m0_ += delta_m0;
266 m1_ += delta_m1;
267 m2_ += delta_m2;
268 drive_ += delta_drive;
269 tick(i, dest, audio_buffer);
270 }
271
272 reset();
273
274 for (; i < buffer_size_; ++i)
275 tick(i, dest, audio_buffer);
276 }
277 else {
278 for (int i = 0; i < buffer_size_; ++i) {
279 m0_ += delta_m0;
280 m1_ += delta_m1;
281 m2_ += delta_m2;
282 drive_ += delta_drive;
283 tick(i, dest, audio_buffer);
284 }
285 }
286 }

317

344 void StateVariableFilter::computePassCoefficients(mopo_float blend,
345 mopo_float cutoff,
346 mopo_float resonance,
347 bool db24) {
348 MOPO_ASSERT(resonance > 0.0);
349 MOPO_ASSERT(cutoff > 0.0);
350
351 if (db24)
352 resonance = sqrt(resonance);
353
354 mopo_float g = tan(PI * utils::min(cutoff / sample_rate_, 0.5));
355 mopo_float k = 1.0 / resonance;
356
357 mopo_float low_pass_amount = sqrt(utils::clamp(1.0 - blend, 0.0, 1.0));
358 mopo_float band_pass_amount = sqrt(utils::clamp(1.0 - fabs(blend - 1.0), 0.0, 1.0));
359 mopo_float high_pass_amount = sqrt(utils::clamp(blend - 1.0, 0.0, 1.0));
360
361 target_m0_ = 0.0;
362 target_m1_ = band_pass_amount;
363 target_m2_ = low_pass_amount;
364
365 target_m0_ += high_pass_amount;
366 target_m1_ += -k * high_pass_amount;
367 target_m2_ += -high_pass_amount;
368
369 a1_ = 1.0 / (1.0 + g * (g + k));
370 a2_ = g * a1_;
371 a3_ = g * a2_;
372 }

374 void StateVariableFilter::computeNotchCoefficients(mopo_float cutoff,
375 mopo_float resonance) {
376 MOPO_ASSERT(resonance > 0.0);
377 MOPO_ASSERT(cutoff > 0.0);
378
379 mopo_float g = tan(PI * utils::min(cutoff / sample_rate_, 0.5));
380 mopo_float k = 1.0 / resonance;
381 target_m0_ = 1.0;
382 target_m1_ = -k;
383 target_m2_ = 0.0;
384
385 a1_ = 1.0 / (1.0 + g * (g + k));
386 a2_ = g * a1_;
387 a3_ = g * a2_;
388 }

390 void StateVariableFilter::computePassCoefficientsJucePath(mopo_float blend,
391 mopo_float cutoff,
392 mopo_float resonance,
393 bool db24) {
394 // Dedicated coefficient path for optional JUCE-style routing.
395 // This keeps processing topology identical while isolating parameter handling.
396 const mopo_float safe_cutoff = utils::clamp(cutoff, MIN_CUTTOFF, sample_rate_);
397 mopo_float safe_resonance = utils::clamp(resonance, MIN_RESONANCE, MAX_RESONANCE);
398 if (db24)
399 safe_resonance = std::sqrt(safe_resonance);
400
401 const mopo_float normalized_cutoff = utils::min(safe_cutoff / sample_rate_, 0.5);
402 const mopo_float g = std::tan(PI * normalized_cutoff);
403 const mopo_float k = 1.0 / safe_resonance;
404
405 const mopo_float low_pass_amount = std::sqrt(utils::clamp(1.0 - blend, 0.0, 1.0));
406 const mopo_float band_pass_amount = std::sqrt(utils::clamp(1.0 - std::fabs(blend - 1.0), 0.0, 1.0));
407 const mopo_float high_pass_amount = std::sqrt(utils::clamp(blend - 1.0, 0.0, 1.0));
408
409 target_m0_ = high_pass_amount;
410 target_m1_ = band_pass_amount - k * high_pass_amount;
411 target_m2_ = low_pass_amount - high_pass_amount;
412
413 a1_ = 1.0 / (1.0 + g * (g + k));
414 a2_ = g * a1_;
415 a3_ = g * a2_;
416 }

418 void StateVariableFilter::computeShelfCoefficients(Shelves choice,
419 mopo_float cutoff,
420 mopo_float gain) {
421 MOPO_ASSERT(gain >= 0.0);
422 MOPO_ASSERT(cutoff > 0.0);
423
424 gain = sqrt(gain);
425
426 mopo_float g = tan(PI * utils::min(cutoff / sample_rate_, 0.5));
427 mopo_float k = 1.0;
428
429 switch(choice) {
430 case Shelves::LowShelf: {
431 g /= sqrt(gain);
432 target_m0_ = 1.0;
433 target_m1_ = k * (gain - 1.0);
434 target_m2_ = gain * gain - 1.0;
435 break;
436 }
437 case Shelves::BandShelf: {
438 k /= gain;
439 target_m0_ = 1.0;
440 target_m1_ = k * (gain * gain - 1.0);
441 target_m2_ = 0.0;
442 break;
443 }
444 case Shelves::HighShelf: {
445 g *= sqrt(gain);
446 target_m0_ = gain * gain;
447 target_m1_ = k * gain * (1.0 - gain);
448 target_m2_ = 1.0 - gain * gain;
449 break;
450 }
451 default: {
452 target_m0_ = 0.0;
453 target_m1_ = 0.0;
454 target_m2_ = 0.0;
455 }
456 }
457
458 a1_ = 1.0 / (1.0 + g * (g + k));
459 a2_ = g * a1_;
460 a3_ = g * a2_;
461
462 if (last_shelf_ != choice) {
463 reset();
464 last_shelf_ = choice;
465 }
466 }

468 inline void StateVariableFilter::tick(int i, mopo_float* dest, const mopo_float* audio_buffer) {
469 mopo_float audio = utils::quickTanh(drive_ * audio_buffer[i]);
470
471 mopo_float v3_a = audio - ic2eq_a_;
472 mopo_float v1_a = a1_ * ic1eq_a_ + a2_ * v3_a;
473 mopo_float v2_a = ic2eq_a_ + a2_ * ic1eq_a_ + a3_ * v3_a;
474 ic1eq_a_ = 2.0 * v1_a - ic1eq_a_;
475 ic2eq_a_ = 2.0 * v2_a - ic2eq_a_;
476
477 dest[i] = m0_ * audio + m1_ * v1_a + m2_ * v2_a;
478 }

480 inline void StateVariableFilter::tick24db(int i, mopo_float* dest,
481 const mopo_float* audio_buffer) {
482 mopo_float audio = drive_ * audio_buffer[i];
483
484 mopo_float v3_a = audio - ic2eq_a_;
485 mopo_float v1_a = a1_ * ic1eq_a_ + a2_ * v3_a;
486 mopo_float v2_a = ic2eq_a_ + a2_ * ic1eq_a_ + a3_ * v3_a;
487 ic1eq_a_ = 2.0 * v1_a - ic1eq_a_;
488 ic2eq_a_ = 2.0 * v2_a - ic2eq_a_;
489 mopo_float out_a = m0_ * audio + m1_ * v1_a + m2_ * v2_a;
490
491 mopo_float distort = utils::quickTanh(out_a);
492
493 mopo_float v3_b = distort - ic2eq_b_;
494 mopo_float v1_b = a1_ * ic1eq_b_ + a2_ * v3_b;
495 mopo_float v2_b = ic2eq_b_ + a2_ * ic1eq_b_ + a3_ * v3_b;
496 ic1eq_b_ = 2.0 * v1_b - ic1eq_b_;
497 ic2eq_b_ = 2.0 * v2_b - ic2eq_b_;
498
499 dest[i] = m0_ * distort + m1_ * v1_b + m2_ * v2_b;
500 }
