118void StutterSection::resized() {
119 int text_width = size_ratio_ * TEXT_WIDTH;
120 int text_height = size_ratio_ * TEXT_HEIGHT;
121 int text_y = size_ratio_ * TEXT_Y;
122 int slider_height = size_ratio_ * SLIDER_HEIGHT;
123
124 float space = (getWidth() - 2.0f * (text_width + text_height)) / 3.0f;
125
126 float title_width = getTitleWidth();
127 on_->setBounds(size_ratio_ * 2.0f, 0, title_width, title_width);
128
129 stutter_frequency_->setBounds(space, text_y, text_width, text_height);
130 stutter_tempo_->setBounds(stutter_frequency_->getBounds());
131 stutter_sync_->setBounds(stutter_frequency_->getBounds().getRight(), text_y,
132 text_height, text_height);
133
134 resample_frequency_->setBounds(stutter_sync_->getBounds().getRight() + space, text_y,
135 text_width, text_height);
136 resample_tempo_->setBounds(resample_frequency_->getBounds());
137 resample_sync_->setBounds(resample_frequency_->getBounds().getRight(), text_y,
138 text_height, text_height);
139
140 int lower_buffer = size_ratio_ * 15;
141 stutter_softness_->setBounds(0, getHeight() - slider_height - lower_buffer,
142 getWidth(), slider_height);
143 SynthSection::resized();
144
145 stutter_frequency_->setPopupDisplayEnabled(false, false, nullptr);
146 stutter_tempo_->setPopupDisplayEnabled(false, false, nullptr);
147 resample_frequency_->setPopupDisplayEnabled(false, false, nullptr);
148 resample_tempo_->setPopupDisplayEnabled(false, false, nullptr);
149}
