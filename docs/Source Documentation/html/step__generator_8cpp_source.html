44 void StepGenerator::process() {
45 using enum Inputs;
46 using enum Outputs;
47 static mopo_float integral;
48 unsigned int num_steps = static_cast<unsigned int>(input(static_cast<int>(Inputs::NumSteps))->at(0));
49 num_steps = static_cast<unsigned int>(utils::iclamp(static_cast<int>(num_steps), 1, static_cast<int>(max_steps_)));
50
51 int i = 0;
52 if (input(static_cast<int>(Inputs::Reset))->source->triggered) {
53 offset_ = 0.0f;
54 current_step_ = 0u;
55 i = input(static_cast<int>(Inputs::Reset))->source->trigger_offset;
56 }
57
58 offset_ += samples_to_process_ * input(static_cast<int>(Inputs::Frequency))->at(0) / sample_rate_;
59 offset_ = utils::mod(offset_, &integral);
60 current_step_ += static_cast<unsigned int>(integral);
61 current_step_ = (current_step_ + num_steps) % num_steps;
62
63 output(static_cast<int>(Outputs::Value))->buffer[0] = input(static_cast<int>(Inputs::Steps) + static_cast<int>(current_step_))->source->buffer[0];
64 output(static_cast<int>(Outputs::Step))->buffer[0] = static_cast<mopo_float>(current_step_);
65 }

67 void StepGenerator::correctToTime(mopo_float samples) {
68 using enum Inputs;
69 static mopo_float integral;
70
71 unsigned int num_steps = static_cast<unsigned int>(input(static_cast<int>(Inputs::NumSteps))->at(0));
72 num_steps = static_cast<unsigned int>(utils::iclamp(static_cast<int>(num_steps), 1, static_cast<int>(max_steps_)));
73
74 offset_ = samples * input(static_cast<int>(Inputs::Frequency))->at(0) / sample_rate_;
75 offset_ = utils::mod(offset_, &integral);
76 current_step_ = static_cast<unsigned int>(integral);
