32constexpr float kMaximumGain = 2.0f;
33constexpr int kMeterBars = 20;

35class CpuPeakMeter : public Component, private Timer {
36 public:
37 explicit CpuPeakMeter(bool left) : left_(left) { startTimerHz(30); }
38
39 ~CpuPeakMeter() override { stopTimer(); }
40
41 void paint(Graphics& g) override {
42 const float width = static_cast<float>(getWidth());
43 const float height = static_cast<float>(getHeight());
44 if (width <= 0.0f || height <= 0.0f)
45 return;
46
47 const float threshold_x = width / std::sqrt(kMaximumGain);
48 g.setColour(Colour(Colors::Color_ff222222));
49 g.fillRect(threshold_x, 0.0f, width - threshold_x, height);
50 g.setColour(Colour(Colors::Color_ff888888));
51 g.fillRect(threshold_x - 1.0f, 0.0f, 2.0f, height);
52
53 const float fill_width = width * std::sqrt(
54 std::clamp(peak_ / kMaximumGain, 0.0f, 1.0f));
55 for (int bar = 0; bar < kMeterBars; ++bar) {
56 const float left = width * bar / kMeterBars;
57 if (left >= fill_width)
58 break;
59 const float right = std::min(width * (bar + 1) / kMeterBars, fill_width);
60 const float position = 2.0f * (bar + 0.5f) / kMeterBars - 1.0f;
61 const float boosted = kMeterBars * position;
62 const float fraction = boosted - std::floor(boosted);
63 const float alpha = std::clamp(3.0f * fraction, 0.0f, 1.0f) *
64 std::clamp(3.0f * (0.95f - fraction), 0.0f, 1.0f);
65 float red = 0.5f * position + 0.5f;
66 float green = 1.0f;
67 float blue = 0.4f;
68 if (position > 0.4142f) {
69 red = 1.0f;
70 green = 1.0f - 0.85f * position;
71 blue = 0.2f;
72 }
73 g.setColour(Colour::fromFloatRGBA(red, green, blue, alpha));
74 g.fillRect(left, 0.0f, right - left, height);
75 }
76 }
77
78 private:
79 void timerCallback() override {
80 if (!isShowing())
81 return;
82 auto* interface = findParentComponentOfClass<FullInterface>();
83 if (interface == nullptr || !interface->isAnimating())
84 return;
85 if (synth_ == nullptr) {
86 if (auto* gui = findParentComponentOfClass<SynthGuiInterface>())
87 synth_ = gui->getSynth();
88 }
89 if (synth_ == nullptr)
90 return;
91 const float next_peak = synth_->getVisualPeak(left_);
92 if (std::isfinite(next_peak) && std::abs(next_peak - peak_) >= 0.001f) {
93 peak_ = next_peak;
94 repaint();
95 }
96 }
97
98 SynthBase* synth_ = nullptr;
99 float peak_ = 0.0f;
100 bool left_;
101};
102} // namespace

149void VolumeSection::resized() {
150 int title_width = getTitleWidth();
151 int height = getHeight() - title_width;
152 int meter_area_height = height / 2;
153 int meter_gap = size_ratio_ * 2.0f;
154 int meter_height = (meter_area_height - meter_gap) / 2;
155 int controls_y = title_width + meter_area_height;
156 int controls_height = height - meter_area_height;
157 int label_width = size_ratio_ * 9.0f;
158 int pan_size = std::min(static_cast<int>(getSmallKnobSize()), controls_height);
159 int controls_gap = size_ratio_ * 4.0f;
160 int pan_label_width = size_ratio_ * 30.0f;
161 int volume_width = getWidth() - pan_size - pan_label_width - 2 * controls_gap;
162 int pan_x = volume_width + controls_gap;
163
164 peak_meter_left_->setBounds(label_width, title_width,
165 getWidth() - label_width, meter_height);
166 peak_meter_right_->setBounds(label_width, title_width + meter_height + meter_gap,
167 getWidth() - label_width, meter_height);
168
169 volume_->setBounds(0, controls_y, volume_width, controls_height);
170 pan_->setBounds(pan_x, controls_y + (controls_height - pan_size) / 2,
171 pan_size, pan_size);
172
173 SynthSection::resized();
174}
