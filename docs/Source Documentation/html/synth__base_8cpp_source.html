83 mopo::ModulationConnection* connection = connection_opt.value_or(nullptr);

101void SynthBase::setModulationAmount(mopo::ModulationConnection* connection,
102 mopo::mopo_float amount) {
103 if (amount == 0.0) {
104 modulation_bank_.recycle(connection);
105 mod_connections_.erase(connection);
106 }
107 else if (mod_connections_.count(connection) == 0)
108 mod_connections_.insert(connection);
109 modulation_change_queue_.enqueue(mopo::modulation_change(connection, amount));
110}

205 auto gui_interface = getGuiInterface();

207 gui_interface.value()->externalPatchLoaded(file);

210 }
211 });

244void SynthBase::processAudio(AudioSampleBuffer* buffer, int channels, int samples, int offset) {
245 mopo::utils::enableDenormalFlushing(true);
246
247 if (engine_.getBufferSize() != samples)
248 engine_.setBufferSize(samples);
249
250 engine_.process();
251
252 auto engine_output_left = std::span<const mopo::mopo_float>(engine_.output(0)->buffer, samples);
253 auto engine_output_right = std::span<const mopo::mopo_float>(engine_.output(1)->buffer, samples);
254 #include <xsimd/xsimd.hpp>
255 using batch = xsimd::batch<float>;
256 constexpr std::size_t simd_size = batch::size;
257 for (int channel = 0; channel < channels; ++channel) {
258 float* channelData = buffer->getWritePointer(channel, offset);
259 auto synth_output = (channel % 2) ? engine_output_right : engine_output_left;
260 int simd_end = samples - (samples % simd_size);
261 int i = 0;
262 alignas(alignof(batch)) float tmp[simd_size];
263 for (; i < simd_end; i += simd_size) {
264 for (std::size_t j = 0; j < simd_size; ++j) tmp[j] = static_cast<float>(synth_output[i + j]);
265 batch s = batch::load_aligned(tmp);
266 s.store_unaligned(&channelData[i]);
267 auto mask = xsimd::isfinite(s);
268 for (std::size_t j = 0; j < simd_size; ++j) {
269 MOPO_ASSERT(mask.get(j));
270 }
271 }
272 for (; i < samples; ++i) {
273 channelData[i] = synth_output[i];
274 MOPO_ASSERT(std::isfinite(synth_output[i]));
275 }
276 }
277
278 updateMemoryOutput(samples, engine_output_left, engine_output_right);
279}

281void SynthBase::processAudioSafe(AudioSampleBuffer* buffer, int channels, int samples, int offset) {
282 mopo::utils::enableDenormalFlushing(true);
283
284 if (engine_.getBufferSize() != samples)
285 engine_.setBufferSize(samples);
286
287 engine_.process();
288
289 // Create safe buffer wrapper
290 helmboy::SafeAudioBuffer safe_buffer(*buffer);
291
292 // Get engine output with bounds checking
293 auto engine_output_left = std::span<const mopo::mopo_float>(engine_.output(0)->buffer, samples);
294 auto engine_output_right = std::span<const mopo::mopo_float>(engine_.output(1)->buffer, samples);
295
296 // Process each channel safely using std::span
297 #include <xsimd/xsimd.hpp>
298 using batch = xsimd::batch<float>;
299 constexpr std::size_t simd_size = batch::size;
300 for (int channel = 0; channel < channels; ++channel) {
301 auto channel_span = safe_buffer.getChannelSpan(channel);
302 auto synth_output = (channel % 2) ? engine_output_right : engine_output_left;
303 auto processing_span = channel_span.subspan(offset, samples);
304 int simd_end = samples - (samples % simd_size);
305 int i = 0;
306 alignas(alignof(batch)) float tmp[simd_size];
307 for (; i < simd_end; i += simd_size) {
308 for (std::size_t j = 0; j < simd_size; ++j) tmp[j] = static_cast<float>(synth_output[i + j]);
309 batch s = batch::load_aligned(tmp);
310 s.store_unaligned(&processing_span[i]);
311 auto mask = xsimd::isfinite(s);
312 for (std::size_t j = 0; j < simd_size; ++j) {
313 MOPO_ASSERT(mask.get(j));
314 }
315 }
316 for (; i < samples; ++i) {
317 processing_span[i] = synth_output[i];
318 MOPO_ASSERT(std::isfinite(synth_output[i]));
319 }
320 }
321
322 updateMemoryOutput(samples, engine_output_left, engine_output_right);
323}

325void SynthBase::processMidi(MidiBuffer& midi_messages, int start_sample, int end_sample) {
326 bool process_all = end_sample == 0;
327
328 for (auto it = midi_messages.cbegin(); it != midi_messages.cend(); ++it) {
329 int midi_sample = (*it).samplePosition;
330 if (process_all || (midi_sample >= start_sample && midi_sample < end_sample))
331 midi_manager_->processMidiMessage((*it).getMessage(), midi_sample - start_sample);
332 }
333}

349void SynthBase::processModulationChanges() {
350 mopo::modulation_change change;
351 while (getNextModulationChange(change)) {
352 mopo::ModulationConnection* connection = change.first;
353 mopo::mopo_float amount = change.second;
354 connection->amount.set(amount);
355
356 bool active = engine_.isModulationActive(connection);
357 if (active && amount == 0.0)
358 engine_.disconnectModulation(connection);
359 else if (!active && amount)
360 engine_.connectModulation(connection);
361 }
362}

364void SynthBase::updateMemoryOutput(int samples, std::span<const mopo::mopo_float> left,
365 std::span<const mopo::mopo_float> right) {
366 mopo::mopo_float last_played = std::max(engine_.getLastActiveNote(), OUTPUT_WINDOW_MIN_NOTE);
367 int num_pressed = engine_.getPressedNotes().size();
368 int output_inc = std::max<int>(1, engine_.getSampleRate() / mopo::MEMORY_SAMPLE_RATE);
369
370 mopo::mopo_float current_last_played_note = last_played_note_.load(std::memory_order_acquire);
371 if (last_played && (current_last_played_note != last_played || num_pressed > last_num_pressed_.load(std::memory_order_acquire))) {
372 last_played_note_.store(last_played, std::memory_order_release);
373
374 mopo::mopo_float frequency = mopo::utils::midiNoteToFrequency(last_played);
375 mopo::mopo_float period = engine_.getSampleRate() / frequency;
376 int window_length = output_inc * mopo::MEMORY_RESOLUTION;
377
378 memory_reset_period_ = period;
379 while (memory_reset_period_ < window_length)
380 memory_reset_period_ += memory_reset_period_;
381
382 memory_reset_period_ = std::min(memory_reset_period_, 2.0 * window_length);
383 memory_index_ = 0;
384 mopo::utils::copyBufferf(
385 std::span<float>(output_memory_, 2 * mopo::MEMORY_RESOLUTION),
386 std::span<const float>(output_memory_write_, 2 * mopo::MEMORY_RESOLUTION)
387 );
388 }
389 last_num_pressed_.store(num_pressed, std::memory_order_release);
390
391 // Load atomics into local temporaries for arithmetic in the audio loop
392 mopo::mopo_float local_input_offset = memory_input_offset_.load(std::memory_order_acquire);
393 int local_memory_index = memory_index_.load(std::memory_order_acquire);
394
395 for (; local_input_offset < samples; local_input_offset += output_inc) {
396 int input_index = mopo::utils::iclamp(static_cast<int>(local_input_offset), 0, samples);
397 local_memory_index = mopo::utils::iclamp(local_memory_index, 0, 2 * mopo::MEMORY_RESOLUTION - 1);
398 MOPO_ASSERT(input_index >= 0);
399 MOPO_ASSERT(input_index < samples);
400 MOPO_ASSERT(local_memory_index >= 0);
401 MOPO_ASSERT(local_memory_index < 2 * mopo::MEMORY_RESOLUTION);
402 output_memory_write_[local_memory_index++] = (left[input_index] + right[input_index]) / 2.0;
403
404 if (local_memory_index * output_inc >= memory_reset_period_) {
405 local_input_offset += memory_reset_period_ - local_memory_index * output_inc;
406 local_memory_index = 0;
407 mopo::utils::copyBufferf(
408 std::span<float>(output_memory_, 2 * mopo::MEMORY_RESOLUTION),
409 std::span<const float>(output_memory_write_, 2 * mopo::MEMORY_RESOLUTION)
410 );
411 }
412 }
413
414 // Store temporaries back to atomics
415 memory_input_offset_.store(local_input_offset - samples, std::memory_order_release);
416 memory_index_.store(local_memory_index, std::memory_order_release);
417}
