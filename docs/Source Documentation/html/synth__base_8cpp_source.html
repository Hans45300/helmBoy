39#if HELMBOY_DEBUG_AUDIO_LEVELS
40float computeStereoPeak(std::span<const mopo::mopo_float> left,
41 std::span<const mopo::mopo_float> right) {
42 float peak = 0.0f;
43 int samples = static_cast<int>(left.size());
44 for (int i = 0; i < samples; ++i) {
45 peak = std::max(peak, std::fabs(static_cast<float>(left[i])));
46 peak = std::max(peak, std::fabs(static_cast<float>(right[i])));
47 }
48 return peak;
49}

113bool SynthBase::ensureGuiPreflight() {
114 if (!JUCEApplicationBase::isStandaloneApp())
115 return true;
116
117 appendStartupTraceBase("SynthBase::ensureGuiPreflight start");
118 const std::lock_guard<std::mutex> preflightLock(gui_preflight_mutex_);
119
120 if (gui_preflight_done_) {
121 appendStartupTraceBase("SynthBase::ensureGuiPreflight already done");
122 return true;
123 }
124
125 try {
126 appendStartupTraceBase("SynthBase::ensureGuiPreflight ensureDeferredModuleInit start");
127 if (!engine_.ensureDeferredModuleInit()) {
128 appendStartupTraceBase("SynthBase::ensureGuiPreflight ensureDeferredModuleInit failed");
129 return false;
130 }
131 appendStartupTraceBase("SynthBase::ensureGuiPreflight ensureDeferredModuleInit done");
132
133 // Standalone runtime can defer submodule initialization until after window creation.
134 // Refresh control cache now that all deferred controls are guaranteed to exist.
135 appendStartupTraceBase("SynthBase::ensureGuiPreflight refresh controls start");
136 controls_ = engine_.getControls();
137 rebuildGuiControlTable();
138 rebuildVisualTelemetryTable();
139 appendStartupTraceBase("SynthBase::ensureGuiPreflight refresh controls done");
140 }
141 catch (const std::exception& ex) {
142 appendStartupTraceBase("SynthBase::ensureGuiPreflight exception: " + String(ex.what()));
143 return false;
144 }
145 catch (...) {
146 appendStartupTraceBase("SynthBase::ensureGuiPreflight unknown exception");
147 return false;
148 }
149
150 gui_preflight_done_ = true;
151 appendStartupTraceBase("SynthBase::ensureGuiPreflight done");
152 return true;
153}

211void SynthBase::setModulationAmount(mopo::ModulationConnection* connection,
212 mopo::mopo_float amount) {
213 const ScopedLock state_lock(getCriticalSection());
214 if (connection == nullptr) {
215 Logger::writeToLog("[Preset Load] Ignored modulation with null connection.");
216 return;
217 }
218
219 if (!std::isfinite(static_cast<double>(amount))) {
220 Logger::writeToLog("[Preset Load] Modulation '" + String(connection->source) + " -> " +
221 String(connection->destination) +
222 "' has non-finite amount. Replaced by 0.");
223 amount = 0.0;
224 }
225
226 if (amount != 0.0) {
227 if (!mopo::Parameters::isParameter(connection->destination)) {
228 Logger::writeToLog("[Modulation] Ignored route to unknown destination '" +
229 String(connection->destination) + "'.");
230 if (mod_connections_.count(connection) == 0)
231 modulation_bank_.recycle(connection);
232 return;
233 }
234
235 const mopo::ValueDetails& destination =
236 mopo::Parameters::getDetails(connection->destination);
237 const mopo::mopo_float clamped_amount =
238 mopo::clampModulationAmount(destination, amount);
239 if (clamped_amount != amount) {
240 Logger::writeToLog("[Modulation] Clamped '" + String(connection->source) + " -> " +
241 String(connection->destination) + "' amount from " +
242 String(amount) + " to " + String(clamped_amount) + ".");
243 amount = clamped_amount;
244 }
245 }
246
247 {
248 const std::lock_guard<std::mutex> lock(modulation_snapshot_mutex_);
249 const auto key = std::make_pair(connection->source, connection->destination);
250 if (amount == 0.0)
251 requested_modulation_amounts_.erase(key);
252 else
253 requested_modulation_amounts_[key] = amount;
254 }
255
256 if (amount == 0.0) {
257 modulation_bank_.recycle(connection);
258 mod_connections_.erase(connection);
259 }
260 else if (mod_connections_.count(connection) == 0)
261 mod_connections_.insert(connection);
262 modulation_change_queue_.enqueue(mopo::modulation_change(connection, amount));
263}

380void SynthBase::processAudio(AudioSampleBuffer* buffer, int channels, int samples, int offset) {
381 mopo::utils::enableDenormalFlushing(true);
382
383 if (engine_.getBufferSize() != samples)
384 engine_.setBufferSize(samples);
385
386 engine_.process();
387
388 auto engine_output_left = std::span<const mopo::mopo_float>(engine_.output(0)->buffer, samples);
389 auto engine_output_right = std::span<const mopo::mopo_float>(engine_.output(1)->buffer, samples);
390 #include <xsimd/xsimd.hpp>
391 using batch = xsimd::batch<float>;
392 constexpr std::size_t simd_size = batch::size;
393 for (int channel = 0; channel < channels; ++channel) {
394 float* channelData = buffer->getWritePointer(channel, offset);
395 auto synth_output = (channel % 2) ? engine_output_right : engine_output_left;
396 int simd_end = samples - (samples % simd_size);
397 int i = 0;
398 alignas(alignof(batch)) float tmp[simd_size];
399 for (; i < simd_end; i += simd_size) {
400 for (std::size_t j = 0; j < simd_size; ++j) tmp[j] = static_cast<float>(synth_output[i + j]);
401 batch s = batch::load_aligned(tmp);
402 s.store_unaligned(&channelData[i]);
403 auto mask = xsimd::isfinite(s);
404 for (std::size_t j = 0; j < simd_size; ++j) {
405 MOPO_ASSERT(mask.get(j));
406 }
407 }
408 for (; i < samples; ++i) {
409 channelData[i] = synth_output[i];
410 MOPO_ASSERT(std::isfinite(synth_output[i]));
411 }
412 }
413
414 updateMemoryOutput(samples, engine_output_left, engine_output_right);
415}

417void SynthBase::processAudioSafe(AudioSampleBuffer* buffer, int channels, int samples, int offset) {
418 mopo::utils::enableDenormalFlushing(true);
419
420 if (engine_.getBufferSize() != samples)
421 engine_.setBufferSize(samples);
422
423 engine_.process();
424
425 // Create safe buffer wrapper
426 helmboy::SafeAudioBuffer safe_buffer(*buffer);
427
428 // Get engine output with bounds checking
429 auto engine_output_left = std::span<const mopo::mopo_float>(engine_.output(0)->buffer, samples);
430 auto engine_output_right = std::span<const mopo::mopo_float>(engine_.output(1)->buffer, samples);
431
432#if HELMBOY_DEBUG_AUDIO_LEVELS
433 static int debug_block_count = 0;
434 if ((++debug_block_count % 512) == 0) {
435 float peak = computeStereoPeak(engine_output_left, engine_output_right);
436 std::fprintf(stderr,
437 "[HELMBOY_DEBUG_AUDIO_LEVELS] block=%d samples=%d peak=%f voices=%d\n",
438 debug_block_count, samples, peak, engine_.getNumActiveVoices());
439 }
440#endif
441
442 // Process each channel safely using std::span
443 #include <xsimd/xsimd.hpp>
444 using batch = xsimd::batch<float>;
445 constexpr std::size_t simd_size = batch::size;
446 for (int channel = 0; channel < channels; ++channel) {
447 auto channel_span = safe_buffer.getChannelSpan(channel);
448 auto synth_output = (channel % 2) ? engine_output_right : engine_output_left;
449 auto processing_span = channel_span.subspan(offset, samples);
450 int simd_end = samples - (samples % simd_size);
451 int i = 0;
452 alignas(alignof(batch)) float tmp[simd_size];
453 for (; i < simd_end; i += simd_size) {
454 for (std::size_t j = 0; j < simd_size; ++j) tmp[j] = static_cast<float>(synth_output[i + j]);
455 batch s = batch::load_aligned(tmp);
456 s.store_unaligned(&processing_span[i]);
457 auto mask = xsimd::isfinite(s);
458 for (std::size_t j = 0; j < simd_size; ++j) {
459 MOPO_ASSERT(mask.get(j));
460 }
461 }
462 for (; i < samples; ++i) {
463 processing_span[i] = synth_output[i];
464 MOPO_ASSERT(std::isfinite(synth_output[i]));
465 }
466 }
467
468 updateMemoryOutput(samples, engine_output_left, engine_output_right);
469}

471void SynthBase::processMidi(MidiBuffer& midi_messages, int start_sample, int end_sample) {
472 if (midi_manager_ == nullptr)
473 return;
474
475 bool process_all = end_sample == 0;
476
477 for (auto it = midi_messages.cbegin(); it != midi_messages.cend(); ++it) {
478 int midi_sample = (*it).samplePosition;
479 if (process_all || (midi_sample >= start_sample && midi_sample < end_sample))
480 midi_manager_->processMidiMessage((*it).getMessage(), midi_sample - start_sample);
481 }
482}

498void SynthBase::processModulationChanges() {
499 mopo::modulation_change change;
500 while (getNextModulationChange(change)) {
501 mopo::ModulationConnection* connection = change.first;
502 mopo::mopo_float amount = change.second;
503 connection->amount.set(amount);
504
505 bool active = engine_.isModulationActive(connection);
506 if (active && amount == 0.0)
507 engine_.disconnectModulation(connection);
508 else if (!active && amount)
509 engine_.connectModulation(connection);
510 }
511}

513void SynthBase::updateMemoryOutput(int samples, std::span<const mopo::mopo_float> left,
514 std::span<const mopo::mopo_float> right) {
515 if (samples <= 0)
516 return;
517
518 publishVisualTelemetry();
519
520 if (visual_peak_source_ != nullptr) {
521 visual_peaks_[0].store(static_cast<float>(visual_peak_source_->buffer[0]), std::memory_order_relaxed);
522 visual_peaks_[1].store(static_cast<float>(visual_peak_source_->buffer[1]), std::memory_order_relaxed);
523 }
524
525 mopo::mopo_float last_played = std::max(engine_.getLastActiveNote(), OUTPUT_WINDOW_MIN_NOTE);
526 int num_pressed = engine_.getPressedNotes().size();
527 int output_inc = std::max<int>(1, engine_.getSampleRate() / mopo::MEMORY_SAMPLE_RATE);
528
529 mopo::mopo_float current_last_played_note = last_played_note_.load(std::memory_order_acquire);
530 if (last_played && (current_last_played_note != last_played || num_pressed > last_num_pressed_.load(std::memory_order_acquire))) {
531 last_played_note_.store(last_played, std::memory_order_release);
532
533 mopo::mopo_float frequency = mopo::utils::midiNoteToFrequency(last_played);
534 mopo::mopo_float period = engine_.getSampleRate() / frequency;
535 int window_length = output_inc * mopo::MEMORY_RESOLUTION;
536
537 memory_reset_period_ = period;
538 while (memory_reset_period_ < window_length)
539 memory_reset_period_ += memory_reset_period_;
540
541 memory_reset_period_ = std::min(memory_reset_period_, 2.0 * window_length);
542 memory_index_ = 0;
543 publishOutputMemory();
544 }
545 last_num_pressed_.store(num_pressed, std::memory_order_release);
546
547 // Load atomics into local temporaries for arithmetic in the audio loop
548 mopo::mopo_float local_input_offset = memory_input_offset_.load(std::memory_order_acquire);
549 int local_memory_index = memory_index_.load(std::memory_order_acquire);
550
551 for (; local_input_offset < samples; local_input_offset += output_inc) {
552 int input_index = mopo::utils::iclamp(static_cast<int>(local_input_offset), 0, samples - 1);
553 local_memory_index = mopo::utils::iclamp(local_memory_index, 0, 2 * mopo::MEMORY_RESOLUTION - 1);
554 MOPO_ASSERT(input_index >= 0);
555 MOPO_ASSERT(input_index < samples);
556 MOPO_ASSERT(local_memory_index >= 0);
557 MOPO_ASSERT(local_memory_index < 2 * mopo::MEMORY_RESOLUTION);
558 output_memory_write_[local_memory_index++] = (left[input_index] + right[input_index]) / 2.0;
559
560 if (local_memory_index * output_inc >= memory_reset_period_) {
561 local_input_offset += memory_reset_period_ - local_memory_index * output_inc;
562 local_memory_index = 0;
563 publishOutputMemory();
564 }
565 }
566
567 // Store temporaries back to atomics
568 memory_input_offset_.store(local_input_offset - samples, std::memory_order_release);
569 memory_index_.store(local_memory_index, std::memory_order_release);
570}

618SynthGuiStateSnapshot SynthBase::captureGuiStateSnapshot() {
619 SynthGuiStateSnapshot snapshot;
620 snapshot.controls.reserve(controls_.size());
621 {
622 const ScopedLock lock(getCriticalSection());
623 GuiControlTable* table = gui_control_table_.load(std::memory_order_acquire);
624 for (const auto& [name, control] : controls_) {
625 if (control == nullptr)
626 continue;
627 mopo::mopo_float value = control->value();
628 if (table != nullptr) {
629 const auto index = table->indices.find(name);
630 if (index != table->indices.end())
631 value = table->values[index->second].load(std::memory_order_acquire);
632 }
633 snapshot.controls.emplace_back(name, value);
634 }
635 snapshot.modulations.reserve(mod_connections_.size());
636 const std::lock_guard<std::mutex> modulation_lock(modulation_snapshot_mutex_);
637 for (const mopo::ModulationConnection* connection : mod_connections_) {
638 if (connection != nullptr) {
639 const auto requested = requested_modulation_amounts_.find(
640 {connection->source, connection->destination});
641 snapshot.modulations.push_back({connection->source, connection->destination,
642 requested != requested_modulation_amounts_.end()
643 ? requested->second
644 : connection->amount.value()});
645 }
646 }
647 }
648 return snapshot;
649}

651SynthGuiStateSnapshot SynthBase::captureInitStateSnapshot() const {
652 SynthGuiStateSnapshot snapshot;
653 snapshot.controls.reserve(controls_.size());
654 for (const auto& [name, control] : controls_) {
655 if (control != nullptr)

661void SynthBase::applyParameterSnapshot(const SynthGuiStateSnapshot& snapshot) {
662 const SynthGuiStateSnapshot current = captureGuiStateSnapshot();
663 for (const auto& [name, value] : snapshot.controls) {
664 const auto control = controls_.find(name);
665 if (control == controls_.end() || control->second == nullptr ||
666 !std::isfinite(static_cast<double>(value)))
667 continue;
668
669 const auto current_value = std::find_if(
670 current.controls.begin(), current.controls.end(),
671 [&name](const auto& entry) { return entry.first == name; });
672 if (current_value != current.controls.end() && current_value->second == value)
673 continue;
674
675 beginChangeGesture(name);
676 valueChangedInternal(name, value);
677 endChangeGesture(name);
678 queueGuiControlChange(name, value);
679 }
680
681 clearModulations();
682 for (const auto& modulation : snapshot.modulations) {
683 if (!mopo::Parameters::isParameter(modulation.destination))
684 continue;
685 auto* connection = modulation_bank_.get(modulation.source, modulation.destination);
686 setModulationAmount(connection, modulation.amount);
687 }
688 requestGuiRefresh();
689}

719SynthBase::GuiControlTable::GuiControlTable(const mopo::control_map& controls) {
720 names.reserve(controls.size());
721 for (const auto& control : controls)
722 names.push_back(control.first);
723
724 const size_t count = names.size();
725 values = std::make_unique<std::atomic<mopo::mopo_float>[]>(count);
726 dirty = std::make_unique<std::atomic<bool>[]>(count);
727 indices.reserve(count);
728 for (size_t i = 0; i < count; ++i) {
729 indices.emplace(names[i], static_cast<int>(i));
730 const auto control = controls.find(names[i]);
731 values[i].store(control != controls.end() && control->second != nullptr
732 ? control->second->value()
733 : 0.0,
734 std::memory_order_relaxed);
735 dirty[i].store(false, std::memory_order_relaxed);
736 }
737}

745void SynthBase::rebuildVisualTelemetryTable() {
746 auto table = std::make_unique<VisualTelemetryTable>();
747 auto addSources = [&table](mopo::output_map& outputs) {
748 for (const auto& entry : outputs) {
749 const mopo::Output* output = entry.second;
750 if (output == nullptr || table->indices.count(output) != 0)
751 continue;
752 int index = static_cast<int>(table->sources.size());
753 table->indices.emplace(output, index);
754 table->sources.push_back(output);
755 }
756 };
757
758 addSources(engine_.getModulationSources());
759 addSources(engine_.getMonoModulations());
760 addSources(engine_.getPolyModulations());
761 addSources(engine_.getVisualTelemetrySources());
762 table->values.resize(table->sources.size(), 0.0f);
763 visual_telemetry_table_.store(table.get(), std::memory_order_release);
764 visual_telemetry_tables_.push_back(std::move(table));

776bool SynthBase::copyVisualTelemetry(std::span<const int> indices,
777 std::span<float> destination) const {
778 if (destination.size() < indices.size())
779 return false;
780
781 VisualTelemetryTable* table = visual_telemetry_table_.load(std::memory_order_acquire);
782 if (table == nullptr || table->busy.test_and_set(std::memory_order_acquire))
783 return false;
784
785 for (size_t i = 0; i < indices.size(); ++i) {
786 if (indices[i] < 0 || static_cast<size_t>(indices[i]) >= table->values.size()) {
787 table->busy.clear(std::memory_order_release);
788 return false;
789 }
790 destination[i] = table->values[indices[i]];
791 }
792 table->busy.clear(std::memory_order_release);
793 return true;
794}

810// Real-time safe: no allocation, no lock, no message posting.
811void SynthBase::queueGuiControlChange(const std::string& name, mopo::mopo_float value) {
812 if (!gui_updates_enabled_.load(std::memory_order_acquire))
813 return;
814
815 GuiControlTable* table = gui_control_table_.load(std::memory_order_acquire);
816 if (table == nullptr)
817 return;
818
819 auto index_it = table->indices.find(name);
820 if (index_it == table->indices.end())
821 return;
822
823 table->values[index_it->second].store(value, std::memory_order_release);
824 table->dirty[index_it->second].store(true, std::memory_order_release);
825 gui_changes_pending_.store(true, std::memory_order_release);
826}

836void SynthBase::deliverGuiNotifications() {
837 if (!gui_updates_enabled_.load(std::memory_order_acquire))
838 return;
839
840 std::vector<std::pair<const std::string*, mopo::mopo_float>> changes;
841 if (gui_changes_pending_.exchange(false, std::memory_order_acq_rel)) {
842 GuiControlTable* table = gui_control_table_.load(std::memory_order_acquire);
843 if (table != nullptr) {
844 for (size_t i = 0; i < table->names.size(); ++i) {
845 if (table->dirty[i].exchange(false, std::memory_order_acquire))
846 changes.emplace_back(&table->names[i], table->values[i].load(std::memory_order_relaxed));
847 }
848 }
849 }
850
851 const bool refresh = gui_refresh_pending_.exchange(false, std::memory_order_acq_rel);
852 const bool fresh = gui_fresh_pending_.exchange(false, std::memory_order_acq_rel);
853 if (!refresh && !fresh && changes.empty())
854 return;
855
856 auto gui_interface = getGuiInterface();
857 if (!gui_interface.has_value())
858 return;
859
860 if (refresh)
861 gui_interface.value()->updateFullGui();
862 else {
863 for (const auto& [name, value] : changes)
864 gui_interface.value()->updateGuiControl(*name, value);
865 }
866 if (fresh)
867 gui_interface.value()->notifyFresh();
868 else if (!changes.empty())
869 gui_interface.value()->notifyChange();
870}
