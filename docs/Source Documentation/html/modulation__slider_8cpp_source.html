39ModulationSlider::ModulationSlider(SynthSlider* destination) : SynthSlider(destination->getName()) {
40 destination_slider_ = destination;
41
42 float destination_range = destination->getMaximum() - destination->getMinimum();
43 setName(destination->getName());
44 if (details_.steps)
45 setRange(-destination_range, destination_range,
46 destination_range / (details_.steps - 1));
47 else
48 setRange(-destination_range, destination_range);
49 setDoubleClickReturnValue(true, 0.0f);
50 setSliderStyle(destination->getSliderStyle());
51 setTextBoxStyle(Slider::NoTextBox, true, 0, 0);
52 setPostMultiply(destination->getPostMultiply());
53 setUnits(destination->getUnits());
54 setScalingType(destination->getScalingType());
55 setPopupPlacement(destination->getPopupPlacement(), destination->getPopupBuffer());
56
57 destination->addListener(this);
58
59 if (destination->isRotary())
60 setMouseDragSensitivity(2.0f * getMouseDragSensitivity());
61 else
62 setVelocityBasedMode(true);
63 setOpaque(false);
64}

74void ModulationSlider::mouseDown(const juce::MouseEvent &e) {
75 if (e.mods.isPopupMenu()) {
76 if (getValue() != 0.0) {
77 PopupMenu m;
78 m.setLookAndFeel(DefaultLookAndFeel::instance());
79 m.addItem(kClearModulation, "Clear Modulation");
80 m.showMenuAsync(PopupMenu::Options(),
81 ModalCallbackFunction::forComponent(sliderPopupCallback, this));
82 }
83 }
84 else
85 SynthSlider::mouseDown(e);
86}
