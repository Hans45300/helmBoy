36ModulationSlider::ModulationSlider(SynthSlider* destination) : SynthSlider(destination->getName()) {
37 destination_slider_ = destination;
38
39 float destination_range =
40 mopo::modulationAmountLimit(destination->getMinimum(), destination->getMaximum());
41 setName(destination->getName());
42 if (details_.steps)
43 setRange(-destination_range, destination_range,
44 destination_range / (details_.steps - 1));
45 else
46 setRange(-destination_range, destination_range);
47 setDoubleClickReturnValue(true, 0.0f);
48 setSliderStyle(destination->getSliderStyle());
49 setTextBoxStyle(Slider::NoTextBox, true, 0, 0);
50 setPostMultiply(destination->getPostMultiply());
51 setUnits(destination->getUnits());
52 setScalingType(destination->getScalingType());
53 setPopupPlacement(destination->getPopupPlacement(), destination->getPopupBuffer());
54
55 destination->addListener(this);
56
57 if (destination->isRotary())
58 setMouseDragSensitivity(2.0f * getMouseDragSensitivity());
59 else
60 setVelocityBasedMode(false);
61 setOpaque(false);
62}

73void ModulationSlider::mouseDown(const juce::MouseEvent &e) {
74 if (e.mods.isPopupMenu()) {
75 PopupMenu m;
76 m.setLookAndFeel(DefaultLookAndFeel::instance());
77 if (getValue() != 0.0)
78 m.addItem(kClearModulation, "Clear Modulation");
79 m.addItem(kEnterValue, "Enter Value...");
80
81 showModulationPopup(m, this, false, kEnterValue + 1,
82 [this](int result) { handlePopupResult(result); });
83 }
84 else
85 SynthSlider::mouseDown(e);
86}

93void ModulationSlider::handlePopupResult(int result) {

97 promptForDirectValueEntry();
