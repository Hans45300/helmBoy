65void ReverbSection::resized() {
66 int title_width = getTitleWidth();
67 on_->setBounds(size_ratio_ * 2.0f, 0, title_width, title_width);
68
69 int knob_width = getStandardKnobSize();
70 float space = (getWidth() - (3.0f * knob_width)) / 4.0f;
71 int y = size_ratio_ * 30;
72
73 feedback_->setBounds(space, y, knob_width, knob_width);
74 damping_->setBounds((knob_width + space) + space, y, knob_width, knob_width);
75 dry_wet_->setBounds(2 * (knob_width + space) + space, y, knob_width, knob_width);
76
77 SynthSection::resized();
78}
