88void ReverbSection::resized() {
89 int title_width = getTitleWidth();
90 on_->setBounds(size_ratio_ * 2.0f, 0, title_width, title_width);
91 int knob_width = getSmallKnobSize();
92 int freeze_y = title_width + size_ratio_ * 2.0f;
93 freeze_->setBounds(size_ratio_ * 2.0f, freeze_y, title_width, title_width);
94
95 int label_height = size_ratio_ * 10.0f;
96 int knob_y = getHeight() - knob_width - label_height - size_ratio_ * 3.0f;
97 int slot_width = getWidth() / 4;
98 feedback_->setBounds((slot_width - knob_width) / 2, knob_y, knob_width, knob_width);
99 damping_->setBounds(slot_width + (slot_width - knob_width) / 2, knob_y, knob_width, knob_width);
100 dry_wet_->setBounds(2 * slot_width + (slot_width - knob_width) / 2, knob_y, knob_width, knob_width);
101 stereo_width_->setBounds(3 * slot_width + (slot_width - knob_width) / 2, knob_y, knob_width, knob_width);
102
103 SynthSection::resized();
104}
