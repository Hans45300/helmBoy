194void OscillatorSection::resized() {
195 int title_width = getTitleWidth();
196 int knob_width = getStandardKnobSize();
197 int tune_width = getSmallKnobSize();
198 int text_width = size_ratio_ * TEXT_WIDTH;
199 int text_height = size_ratio_ * TEXT_HEIGHT;
200 int trans_width = size_ratio_ * TRANS_WIDTH;
201 int wave_selector_height = size_ratio_ * WAVE_SELECTOR_HEIGHT;
202 int knob_section_height = size_ratio_ * KNOB_SECTION_HEIGHT;
203
204 float cross_mod_width = CROSS_MOD_WIDTH_PERCENT * getWidth();
205 float osc_width = (getWidth() - cross_mod_width) / 2.0f;
206 float osc_height = getHeight() - title_width - wave_selector_height - knob_section_height;
207 float osc_y = title_width + wave_selector_height;
208
209 wave_selector_1_->setBounds(0.0f, title_width, osc_width, wave_selector_height);
210 wave_selector_2_->setBounds(getWidth() - osc_width, title_width, osc_width, wave_selector_height);
211 wave_viewer_1_->setBounds(0.0f, osc_y, osc_width, osc_height);
212 wave_viewer_2_->setBounds(getWidth() - osc_width, osc_y, osc_width, osc_height);
213
214 cross_modulation_->setBounds((getWidth() - knob_width) / 2.0f,
215 osc_y + (osc_height - knob_width) / 2.0f,
216 knob_width, knob_width);
217
218 float space = (getWidth() - (2.0f * text_width + 2.0f * trans_width + 2.0f * tune_width)) / 5.0f;
219 int knob_bottom = getHeight() - size_ratio_ * 22.0f;
220
221 tune_1_->setBounds(space * 1.75f, knob_bottom - tune_width, tune_width, tune_width);
222
223 transpose_1_->setBounds(tune_width + space * 1.75f, knob_bottom - trans_width,
224 trans_width, trans_width);
225
226 unison_detune_1_->setBounds(trans_width + tune_width + 2.45f * space,
227 knob_bottom - text_height,
228 text_width, text_height);
229
230 unison_voices_1_->setBounds(trans_width + tune_width + 2.45f * space,
231 knob_bottom - 2 * text_height,
232 text_width - text_height, text_height);
233
234 unison_harmonize_1_->setBounds(unison_voices_1_->getX() + unison_voices_1_->getWidth(),
235 unison_voices_1_->getY(),
236 text_height, text_height);
237
238 unison_detune_2_->setBounds(trans_width + text_width + tune_width + 2.55f * space,
239 knob_bottom - text_height,
240 text_width, text_height);
241
242 unison_voices_2_->setBounds(trans_width + text_width + tune_width + 2.55f * space,
243 knob_bottom - 2 * text_height,
244 text_width - text_height, text_height);
245
246 unison_harmonize_2_->setBounds(unison_voices_2_->getX() + unison_voices_2_->getWidth(),
247 unison_voices_2_->getY(),
248 text_height, text_height);
249
250 transpose_2_->setBounds(trans_width + 2 * text_width + tune_width + 3.25f * space,
251 knob_bottom - trans_width, trans_width, trans_width);
252
253 tune_2_->setBounds(2 * trans_width + 2 * text_width + tune_width + 3.25f * space,
254 knob_bottom - tune_width, tune_width, tune_width);
255
256 fm_amount_->setBounds((getWidth() - (knob_width / 2.0f)),
257 getHeight() - knob_width*.85f,
258 knob_width / 2.0f, knob_width / 2.0f);
259
260 ring_mod_->setBounds(0.0f,
261 getHeight() - knob_width*.85f,
262 knob_width / 2.0f, knob_width / 2.0f);
263
264 phase_stretch_1_->setBounds(0.0f,
265 osc_y + osc_height,
266 knob_width / 2.0f, knob_width / 2.0f);
267
268 phase_stretch_2_->setBounds((getWidth() - (knob_width / 2.0f)),
269 osc_y + osc_height,
270 knob_width / 2.0f, knob_width / 2.0f);
271
272 float cross_x_padding = size_ratio_ * 8.0f;
273 float cross_y_padding = size_ratio_ * 8.0f;
274 float cross_width = cross_mod_width - 2.0f * cross_x_padding;
275 float cross_height = osc_height - 2.0f * cross_y_padding;
276 float cross_percent = 0.2f;
277
278 top_left_cross_path_.clear();
279 top_left_cross_path_.startNewSubPath(osc_width, osc_y + cross_y_padding);
280 top_left_cross_path_.lineTo(osc_width + cross_x_padding, osc_y + cross_y_padding);
281 top_left_cross_path_.lineTo(osc_width + cross_x_padding + cross_percent * cross_width,
282 osc_y + cross_y_padding + cross_percent * cross_height);
283
284 top_right_cross_path_.clear();
285 top_right_cross_path_.startNewSubPath(getWidth() - osc_width, osc_y + cross_y_padding);
286 top_right_cross_path_.lineTo(getWidth() - osc_width - cross_x_padding, osc_y + cross_y_padding);
287 top_right_cross_path_.lineTo(getWidth() - osc_width - cross_x_padding -
288 cross_percent * cross_width,
289 osc_y + cross_y_padding + cross_percent * cross_height);
290
291 bottom_left_cross_path_.clear();
292 bottom_left_cross_path_.startNewSubPath(osc_width, osc_y + osc_height - cross_y_padding);
293 bottom_left_cross_path_.lineTo(osc_width + cross_x_padding, osc_y + osc_height - cross_y_padding);
294 bottom_left_cross_path_.lineTo(osc_width + cross_x_padding + cross_percent * cross_width,
295 osc_y + osc_height - cross_y_padding -
296 cross_percent * cross_height);
297
298 bottom_right_cross_path_.clear();
299 bottom_right_cross_path_.startNewSubPath(getWidth() - osc_width,
300 osc_y + osc_height - cross_y_padding);
301 bottom_right_cross_path_.lineTo(getWidth() - osc_width - cross_x_padding,
302 osc_y + osc_height - cross_y_padding);
303 bottom_right_cross_path_.lineTo(getWidth() - osc_width - cross_x_padding -
304 cross_percent * cross_width,
305 osc_y + osc_height - cross_y_padding -
306 cross_percent * cross_height);
307
308 SynthSection::resized();
309
310 unison_detune_1_->setPopupDisplayEnabled(false, false, nullptr);
311 unison_detune_2_->setPopupDisplayEnabled(false, false, nullptr);
312 unison_voices_1_->setPopupDisplayEnabled(false, false, nullptr);
313 unison_voices_2_->setPopupDisplayEnabled(false, false, nullptr);
314}
