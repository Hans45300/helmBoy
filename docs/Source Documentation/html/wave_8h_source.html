56 class WaveLookup {
57 public:
61 WaveLookup() {
62 // Sin lookup table.
63 for (int i = 0; i < mopo::LOOKUP_SIZE + 1; ++i)
64 sin_[i] = std::sin((2.0 * mopo::PI * i) / mopo::LOOKUP_SIZE);
65
66 // Square lookup table.
67 for (int i = 0; i < mopo::LOOKUP_SIZE + 1; ++i) {
68 int p = i;
69 constexpr mopo_float scale = 4.0 / mopo::PI;
70 square_[0][i] = scale * sin_[p];
71 for (int h = 1; h < mopo::MAX_HARMONICS; ++h) {
72 p = (p + i) % mopo::LOOKUP_SIZE;
73 square_[h][i] = square_[h - 1][i];
74 if (h % 2 == 0)
75 square_[h][i] += scale * sin_[p] / static_cast<mopo_float>(h + 1);
76 }
77 }
78
79 // Saw lookup table.
80 for (int i = 0; i < mopo::LOOKUP_SIZE + 1; ++i) {
81 int index = (i + (mopo::LOOKUP_SIZE / 2)) % mopo::LOOKUP_SIZE;
82 int p = i;
83 constexpr mopo_float scale = 2.0 / mopo::PI;
84 saw_[0][index] = scale * sin_[p];
85 for (int h = 1; h < mopo::MAX_HARMONICS; ++h) {
86 p = (p + i) % mopo::LOOKUP_SIZE;
87 mopo_float harmonic = scale * sin_[p] / static_cast<mopo_float>(h + 1);
88 if (h % 2 == 0)
89 saw_[h][index] = saw_[h - 1][index] + harmonic;
90 else
91 saw_[h][index] = saw_[h - 1][index] - harmonic;
92 }
93 }
94
95 // Triangle lookup table.
96 for (int i = 0; i < mopo::LOOKUP_SIZE + 1; ++i) {
97 int p = i;
98 constexpr mopo_float scale = 8.0 / (mopo::PI * mopo::PI);
99 triangle_[0][i] = scale * sin_[p];
100 for (int h = 1; h < mopo::MAX_HARMONICS; ++h) {
101 p = (p + i) % mopo::LOOKUP_SIZE;
102 triangle_[h][i] = triangle_[h - 1][i];
103 mopo_float harmonic = scale * sin_[p] / (static_cast<mopo_float>((h + 1) * (h + 1)));
104 if (h % 4 == 0)
105 triangle_[h][i] += harmonic;
106 else if (h % 2 == 0)
107 triangle_[h][i] -= harmonic;
108 }
109 }
110 }

173 template<size_t steps> requires (steps > 1)
174 [[nodiscard]] inline mopo_float step(mopo_float t, int harmonics) const {
175 constexpr mopo_float step_size = (1.0 * steps) / (steps - 1);
176 return step_size * (upsaw(t, harmonics) +
177 downsaw(steps * t, harmonics / steps) / steps);
178 }
179
183 template<size_t steps> requires (steps > 1)
184 [[nodiscard]] inline mopo_float pyramid(mopo_float t, int harmonics) const {
185 constexpr size_t squares = steps - 1;
186 constexpr mopo_float phase_increment = 1.0 / (2.0 * squares);
187
188 mopo_float phase = 0.75 + t;
189 mopo_float out = 0.0;
190
191 mopo_float integral;
192 for (size_t i = 0; i < squares; ++i) {
193 out += square(utils::mod(phase, &integral), harmonics);
194 phase += phase_increment;
195 }
196 out /= squares;
197 return out;
198 }

204 mopo_float saw_[MAX_HARMONICS][LOOKUP_SIZE + 1];
205 mopo_float triangle_[MAX_HARMONICS][LOOKUP_SIZE + 1];
206 };

251 [[nodiscard]] static inline mopo_float blwave(Type waveform, mopo_float t,
252 mopo_float frequency) {
253 const WaveLookup* lookup = WaveLookup::instance();
254 if (std::fabs(frequency) < 1)
255 return Wave::wave(waveform, t);
256 int harmonics = mopo::HIGH_FREQUENCY / static_cast<int>(std::fabs(frequency)) - 1;
257 if (harmonics >= mopo::MAX_HARMONICS)
258 return Wave::wave(waveform, t);
259
260 switch (waveform) {
261 case Type::Sin:
262 return lookup->fullsin(t);
263 case Type::Triangle:
264 return lookup->triangle(t, harmonics);
265 case Type::Square:
266 return lookup->square(t, harmonics);
267 case Type::DownSaw:
268 return lookup->downsaw(t, harmonics);
269 case Type::UpSaw:
270 return lookup->upsaw(t, harmonics);
271 case Type::ThreeStep:
272 return lookup->step<3>(t, harmonics);
273 case Type::FourStep:
274 return lookup->step<4>(t, harmonics);
275 case Type::EightStep:
276 return lookup->step<8>(t, harmonics);
277 case Type::ThreePyramid:
278 return lookup->pyramid<3>(t, harmonics);
279 case Type::FivePyramid:
280 return lookup->pyramid<5>(t, harmonics);
281 case Type::NinePyramid:
282 return lookup->pyramid<9>(t, harmonics);
283 default:
284 return Wave::wave(waveform, t);
285 }
286 }

325 [[nodiscard]] static inline mopo_float sampleAndGlideValue(
326 mopo_float from, mopo_float to, mopo_float phase) {
327 const mopo_float interpolation = (1.0 - std::cos(PI * phase)) / 2.0;
328 return utils::interpolate(from, to, interpolation);

331 [[nodiscard]] static inline mopo_float sampleAndHoldValue(
332 mopo_float held, mopo_float next, mopo_float phase) {
333 constexpr mopo_float transition_start = 0.9;
334 if (phase < transition_start)
335 return held;
336 const mopo_float transition_phase = (phase - transition_start) /
337 (1.0 - transition_start);
338 return sampleAndGlideValue(held, next, transition_phase);
339 }

341 [[nodiscard]] static inline mopo_float smoothRandomValue(
342 const mopo_float* points, mopo_float phase) {
343 const mopo_float position = phase * 4.0;
344 const int index = utils::iclamp(static_cast<int>(position), 0, 3);
345 return sampleAndGlideValue(points[index], points[index + 1], position - index);
346 }
