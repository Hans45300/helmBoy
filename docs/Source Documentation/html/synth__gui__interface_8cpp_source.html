41bool SynthGuiInterface::showExportDialog() {
42 if (gui_ == nullptr)
43 return false;
44
45 export_chooser_ = std::make_unique<FileChooser>("Export Patch", File(), String("*.") + mopo::PATCH_EXTENSION);
46 export_chooser_->launchAsync(FileBrowserComponent::saveMode | FileBrowserComponent::canSelectFiles,
47 [safe_gui = Component::SafePointer<FullInterface>(gui_.get()),
48 weak_lifetime = synth_->getGuiLifetime()](const FileChooser& chooser) {
49 auto file = chooser.getResult();
50 auto lifetime = weak_lifetime.lock();
51 if (file == File() || lifetime == nullptr || safe_gui == nullptr)
52 return;
53
54 const std::lock_guard<std::mutex> lock(lifetime->mutex);
55 if (lifetime->synth == nullptr)
56 return;
57
58 if (lifetime->synth->saveToFile(file) && safe_gui != nullptr)
59 safe_gui->externalPatchLoaded(file);
60 });
61 return true;
62}

64bool SynthGuiInterface::initializeGui() {
65 if (gui_ != nullptr)
66 return true;
67
68 appendStartupTraceGui("SynthGuiInterface::initializeGui preflight start");
69 // Keep GUI path lightweight: only ensure deferred module/control maps needed
70 // by FullInterface construction. Full startup checks (MIDI/config) run later
71 // in startAudioSubsystem.
72 if (!synth_->ensureGuiPreflight()) {
73 appendStartupTraceGui("SynthGuiInterface::initializeGui preflight failed");
74 gui_ = nullptr;
75 return false;
76 }
77 appendStartupTraceGui("SynthGuiInterface::initializeGui preflight done");
78
79 try {
80 appendStartupTraceGui("SynthGuiInterface::initializeGui FullInterface ctor start");
81 auto controls = synth_->getControls();
82 auto mod_sources = synth_->getEngine()->getModulationSources();
83 auto mono_mods = synth_->getEngine()->getMonoModulations();
84 auto poly_mods = synth_->getEngine()->getPolyModulations();
85 auto kbd_state = synth_->getKeyboardState();
86 gui_ = std::make_unique<FullInterface>(synth_, controls, mod_sources, mono_mods, poly_mods, kbd_state);
87 synth_->startGuiUpdates();
88 appendStartupTraceGui("SynthGuiInterface::initializeGui FullInterface ctor done");
89 return true;
90 }
91 catch (const std::exception& ex) {
92 appendStartupTraceGui("SynthGuiInterface::initializeGui exception: " + String(ex.what()));
93 Logger::writeToLog("[Standalone Startup] SynthGuiInterface::initializeGui exception: " + String(ex.what()));
94 }
95 catch (...) {
96 appendStartupTraceGui("SynthGuiInterface::initializeGui unknown exception");
97 Logger::writeToLog("[Standalone Startup] SynthGuiInterface::initializeGui unknown exception.");
98 }
99
100 gui_ = nullptr;
101 return false;
102}
