48 [[nodiscard]] mopo_float centsLookup(mopo_float cents_from_0) const {
49 mopo_float clamped_cents = utils::clamp(cents_from_0, 0.0, MAX_CENTS);
50 int full_cents = static_cast<int>(clamped_cents);
51 mopo_float fraction_cents = clamped_cents - full_cents;
52
53 return utils::interpolate(frequency_lookup_[full_cents],
54 frequency_lookup_[full_cents + 1], fraction_cents);
55 }

57 private:
58 mopo_float frequency_lookup_[MAX_CENTS + 2];
59 };
