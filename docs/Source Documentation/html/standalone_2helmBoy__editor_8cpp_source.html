35HelmBoyEditor::HelmBoyEditor(bool use_gui) : SynthGuiInterface(this, use_gui) {
36 computer_keyboard_ = std::make_unique<HelmBoyComputerKeyboard>(&engine_, keyboard_state_.get());
37
38 setAudioChannels(0, mopo::NUM_CHANNELS);
39
40 AudioDeviceManager::AudioDeviceSetup setup;
41 deviceManager.getAudioDeviceSetup(setup);
42 setup.sampleRate = mopo::DEFAULT_SAMPLE_RATE;
43 deviceManager.initialise(0, mopo::NUM_CHANNELS, nullptr, true, "", &setup);
44
45 if (deviceManager.getCurrentAudioDevice() == nullptr) {
46 const OwnedArray<AudioIODeviceType>& device_types = deviceManager.getAvailableDeviceTypes();
47
48 for (auto* device_type : device_types) {
49 deviceManager.setCurrentAudioDeviceType(device_type->getTypeName(), true);
50 if (deviceManager.getCurrentAudioDevice())
51 break;
52 }
53 }
54
55
56// Récupère la liste des périphériques MIDI disponibles
57juce::Array<juce::MidiDeviceInfo> all_midi_ins = juce::MidiInput::getAvailableDevices();
58
59// Active chaque périphérique MIDI
60std::ranges::for_each(std::views::iota(0, all_midi_ins.size()), [&](int i) {
61 deviceManager.setMidiInputEnabled(all_midi_ins[i].identifier, true);
62});
63
64// Ajoute le callback MIDI ("" = tous les périphériques)
65deviceManager.addMidiInputCallback({}, midi_manager_.get());
66
67 if (use_gui) {
68 setLookAndFeel(DefaultLookAndFeel::instance());
69 addAndMakeVisible(gui_.get());
70 gui_->setOutputMemory(getOutputMemory());
71 float window_size = LoadSave::loadWindowSize();
72 setSize(window_size * mopo::DEFAULT_WINDOW_WIDTH, window_size * mopo::DEFAULT_WINDOW_HEIGHT);
73
74 setWantsKeyboardFocus(true);
75 addKeyListener(computer_keyboard_.get());
76 setOpaque(true);
77 }
78}

95void HelmBoyEditor::getNextAudioBlock(const AudioSourceChannelInfo& buffer) {
96 ScopedLock lock(getCriticalSection());
97
98 int num_samples = buffer.buffer->getNumSamples();
99 int synth_samples = std::min(num_samples, MAX_BUFFER_PROCESS);
100
101 processControlChanges();
102 processModulationChanges();
103 MidiBuffer midi_messages;
104 midi_manager_->removeNextBlockOfMessages(midi_messages, num_samples);
105 processMidi(midi_messages);
106 processKeyboardEvents(midi_messages, num_samples);
107
108 for (auto b = 0; b < num_samples; b += synth_samples) {
109 auto current_samples = std::min<int>(synth_samples, num_samples - b);
110
111 processAudio(buffer.buffer, mopo::NUM_CHANNELS, current_samples, b);
112 }
113}
