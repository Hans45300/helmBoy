89void ArpSection::paintBackground(Graphics& g) {
90 SynthSection::paintContainer(g);
91 float knob_width = getStandardKnobSize();
92 float text_height = size_ratio_ * TEXT_HEIGHT;
93
94 int title_width = getTitleWidth();
95 float shadow_left = getTitleWidth() - size_ratio_ * SHADOW_WIDTH;
96
97 g.setGradientFill(ColourGradient(Colour(Colors::Color_22000000), shadow_left, 0.0f,
98 Colour(Colors::Color_66000000), title_width, 0.0f,
99 false));
100 g.fillRoundedRectangle(0, 0, title_width, getHeight(), 1.0f);
101
102 g.setColour(Colors::control_label_text);
103 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(size_ratio_ * 10.0f));
104
105 drawTextForComponent(g, TRANS("GATE"), gate_.get());
106 drawTextForComponent(g, TRANS("OCTAVES"), octaves_.get());
107
108 int font_y = gate_->getBounds().getY() + knob_width + size_ratio_ * 4;
109 g.drawText(TRANS("FREQUENCY"),
110 frequency_->getBounds().getX() - size_ratio_ * 5, font_y,
111 frequency_->getBounds().getWidth() + text_height + size_ratio_ * 10,
112 size_ratio_ * 10, Justification::centred, false);
113 g.drawText(TRANS("PATTERN"),
114 pattern_->getBounds().getX() - size_ratio_ * 5, font_y,
115 pattern_->getBounds().getWidth() + size_ratio_ * 10,
116 size_ratio_ * 10, Justification::centred, false);
117
118 g.saveState();
119 g.addTransform(AffineTransform::rotation(-static_cast<float>(mopo::PI) / 2.0f, 0, 0));
120 g.setColour(Colors::tab_heading_text);
121 g.setFont(Fonts::instance()->proportional_light().withPointHeight(size_ratio_ * 13.40f));
122 g.drawText(getName(), -getHeight(), 0, getHeight() - title_width, title_width,
123 Justification::centred, false);
124 g.restoreState();
125
126 paintKnobShadows(g);
127}

129void ArpSection::resized() {
130 int knob_width = getStandardKnobSize();
131 int title_width = getTitleWidth();
132 int text_width = size_ratio_ * TEXT_WIDTH;
133 int text_height = size_ratio_ * TEXT_HEIGHT;
134
135 float space = (getWidth() - 2 * (text_width + text_height + knob_width) - title_width) / 5.0f;
136 int y = size_ratio_ * 4;
137
138 int text_y = y + size_ratio_ * 12;
139 frequency_->setBounds(title_width + space, text_y, text_width, text_height);
140 sync_->setBounds(title_width + space + text_width, text_y, text_height, text_height);
141 gate_->setBounds(title_width + 2 * space + text_height + text_width, y, knob_width, knob_width);
142 octaves_->setBounds(title_width + 3 * space + text_height + text_width + knob_width, y,
143 knob_width, knob_width);
144 pattern_->setBounds(title_width + 4 * space + text_height + text_width + 2 * knob_width, text_y,
145 text_width + text_height, text_height);
146 tempo_->setBounds(frequency_->getBounds());
147 on_->setBounds(0, size_ratio_ * 2, title_width, title_width);
148
149 SynthSection::resized();
150
151 frequency_->setPopupDisplayEnabled(false, false, nullptr);
152 tempo_->setPopupDisplayEnabled(false, false, nullptr);
153}
