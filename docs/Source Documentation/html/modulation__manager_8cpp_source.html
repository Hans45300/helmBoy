38ModulationManager::ModulationManager(
39 mopo::output_map modulation_sources,
40 std::map<std::string, ModulationButton*> modulation_buttons,
41 std::map<std::string, SynthSlider*> sliders,
42 mopo::output_map mono_modulations,
43 mopo::output_map poly_modulations) : SynthSection("modulation") {
44 modulation_buttons_ = modulation_buttons;
45 modulation_sources_ = modulation_sources;
46 setInterceptsMouseClicks(false, true);
47 // startTimerHz(FRAMES_PER_SECOND);
48
49 current_modulator_ = "";
50
51 polyphonic_destinations_ = std::make_unique<Component>();
52 polyphonic_destinations_->setInterceptsMouseClicks(false, true);
53
54 monophonic_destinations_ = std::make_unique<Component>();
55 monophonic_destinations_->setInterceptsMouseClicks(false, true);
56
57 for (auto& mod_button : modulation_buttons_) {
58 mod_button.second->addListener(this);
59 mod_button.second->addDisconnectListener(this);
60
61 // Create modulation highlight overlays.
62 std::string name = mod_button.second->getName().toStdString();
63 ModulationHighlight* overlay = new ModulationHighlight();
64 addChildComponent(overlay);
65 overlay_lookup_[name] = overlay;
66 overlay->setName(name);
67 Rectangle<int> local_bounds = mod_button.second->getBoundsInParent();
68 overlay->setBounds(mod_button.second->getParentComponent()->localAreaToGlobal(local_bounds));
69 }
70
71 slider_model_lookup_ = sliders;
72 for (auto& slider : slider_model_lookup_) {
73 std::string name = slider.first;
74 const mopo::Output* mono_total = mono_modulations[name];
75 const mopo::Output* poly_total = poly_modulations[name];
76
77 slider.second->addSliderListener(this);
78
79 // Create modulation meter.
80 if (mono_total) {
81 std::string name = slider.second->getName().toStdString();
82 ModulationMeter* meter = new ModulationMeter(mono_total, poly_total, slider.second);
83 addChildComponent(meter);
84 meter_lookup_[name] = meter;
85 meter->setName(name);
86 Rectangle<int> local_bounds = slider.second->getBoundsInParent();
87 meter->setBounds(slider.second->getParentComponent()->localAreaToGlobal(local_bounds));
88 }
89
90 // Create modulation slider.
91 ModulationSlider* mod_slider = new ModulationSlider(slider.second);
92 mod_slider->setLookAndFeel(ModulationLookAndFeel::instance());
93 mod_slider->addListener(this);
94 if (poly_total)
95 polyphonic_destinations_->addAndMakeVisible(mod_slider);
96 else
97 monophonic_destinations_->addAndMakeVisible(mod_slider);
98
99 slider_lookup_[name] = mod_slider;
100 owned_sliders_.push_back(mod_slider);
101 }
102
103 addAndMakeVisible(polyphonic_destinations_.get());
104 addAndMakeVisible(monophonic_destinations_.get());
105
106 forgetModulator();
107}

109ModulationManager::~ModulationManager() {

112 for (auto& overlay : overlay_lookup_)

115 delete slider;

121void ModulationManager::resized() {
122 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
123 polyphonic_destinations_->setBounds(getBounds());
124 monophonic_destinations_->setBounds(getBounds());
125
126 // Update modulation slider locations.
127 for (auto& slider : slider_lookup_) {
128 SynthSlider* model = slider_model_lookup_[slider.first];
129 Point<float> local_top_left = getLocalPoint(model, Point<float>(0.0f, 0.0f));
130 slider.second->setVisible(model->isVisible());
131 slider.second->setBounds(local_top_left.x, local_top_left.y,
132 model->getWidth(), model->getHeight());
133 }
134
135 // Update modulation meter locations.
136 for (auto& meter : meter_lookup_) {
137 Slider* model = slider_model_lookup_[meter.first];
138 Point<float> local_top_left = getLocalPoint(model, Point<float>(0.0f, 0.0f));
139 meter.second->setBounds(local_top_left.x, local_top_left.y,
140 model->getWidth(), model->getHeight());
141 if (parent) {
142 int num_modulations = parent->getSynth()->getNumModulations(meter.first);
143 meter.second->setModulated(num_modulations);
144 }
145 }
146
147 // Update modulation highlight overlay locations.
148 for (auto& overlay : overlay_lookup_) {
149 ModulationButton* model = modulation_buttons_[overlay.first];
150 Point<float> local_top_left = getLocalPoint(model, Point<float>(0.0f, 0.0f));
151 overlay.second->setBounds(local_top_left.x, local_top_left.y,
152 model->getWidth(), model->getHeight());
153 }
154
155 SynthSection::resized();
156}

158void ModulationManager::buttonClicked(juce::Button *clicked_button) {
159 std::string name = clicked_button->getName().toStdString();
160 if (clicked_button->getToggleState()) {
161 if (current_modulator_ != "") {
162 Button* modulator = modulation_buttons_[current_modulator_];
163 modulator->setToggleState(false, NotificationType::dontSendNotification);
164 }
165 changeModulator(name);
166 }
167 else
168 forgetModulator();
169}

217void ModulationManager::timerCallback() {
218 for (auto& slider : slider_lookup_) {
219 SynthSlider* model = slider_model_lookup_[slider.first];
220 slider.second->setVisible(model->isVisible());
221 }
222
223 updateModulationValues();
224
225 for (auto& meter : meter_lookup_) {
226 bool show = meter.second->isModulated() && slider_model_lookup_[meter.first]->isVisible();
227 meter.second->setVisible(show);
228 if (show)
229 meter.second->updateDrawing();
230 }
231}

270void ModulationManager::setSliderValues() {
271 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
272 if (parent == nullptr)
273 return;
274
275 std::vector<mopo::ModulationConnection*> connections =
276 parent->getSynth()->getSourceConnections(current_modulator_);
277 for (auto& slider : slider_lookup_) {
278 std::string destination_name = slider.second->getName().toStdString();
279 float value = 0.0f;
280
281 for (auto* connection : connections) {
282 if (connection->destination == destination_name) {
283 value = connection->amount.value();
284 break;
285 }
286 }
287 slider.second->setValue(value, NotificationType::dontSendNotification);
288 slider.second->repaint();
289 }
290}
