49 void NoiseOscillator::resetColorFilter() {
50 const mopo_float safe_sample_rate = std::max(1, sample_rate_);
51 color_filter_coefficient_ =
52 1.0 - std::exp(-2.0 * PI * kNoiseColorFilterCutoff / safe_sample_rate);
53 color_filter_state_ = 0.0;
54 color_smoothing_samples_ = std::max(1, sample_rate_ / 100);
55 color_smoothing_remaining_ = 0;
56 }

58 void NoiseOscillator::process() {
59 mopo_float amplitude = input(kAmplitude)->source->buffer[0];
60 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
61
62 if (amplitude == 0.0) {
63 if (dest[0] != 0.0 || dest[dest.size() - 1] != 0.0)
64 utils::zeroBuffer(dest);
65 return;
66 }
67
68 const Output* color_source = input(kColor)->source;
69 const mopo_float requested_color =
70 color_source == &null_source_ ? kNoiseColorNeutral : input(kColor)->at(0);
71 const mopo_float target_color = std::isfinite(requested_color)
72 ? utils::clamp(requested_color, 0.0, 1.0)
73 : kNoiseColorNeutral;
74 const mopo_float brown_gain = std::sqrt((2.0 - color_filter_coefficient_) /
75 color_filter_coefficient_);
76 const mopo_float low_pass_variance = color_filter_coefficient_ /
77 (2.0 - color_filter_coefficient_);
78 const mopo_float violet_variance = 1.0 + low_pass_variance -
79 2.0 * color_filter_coefficient_;
80 const mopo_float violet_gain = 1.0 / std::sqrt(std::max(violet_variance, 1.0e-9));
81 if (target_color != target_color_) {
82 target_color_ = target_color;
83 color_smoothing_remaining_ = color_smoothing_samples_;
84 }
85
86 int i = 0;
87 if (input(kReset)->source->triggered) {
88 int trigger_offset = input(kReset)->source->trigger_offset;
89 trigger_offset = std::clamp(trigger_offset, 0, buffer_size_);
90 for (; i < trigger_offset; ++i) {
91 if (color_smoothing_remaining_ > 0) {
92 current_color_ += (target_color_ - current_color_) / color_smoothing_remaining_;
93 if (--color_smoothing_remaining_ == 0)
94 current_color_ = target_color_;
95 }
96 const mopo_float white = nextWhiteSample();
97 color_filter_state_ += color_filter_coefficient_ *
98 (white - color_filter_state_);
99 const mopo_float colored = colorSample(white, current_color_, brown_gain,
100 violet_gain);
101 dest[i] = amplitude * colored;
102 }
103
104 current_noise_value_ = rand() / mopo_float(RAND_MAX);
105 color_filter_state_ = 0.0;
106 }
107 for (; i < buffer_size_; ++i) {
108 if (color_smoothing_remaining_ > 0) {
109 current_color_ += (target_color_ - current_color_) / color_smoothing_remaining_;
110 if (--color_smoothing_remaining_ == 0)
111 current_color_ = target_color_;
112 }
113 const mopo_float white = nextWhiteSample();
114 color_filter_state_ += color_filter_coefficient_ *
115 (white - color_filter_state_);
116 const mopo_float colored = colorSample(white, current_color_, brown_gain,
117 violet_gain);
118 dest[i] = amplitude * colored;
119 }
120
121 }

123 mopo_float NoiseOscillator::colorSample(mopo_float white, mopo_float color,
124 mopo_float brown_gain,
125 mopo_float violet_gain) const {
126 if (color == kNoiseColorNeutral)
127 return white;
128
129 const mopo_float brown = color_filter_state_ * brown_gain;
130 const mopo_float violet = (white - color_filter_state_) * violet_gain;
131 if (color < kNoiseColorNeutral) {
132 const mopo_float position = color * 2.0;
133 const mopo_float blend = position * position * (3.0 - 2.0 * position);
134 return brown * (1.0 - blend) + white * blend;
135 }
136
137 const mopo_float position = (color - kNoiseColorNeutral) * 2.0;
138 const mopo_float blend = position * position * (3.0 - 2.0 * position);
139 return white * (1.0 - blend) + violet * blend;
140 }
