38 void PeakMeter::process() {
39 auto left = std::span<const mopo_float>(input(0)->source->buffer, buffer_size_);
40 auto right = std::span<const mopo_float>(input(1)->source->buffer, buffer_size_);
41 mopo_float peak_left = utils::peak(left, 8);
42 mopo_float peak_right = utils::peak(right, 8);
43
44 mopo_float exponent = buffer_size_ * (1.0 * mopo::DEFAULT_SAMPLE_RATE) / sample_rate_;
45 mopo_float movement = MIN_MOVEMENT * exponent;
46
47 mopo_float delta_left = fabs(peak_left - current_peak_left_);
48 mopo_float delta_right = fabs(peak_right - current_peak_right_);
49 mopo_float decay_left = pow(1.0 - PEAK_DECAY * (1.0 + DELTA_SCALE * delta_left), exponent);
50 mopo_float decay_right = pow(1.0 - PEAK_DECAY * (1.0 + DELTA_SCALE * delta_right), exponent);
51 current_peak_left_ *= decay_left;
52 current_peak_right_ *= decay_right;
53
54 current_peak_left_ = utils::max(current_peak_left_ - movement, peak_left);
55 current_peak_right_ = utils::max(current_peak_right_ - movement, peak_right);
56 output()->buffer[0] = current_peak_left_;
57 output()->buffer[1] = current_peak_right_;
58 }
