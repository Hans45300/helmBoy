49void ButtonModulationSlider::mouseDown(const MouseEvent& e) {
50 if (e.mods.isPopupMenu()) {
51 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
52 if (parent == nullptr)
53 return;
54
55 std::vector<mopo::ModulationConnection*> connections =
56 parent->getSynth()->getDestinationConnections(getName().toStdString());
57
58 if (connections.empty())
59 return;
60
61 PopupMenu m;
62 m.setLookAndFeel(DefaultLookAndFeel::instance());
63
64 String disconnect("Disconnect from ");
65 for (int i = 0; i < static_cast<int>(connections.size()); ++i)
66 m.addItem(kModulationList + i, disconnect + connections[i]->source);
67
68 if (connections.size() > 1)
69 m.addItem(kClearModulations, "Disconnect all modulations");
70
71 showModulationPopup(m, this, false,
72 kModulationList + static_cast<int>(connections.size()),
73 [this](int result) { handlePopupResult(result); });
74 }
75 else
76 SynthSlider::mouseDown(e);
77}

84void ButtonModulationSlider::handlePopupResult(int result) {
85 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
86 if (parent == nullptr)
87 return;
88
89 SynthBase* synth = parent->getSynth();
90 std::vector<mopo::ModulationConnection*> connections =
91 synth->getDestinationConnections(getName().toStdString());
92
93 bool disconnected = false;
94 if (result == kClearModulations) {
95 for (auto* connection : connections) {
96 synth->disconnectModulation(connection);
97 disconnected = true;
98 }
99 }
100 else if (result >= kModulationList) {
101 int connection_index = result - kModulationList;
102 if (connection_index >= 0 && connection_index < static_cast<int>(connections.size())) {
103 synth->disconnectModulation(connections[connection_index]);
104 disconnected = true;
105 }
106 }
107
108 if (disconnected)
109 destination_button_->notifyModulationsChanged();
110}
