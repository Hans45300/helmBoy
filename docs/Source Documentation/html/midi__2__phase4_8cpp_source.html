48bool Midi2Manager::enableProfile(const ProfileId& profile, int channel) {
49 juce::ScopedLock lock(profiles_lock_);
50
51 if (channel == -1) {
52 // Enable for all channels
53 for (int ch = 0; ch < 16; ++ch) {
54 auto& profiles = active_profiles_[ch];
55 if (std::find(profiles.begin(), profiles.end(), profile) == profiles.end()) {
56 profiles.push_back(profile);
57 }
58 }
59 } else if (channel >= 0 && channel < 16) {
60 auto& profiles = active_profiles_[channel];
61 if (std::find(profiles.begin(), profiles.end(), profile) == profiles.end()) {
62 profiles.push_back(profile);
63 }
64 } else {
65 return false;
66 }
67
68 // Send Profile Enabled message to controller
69 if (midi_output_ && isMidi2Enabled()) {
70 uint32_t ump[4];
71 buildProfileMessage(profile, channel, true, ump);
72
73 juce::ScopedLock output_lock(output_lock_);
74 if (midi_output_) {
75 midi_output_->sendMessageNow(juce::MidiMessage::createSysExMessage(
76 reinterpret_cast<const uint8_t*>(ump), 16));
77 }
78 }
79
80 // Notify callback
81 if (onProfileChanged && channel != -1) {
82 onProfileChanged(profile, channel, true);
83 }
84
85 DBG("MIDI2: Profile enabled - Bank=" << (int)profile.profile_byte_1
86 << " Number=" << (int)profile.profile_byte_2
87 << " Channel=" << (channel == -1 ? "All" : std::to_string(channel)));
88
89 return true;
90}

92void Midi2Manager::disableProfile(const ProfileId& profile, int channel) {
93 juce::ScopedLock lock(profiles_lock_);
94
95 if (channel == -1) {
96 // Disable for all channels
97 for (int ch = 0; ch < 16; ++ch) {
98 auto& profiles = active_profiles_[ch];
99 profiles.erase(std::remove(profiles.begin(), profiles.end(), profile), profiles.end());
100 }
101 } else if (channel >= 0 && channel < 16) {
102 auto& profiles = active_profiles_[channel];
103 profiles.erase(std::remove(profiles.begin(), profiles.end(), profile), profiles.end());
104 }
105
106 // Send Profile Disabled message to controller
107 if (midi_output_ && isMidi2Enabled()) {
108 uint32_t ump[4];
109 buildProfileMessage(profile, channel, false, ump);
110
111 juce::ScopedLock output_lock(output_lock_);
112 if (midi_output_) {
113 midi_output_->sendMessageNow(juce::MidiMessage::createSysExMessage(
114 reinterpret_cast<const uint8_t*>(ump), 16));
115 }
116 }
117
118 // Notify callback
119 if (onProfileChanged && channel != -1) {
120 onProfileChanged(profile, channel, false);
121 }
122
123 DBG("MIDI2: Profile disabled - Bank=" << (int)profile.profile_byte_1
124 << " Number=" << (int)profile.profile_byte_2
125 << " Channel=" << (channel == -1 ? "All" : std::to_string(channel)));
126}

134bool Midi2Manager::isProfileEnabled(const ProfileId& profile, int channel) const {
135 if (channel < 0 || channel >= 16)
136 return false;
137
138 juce::ScopedLock lock(const_cast<juce::CriticalSection&>(profiles_lock_));
139
140 auto it = active_profiles_.find(channel);
141 if (it == active_profiles_.end())
142 return false;
143
144 const auto& profiles = it->second;
145 return std::find(profiles.begin(), profiles.end(), profile) != profiles.end();
146}

150void Midi2Manager::sendProperty(const juce::String& property_id, const juce::var& value) {
151 if (!midi_output_ || !isMidi2Enabled())
152 return;
153
154 {
155 juce::ScopedLock lock(properties_lock_);
156 properties_[property_id] = value;
157 }
158
159 std::vector<uint32_t> ump;
160 buildPropertyMessage(PropertyRequest::SET_PROPERTY, property_id, value, ump);
161
162 if (!ump.empty()) {
163 juce::ScopedLock output_lock(output_lock_);
164 if (midi_output_) {
165 midi_output_->sendMessageNow(juce::MidiMessage::createSysExMessage(
166 reinterpret_cast<const uint8_t*>(ump.data()), static_cast<int>(ump.size() * 4)));
167 }
168 }
169
170 DBG("MIDI2: Property sent - ID=" << property_id << " Value=" << value.toString());
171}

250void Midi2Manager::buildProfileMessage(const ProfileId& profile, int channel,
251 bool enable, uint32_t* ump_out) {
252 // UMP Stream message (128-bit) - Profile Configuration
253 // Word 1: [MT=0xF][Group=0][Status][Reserved]
254 // Words 2-4: Profile ID and channel info
255
256 uint8_t status = enable ? 0x24 : 0x25; // Profile Enable/Disable
257
258 ump_out[0] = (0xF << 28) | (status << 16);
259 ump_out[1] = (profile.profile_byte_1 << 24) | (profile.profile_byte_2 << 16) |
260 (profile.profile_byte_3 << 8) | profile.profile_byte_4;
261 ump_out[2] = (channel == -1 ? 0xFF : channel) << 24; // 0xFF = all channels
262 ump_out[3] = 0;
263}

265void Midi2Manager::buildPropertyMessage(PropertyRequest request, const juce::String& property_id,
266 const juce::var& value, std::vector<uint32_t>& ump_out) {
267 // UMP Stream message - Property Exchange
268 // Encoding: JSON data in multiple 128-bit packets
269
270 juce::String json_data;
271 if (request == PropertyRequest::SET_PROPERTY || request == PropertyRequest::NOTIFY) {
272 juce::DynamicObject::Ptr obj = new juce::DynamicObject();
273 obj->setProperty(property_id, value);
274 json_data = juce::JSON::toString(juce::var(obj.get()));
275 } else {
276 juce::DynamicObject::Ptr obj = new juce::DynamicObject();
277 obj->setProperty(property_id, juce::var());
278 json_data = juce::JSON::toString(juce::var(obj.get()));
279 }
280
281 // Convert to UTF-8 bytes
282 const char* utf8_data = json_data.toUTF8();
283 int data_length = static_cast<int>(strlen(utf8_data));
284
285 // Pack into 128-bit UMP packets
286 int num_packets = (data_length + 11) / 12; // 12 bytes per packet
287
288 for (int i = 0; i < num_packets; ++i) {
289 uint32_t packet[4] = {0, 0, 0, 0};
290
291 uint8_t status;
292 switch (request) {
293 case PropertyRequest::GET_PROPERTY: status = 0x30; break;
294 case PropertyRequest::SET_PROPERTY: status = 0x31; break;
295 case PropertyRequest::SUBSCRIBE: status = 0x32; break;
296 case PropertyRequest::NOTIFY: status = 0x33; break;
297 default: status = 0x30; break;
298 }
299
300 packet[0] = (0xF << 28) | (status << 16) | (i & 0xFFFF);
301
302 // Copy up to 12 bytes of data
303 uint8_t* data_ptr = reinterpret_cast<uint8_t*>(&packet[1]);
304 int offset = i * 12;
305 for (int j = 0; j < 12 && (offset + j) < data_length; ++j) {
306 data_ptr[j] = utf8_data[offset + j];
307 }
308
309 for (int j = 0; j < 4; ++j) {
310 ump_out.push_back(packet[j]);
311 }
312 }
313}

315void Midi2Manager::buildFeedbackMessage(int channel, int controller, float value,
316 bool is_per_note, int note, uint32_t* ump_out) {
317 // Build MIDI 2.0 64-bit Control Change or Per-Note Controller
318
319 if (is_per_note) {
320 // Per-Note Controller (64-bit)
321 ump_out[0] = (0x4 << 28) | (0x0 << 20) | (channel << 16) | (note << 8);
322 ump_out[1] = static_cast<uint32_t>(value * 4294967295.0f);
323 } else {
324 // Standard Control Change (64-bit)
325 ump_out[0] = (0x4 << 28) | (0xB << 20) | (channel << 16) | (controller << 8);
326 ump_out[1] = static_cast<uint32_t>(value * 4294967295.0f);
327 }
328}
