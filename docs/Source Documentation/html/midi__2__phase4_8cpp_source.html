78bool Midi2Manager::enableProfile(const ProfileId& profile, int channel) {
79 juce::ScopedLock lock(profiles_lock_);
80
81 if (channel == -1) {
82 // Enable for all channels
83 for (int ch = 0; ch < 16; ++ch) {
84 auto& profiles = active_profiles_[ch];
85 if (std::find(profiles.begin(), profiles.end(), profile) == profiles.end()) {
86 profiles.push_back(profile);
87 }
88 }
89 } else if (channel >= 0 && channel < 16) {
90 auto& profiles = active_profiles_[channel];
91 if (std::find(profiles.begin(), profiles.end(), profile) == profiles.end()) {
92 profiles.push_back(profile);
93 }
94 } else {
95 return false;
96 }
97
98 // Send Profile Enabled message to controller
99 if (midi_output_ && isMidi2Enabled()) {
100 uint32_t ump[4];
101 buildProfileMessage(profile, channel, true, ump);
102
103 juce::ScopedLock output_lock(output_lock_);
104 if (midi_output_) {
105 midi_output_->sendMessageNow(juce::MidiMessage::createSysExMessage(
106 reinterpret_cast<const uint8_t*>(ump), 16));
107 }
108 }
109
110 // Notify callback
111 if (onProfileChanged && channel != -1) {
112 onProfileChanged(profile, channel, true);
113 }
114
115 DBG("MIDI2: Profile enabled - Bank=" << (int)profile.profile_byte_1
116 << " Number=" << (int)profile.profile_byte_2
117 << " Channel=" << (channel == -1 ? "All" : std::to_string(channel)));
118
119 return true;
120}

122void Midi2Manager::disableProfile(const ProfileId& profile, int channel) {
123 juce::ScopedLock lock(profiles_lock_);
124
125 if (channel == -1) {
126 // Disable for all channels
127 for (int ch = 0; ch < 16; ++ch) {
128 auto& profiles = active_profiles_[ch];
129 profiles.erase(std::remove(profiles.begin(), profiles.end(), profile), profiles.end());
130 }
131 } else if (channel >= 0 && channel < 16) {
132 auto& profiles = active_profiles_[channel];
133 profiles.erase(std::remove(profiles.begin(), profiles.end(), profile), profiles.end());
134 }
135
136 // Send Profile Disabled message to controller
137 if (midi_output_ && isMidi2Enabled()) {
138 uint32_t ump[4];
139 buildProfileMessage(profile, channel, false, ump);
140
141 juce::ScopedLock output_lock(output_lock_);
142 if (midi_output_) {
143 midi_output_->sendMessageNow(juce::MidiMessage::createSysExMessage(
144 reinterpret_cast<const uint8_t*>(ump), 16));
145 }
146 }
147
148 // Notify callback
149 if (onProfileChanged && channel != -1) {
150 onProfileChanged(profile, channel, false);
151 }
152
153 DBG("MIDI2: Profile disabled - Bank=" << (int)profile.profile_byte_1
154 << " Number=" << (int)profile.profile_byte_2
155 << " Channel=" << (channel == -1 ? "All" : std::to_string(channel)));
156}

164bool Midi2Manager::isProfileEnabled(const ProfileId& profile, int channel) const {
165 if (channel < 0 || channel >= 16)
166 return false;
167
168 juce::ScopedLock lock(const_cast<juce::CriticalSection&>(profiles_lock_));
169
170 auto it = active_profiles_.find(channel);
171 if (it == active_profiles_.end())
172 return false;
173
174 const auto& profiles = it->second;
175 return std::find(profiles.begin(), profiles.end(), profile) != profiles.end();
176}

181 if (!midi_output_ || !isMidi2Enabled())

196 reinterpret_cast<const uint8_t*>(ump.data()), static_cast<int>(ump.size() * 4)));
197 }
198 }
199
200 DBG("MIDI2: Property sent - ID=" << property_id << " Value=" << value.toString());

280void Midi2Manager::buildProfileMessage(const ProfileId& profile, int channel,
281 bool enable, uint32_t* ump_out) {
282 // UMP Stream message (128-bit) - Profile Configuration
283 // Word 1: [MT=0xF][Group=0][Status][Reserved]
284 // Words 2-4: Profile ID and channel info
285
286 uint8_t status = enable ? 0x24 : 0x25; // Profile Enable/Disable
287
288 ump_out[0] = (0xF << 28) | (status << 16);
289 ump_out[1] = (profile.profile_byte_1 << 24) | (profile.profile_byte_2 << 16) |
290 (profile.profile_byte_3 << 8) | profile.profile_byte_4;
291 ump_out[2] = (channel == -1 ? 0xFF : channel) << 24; // 0xFF = all channels
292 ump_out[3] = 0;
293}

295void Midi2Manager::buildPropertyMessage(PropertyRequest request, const juce::String& property_id,
296 const juce::var& value, std::vector<uint32_t>& ump_out) {
297 // UMP Stream message - Property Exchange
298 // Encoding: JSON data in multiple 128-bit packets
299
300 juce::String json_data;
301 if (request == PropertyRequest::SET_PROPERTY || request == PropertyRequest::NOTIFY) {
302 juce::DynamicObject::Ptr obj = new juce::DynamicObject();
303 obj->setProperty(property_id, value);
304 json_data = juce::JSON::toString(juce::var(obj.get()));
305 } else {
306 juce::DynamicObject::Ptr obj = new juce::DynamicObject();
307 obj->setProperty(property_id, juce::var());
308 json_data = juce::JSON::toString(juce::var(obj.get()));
309 }
310
311 // Convert to UTF-8 bytes
312 const char* utf8_data = json_data.toUTF8();
313 int data_length = static_cast<int>(strlen(utf8_data));
314
315 // Pack into 128-bit UMP packets
316 int num_packets = (data_length + 11) / 12; // 12 bytes per packet
317
318 for (int i = 0; i < num_packets; ++i) {
319 uint32_t packet[4] = {0, 0, 0, 0};
320
321 uint8_t status;
322 switch (request) {
323 case PropertyRequest::GET_PROPERTY: status = 0x30; break;
324 case PropertyRequest::SET_PROPERTY: status = 0x31; break;
325 case PropertyRequest::SUBSCRIBE: status = 0x32; break;
326 case PropertyRequest::NOTIFY: status = 0x33; break;
327 default: status = 0x30; break;
328 }
329
330 packet[0] = (0xF << 28) | (status << 16) | (i & 0xFFFF);
331
332 // Copy up to 12 bytes of data
333 uint8_t* data_ptr = reinterpret_cast<uint8_t*>(&packet[1]);
334 int offset = i * 12;
335 for (int j = 0; j < 12 && (offset + j) < data_length; ++j) {
336 data_ptr[j] = utf8_data[offset + j];
337 }
338
339 for (int j = 0; j < 4; ++j) {
340 ump_out.push_back(packet[j]);
341 }
342 }
343}

345void Midi2Manager::buildFeedbackMessage(int channel, int controller, float value,
346 bool is_per_note, int note, uint32_t* ump_out) {
347 // Build MIDI 2.0 64-bit Control Change or Per-Note Controller
348
349 if (is_per_note) {
350 // Per-Note Controller (64-bit)
351 ump_out[0] = (0x4 << 28) | (0x0 << 20) | (channel << 16) | (note << 8);
352 ump_out[1] = static_cast<uint32_t>(value * 4294967295.0f);
353 } else {
354 // Standard Control Change (64-bit)
355 ump_out[0] = (0x4 << 28) | (0xB << 20) | (channel << 16) | (controller << 8);
356 ump_out[1] = static_cast<uint32_t>(value * 4294967295.0f);
357 }
358}
