52 void HelmBoyLfo::advanceRandomCycle(Wave::Type waveform) {
53 if (waveform == Wave::Type::WhiteNoise) {
54 current_random_value_ = next_random_value_;
55 next_random_value_ = nextRandomValue();
56 return;
57 }
58
59 if (waveform == Wave::Type::SampleAndGlide) {
60 last_random_value_ = current_random_value_;
61 current_random_value_ = next_random_value_;
62 next_random_value_ = nextRandomValue();
63 return;
64 }
65
66 if (waveform == Wave::Type::SmoothRandom) {
67 smooth_random_values_[0] = smooth_random_values_.back();
68 for (std::size_t index = 1; index < smooth_random_values_.size(); ++index)
69 smooth_random_values_[index] = nextRandomValue();
70 last_random_value_ = smooth_random_values_.front();
71 current_random_value_ = smooth_random_values_.back();
72 }
73 }

75 void HelmBoyLfo::process() {
76 int num_samples = samples_to_process_;
77 int waveform_index = utils::truncateAndClampWaveformIndex(
78 input(kWaveform)->at(0), 0, static_cast<int>(Wave::Type::NumWaveforms) - 1);
79 const Wave::Type waveform = static_cast<Wave::Type>(waveform_index);
80 mopo_float phase = input(kPhase)->at(0) / 360.0;
81 if (!std::isfinite(phase))
82 phase = 0.0;
83
84 if (input(kReset)->source->triggered) {
85 num_samples = samples_to_process_ - input(kReset)->source->trigger_offset;
86 offset_ = 0.0;
87 previous_phase_offset_ = phase;
88 last_random_value_ = current_random_value_;
89 current_random_value_ = next_random_value_;
90 next_random_value_ = nextRandomValue();
91 smooth_random_values_[0] = last_random_value_;
92 for (std::size_t index = 1; index < smooth_random_values_.size(); ++index)
93 smooth_random_values_[index] = nextRandomValue();
94 }
95
96 mopo_float frequency = input(kFrequency)->at(0);
97 mopo_float stretch = input(kPhaseStretch)->at(0);
98 if (!std::isfinite(frequency))
99 frequency = 0.0;
100 if (!std::isfinite(stretch))
101 stretch = 0.5;
102 stretch = utils::clamp(stretch, 0.01, 0.99);
103
104 const mopo_float requested_phase_advance = num_samples * frequency / sample_rate_;
105 const mopo_float phase_advance = std::isfinite(requested_phase_advance)
106 ? requested_phase_advance : 0.0;
107 const mopo_float unwrapped_offset = offset_ + phase_advance;
108 const mopo_float whole_cycles = std::floor(unwrapped_offset);
109 const mopo_float previous_effective_phase = offset_ + previous_phase_offset_;
110 const mopo_float current_effective_phase = unwrapped_offset + phase;
111 const mopo_float crossed_phase_cycles = std::floor(current_effective_phase) -
112 std::floor(previous_effective_phase);
113 offset_ = unwrapped_offset - whole_cycles;
114 previous_phase_offset_ = phase;
115 if (waveform >= Wave::Type::WhiteNoise && crossed_phase_cycles != 0.0) {
116 const auto crossed_cycles = static_cast<std::uint64_t>(std::fabs(crossed_phase_cycles));
117 for (std::uint64_t cycle = 0; cycle < crossed_cycles; ++cycle)
118 advanceRandomCycle(waveform);
119 }
120
121 mopo_float phase_position = offset_ + phase;
122 mopo_float phased_offset = phase_position - std::floor(phase_position);
123 mopo_float warped_offset = utils::remapPhase(phased_offset, stretch);
124
125 output(kOscPhase)->buffer[0] = phased_offset;
126 output(kVisualPhase)->buffer[0] = offset_;
127
128 if (waveform < Wave::Type::WhiteNoise)
129 output(kValue)->buffer[0] = Wave::wave(waveform, warped_offset);
130 else if (waveform == Wave::Type::WhiteNoise)
131 output(kValue)->buffer[0] = Wave::sampleAndHoldValue(
132 current_random_value_, next_random_value_, warped_offset);
133 else if (waveform == Wave::Type::SampleAndGlide) {
134 output(kValue)->buffer[0] = Wave::sampleAndGlideValue(
135 last_random_value_, current_random_value_, warped_offset);
136 }
137 else if (waveform == Wave::Type::SmoothRandom) {
138 output(kValue)->buffer[0] = Wave::smoothRandomValue(
139 smooth_random_values_.data(), warped_offset);
140 }
141
142 if (waveform == Wave::Type::WhiteNoise) {
143 output(kRandomPoint0)->buffer[0] = current_random_value_;
144 output(kRandomPoint1)->buffer[0] = next_random_value_;
145 }
146 else if (waveform == Wave::Type::SampleAndGlide) {
147 output(kRandomPoint0)->buffer[0] = last_random_value_;
148 output(kRandomPoint1)->buffer[0] = current_random_value_;
149 }
150 else if (waveform == Wave::Type::SmoothRandom) {
151 for (int index = 0; index < static_cast<int>(smooth_random_values_.size()); ++index)
152 output(kRandomPoint0 + index)->buffer[0] = smooth_random_values_[index];
153 }
154 }

156 void HelmBoyLfo::correctToTime(mopo_float samples) {
157 mopo_float frequency = input(kFrequency)->at(0);
158 if (!std::isfinite(frequency))
159 frequency = 0.0;
160 const mopo_float phase = samples * frequency / sample_rate_;
161 offset_ = std::isfinite(phase) ? phase - std::floor(phase) : 0.0;
162 }
