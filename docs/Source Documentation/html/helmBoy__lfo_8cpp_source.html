44 void HelmBoyLfo::process() {
45 int num_samples = samples_to_process_;
46
47 if (input(kReset)->source->triggered) {
48 num_samples = samples_to_process_ - input(kReset)->source->trigger_offset;
49 offset_ = 0.0;
50 last_random_value_ = current_random_value_;
51 current_random_value_ = randomLfoValue();
52 }
53
54 Wave::Type waveform = static_cast<Wave::Type>(static_cast<int>(input(kWaveform)->at(0)));
55 mopo_float frequency = input(kFrequency)->at(0);
56 mopo_float phase = input(kPhase)->at(0);
57 mopo_float stretch = input(kPhaseStretch)->at(0);
58
59 offset_ += num_samples * frequency / sample_rate_;
60
61 mopo_float offset_integral;
62 offset_ = utils::mod(offset_, &offset_integral);
63
64 mopo_float phase_integral;
65 mopo_float phased_offset = utils::mod(offset_ + phase, &phase_integral);
66 mopo_float warped_offset = remapPhase(phased_offset, stretch);
67
68 output(kOscPhase)->buffer[0] = phased_offset;
69
70 if (waveform < Wave::Type::WhiteNoise)
71 output(kValue)->buffer[0] = Wave::wave(waveform, warped_offset);
72 else {
73 if (offset_integral) {
74 last_random_value_ = current_random_value_;
75 current_random_value_ = randomLfoValue();
76 }
77 if (waveform == Wave::Type::WhiteNoise)
78 output(kValue)->buffer[0] = current_random_value_;
79 else {
80 // Smooth random value.
81 mopo_float t = (1.0 - cos(PI * warped_offset)) / 2.0;
82 output(kValue)->buffer[0] = utils::interpolate(last_random_value_,
83 current_random_value_, t);
84 }
85 }
86 }

88 void HelmBoyLfo::correctToTime(mopo_float samples) {
89 mopo_float frequency = input(kFrequency)->at(0);
90 offset_ = samples * frequency / sample_rate_;
91 mopo_float integral;
92 offset_ = utils::mod(offset_, &integral);
93 }
