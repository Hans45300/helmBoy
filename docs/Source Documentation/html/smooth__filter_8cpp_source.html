39 void SmoothFilter::process() {
40 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Target)));
41
42 mopo_float half_life = input(static_cast<int>(Inputs::HalfLife))->at(0);
43 mopo_float decay = 0.0;
44 if (half_life > 0.0)
45 decay = std::pow(0.5, 1.0 / (half_life * sample_rate_));
46
47 // Prï¿½paration SIMD du buffer cible (lecture input)
48 std::vector<mopo_float> target_buffer(buffer_size_);
49 using batch = xsimd::batch<mopo_float>;
50 constexpr std::size_t simd_size = batch::size;
51 int simd_end = buffer_size_ - (buffer_size_ % simd_size);
52 int i = 0;
53 for (; i < simd_end; i += simd_size) {
54 for (std::size_t j = 0; j < simd_size; ++j) {
55 target_buffer[i + j] = input(static_cast<int>(Inputs::Target))->at(i + j);
56 }
57 }
58 for (; i < buffer_size_; ++i) {
59 target_buffer[i] = input(static_cast<int>(Inputs::Target))->at(i);
60 }
61 // Boucle sï¿½quentielle de lissage
62 for (int k = 0; k < buffer_size_; ++k) {
63 last_value_ = utils::interpolate(target_buffer[k], last_value_, decay);
64 output(0)->buffer[k] = last_value_;
65 }
66 }
