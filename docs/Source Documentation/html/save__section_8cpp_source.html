126void SaveSection::paint(Graphics& g) {
127 static const DropShadow shadow(Colour(Colors::Color_ff000000), 5, Point<int>(0, 0));
128
129 g.setColour(Colors::overlay_screen);
130 g.fillAll();
131
132 Rectangle<int> save_rect = getSaveRect();
133 shadow.drawForRectangle(g, save_rect);
134 g.setColour(Colour(Colors::Color_ff303030));
135 g.fillRect(save_rect);
136
137 g.saveState();
138 g.setOrigin(save_rect.getX() + PADDING_X, save_rect.getY() + PADDING_Y);
139
140 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(14.0f));
141 g.setColour(Colour(Colors::Color_ff888888));
142
143 g.drawText(TRANS("PATCH NAME"),
144 0, PADDING_Y, DIVISION - 10, TEXT_EDITOR_HEIGHT,
145 Justification::centredRight, false);
146 g.drawText(TRANS("AUTHOR"),
147 0, 2 * PADDING_Y + TEXT_EDITOR_HEIGHT, DIVISION - 10, TEXT_EDITOR_HEIGHT,
148 Justification::centredRight, false);
149 g.drawText(TRANS("BANK"),
150 0, banks_view_->getY() - save_rect.getY() - PADDING_Y,
151 DIVISION - 10, TEXT_EDITOR_HEIGHT,
152 Justification::centredRight, false);
153 g.drawText(TRANS("FOLDER"),
154 0, folders_view_->getY() - save_rect.getY() - PADDING_Y,
155 DIVISION - 10, TEXT_EDITOR_HEIGHT,
156 Justification::centredRight, false);
157
158 g.restoreState();
159}

161void SaveSection::resized() {
162 Rectangle<int> save_rect = getSaveRect();
163 patch_name_->setBounds(save_rect.getX() + PADDING_X + DIVISION,
164 save_rect.getY() + 2 * PADDING_Y,
165 save_rect.getWidth() - 2 * PADDING_X - DIVISION,
166 TEXT_EDITOR_HEIGHT);
167 author_->setBounds(save_rect.getX() + PADDING_X + DIVISION,
168 save_rect.getY() + 3 * PADDING_Y + TEXT_EDITOR_HEIGHT,
169 save_rect.getWidth() - 2 * PADDING_X - DIVISION,
170 TEXT_EDITOR_HEIGHT);
171
172 float button_width = (save_rect.getWidth() - 3 * PADDING_X) / 2.0f;
173 save_button_->setBounds(save_rect.getX() + button_width + 2 * PADDING_X,
174 save_rect.getBottom() - PADDING_Y - BUTTON_HEIGHT,
175 button_width, BUTTON_HEIGHT);
176 cancel_button_->setBounds(save_rect.getX() + PADDING_X,
177 save_rect.getBottom() - PADDING_Y - BUTTON_HEIGHT,
178 button_width, BUTTON_HEIGHT);
179
180 float folder_x = save_rect.getX() + PADDING_X + DIVISION;
181 float folder_width = save_rect.getWidth() - 2 * PADDING_X - DIVISION;
182 float add_folder_y = save_button_->getY() - PADDING_Y - ADD_FOLDER_HEIGHT;
183 add_folder_button_->setBounds(folder_x, add_folder_y, ADD_FOLDER_HEIGHT, ADD_FOLDER_HEIGHT);
184 add_folder_name_->setBounds(folder_x + ADD_FOLDER_HEIGHT, add_folder_y,
185 folder_width - ADD_FOLDER_HEIGHT, ADD_FOLDER_HEIGHT);
186
187 float bank_y = save_rect.getY() + 4 * PADDING_Y + 2 * TEXT_EDITOR_HEIGHT;
188 float bank_folder_height = (add_folder_y - bank_y - PADDING_Y - ADD_FOLDER_HEIGHT) / 2.0f;
189 float folder_y = bank_y + bank_folder_height + PADDING_Y + ADD_FOLDER_HEIGHT;
190 banks_view_->setBounds(folder_x, bank_y, folder_width, bank_folder_height);
191 add_bank_button_->setBounds(folder_x, bank_y + bank_folder_height, ADD_FOLDER_HEIGHT, ADD_FOLDER_HEIGHT);
192 add_bank_name_->setBounds(folder_x + ADD_FOLDER_HEIGHT, bank_y + bank_folder_height,
193 folder_width - ADD_FOLDER_HEIGHT, ADD_FOLDER_HEIGHT);
194 folders_view_->setBounds(folder_x, folder_y, folder_width, bank_folder_height);
195}

248void SaveSection::save() {
249 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
250 SparseSet<int> selected_rows = folders_view_->getSelectedRows();
251 if (selected_rows.size() == 0)
252 return;
253
254 File folder = folders_model_->getFileAtRow(selected_rows[0]);
255 String patch_name = patch_name_->getText();
256 if (patch_name.length() == 0)
257 return;
258
259 File save_file = folder.getChildFile(patch_name);
260 parent->getSynth()->setAuthor(author_->getText());
261 (void)parent->getSynth()->saveToFile(save_file);
262
263 patch_name_->clear();
264 setVisible(false);
265 if (listener_)
266 listener_->fileSaved(save_file);
267}

269void SaveSection::createNewBank() {
270 String bank_name = add_bank_name_->getText();
271 if (bank_name.length() == 0)
272 return;
273
274 File banks_dir = LoadSave::getBankDirectory();
275 File new_bank = banks_dir.getChildFile(bank_name);
276 if (!new_bank.exists())
277 new_bank.createDirectory();
278
279 add_bank_name_->clear();
280
281 rescanFolders();
282 int row = banks_model_->getIndexOfFile(new_bank);
283 banks_view_->selectRow(row);
284 banks_view_->updateContent();
285}
