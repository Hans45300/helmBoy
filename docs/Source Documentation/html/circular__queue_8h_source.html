189 void reserve(int capcity) {
190 // Free previously allocated storage. Use delete[] because `data_`
191 // points to an array allocated with new T[]. Using plain delete
192 // is undefined behaviour.
193 delete[] data_;
194
195 capacity_ = capcity + 1;
196 data_ = new T[capacity_];
197 // Reset the indices to an empty state. We use release ordering to
198 // ensure any prior modifications to `data_` are visible to other
199 // threads that perform an acquire load on `start_`/`end_`.
200 start_.store(0, std::memory_order_release);
201 end_.store(0, std::memory_order_release);
202 }

324 void removeAt(int index) {
325 int current_start = start_.load(std::memory_order_acquire);
326 int current_end = end_.load(std::memory_order_acquire);
327 int i = (index + current_start) % capacity_;
328 int new_end = (current_end - 1 + capacity_) % capacity_;
329 while (i != new_end) {
330 int next = (i + 1) % capacity_;
331 data_[i] = data_[next];
332 i = next;
333 }
334 end_.store(new_end, std::memory_order_release);
335 }
336
341 void remove(T entry) {
342 int current_start = start_.load(std::memory_order_acquire);
343 int current_end = end_.load(std::memory_order_acquire);
344 for (int i = current_start; i != current_end; i = (i + 1) % capacity_) {
345 if (data_[i] == entry) {
346 removeAt((i - current_start + capacity_) % capacity_);
347 return;
348 }
349 }
350 }
351
359 void removeAll(T entry) {
360 int current_start = start_.load(std::memory_order_acquire);
361 int current_end = end_.load(std::memory_order_acquire);
362 for (int i = current_start; i != current_end; i = (i + 1) % capacity_) {
363 if (data_[i] == entry) {
364 removeAt((i - current_start + capacity_) % capacity_);
365 i--;
366 current_end = end_.load(std::memory_order_acquire);
367 }
368 }
369 }

465 private:
466 // Raw pointer to the allocated storage block. The memory layout is a
467 // circular buffer of size `capacity_` (where capacity_ == requested
468 // capacity + 1). Elements are stored contiguously for cache efficiency.
469 T* data_;
470
471 // Allocated capacity (internal size = user_capacity + 1).
472 int capacity_;
473
474 // Atomic indices indicating the logical start (read index) and end
475 // (write index) into `data_`. These are manipulated with
476 // acquire/release ordering to ensure correct visibility between the
477 // producer and consumer without locks.
478 std::atomic<int> start_;
479 std::atomic<int> end_;
