43 inline mopo_float linearFold(mopo_float t) {
44 mopo_float adjust = 0.25 * t + 0.75;
45 mopo_float range = adjust - floor(adjust);
46 return fabs(2.0 - 4.0 * range) - 1.0;
47 }

56 inline mopo_float sinFold(mopo_float t) {
57 mopo_float adjust = -0.25 * t + 0.5;
58 mopo_float range = adjust - floor(adjust);

85 void Distortion::processSoftClip() {
86 auto audio = std::span<const mopo_float>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
87 mopo_float next_drive = input(static_cast<int>(Inputs::Drive))->at(0);
88 mopo_float next_mix = input(static_cast<int>(Inputs::Mix))->at(0);
89 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
90 int buffer_size = buffer_size_;
91
92 // SIMD path when drive and mix are constant for the block. This uses
93 // xsimd for vector loads/stores, but falls back to scalar quickTanh
94 // computations because quickTanh isn't vectorized here.
95 if (last_drive_ == next_drive && last_mix_ == next_mix) {
96 #include <xsimd/xsimd.hpp>
97 using batch = xsimd::batch<mopo_float>;
98 constexpr std::size_t simd_size = batch::size;
99 int simd_end = buffer_size - (buffer_size % simd_size);
100 int i = 0;
101 batch drive = batch(next_drive);
102 batch mix = batch(next_mix);
103 for (; i < simd_end; i += simd_size) {
104 batch a = batch::load_unaligned(&audio[i]);
105 batch distort = drive * a;
106 // quickTanh is not vectorized, do scalar fallback per component
107 alignas(alignof(batch)) mopo_float distort_arr[simd_size];
108 distort.store_unaligned(distort_arr);
109 mopo_float tanh_arr[simd_size];
110 for (std::size_t j = 0; j < simd_size; ++j) {
111 tanh_arr[j] = utils::quickTanh(distort_arr[j]);
112 }
113 batch tanh_batch = batch::load_unaligned(tanh_arr);
114 batch result = a + mix * (tanh_batch - a);
115 result.store_unaligned(&dest[i]);
116 }
117 for (; i < buffer_size; ++i) {
118 mopo_float distort = next_drive * audio[i];
119 dest[i] = utils::interpolate(audio[i], utils::quickTanh(distort), next_mix);
120 }
121 } else {
122 // Scalar fallback: interpolate mix and drive across the block to
123 // prevent zipper noise when parameters change.
124 for (int i = 0; i < buffer_size; ++i) {
125 mopo_float mix = last_mix_ + i * (next_mix - last_mix_) / buffer_size;
126 mopo_float drive = last_drive_ + i * (next_drive - last_drive_) / buffer_size;
127 mopo_float distort = drive * audio[i];
128 dest[i] = utils::interpolate(audio[i], utils::quickTanh(distort), mix);
129 }
130 }
131 // Store the last block values for next block's interpolation
132 last_mix_ = next_mix;
133 last_drive_ = next_drive;
134 }

142 void Distortion::processHardClip() {
143 const mopo_float* audio = input(static_cast<int>(Inputs::Audio))->source->buffer;
144 mopo_float next_drive = input(static_cast<int>(Inputs::Drive))->at(0);
145 mopo_float mult_drive = (next_drive - last_drive_) / buffer_size_;
146 mopo_float next_mix = input(static_cast<int>(Inputs::Mix))->at(0);
147 mopo_float mult_mix = (next_mix - last_mix_) / buffer_size_;
148
149 mopo_float* dest = output()->buffer;
150 int buffer_size = buffer_size_;
151
152 for (int i = 0; i < buffer_size; ++i) {
153 mopo_float mix = last_mix_ + i * mult_mix;
154 mopo_float drive = last_drive_ + i * mult_drive;
155 mopo_float distort = utils::clamp(drive * audio[i], -1.0, 1.0);
156 dest[i] = utils::interpolate(audio[i], distort, mix);
157 }
158
159 last_mix_ = next_mix;
160 last_drive_ = next_drive;
161 }

228 void Distortion::process() {
229 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
230
231 Type type = static_cast<Type>(static_cast<int>(input(static_cast<int>(Inputs::Type))->at(0)));
232 if (input(static_cast<int>(Inputs::On))->at(0) == 0.0) {
233 utils::copyBuffer(
234 std::span<mopo_float, MAX_BUFFER_SIZE>(output()->buffer, buffer_size_),
235 std::span<const mopo_float, MAX_BUFFER_SIZE>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_)
236 );
237 return;
238 }
239
240 switch(type) {
241 case Type::SoftClip:
242 processSoftClip();
243 break;
244 case Type::HardClip:
245 processHardClip();
246 break;
247 case Type::LinearFold:
248 processLinearFold();
249 break;
250 case Type::SinFold:
251 processSinFold();
252 break;
253 default:
254 utils::copyBuffer(
255 std::span<mopo_float, MAX_BUFFER_SIZE>(output()->buffer, buffer_size_),
256 std::span<const mopo_float, MAX_BUFFER_SIZE>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_)
257 );
258 }
259 }
