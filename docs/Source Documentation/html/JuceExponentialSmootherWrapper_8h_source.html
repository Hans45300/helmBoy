55class JuceExponentialSmootherWrapper : public Processor {
56 public:
60 enum class Inputs : int {

70 explicit JuceExponentialSmootherWrapper(mopo_float start_value = 0.0)
71 : Processor(static_cast<int>(Inputs::NumInputs), 1, true),
72 last_value_(start_value) {
73 // Processor constructor sets up input/output buffers in mopo style
74 }
75
80 [[nodiscard]] Processor* clone() const override {
81 return new JuceExponentialSmootherWrapper(*this);
82 }
83
92 void process() override {
93 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Target)));
94
95 // Get target and half-life from input buffers
96 mopo_float target = input(static_cast<int>(Inputs::Target))->at(0);
97 mopo_float half_life = input(static_cast<int>(Inputs::HalfLife))->at(0);
98
99 // Compute decay coefficient: 0.5^(1.0 / (half_life * sample_rate))
100 mopo_float decay = 0.0;
101 if (half_life > 0.0) {
102 decay = std::pow(0.5, 1.0 / (half_life * sample_rate_));
103 }
104
105 // Exponential interpolation toward target
106 // last_value += decay * (target - last_value)
107 last_value_ = utils::interpolate(target, last_value_, decay);
108 output(0)->buffer[0] = last_value_;
109 }
110
115 [[nodiscard]] mopo_float get() const noexcept {
116 return last_value_;
117 }
118
119 private:
120 mopo_float last_value_;
121};
