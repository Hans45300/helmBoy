91 BiquadFilter::Coefficients BiquadFilter::lowPassCoefficients(mopo_float cutoff,
92 mopo_float resonance,
93 mopo_float sample_rate) {
94 MOPO_ASSERT(cutoff > 0.0);
95 MOPO_ASSERT(resonance > 0.0);
96 MOPO_ASSERT(sample_rate > 0.0);
97
98 const mopo_float phase_delta = 2.0 * PI * cutoff / sample_rate;
99 const mopo_float real_delta = cos(phase_delta);
100 const mopo_float imag_delta = sin(phase_delta);
101 const mopo_float alpha = imag_delta / (2.0 * resonance);
102 const mopo_float norm = 1.0 + alpha;
103
104 return {
105 (1.0 - real_delta) / (2.0 * norm),
106 (1.0 - real_delta) / norm,
107 (1.0 - real_delta) / (2.0 * norm),
108 -2.0 * real_delta / norm,
109 (1.0 - alpha) / norm
110 };
111 }

129 std::complex<mopo_float> BiquadFilter::getResponse(mopo_float frequency) {
130 static const std::complex<mopo_float> one(1.0, 0.0);
131 const mopo_float phase_delta = 2.0 * PI * frequency / sample_rate_;
132 const std::complex<mopo_float> freq_tick1 = std::polar(mopo_float(1.0), -phase_delta);
133 const std::complex<mopo_float> freq_tick2 = std::polar(mopo_float(1.0), -2 * phase_delta);
134
135 return (target_in_0_ * one + target_in_1_ * freq_tick1 + target_in_2_ * freq_tick2) /
136 (one + target_out_1_ * freq_tick1 + target_out_2_ * freq_tick2);
137 }

139 void BiquadFilter::process() {
140 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
141
142 current_type_ = static_cast<Type>(static_cast<int>(input(static_cast<int>(Inputs::Type))->at(0)));
143 mopo_float cutoff = utils::clamp(input(static_cast<int>(Inputs::Cutoff))->at(0), MIN_CUTTOFF, sample_rate_);
144 mopo_float resonance = utils::clamp(input(static_cast<int>(Inputs::Resonance))->at(0),
145 MIN_RESONANCE, MAX_RESONANCE);
146 mopo_float gain = input(static_cast<int>(Inputs::Gain))->at(0);
147 const bool use_juce_path = shouldUseJucePath(current_type_);
148#if defined(HELMBOY_DEBUG_BIQUAD_ROUTING)
149 static Type last_logged_type = Type::NumTypes;
150 static bool last_logged_route_juce = false;
151 if (current_type_ != last_logged_type || use_juce_path != last_logged_route_juce) {
152 std::fprintf(stderr,
153 "[HelmBoy][Biquad] type=%s route=%s\n",
154 getTypeName(current_type_),
155 use_juce_path ? "juce-compatible" : "legacy-fallback");
156 last_logged_type = current_type_;
157 last_logged_route_juce = use_juce_path;
158 }
159#endif
160 if (use_juce_path)
161 computeCoefficientsJucePath(current_type_, cutoff, resonance, gain);
162 else
163 computeCoefficients(current_type_, cutoff, resonance, gain);
164
165 mopo_float delta_in_0 = (target_in_0_ - in_0_) / buffer_size_;
166 mopo_float delta_in_1 = (target_in_1_ - in_1_) / buffer_size_;
167 mopo_float delta_in_2 = (target_in_2_ - in_2_) / buffer_size_;
168 mopo_float delta_out_1 = (target_out_1_ - out_1_) / buffer_size_;
169 mopo_float delta_out_2 = (target_out_2_ - out_2_) / buffer_size_;
170
171 std::span<const mopo_float> audio_buffer(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
172 std::span<mopo_float> dest(output()->buffer, buffer_size_);
173 if (input(static_cast<int>(Inputs::Reset))->source->triggered &&
174 input(static_cast<int>(Inputs::Reset))->source->trigger_value == static_cast<mopo_float>(kVoiceReset)) {
175 int trigger_offset = input(static_cast<int>(Inputs::Reset))->source->trigger_offset;
176 int i = 0;
177 std::ranges::for_each(std::views::iota(0, trigger_offset), [&](int idx) {
178 in_0_ += delta_in_0;
179 in_1_ += delta_in_1;
180 in_2_ += delta_in_2;
181 out_1_ += delta_out_1;
182 out_2_ += delta_out_2;
183 tick(idx, dest, audio_buffer);
184 });
185
186 reset();
187
188 std::ranges::for_each(std::views::iota(trigger_offset, buffer_size_), [&](int idx) {
189 tick(idx, dest, audio_buffer);
190 });
191 }
192 else {
193 std::ranges::for_each(std::views::iota(0, buffer_size_), [&](int i) {
194 in_0_ += delta_in_0;
195 in_1_ += delta_in_1;
196 in_2_ += delta_in_2;
197 out_1_ += delta_out_1;
198 out_2_ += delta_out_2;
199 tick(i, dest, audio_buffer);
200 });
201 }
202 }

204 void BiquadFilter::computeCoefficients(Type type,
205 mopo_float cutoff,
206 mopo_float resonance,
207 mopo_float gain) {
208 MOPO_ASSERT(resonance > 0.0);
209 MOPO_ASSERT(cutoff > 0.0);
210 MOPO_ASSERT(gain >= 0.0);
211
212 mopo_float phase_delta = 2.0 * PI * cutoff / sample_rate_;
213 mopo_float real_delta = cos(phase_delta);
214 mopo_float imag_delta = sin(phase_delta);
215
216 switch(type) {
217 case Type::LowPass: {
218 const Coefficients coefficients = lowPassCoefficients(cutoff, resonance, sample_rate_);
219 target_in_0_ = coefficients.in_0;
220 target_in_1_ = coefficients.in_1;
221 target_in_2_ = coefficients.in_2;
222 target_out_1_ = coefficients.out_1;
223 target_out_2_ = coefficients.out_2;
224 break;
225 }
226 case Type::HighPass: {
227 mopo_float alpha = imag_delta / (2.0 * resonance);
228 mopo_float norm = 1.0 + alpha;
229 target_in_0_ = (1.0 + real_delta) / (2.0 * norm);
230 target_in_1_ = -(1.0 + real_delta) / norm;
231 target_in_2_ = target_in_0_;
232 target_out_1_ = -2.0 * real_delta / norm;
233 target_out_2_ = (1.0 - alpha) / norm;
234 break;
235 }
236 case Type::BandPass: {
237 mopo_float alpha = imag_delta / (2.0 * resonance);
238 mopo_float norm = 1.0 + alpha;
239 target_in_0_ = (imag_delta / 2.0) / norm;
240 target_in_1_ = 0;
241 target_in_2_ = -target_in_0_;
242 target_out_1_ = -2.0 * real_delta / norm;
243 target_out_2_ = (1.0 - alpha) / norm;
244 break;
245 }
246 case Type::LowShelf: {
247 mopo_float g = std::sqrt(gain);
248 mopo_float alpha = (imag_delta / 2.0) * std::sqrt((g + 1.0 / g) *
249 (1.0 / resonance - 1) + 2.0);
250 mopo_float sq = 2 * std::sqrt(g) * alpha;
251 mopo_float norm = (g + 1) + (g - 1) * real_delta + sq;
252
253 target_in_0_ = ((g + 1) - (g - 1) * real_delta + sq) * (g / norm);
254 target_in_1_ = 2 * ((g - 1) - (g + 1) * real_delta) * (g / norm);
255 target_in_2_ = ((g + 1) - (g - 1) * real_delta - sq) * (g / norm);
256 target_out_1_ = -2 * ((g - 1) + (g + 1) * real_delta) / norm;
257 target_out_2_ = ((g + 1) + (g - 1) * real_delta - sq) / norm;
258 break;
259 }
260 case Type::HighShelf: {
261 mopo_float g = std::sqrt(gain);
262 mopo_float alpha = (imag_delta / 2.0) * std::sqrt((g + 1.0 / g) *
263 (1.0 / resonance - 1) + 2.0);
264 mopo_float sq = 2 * std::sqrt(g) * alpha;
265 mopo_float norm = (g + 1) - (g - 1) * real_delta + sq;
266
267 target_in_0_ = ((g + 1) + (g - 1) * real_delta + sq) * (g / norm);
268 target_in_1_ = -2 * ((g - 1) + (g + 1) * real_delta) * (g / norm);
269 target_in_2_ = ((g + 1) + (g - 1) * real_delta - sq) * (g / norm);
270 target_out_1_ = 2 * ((g - 1) - (g + 1) * real_delta) / norm;
271 target_out_2_ = ((g + 1) - (g - 1) * real_delta - sq) / norm;
272 break;
273 }
274 case Type::BandShelf: {
275 mopo_float g = std::sqrt(gain);
276 mopo_float alpha = imag_delta *
277 sinh(log(2.0) * resonance * phase_delta / (2.0 * imag_delta));
278 mopo_float norm = 1.0 + alpha / g;
279
280 target_in_0_ = (1.0 + alpha * g) / norm;
281 target_in_1_ = -2.0 * real_delta / norm;
282 target_in_2_ = (1.0 - alpha * g) / norm;
283 target_out_1_ = -2.0 * real_delta / norm;
284 target_out_2_ = (1.0 - alpha / g) / norm;
285 break;
286 }
287 case Type::AllPass: {
288 mopo_float alpha = imag_delta / (2.0 * resonance);
289 mopo_float norm = 1.0 + alpha;
290 target_in_0_ = (1.0 - alpha) / norm;
291 target_in_1_ = -2.0 * real_delta / norm;
292 target_in_2_ = 1.0;
293 target_out_1_ = -2.0 * real_delta / norm;
294 target_out_2_ = (1.0 - alpha) / norm;
295 break;
296 }
297 case Type::Notch: {
298 mopo_float alpha = imag_delta / (2.0 * resonance);
299 mopo_float norm = 1.0 + alpha;
300 target_in_0_ = 1.0 / norm;
301 target_in_1_ = -2.0 * real_delta / norm;
302 target_in_2_ = target_in_0_;
303 target_out_1_ = target_in_1_;
304 target_out_2_ = (1.0 - alpha) / norm;
305 break;
306 }
307 case Type::GainedBandPass: {
308 mopo_float alpha = imag_delta / (2.0 * resonance);
309 mopo_float norm = 1.0 + alpha;
310 target_in_0_ = gain * (imag_delta / (2.0 * resonance)) / norm;
311 target_in_1_ = 0;
312 target_in_2_ = -target_in_0_;
313 target_out_1_ = -2.0 * real_delta / norm;
314 target_out_2_ = (1.0 - alpha) / norm;
315 break;
316 }
317 default: {
318 target_in_0_ = 1.0;
319 target_in_2_ = target_in_1_ = target_out_2_ = target_out_1_ = 0.0;
320 }
321 }
322
323 current_cutoff_ = cutoff;
324 current_resonance_ = resonance;
325 }

335 const mopo_float phase_delta = 2.0 * PI * cutoff / sample_rate_;

337 const mopo_float imag_delta = sin(phase_delta);
338
339 switch (type) {
340 case Type::LowPass: {
341 const Coefficients coefficients = lowPassCoefficients(cutoff, resonance, sample_rate_);
342 target_in_0_ = coefficients.in_0;
343 target_in_1_ = coefficients.in_1;
344 target_in_2_ = coefficients.in_2;
345 target_out_1_ = coefficients.out_1;
346 target_out_2_ = coefficients.out_2;

358 }

360 const mopo_float alpha = imag_delta / (2.0 * resonance);
361 const mopo_float norm = 1.0 + alpha;
362 target_in_0_ = (imag_delta / 2.0) / norm;
363 target_in_1_ = 0;
364 target_in_2_ = -target_in_0_;
365 target_out_1_ = -2.0 * real_delta / norm;
366 target_out_2_ = (1.0 - alpha) / norm;
367 break;
368 }

371 const mopo_float alpha = (imag_delta / 2.0) * std::sqrt((g + 1.0 / g) *

376 target_in_0_ = ((g + 1) - (g - 1) * real_delta + sq) * (g / norm);
377 target_in_1_ = 2 * ((g - 1) - (g + 1) * real_delta) * (g / norm);
378 target_in_2_ = ((g + 1) - (g - 1) * real_delta - sq) * (g / norm);
379 target_out_1_ = -2 * ((g - 1) + (g + 1) * real_delta) / norm;
380 target_out_2_ = ((g + 1) + (g - 1) * real_delta - sq) / norm;
381 break;
382 }
383 case Type::HighShelf: {
384 const mopo_float g = std::sqrt(gain);
385 const mopo_float alpha = (imag_delta / 2.0) * std::sqrt((g + 1.0 / g) *
386 (1.0 / resonance - 1) + 2.0);
387 const mopo_float sq = 2 * std::sqrt(g) * alpha;
388 const mopo_float norm = (g + 1) - (g - 1) * real_delta + sq;
389
390 target_in_0_ = ((g + 1) + (g - 1) * real_delta + sq) * (g / norm);
391 target_in_1_ = -2 * ((g - 1) + (g + 1) * real_delta) * (g / norm);
392 target_in_2_ = ((g + 1) + (g - 1) * real_delta - sq) * (g / norm);
393 target_out_1_ = 2 * ((g - 1) - (g + 1) * real_delta) / norm;
394 target_out_2_ = ((g + 1) - (g - 1) * real_delta - sq) / norm;
395 break;
396 }
397 case Type::AllPass: {
398 const mopo_float alpha = imag_delta / (2.0 * resonance);
399 const mopo_float norm = 1.0 + alpha;
400 target_in_0_ = (1.0 - alpha) / norm;
401 target_in_1_ = -2.0 * real_delta / norm;
402 target_in_2_ = 1.0;
403 target_out_1_ = -2.0 * real_delta / norm;
404 target_out_2_ = (1.0 - alpha) / norm;
405 break;
406 }
407 case Type::Notch: {
408 const mopo_float alpha = imag_delta / (2.0 * resonance);
409 const mopo_float norm = 1.0 + alpha;
410 target_in_0_ = 1.0 / norm;
411 target_in_1_ = -2.0 * real_delta / norm;
412 target_in_2_ = target_in_0_;
413 target_out_1_ = target_in_1_;
414 target_out_2_ = (1.0 - alpha) / norm;
415 break;
416 }
417 default: {
418 // Explicit fallback keeps unsupported modes on legacy coefficient logic.
419 computeCoefficients(type, cutoff, resonance, gain);
420 return;
421 }
422 }
423
424 current_cutoff_ = cutoff;
425 current_resonance_ = resonance;
426 }

428 inline void BiquadFilter::tick(int i, std::span<mopo_float> dest, std::span<const mopo_float> audio_buffer) {
429 mopo_float audio = audio_buffer[i];
430 mopo_float out = audio * in_0_ +
431 past_in_1_ * in_1_ +
432 past_in_2_ * in_2_ -
433 past_out_1_ * out_1_ -
434 past_out_2_ * out_2_;
435 past_in_2_ = past_in_1_;
436 past_in_1_ = audio;
437 past_out_2_ = past_out_1_;
438 past_out_1_ = out;
439 dest[i] = out;
440 }
