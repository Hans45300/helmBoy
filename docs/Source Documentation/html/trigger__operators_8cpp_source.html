53 void TriggerWait::process() {
54 using enum Inputs;
55 output()->clearTrigger();
56
57 if (input(static_cast<int>(Inputs::Wait))->source->triggered &&
58 input(static_cast<int>(Inputs::Trigger))->source->triggered) {
59
60 if (input(static_cast<int>(Inputs::Wait))->source->trigger_offset <=
61 input(static_cast<int>(Inputs::Trigger))->source->trigger_offset) {
62 waitTrigger(input(static_cast<int>(Inputs::Wait))->source->trigger_value);
63 sendTrigger(input(static_cast<int>(Inputs::Trigger))->source->trigger_offset);
64 }
65 else {
66 sendTrigger(input(static_cast<int>(Inputs::Trigger))->source->trigger_offset);
67 waitTrigger(input(static_cast<int>(Inputs::Wait))->source->trigger_value);
68 }
69 }
70 else if (input(static_cast<int>(Inputs::Wait))->source->triggered)
71 waitTrigger(input(static_cast<int>(Inputs::Wait))->source->trigger_value);
72 else if (input(static_cast<int>(Inputs::Trigger))->source->triggered)
73 sendTrigger(input(static_cast<int>(Inputs::Trigger))->source->trigger_offset);
74 }

117 void LegatoFilter::process() {
118 using enum Inputs;
119 using enum Outputs;
120 output(static_cast<int>(Outputs::Retrigger))->clearTrigger();
121 output(static_cast<int>(Outputs::Remain))->clearTrigger();
122 if (!input(static_cast<int>(Inputs::Trigger))->source->triggered)
123 return;
124
125 if (static_cast<int>(input(static_cast<int>(Inputs::Trigger))->source->trigger_value) == static_cast<int>(kVoiceOn) &&
126 static_cast<int>(last_value_) == static_cast<int>(kVoiceOn) && input(static_cast<int>(Inputs::Legato))->at(0)) {
127 output(static_cast<int>(Outputs::Remain))->trigger(
128 input(static_cast<int>(Inputs::Trigger))->source->trigger_value,
129 input(static_cast<int>(Inputs::Trigger))->source->trigger_offset);
130 }
131 else {
132 output(static_cast<int>(Outputs::Retrigger))->trigger(
133 input(static_cast<int>(Inputs::Trigger))->source->trigger_value,
134 input(static_cast<int>(Inputs::Trigger))->source->trigger_offset);
135 }
136 last_value_ = input(static_cast<int>(Inputs::Trigger))->source->trigger_value;
137 }

142 void PortamentoFilter::updateReleased() {
143 using enum Inputs;
144 if (!input(static_cast<int>(Inputs::VoiceTrigger))->source->triggered)
145 return;
146
147 int voice_trigger = input(static_cast<int>(Inputs::VoiceTrigger))->source->trigger_value;
148 if (voice_trigger == kVoiceOff)
149 released_ = true;
150 }
151
152 void PortamentoFilter::updateTrigger() {
153 using enum Inputs;
154 using enum State;
155 output()->clearTrigger();
156 if (!input(static_cast<int>(Inputs::FrequencyTrigger))->source->triggered)
157 return;
158
159 int state = static_cast<int>(input(static_cast<int>(Inputs::Portamento))->at(0));
160 if (state == static_cast<int>(State::PortamentoOff) || (state == static_cast<int>(State::PortamentoAuto) && released_)) {
161 output()->trigger(input(static_cast<int>(Inputs::FrequencyTrigger))->source->trigger_value,
162 input(static_cast<int>(Inputs::FrequencyTrigger))->source->trigger_offset);
163 released_ = false;
164 }
165 }
