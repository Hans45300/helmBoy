52SynthSlider::SynthSlider(String name) : Slider(name), bipolar_(false), flip_coloring_(false),
53 active_(true), snap_to_value_(false), snap_value_(0.0),
54 string_lookup_(nullptr), parent_(nullptr) {
55 popup_placement_ = BubbleComponent::below;
56 popup_buffer_ = DEFAULT_POPUP_BUFFER;
57
58 if (!mopo::Parameters::isParameter(name.toStdString()))
59 return;
60
61 setRotaryParameters(2.0f * mopo::PI - rotary_angle, 2.0f * mopo::PI + rotary_angle, true);
62 details_ = mopo::Parameters::getDetails(name.toStdString());
63 if (details_.steps)
64 setRange(details_.min, details_.max, (details_.max - details_.min) / (details_.steps - 1));
65 else
66 setRange(details_.min, details_.max);
67
68 setDoubleClickReturnValue(true, details_.default_value);
69 setTextBoxStyle(Slider::NoTextBox, true, 0, 0);
70
71 // Avoid stale cached bitmaps when controls are toggled or relayouted.
72 setBufferedToImage(false);
73 setColour(Slider::backgroundColourId, Colour(Colors::Color_ff303030));
74 setColour(Slider::textBoxOutlineColourId, Colour(Colors::Color_00000000));
75}

85void SynthSlider::mouseDown(const MouseEvent& e) {
86 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
87 if (parent == nullptr)
88 return;
89 SynthBase* synth = parent->getSynth();
90
91 if (e.mods.isPopupMenu()) {
92 PopupMenu m;
93 m.setLookAndFeel(DefaultLookAndFeel::instance());
94
95 if (isDoubleClickReturnEnabled())
96 m.addItem(kDefaultValue, "Set to Default Value");
97
98 m.addItem(kEnterValue, "Enter Value...");
99
100 std::vector<mopo::ModulationConnection*> connections;
101 m.addItem(kArmMidiLearn, "Learn MIDI Assignment");
102 if (parent->getSynth()->isMidiMapped(getName().toStdString()))
103 m.addItem(kClearMidiLearn, "Clear MIDI Assignment");
104
105 connections = parent->getSynth()->getDestinationConnections(getName().toStdString());
106
107 String disconnect("Disconnect from ");
108 for (int i = 0; i < connections.size(); ++i)
109 m.addItem(kModulationList + i, disconnect + connections[i]->source);
110
111 if (connections.size() > 1)
112 m.addItem(kClearModulations, "Disconnect all modulations");
113
114 showModulationPopup(m, this, false, kModulationList + static_cast<int>(connections.size()),
115 [this](int result) { handlePopupResult(result); });
116 }
117 else {
118 Slider::mouseDown(e);
119
120 if (parent)
121 synth->beginChangeGesture(getName().toStdString());
122 if (isRotary()) {
123 click_position_ = e.getScreenPosition().toFloat();
124 setMouseCursor(MouseCursor::NoCursor);
125 }
126 }
127}

175String SynthSlider::getTextFromValue(double value) {
176 if (string_lookup_) {
177 int lookup = mopo::utils::iclamp(value, 0, getMaximum());
178 return string_lookup_[lookup];
179 }
180
181 float display_value = value;
182 switch (details_.display_skew) {
183 case mopo::DisplaySkew::kQuadratic:
184 display_value = powf(display_value, 2.0f);
185 break;
186 case mopo::DisplaySkew::kExponential:
187 display_value = powf(2.0f, display_value);
188 break;
189 case mopo::DisplaySkew::kSquareRoot:
190 display_value = sqrt(display_value);
191 break;
192 default:
193 break;
194 }
195 display_value += details_.post_offset;
196 if (details_.display_invert)
197 display_value = 1.0 / display_value;
198 display_value *= details_.display_multiply;
199
200 return formatValue(display_value);
201}

203double SynthSlider::snapValue(double attempted_value, DragMode drag_mode) {
204 const double percent = 0.05;
205 if (!snap_to_value_ || drag_mode != DragMode::absoluteDrag)
206 return attempted_value;
207
208 double range = getMaximum() - getMinimum();
209 double radius = percent * range;
210 if (attempted_value - snap_value_ <= radius && attempted_value - snap_value_ >= -radius)
211 return snap_value_;
212 return attempted_value;
213}

226void SynthSlider::drawRotaryShadow(Graphics &g) {
227 static const DropShadow shadow(Colour(Colors::Color_ee000000), 3, Point<int>(0, 0));
228 static const float stroke_percent = 0.12f;
229
230 g.saveState();
231 g.setOrigin(getX(), getY());
232
233 float full_radius = std::min(getWidth() / 2.0f, getHeight() / 2.0f);
234 float stroke_width = 2.0f * full_radius * stroke_percent;
235 Path shadow_path;
236 float outer_radius = full_radius - stroke_width;
237 shadow_path.addCentredArc(full_radius, full_radius,
238 0.89f * full_radius, 0.87f * full_radius,
239 0, -rotary_angle, rotary_angle, true);
240 shadow.drawForPath(g, shadow_path);
241
242 Path rail_outer;
243 rail_outer.addCentredArc(full_radius, full_radius, outer_radius, outer_radius,
244 0.0f, -rotary_angle, rotary_angle, true);
245
246 g.setColour(Colour(Colors::Color_ff333333));
247
248 PathStrokeType outer_stroke =
249 PathStrokeType(stroke_width, PathStrokeType::beveled, PathStrokeType::butt);
250 g.strokePath(rail_outer, outer_stroke);
251
252 g.restoreState();
253}

286String SynthSlider::formatValue(float value) {
287 static const int number_length = 5;
288 static const int max_decimals = 3;
289
290 if (details_.steps)
291 return String(value) + " " + details_.display_units;
292
293 String format = String(value, max_decimals);
294 format = format.substring(0, number_length);
295 int spaces = number_length - format.length();
296
297 for (int i = 0; i < spaces; ++i)
298 format = " " + format;
299
300 return format + " " + details_.display_units;
301}

308void SynthSlider::handlePopupResult(int result) {
309 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
310 if (parent == nullptr)
311 return;
312
313 SynthBase* synth = parent->getSynth();
314 std::vector<mopo::ModulationConnection*> connections =
315 parent->getSynth()->getDestinationConnections(getName().toStdString());
316
317 if (result == kArmMidiLearn)
318 synth->armMidiLearn(getName().toStdString());
319 else if (result == kClearMidiLearn)
320 synth->clearMidiLearn(getName().toStdString());
321 else if (result == kDefaultValue)
322 setValue(getDoubleClickReturnValue());
323 else if (result == kEnterValue)
324 promptForDirectValueEntry();
325 else if (result == kClearModulations) {
326 for (auto* connection : connections) {
327 std::string source = connection->source;
328 synth->disconnectModulation(connection);
329 }
330 for (auto* listener : slider_listeners_)
331 listener->modulationsChanged(getName().toStdString());
332 }
333 else if (result >= kModulationList) {
334 int connection_index = result - kModulationList;
335 if (connection_index < 0 || connection_index >= static_cast<int>(connections.size()))
336 return;
337 std::string source = connections[connection_index]->source;
338 synth->disconnectModulation(connections[connection_index]);
339
340 for (auto* listener : slider_listeners_)
341 listener->modulationsChanged(getName().toStdString());
342 }
343}

345void SynthSlider::showModulationPopup(PopupMenu& menu, Component* owner, bool source_menu,
346 int first_value_id,
347 std::function<void(int)> handle_existing) {
348 auto* parent = owner->findParentComponentOfClass<SynthGuiInterface>();
349 if (parent == nullptr)
350 return;
351
352 const std::string name = owner->getName().toStdString();
353 const auto connections = source_menu ? parent->getSynth()->getSourceConnections(name) :
354 parent->getSynth()->getDestinationConnections(name);
355 std::vector<std::pair<std::string, std::string>> routes;
356 for (auto* connection : connections) {
357 menu.addItem(first_value_id + static_cast<int>(routes.size()),
358 "Enter Value for " + String(source_menu ? connection->destination : connection->source));
359 routes.emplace_back(connection->source, connection->destination);
360 }
361
362 SafePointer<Component> safe_owner(owner);
363 menu.showMenuAsync(PopupMenu::Options(), ModalCallbackFunction::create(
364 [safe_owner, routes, first_value_id, handle_existing](int result) {
365 if (safe_owner == nullptr || result == 0)
366 return;
367 const int route_index = result - first_value_id;
368 if (route_index >= 0 && route_index < static_cast<int>(routes.size())) {
369 auto* interface = safe_owner->findParentComponentOfClass<FullInterface>();
370 if (interface != nullptr)
371 interface->promptForModulationValueEntry(routes[route_index].first,
372 routes[route_index].second);
373 }
374 else
375 handle_existing(result);
376 }));
377}

379void SynthSlider::promptForModulationValueEntry(const std::string& source,
380 std::function<void(double)> apply_value,
381 Component* owner) {
382 auto* parent = findParentComponentOfClass<SynthGuiInterface>();
383 if (parent == nullptr)
384 return;
385 mopo::ModulationConnection* connection = nullptr;
386 for (auto* candidate : parent->getSynth()->getDestinationConnections(getName().toStdString())) {
387 if (candidate->source == source) {
388 connection = candidate;
389 break;
390 }
391 }
392 if (connection == nullptr)
393 return;
394
395 auto entry = std::make_shared<SynthSlider>(getName());
396 entry->details_ = details_;
397 entry->setStringLookup(string_lookup_);
398 entry->setRange(getMinimum(), getMaximum(), getInterval());
399 entry->setDoubleClickReturnValue(true, 0.0);
400 entry->setValue(connection->amount.value(), dontSendNotification);
401 entry->promptForDirectValueEntry(
402 [entry, apply_value](double amount) { apply_value(amount); },
403 String(source) + " -> " + getSliderDisplayName(), owner);
404}

406void SynthSlider::promptForDirectValueEntry(std::function<void(double)> apply_value,
407 String display_name, Component* owner) {
408 String current_text;
409 if (string_lookup_)
410 current_text = String(static_cast<int>(std::round(getValue())));
411 else {
412 current_text = getTextFromValue(getValue()).trim();
413 const String units = getUnits().trim();
414 if (units.isNotEmpty() && current_text.endsWith(units))
415 current_text = current_text.dropLastCharacters(units.length()).trimEnd();
416 }
417
418 const String message = (display_name.isEmpty() ? getSliderDisplayName() : display_name) +
419 "\n" + getAcceptedRangeText();
420 auto* alert = new AlertWindow("Enter Value", message, AlertWindow::NoIcon,
421 owner != nullptr ? owner : this);
422 alert->addTextEditor("value", current_text, "Value");
423 alert->addButton("OK", 1, KeyPress(KeyPress::returnKey));
424 alert->addButton("Cancel", 0, KeyPress(KeyPress::escapeKey));
425
426 SafePointer<SynthSlider> safe_this(this);
427 alert->enterModalState(true,
428 ModalCallbackFunction::create([safe_this, alert, apply_value](int result) {
429 std::unique_ptr<AlertWindow> alert_cleanup(alert);
430 if (result == 1 && safe_this != nullptr)
431 safe_this->applyDirectValueEntry(alert->getTextEditorContents("value"),
432 apply_value);
433 }),
434 true);
435}

437void SynthSlider::applyDirectValueEntry(const String& input_text,
438 const std::function<void(double)>& apply_value) {
439 double parsed_display_value = 0.0;
440 if (!tryParseDisplayValue(input_text, parsed_display_value)) {
441 AlertWindow::showMessageBoxAsync(AlertWindow::WarningIcon,
442 "Invalid value",
443 "Please enter a valid numeric value.");
444 return;
445 }
446
447 double internal_value = 0.0;
448 if (!tryConvertDisplayToInternalValue(parsed_display_value, internal_value)) {
449 AlertWindow::showMessageBoxAsync(AlertWindow::WarningIcon,
450 "Invalid value",
451 "This value cannot be represented for this control.");
452 return;
453 }
454
455 const double safe_value = clampAndSnapToRange(internal_value);
456
457 if (apply_value) {
458 apply_value(safe_value);
459 return;
460 }
461
462 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
463 if (parent != nullptr)
464 parent->getSynth()->beginChangeGesture(getName().toStdString());
465
466 setValue(safe_value);
467
468 if (parent != nullptr)
469 parent->getSynth()->endChangeGesture(getName().toStdString());
470}

472bool SynthSlider::tryParseDisplayValue(const String& input_text, double& parsed_value) const {
473 String normalized = input_text.trim().replaceCharacter(',', '.');
474 if (normalized.isEmpty())
475 return false;
476
477 const std::string utf8 = normalized.toStdString();
478 const char* begin = utf8.c_str();
479 char* end = nullptr;
480 const double value = std::strtod(begin, &end);
481 if (end == begin || !std::isfinite(value))
482 return false;
483
484 while (*end != '\0') {
485 const unsigned char current = static_cast<unsigned char>(*end);
486 if (std::isspace(current) || std::isalpha(current) || *end == '%' || *end == '/' ||
487 *end == '-' || *end == '_') {
488 ++end;
489 continue;
490 }
491 return false;
492 }
493
494 parsed_value = value;
495 return true;
496}

498bool SynthSlider::tryConvertDisplayToInternalValue(double display_value, double& internal_value) const {
499 if (!std::isfinite(display_value))
500 return false;
501
502 if (string_lookup_) {
503 internal_value = display_value;
504 return true;
505 }
506
507 if (std::abs(details_.display_multiply) < std::numeric_limits<double>::epsilon())
508 return false;
509
510 double value = display_value / details_.display_multiply;
511 if (details_.display_invert) {
512 if (std::abs(value) < std::numeric_limits<double>::epsilon())
513 return false;
514 value = 1.0 / value;
515 }
516
517 value -= details_.post_offset;
518
519 switch (details_.display_skew) {
520 case mopo::DisplaySkew::kQuadratic:
521 if (value < 0.0)
522 return false;
523 value = std::sqrt(value);
524 break;
525 case mopo::DisplaySkew::kExponential:
526 if (value <= 0.0)
527 return false;
528 value = std::log2(value);
529 break;
530 case mopo::DisplaySkew::kSquareRoot:
531 value = value * value;
532 break;
533 default:
534 break;
535 }
536
537 if (!std::isfinite(value))
538 return false;
539
540 internal_value = value;
541 return true;
542}

544bool SynthSlider::tryConvertInternalToDisplayValue(double internal_value, double& display_value) const {
545 if (!std::isfinite(internal_value))
546 return false;
547
548 if (string_lookup_) {
549 display_value = std::round(internal_value);
550 return true;
551 }
552
553 double value = internal_value;
554 switch (details_.display_skew) {
555 case mopo::DisplaySkew::kQuadratic:
556 value = std::pow(value, 2.0);
557 break;
558 case mopo::DisplaySkew::kExponential:
559 value = std::pow(2.0, value);
560 break;
561 case mopo::DisplaySkew::kSquareRoot:
562 if (value < 0.0)
563 return false;
564 value = std::sqrt(value);
565 break;
566 default:
567 break;
568 }
569
570 value += details_.post_offset;
571 if (details_.display_invert) {
572 if (std::abs(value) < std::numeric_limits<double>::epsilon())
573 return false;
574 value = 1.0 / value;
575 }
576
577 value *= details_.display_multiply;
578 if (!std::isfinite(value))
579 return false;
580
581 display_value = value;
582 return true;
583}

585String SynthSlider::getAcceptedRangeText() const {
586 const double minimum = getMinimum();
587 const double maximum = getMaximum();
588
589 if (string_lookup_)
590 return "Accepted range: " + String(static_cast<int>(minimum)) + " to " +
591 String(static_cast<int>(maximum));
592
593 double display_min = 0.0;
594 double display_max = 0.0;
595 if (!tryConvertInternalToDisplayValue(minimum, display_min) ||
596 !tryConvertInternalToDisplayValue(maximum, display_max)) {
597 display_min = minimum;
598 display_max = maximum;
599 }
600
601 if (display_min > display_max)
602 std::swap(display_min, display_max);
603
604 const String units = getUnits().trim();
605 String range = "Accepted range: " + String(display_min, 4) + " to " + String(display_max, 4);
606 if (units.isNotEmpty())
607 range += " " + units;
608
609 return range;
610}

619double SynthSlider::clampAndSnapToRange(double value) const {
620 if (!std::isfinite(value))
621 value = getDoubleClickReturnValue();
622
623 value = jlimit(getMinimum(), getMaximum(), value);
624
625 const double interval = getInterval();
626 if (interval > 0.0) {
627 value = getMinimum() + std::round((value - getMinimum()) / interval) * interval;
628 value = jlimit(getMinimum(), getMaximum(), value);
629 }
630
631 return value;
632}
