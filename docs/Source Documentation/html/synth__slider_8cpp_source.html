50SynthSlider::SynthSlider(String name) : Slider(name), bipolar_(false), flip_coloring_(false),
51 active_(true), snap_to_value_(false), snap_value_(0.0),
52 string_lookup_(nullptr), parent_(nullptr) {
53 popup_placement_ = BubbleComponent::below;
54 popup_buffer_ = DEFAULT_POPUP_BUFFER;
55
56 if (!mopo::Parameters::isParameter(name.toStdString()))
57 return;
58
59 setRotaryParameters(2.0f * mopo::PI - rotary_angle, 2.0f * mopo::PI + rotary_angle, true);
60 details_ = mopo::Parameters::getDetails(name.toStdString());
61 if (details_.steps)
62 setRange(details_.min, details_.max, (details_.max - details_.min) / (details_.steps - 1));
63 else
64 setRange(details_.min, details_.max);
65
66 setDoubleClickReturnValue(true, details_.default_value);
67 setTextBoxStyle(Slider::NoTextBox, true, 0, 0);
68
69 setBufferedToImage(true);
70 setColour(Slider::backgroundColourId, Colour(Colors::Color_ff303030));
71 setColour(Slider::textBoxOutlineColourId, Colour(Colors::Color_00000000));
72}

82void SynthSlider::mouseDown(const MouseEvent& e) {
83 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
84 if (parent == nullptr)
85 return;
86 SynthBase* synth = parent->getSynth();
87
88 if (e.mods.isPopupMenu()) {
89 PopupMenu m;
90 m.setLookAndFeel(DefaultLookAndFeel::instance());
91
92 if (isDoubleClickReturnEnabled())
93 m.addItem(kDefaultValue, "Set to Default Value");
94
95 std::vector<mopo::ModulationConnection*> connections;
96 m.addItem(kArmMidiLearn, "Learn MIDI Assignment");
97 if (parent->getSynth()->isMidiMapped(getName().toStdString()))
98 m.addItem(kClearMidiLearn, "Clear MIDI Assignment");
99
100 connections = parent->getSynth()->getDestinationConnections(getName().toStdString());
101
102 String disconnect("Disconnect from ");
103 for (int i = 0; i < connections.size(); ++i)
104 m.addItem(kModulationList + i, disconnect + connections[i]->source);
105
106 if (connections.size() > 1)
107 m.addItem(kClearModulations, "Disconnect all modulations");
108
109 m.showMenuAsync(PopupMenu::Options(),
110 ModalCallbackFunction::forComponent(sliderPopupCallback, this));
111 }
112 else {
113 Slider::mouseDown(e);
114
115 if (parent)
116 synth->beginChangeGesture(getName().toStdString());
117 if (isRotary()) {
118 click_position_ = e.getScreenPosition().toFloat();
119 setMouseCursor(MouseCursor::NoCursor);
120 }
121 }
122}

170String SynthSlider::getTextFromValue(double value) {
171 if (string_lookup_) {
172 int lookup = mopo::utils::iclamp(value, 0, getMaximum());
173 return string_lookup_[lookup];
174 }
175
176 float display_value = value;
177 switch (details_.display_skew) {
178 case mopo::DisplaySkew::kQuadratic:
179 display_value = powf(display_value, 2.0f);
180 break;
181 case mopo::DisplaySkew::kExponential:
182 display_value = powf(2.0f, display_value);
183 break;
184 case mopo::DisplaySkew::kSquareRoot:
185 display_value = sqrt(display_value);
186 break;
187 default:
188 break;
189 }
190 display_value += details_.post_offset;
191 if (details_.display_invert)
192 display_value = 1.0 / display_value;
193 display_value *= details_.display_multiply;
194
195 return formatValue(display_value);
196}

198double SynthSlider::snapValue(double attempted_value, DragMode drag_mode) {
199 const double percent = 0.05;
200 if (!snap_to_value_ || drag_mode != DragMode::absoluteDrag)
201 return attempted_value;
202
203 double range = getMaximum() - getMinimum();
204 double radius = percent * range;
205 if (attempted_value - snap_value_ <= radius && attempted_value - snap_value_ >= -radius)
206 return snap_value_;
207 return attempted_value;
208}

221void SynthSlider::drawRotaryShadow(Graphics &g) {
222 static const DropShadow shadow(Colour(Colors::Color_ee000000), 3, Point<int>(0, 0));
223 static const float stroke_percent = 0.12f;
224
225 g.saveState();
226 g.setOrigin(getX(), getY());
227
228 float full_radius = std::min(getWidth() / 2.0f, getHeight() / 2.0f);
229 float stroke_width = 2.0f * full_radius * stroke_percent;
230 Path shadow_path;
231 float outer_radius = full_radius - stroke_width;
232 shadow_path.addCentredArc(full_radius, full_radius,
233 0.89f * full_radius, 0.87f * full_radius,
234 0, -rotary_angle, rotary_angle, true);
235 shadow.drawForPath(g, shadow_path);
236
237 Path rail_outer;
238 rail_outer.addCentredArc(full_radius, full_radius, outer_radius, outer_radius,
239 0.0f, -rotary_angle, rotary_angle, true);
240
241 g.setColour(Colour(Colors::Color_ff333333));
242
243 PathStrokeType outer_stroke =
244 PathStrokeType(stroke_width, PathStrokeType::beveled, PathStrokeType::butt);
245 g.strokePath(rail_outer, outer_stroke);
246
247 g.restoreState();
248}

281String SynthSlider::formatValue(float value) {
282 static const int number_length = 5;
283 static const int max_decimals = 3;
284
285 if (details_.steps)
286 return String(value) + " " + details_.display_units;
287
288 String format = String(value, max_decimals);
289 format = format.substring(0, number_length);
290 int spaces = number_length - format.length();
291
292 for (int i = 0; i < spaces; ++i)
293 format = " " + format;
294
295 return format + " " + details_.display_units;
296}

303void SynthSlider::handlePopupResult(int result) {
304 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
305 if (parent == nullptr)
306 return;
307
308 SynthBase* synth = parent->getSynth();
309 std::vector<mopo::ModulationConnection*> connections =
310 parent->getSynth()->getDestinationConnections(getName().toStdString());
311
312 if (result == kArmMidiLearn)
313 synth->armMidiLearn(getName().toStdString());
314 else if (result == kClearMidiLearn)
315 synth->clearMidiLearn(getName().toStdString());
316 else if (result == kDefaultValue)
317 setValue(getDoubleClickReturnValue());
318 else if (result == kClearModulations) {
319 for (auto* connection : connections) {
320 std::string source = connection->source;
321 synth->disconnectModulation(connection);
322 }
323 for (auto* listener : slider_listeners_)
324 listener->modulationsChanged(getName().toStdString());
325 }
326 else if (result >= kModulationList) {
327 int connection_index = result - kModulationList;
328 std::string source = connections[connection_index]->source;
329 synth->disconnectModulation(connections[connection_index]);
330
331 for (auto* listener : slider_listeners_)
332 listener->modulationsChanged(getName().toStdString());
333 }
334}
