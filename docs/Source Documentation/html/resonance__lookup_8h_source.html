56 [[nodiscard]] mopo_float qLookup(mopo_float magnitude) const {
57 mopo_float index = Q_RESOLUTION * utils::clamp(magnitude, 0.0, 1.0);
58 int int_index = static_cast<int>(index);
59 mopo_float fraction = index - int_index;
60 return utils::interpolate(q_lookup_[int_index],
61 q_lookup_[int_index + 1], fraction);
62 }
